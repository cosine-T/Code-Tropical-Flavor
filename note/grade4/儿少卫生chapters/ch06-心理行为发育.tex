% ----- 6.1 大脑发育 -----
\section{大脑发育}

\begin{defbox}
\textbf{心理行为发育（Psychological and Behavior Development）}：指从受精卵开始直至发育成熟，儿童心理和行为方面的变化过程，包括认知、语言、情绪情感、个性和社会化适应等多个方面。在整个发育过程中，\imp{脑发育}是个体心理和行为发育的物质基础。
\end{defbox}

\subsection{大脑发育的阶段特点}

\begin{keybox}
\textbf{（一）生命早期的大脑发育（0--2岁）}

生命早期大脑以惊人的速度成长，胎儿的脑和婴儿的脑通常被认为是"大脑发育的黄金期"。

\begin{itemize}[leftmargin=*]
    \item \imp{髓鞘化（Myelination）}：神经纤维被髓鞘包裹的过程，可提高神经元之间信号传递速度，促进神经传导效率
    \item 神经纤维髓鞘化从胎儿期和婴儿期开始，持续到青春期甚至成年早期
\end{itemize}

\textbf{（二）儿童期大脑发育}

\begin{itemize}[leftmargin=*]
    \item 儿童期是大脑结构和功能快速发育的重要时期
    \item \imp{2--6岁}是儿童大脑发育的关键时期，6岁时儿童的大脑重量已达成人90\%
    \item \imp{大脑网络（Cerebral Network）}：大脑不同空间位置的皮层区域通过结构和功能连接相互联系形成的网络模式。儿童期—青春期是大脑网络动态重塑的关键阶段
    \item \imp{大脑可塑性（Brain Plasticity）}：大脑对环境和经验做出反应而改变结构和功能的能力。儿童期大脑经历从较不成熟到接近成熟的过程，是可塑性最强的时期
\end{itemize}

\textbf{（三）青春期大脑发育}

\begin{itemize}[leftmargin=*]
    \item 青春期大脑结构与功能持续完善，皮层下边缘系统（与认知、情绪和社会功能相关）在青春期达到高峰
    \item 突触修剪（Synaptic Pruning）加速，使神经连接更加精确高效
    \item 脑网络大规模重组，从儿童期短程连接模式过渡到成人远距离连接模式
    \item 三大核心网络：\imp{默认模式网络（DMN）}、\imp{突显网络（SN）}、\imp{中央执行网络（CEN）}，分别与思维记忆、情绪注意处理和高层认知相关
\end{itemize}
\end{keybox}

% ----- 6.2 认知能力发育 -----
\section{认知能力发育}

\begin{defbox}
\textbf{认知（Cognition）}：一般指认识活动或认识过程，是大脑反映和把握客观事物的属性状态及相互关系，揭示事物对人的意义与作用的一种高级心理活动。认知能力主要包括感知觉、注意、记忆、思维等方面。
\end{defbox}

\subsection{Piaget认知发展理论}

\begin{keybox}
\textbf{Piaget的认知发展阶段理论：}

\begin{table}[H]
\centering
\caption{Piaget认知发展的四个阶段}
\begin{tabularx}{\textwidth}{l>{\raggedright\arraybackslash}p{2cm}>{\raggedright\arraybackslash}X}
\hline
\textbf{阶段} & \textbf{大致年龄} & \textbf{主要特征} \\
\hline
\imp{感知运动阶段} & 0--2岁 & 通过感知和动作认识世界；客体永久性（object permanence）概念形成 \\
\imp{前运算阶段} & 2--7岁 & 符号功能发展；以自我为中心；思维不可逆；泛灵论 \\
\imp{具体运算阶段} & 7--11岁 & 逻辑思维开始发展；守恒概念形成；可逆性思维；能进行简单分类和排序 \\
\imp{形式运算阶段} & 11岁以上 & 抽象逻辑思维；假设-演绎推理；能处理假设性问题和系统性思维 \\

\end{tabularx}
\end{table}
\end{keybox}

\subsection{Vygotsky的社会文化认知理论}

\begin{defbox}
\textbf{Vygotsky的社会文化认知理论}强调社会文化环境对儿童认知发展的决定性作用，与Piaget的个人建构论形成互补。

\textbf{核心概念：}
\begin{enumerate}[leftmargin=*]
    \item \imp{最近发展区（Zone of Proximal Development, ZPD）}：儿童独立解决问题所决定的实际发展水平与在成人指导或与能力更强的同伴合作下解决问题所决定的潜在发展水平之间的差距。教学应着眼于最近发展区，提供适度挑战。
    \item \imp{脚手架（Scaffolding）}：成人或能力更强的同伴为儿童提供的暂时性支持和指导，帮助儿童完成超出其当前能力水平的任务，随着儿童能力提高逐渐减少支持。
    \item \imp{语言作为思维工具}：Vygotsky认为语言不仅是交流工具，更是认知发展的核心工具。儿童通过"自我中心言语"（自言自语）来指导自己的行为，逐渐内化为内部语言（思维）。
\end{enumerate}
\end{defbox}

\subsection{Erikson的心理社会发展理论}

\begin{defbox}
\textbf{Erikson的心理社会发展理论}将人的一生划分为八个阶段，每个阶段都有其特定的发展危机（心理社会危机）。儿童少年期涵盖前五个阶段：

\begin{enumerate}[leftmargin=*]
    \item \imp{婴儿期（0--1岁）：信任 vs 不信任}——主要照顾者的敏感回应建立基本信任感
    \item \imp{幼儿期（1--3岁）：自主性 vs 羞耻与怀疑}——探索和独立尝试的能力，建立自主意识
    \item \imp{学龄前期（3--6岁）：主动性 vs 内疚感}——主动创造和计划活动，发展目的感
    \item \imp{学龄期（6--12岁）：勤奋感 vs 自卑感}——通过学习和完成任务获得能力感；无法完成则产生自卑
    \item \imp{青春期（12--18岁）：自我同一性 vs 角色混乱}——整合不同角色和自我认识形成稳定的自我同一性
\end{enumerate}

Erikson理论强调\imp{环境和社会文化}对个性发展的影响，补充了Freud性心理理论的不足。
\end{defbox}

\subsection{感知觉发育}

\begin{defbox}
\textbf{感知觉（Sensory Perception）}：大脑对当前作用于感觉器官的客观事物的反映。感觉和知觉是认知活动的基础。

\begin{itemize}[leftmargin=*]
    \item \imp{视觉}：新生儿只能看清15--20cm内物体；出生后半年知觉发展迅速；学龄期眼的结构和功能基本完善
    \item \imp{听觉}：新生儿有一定听觉能力，3--7天听觉已起作用；4岁时听觉已相当完善
    \item \imp{嗅觉}：出生时已较成熟，1--2岁即可辨别不同气味
    \item \imp{味觉}：出生时已较完善；4--5个月婴儿对食物味道微小改变表现出敏感反应，此时期是味觉发育关键期
    \item \imp{皮肤感觉}：包括触觉、痛觉、温度觉。新生儿触觉较发达；5--6岁时皮肤辨别的精细程度已较高
    \item \imp{知觉}：分为时间知觉、空间知觉和运动知觉。\textbf{视崖实验（Gibson \& Walk, 1960）}证明婴儿已具备深度知觉能力
\end{itemize}
\end{defbox}

\subsection{注意与记忆发育}

\begin{defbox}
\textbf{注意（Attention）的发展：}
\begin{itemize}[leftmargin=*]
    \item 新生儿期仅有定向注意（原始的不随意注意）
    \item 学龄前期以不随意注意占优势，注意稳定时间短、易分散；3岁开始出现随意注意
    \item 学龄期注意稳定性提高：7--10岁约20分钟，10--12岁约25分钟，12岁以上约30分钟
    \item 青少年期：注意稳定性可维持40分钟以上，可根据预定目的和任务自觉控制注意
\end{itemize}

\textbf{记忆（Memory）的发展：}
\begin{itemize}[leftmargin=*]
    \item 婴儿期是记忆迅速发展的第一个时期，以机械记忆为主
    \item 学龄期以后：有意识记逐渐超过无意识记成为主要方式；意义识记在记忆活动中占主导地位
    \item 青少年期：记忆达到高峰时期，有意识记占绝对主导，意义识记和抽象记忆能力增强
\end{itemize}
\end{defbox}

\subsection{思维发育}

\begin{defbox}
\textbf{思维（Thinking）的发展阶段：}
\begin{enumerate}[leftmargin=*]
    \item \imp{婴儿期}：直觉行动思维——通过感知和动作认识世界
    \item \imp{学龄前期}：具体形象思维——以自我为中心，"万物有灵论"
    \item \imp{学龄期}：从具体形象思维向抽象逻辑思维过渡——概念化和推理能力的发展
    \item \imp{青春期}：抽象逻辑思维占主导——能处理抽象命题，运用假设-演绎推理
\end{enumerate}
\end{defbox}

% ----- 6.3 语言发育 -----
\section{语言发育}

\begin{defbox}
\textbf{语言（Language）}：人类社会中客观存在的语言现象以及人们使用的符号系统，由词汇（音、形、义）和语法规则构成，是认知发展的产物。

\textbf{语言发育阶段：}
\begin{enumerate}[leftmargin=*]
    \item \imp{前语言阶段（0--12个月）}：哭叫→"咿呀学语"（6--10个月）→语音知觉和手势交流→10--14个月说出第一个词
    \item \imp{单词句阶段（1--1.5岁）}：单词语表达完整意思→18--24个月"词汇爆炸"（每周可学10--20个新词）→2岁词汇量约200个
    \item \imp{电报句阶段（1.5--2岁）}：两个词组成的简单句（如"爸爸吃"、"妈妈抱"），电报式语言
    \item \imp{学龄前语法学习（3--5岁）}：词汇量迅速增加，是一生中词汇量增长最快时期；1.5--3岁简单句发展为复合句；4岁能说较多复杂句子
    \item \imp{儿童后期和青少年期}：口语表达能力和内部语言持续发展，掌握书面语言，能维持较长时间对话
\end{enumerate}
\end{defbox}

% ----- 6.4 情绪情感发育 -----
\section{情绪情感发育}

\begin{defbox}
\textbf{情绪（Emotion）}：对客观事物是否满足自身需要而产生的态度体验。\textbf{情感（Emotional Feeling）}：与社会交往和实践活动中形成的与人的高级社会需要相联系的体验，如美感、道德感、理智感等。

\textbf{情绪情感发展的阶段性：}

\begin{enumerate}[leftmargin=*]
    \item \imp{婴儿期}：出生即表现基本情绪（哭、笑），2个月出现自发性微笑（社交微笑）——是婴儿与照顾者互动的重要标志；2--7个月出现基本情绪，12--24个月出现更复杂的情绪（愤怒、恐惧、害羞、骄傲、嫉妒）

    \item \imp{学龄前期}：情绪丰富但不稳定且多变，3岁儿童情绪易受情境影响，应对比成人强烈易冲动；3--7岁情绪内容相当丰富，逐渐发展同情、友谊、美感等高级情感

    \item \imp{学龄期}：情绪体验已基本具备成人情绪的各种表现形式，情感趋于稳定且水平提高

    \item \imp{青春期}：情绪情感迅速发展，保留儿童期痕迹又萌发成人成熟行为；情绪常表现为成熟与幼稚、主动与被动矛盾的辩证特征；容易出现情绪波动和冲动行为
\end{enumerate}
\end{defbox}

% ----- 6.5 个性与社会化 -----
\section{个性及社会化发展}

\subsection{个性发展}

\begin{defbox}
\textbf{个性（Personality）}：在一定生物遗传基础上，与社会环境相互作用中形成的独特、稳定且具有倾向性的心理特征总和，包括气质、性格、自我意识等方面。

\textbf{气质（Temperament）}：婴儿期最早表现出的较为明显而稳定的个人特征，与遗传密切相关，也可随环境改变。Thomas和Chess将婴儿气质分为三种类型：
\begin{enumerate}[leftmargin=*]
    \item \imp{易养型}——约40\%，节律性强，情绪积极，适应快
    \item \imp{难养型}——约10\%，节律性差，情绪消极，适应慢
    \item \imp{启动缓慢型}——约15\%，活动水平低，反应温和，适应需时间
    \item 中间型——约35\%
\end{enumerate}

\textbf{自我意识（Self-Consciousness）}：
\begin{itemize}[leftmargin=*]
    \item \imp{自我概念（Self-Concept）}：个人心目中对自己的印象
    \item \imp{自我评价（Self-Evaluation）}：对自身状况和行为表现某一维度上的评判
    \item \imp{自我体验（Self-Experience）}：对自身状况是否满意的情绪体验
\end{itemize}
\end{defbox}

\subsection{社会化发展}

\begin{defbox}
\textbf{社会化（Socialization）}：个体在特定社会环境中，学习社会规范、习惯、行为方式和价值观，从而成为社会合格成员的过程。儿童社会化发展包括：
\begin{itemize}[leftmargin=*]
    \item 学习和遵守社会规范和道德准则
    \item 发展人际关系（同伴关系、师生关系、家庭关系）
    \item 形成社会角色认同
    \item 发展社会适应能力
\end{itemize}
\end{defbox}

\subsection*{参考答案要点}

\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item \textbf{心理行为发育}：从受精卵至成熟，心理行为的变化过程，包括认知、语言、情绪、个性、社会化等。\textbf{大脑可塑性}：大脑对环境和经验做出反应改变结构和功能的能力，儿童期最强。\textbf{髓鞘化}：神经纤维被髓鞘包裹的过程，提高信号传递速度。

    \item \textbf{认知}：大脑反映和把握客观事物属性及其关系的高级心理活动，包括感知觉、注意、记忆、思维等。

    \item \textbf{自我意识}：个体对自己的认识，包括自我概念、自我评价和自我体验。\textbf{气质}：婴儿期最早表现的个人特征，Thomas和Chess分为易养型、难养型、启动缓慢型和中间型。\textbf{社会化}：个体学习社会规范和价值观成为社会合格成员的过程。

    \item (1)感知运动阶段（0--2岁）：通过感知和动作认识世界；(2)前运算阶段（2--7岁）：符号功能、自我中心、不可逆；(3)具体运算阶段（7--11岁）：守恒、可逆性、逻辑分类；(4)形式运算阶段（11岁以上）：抽象逻辑思维、假设-演绎推理。

    \item (1)生命早期（0--2岁）：大脑以惊人速度成长，髓鞘化开始；(2)儿童期：2--6岁为关键期，6岁脑重达成人90\%，脑网络动态重塑；(3)青春期：突触修剪加速，脑网络重组（DMN/SN/CEN），皮层下边缘系统达发育高峰。

    \item 新生儿期仅有定向注意→学龄前期不随意注意占优→学龄期7--10岁约20分钟，10--12岁约25分钟，12岁以上约30分钟→青少年期可维持40分钟以上。

    \item (1)前语言阶段（0--12个月）：咿呀学语→第一个词；(2)单词句阶段（1--1.5岁）：单词语→词汇爆炸；(3)电报句阶段（1.5--2岁）：简单双词语；(4)学龄前（3--5岁）：词汇激增、语法学习；(5)儿童后期和青少年期：内部语言发展、书面语言掌握。

    \item 婴儿期：基本情绪+社交微笑→复杂情绪；学龄前期：丰富但不稳定→逐渐出现高级情感；学龄期：情绪形式较完善，情感趋于稳定；青春期：成熟与幼稚矛盾交织，容易情绪波动。

    \item 三种类型：(1)易养型（约40\%）：节律强、情绪积极、适应快；(2)难养型（约10\%）：节律差、情绪消极、适应慢；(3)启动缓慢型（约15\%）：活动水平低、反应温和。其余约35\%为中间型。
\end{enumerate}
