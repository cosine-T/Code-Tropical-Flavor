% ----- 8.1 概述 -----
\section{儿童少年常见病概述}

\begin{defbox}
\textbf{儿童少年常见病（Common Diseases in Children and Adolescents）}：也称学生常见病，指在儿童少年中发病率较高的一类疾病，其发生发展与生长发育阶段有关，并受阶段性和学校生活环境的影响。

\textbf{主要种类：}
\begin{itemize}[leftmargin=*]
    \item 近视、肥胖、营养不良/营养缺乏症
    \item 脊柱弯曲异常
    \item 心理行为问题（注意力缺陷多动障碍等）
    \item 龋齿和牙周病
    \item 沙眼等
\end{itemize}
\end{defbox}

% ----- 8.2 近视 -----
\section{近视（难点*）}

\subsection{概念与分类}

\begin{defbox}
\textbf{近视（Myopia）}：在眼调节放松状态下，平行光线经眼球屈光系统后聚焦在视网膜之前的一种屈光不正类型。

\textbf{等效球镜（Spherical Equivalent, SE）}：SE = 球镜度数 + 1/2 × 柱镜度数。\\
流行病学中常用判定标准：SE $\leq$ -0.50D为近视；SE $\leq$ -6.0D为高度近视。

\textbf{近视的分类：}
\begin{enumerate}[leftmargin=*]
    \item \imp{按屈光成分}：屈光性近视（角膜/晶状体曲率过大）、轴性近视（眼轴过长，最常见）
    \item \imp{按病程进展和病理变化}：单纯性近视、病理性近视
    \item \imp{按近视度数}：低度近视、中度近视、高度近视（SE $\leq$ -6.0D）
    \item \imp{按公共卫生防控策略}：近视前期、近视发展期、高度近视期、病理性近视期
\end{enumerate}
\end{defbox}

\subsection{流行病学}

\begin{keybox}
\textbf{中国儿童青少年近视流行特点：}
\begin{itemize}[leftmargin=*]
    \item 中国是全球儿童青少年近视患病率最高的国家之一
    \item 近视检出率随年龄、学段增高而显著上升
    \item \imp{城市高于农村}，但差距在缩小
    \item 女童近视率高于男童
    \item 近视低龄化趋势明显，高度近视比例增加
\end{itemize}
\end{keybox}

\subsection{影响因素}

\begin{keybox}
\textbf{近视的影响因素（难点*）：}

\begin{enumerate}[leftmargin=*]
    \item \imp{遗传因素}：双生子和家系研究估计近视遗传度约70\%--80\%。父母近视（特别是高度近视）的儿童患病风险显著增高

    \item \imp{环境因素（最重要可控因素）}：
    \begin{itemize}
        \item \keyword{户外活动时间}——是近视发生发展的最重要作用保护因素。\imp{户外活动对近视的保护作用关键在于户外时间长短，而非运动类型或强度}。
        \item 近距离用眼——持续长时间读写、使用电子产品
        \item 光照不足——室内光线质量和照度不达标
    \end{itemize}

    \item \imp{生活方式因素}：静态行为增多、睡眠不足、营养不均衡（高糖饮食）等都是促进近视的危险因素

    \item \imp{生长发育因素}：身高突增期（青春期）眼轴快速增长，是近视度数加深的关键期。发育越早、增长越快，近视进展越迅速
\end{enumerate}
\end{keybox}

\subsection{预防与控制}

\begin{keybox}
\textbf{近视防控综合策略（难点*）：}

\begin{enumerate}[leftmargin=*]
    \item \imp{筛查与建立视力档案}——对0--6岁儿童和中小学生定期进行视力检查，建立屈光发育档案，动态观察屈光状态变化

    \item \imp{增加户外活动}——\keyword{每天白天户外活动不少于2小时}。户外强光可促进视网膜多巴胺释放，抑制眼轴增长，延缓近视进展。开阔视野可减少调节负荷

    \item \imp{培养良好用眼行为}——保持正确读写姿势（"一尺一拳一寸"）、适量控制近距离用眼时间、保持充足光照

    \item \imp{建设视觉友好环境}——家庭、学校、医疗卫生机构等多方共同参与

    \item \imp{规范开展近视筛查监测}——定期开展学生近视筛查监测，确定高危人群和危险因素

    \item \imp{分级防治管理}：
    \begin{itemize}
        \item 高危因素或远视储备不足者：针对性改变危险行为
        \item 视力下降者：到医疗机构接受医学验光检查，明确诊断
        \item 已近视者：配戴眼镜矫正屈光不正
        \item 近视进展快者：在医生指导下使用低浓度阿托品、角膜塑形镜（OK镜）等干预手段
    \end{itemize}
\end{enumerate}
\end{keybox}

% ----- 8.3 单纯性肥胖 -----
\section{单纯性肥胖}

\subsection{概念与判定}

\begin{defbox}
\textbf{单纯性肥胖（Simple Obesity）}：排除内分泌疾病和药物等继发性因素后，由于能量摄入超过能量消耗导致体内脂肪过度积聚，且达到危害健康程度的一种慢性代谢性疾病。

\textbf{判定标准（中国学龄儿童）：}
\begin{itemize}[leftmargin=*]
    \item 使用BMI作为筛查指标
    \item 按性别、年龄别的BMI百分位数判定：BMI $\geq$ P85为超重，BMI $\geq$ P95为肥胖
    \item 也可结合腰围（中心性肥胖）综合判断
\end{itemize}
\end{defbox}

\subsection{流行病学与影响因素}

\begin{keybox}
\textbf{流行趋势：}
中国儿童超重肥胖率持续上升。2020年6岁以下儿童超重率达10.4\%，6--17岁达19\%，预计2027年农村将接近城市水平。

\textbf{主要影响因素：}
\begin{enumerate}[leftmargin=*]
    \item \imp{遗传因素}——肥胖有明显的家族聚集性，遗传度约40\%--70\%
    \item \imp{膳食因素}——高能量密度饮食、含糖饮料、不吃早餐等
    \item \imp{体力活动不足}——静态行为时间过长（日均$>$8h）
    \item \imp{睡眠不足}
    \item \imp{环境因素}——致胖环境（obesogenic environment）：高热量食物的易获得性、缺乏运动空间等
    \item \imp{社会心理因素}
\end{enumerate}
\end{keybox}

\subsection{防治措施}

\begin{enumerate}[leftmargin=*]
    \item \imp{平衡膳食}——控制总能量摄入，增加蔬菜水果，减少含糖饮料和高脂食物
    \item \imp{增加体力活动}——每天至少60分钟中等强度以上体力活动
    \item \imp{减少静态行为}——限制屏幕时间$<$2h/天
    \item \imp{行为干预}——自我监测、目标设定、正向强化
    \item \imp{家庭参与}——家长以身作则，全家改变生活方式
    \item \imp{学校为基础的综合干预}——健康教育+健康环境+体育活动
\end{enumerate}

% ----- 8.4 龋齿 -----
\section{龋齿}

\begin{defbox}
\textbf{龋齿（Dental Caries）}：在以细菌为主的多种因素影响下，牙体硬组织（牙釉质、牙本质和牙骨质）发生慢性进行性破坏的感染性疾病。

\textbf{四联因素学说：}
\begin{enumerate}[leftmargin=*]
    \item \imp{宿主（牙齿）}——牙齿发育矿化不良、排列不齐者易患龋
    \item \imp{致龋微生物}——变形链球菌为主要致龋菌
    \item \imp{食物（糖类）}——蔗糖是致龋性最强的碳水化合物
    \item \imp{时间}——上述三因素持续作用足够时间
\end{enumerate}
\end{defbox}

\begin{tipbox}
\textbf{龋齿的预防措施：}
\begin{enumerate}[leftmargin=*]
    \item \imp{窝沟封闭}——针对磨牙深窝沟的最有效预防方法
    \item \imp{氟化物应用}——含氟牙膏、氟化水、局部涂氟
    \item \imp{口腔卫生}——正确刷牙（早晚各一次）、使用牙线
    \item \imp{控制糖摄入}——减少摄糖频次和数量
    \item \imp{定期口腔检查}——早发现、早治疗
\end{enumerate}
\end{tipbox}

% ----- 8.5 脊柱弯曲异常 -----
\section{脊柱弯曲异常}

\begin{defbox}
\textbf{脊柱弯曲异常（Spinal Curvature Abnormality）}：脊柱在冠状面、矢状面或轴位上偏离正常位置，形态和弯曲度出现异常。

\textbf{常见类型：}
\begin{enumerate}[leftmargin=*]
    \item \imp{脊柱侧弯（Scoliosis）}——脊柱向侧方弯曲（$>$10$^\circ$为阳性），最常见
    \item \imp{脊柱后凸异常（驼背）}
    \item \imp{脊柱前凸异常}
\end{enumerate}

\textbf{特发性脊柱侧弯}：最常见类型，约占80\%--85\%，病因不明，多见于青春期女性。
\end{defbox}

\begin{keybox}
\textbf{脊柱弯曲异常的防控（难点*）：}

\begin{enumerate}[leftmargin=*]
    \item \imp{筛查}——学校定期开展脊柱健康筛查（Adam前屈试验是主要筛查方法）
    \item \imp{早期干预}——轻度侧弯（Cobb角10$^\circ$--25$^\circ$）可进行姿势矫正和特异性运动治疗
    \item \imp{矫形支具}——中度侧弯（25$^\circ$--45$^\circ$）佩戴支具
    \item \imp{手术治疗}——重度侧弯（$>$45$^\circ$）可能需要手术
    \item \imp{预防措施}：保证充足营养（钙和维生素D）、正确坐姿站姿、减轻书包重量（使用双肩背包）、增加体育锻炼
\end{enumerate}
\end{keybox}

    \item \textbf{【简答题】}简述户外活动预防近视的机制。
    \item \textbf{【简答题】}简述儿童单纯性肥胖的主要影响因素及防治措施。
    \item \textbf{【简答题】}简述龋齿的四联因素学说及预防措施。
    \item \textbf{【简答题】}简述脊柱弯曲异常的筛查与防控策略。
\end{enumerate}

\subsection*{参考答案要点}

\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item \textbf{近视}：眼调节放松状态下，平行光线聚焦在视网膜之前的屈光不正。\textbf{SE}：球镜度数 + 1/2 × 柱镜度数。\textbf{单纯性肥胖}：排除继发性因素后，能量摄入超过消耗导致脂肪过度积聚的慢性代谢病。\textbf{四联因素}：宿主（牙齿）、微生物（变形链球菌）、食物（糖）、时间。

    \item \textbf{脊柱侧弯}：脊柱向侧方弯曲$>$10$^\circ$为阳性，常见于青春期女性。\textbf{Adam前屈试验}：患者弯腰向前，观察背部是否对称的脊柱侧弯筛查方法。

    \item 影响因素：遗传（遗传度70\%--80\%）、环境（户外活动少是最关键危险因素、近距离用眼过多、光照不足）、生活方式（静态行为多、睡眠不足）、生长发育（青春期眼轴快增）。策略：筛查建档、增加户外活动（$\geq$2h/天）、养成良好用眼习惯、视觉友好环境、分级防治管理。

    \item 户外强光刺激视网膜多巴胺释放→抑制眼轴增长→延缓近视进展。开阔视野减少调节负荷也是可能机制。\textbf{关键在于户外时间长短，而非运动类型}。

    \item 影响因素：遗传（40\%--70\%）、膳食（高能量密度、含糖饮料）、体力活动不足、睡眠不足、致胖环境、社会心理因素。防治：平衡膳食、增加体力活动（$\geq$60min/天）、减少静态行为（屏幕$<$2h/天）、行为干预、家庭参与、学校综合干预。

    \item 四联因素：宿主（牙齿矿化不良）、致龋菌（变形链球菌）、糖（蔗糖致龋性最强）、时间。预防：窝沟封闭、氟化物应用、正确刷牙、控制糖摄入、定期口腔检查。

    \item 筛查（Adam前屈试验）→分层管理：轻度（姿势矫正+运动治疗）、中度（支具矫正）、重度（手术）。预防：营养充足、正确姿势、减负书包、加强体育锻炼。
\end{enumerate}
