% ----- 12.1 概述 -----
\section{学校突发公共卫生事件概述}

\begin{defbox}
\textbf{突发公共卫生事件（Public Health Emergency）}：突然发生，造成或者可能造成社会公众健康严重损害的重大传染病疫情、群体性不明原因疾病、重大食物和职业中毒以及其他严重影响公众健康的事件。

\textbf{学校突发公共卫生事件}：在学校（包括托幼机构）发生的，对师生健康和生命安全造成或可能造成严重损害的突发公共卫生事件。
\end{defbox}

\subsection{分类与特点}

\begin{keybox}
\textbf{学校突发公共卫生事件的分类：}

\begin{enumerate}[leftmargin=*]
    \item \imp{重大传染病疫情}——如流感暴发、手足口病暴发、COVID-19等
    \item \imp{群体性食物中毒}——学校食堂或集体用餐引起的食物中毒事件
    \item \imp{群体性不明原因疾病}——短时间内出现多例原因不明的相似病例
    \item \imp{职业中毒}——化学实验、清洁消毒等引起的急性中毒
    \item \imp{其他严重影响师生健康的事件}——饮用水污染、环境因素引起的群体性不良反应等
\end{enumerate}

\textbf{学校突发公共卫生事件的特点：}
\begin{enumerate}[leftmargin=*]
    \item \imp{突发性和不确定性}——难以预测，发生突然
    \item \imp{群体性}——学校人口密集，容易短时间内波及大量师生
    \item \imp{社会影响大}——牵涉家长、社会，易引发恐慌和关注
    \item \imp{处理复杂性}——需要卫生、教育、疾控等多部门协作
    \item \imp{低龄化特点}——学生自我保护能力弱，需要我们给予更多保护
\end{enumerate}
\end{keybox}

\subsection{预防与应急准备}

\begin{defbox}
\textbf{学校突发公共卫生事件的常态化预防措施：}

\begin{enumerate}[leftmargin=*]
    \item \imp{建立健全应急预案}——每所学校应制定突发公共卫生事件应急预案
    \item \imp{完善组织管理体系}——成立由校长负责的学校公共卫生工作领导小组
    \item \imp{落实日常监测制度}——晨检、因病缺课追踪、传染病报告
    \item \imp{加强健康教育}——提高师生自我防护意识和能力
    \item \imp{保障物资储备}——储备必要的防护用品、消毒药品和器械
    \item \imp{定期开展应急演练}——检验预案可操作性，提高应急处置能力
    \item \imp{改善学校卫生条件}——保障食堂卫生、饮用水安全、环境卫生
\end{enumerate}
\end{defbox}

% ----- 12.2 传染病事件应急 -----
\section{学校传染病事件的应急行动}

\begin{keybox}
\textbf{学校传染病暴发应急处置流程：}

\begin{enumerate}[leftmargin=*]
    \item \imp{第一时间报告}
    \begin{itemize}
        \item 发现聚集性病例（班级内一天出现3例或3天内出现5例及以上相似症状）应立即报告
        \item 同时向\keyword{当地疾病预防控制中心}和\keyword{上级教育行政部门}报告
        \item 不得隐瞒、缓报、谎报
    \end{itemize}

    \item \imp{现场初步处置}
    \begin{itemize}
        \item 隔离患者：将疑似或确诊患者及时隔离
        \item 通风消毒：教室、宿舍、食堂等公共场所加强通风和消毒
        \item 排查密切接触者
        \item 停止大型集会活动，必要时暂停上课（停课标准由疾控部门评估）
    \end{itemize}

    \item \imp{配合流行病学调查}
    \begin{itemize}
        \item 协助疾控中心开展个案调查和暴发调查
        \item 提供学生名单、出勤记录、座位表等信息
        \item 采集必要的生物标本
    \end{itemize}

    \item \imp{落实防控措施}
    \begin{itemize}
        \item 应急接种（如麻疹、流腮等有疫苗可预防的传染病）
        \item 预防性服药（需在医生指导下）
        \item 加强健康监测（增加晨检频次）
    \end{itemize}

    \item \imp{信息发布与风险沟通}
    \begin{itemize}
        \item 及时向师生、家长通报疫情信息
        \item 消除恐慌情绪，传播科学的防控知识
        \item 做好舆情监测和引导
    \end{itemize}

    \item \imp{恢复与评估}
    \begin{itemize}
        \item 经疾控部门评估确认疫情结束后恢复常态
        \item 总结评估应急响应的经验教训
        \item 修订完善应急预案
    \end{itemize}
\end{enumerate}
\end{keybox}

% ----- 12.3 食物中毒应对 -----
\section{群体性食物中毒事件的应对}

\begin{defbox}
\textbf{学校群体性食物中毒}：学生在学校食堂或集体用餐后，多人出现相似消化道症状（恶心、呕吐、腹痛、腹泻等）的事件。

\textbf{常见原因：}
\begin{itemize}[leftmargin=*]
    \item 细菌性食物中毒——沙门氏菌、金黄色葡萄球菌、副溶血性弧菌等
    \item 化学性食物中毒——亚硝酸盐、有机磷农药、洗涤剂污染
    \item 有毒动植物中毒——未煮熟菜豆、发芽马铃薯、毒蘑菇等
\end{itemize}
\end{defbox}

\begin{keybox}
\textbf{食物中毒事件的应急处置要点：}

\begin{enumerate}[leftmargin=*]
    \item \imp{立即送医}——出现症状的学生立即送医院诊治
    \item \imp{停止供餐}——立即停止食堂或可疑供餐单位的供餐
    \item \imp{封存留样}——\keyword{封存48小时内所有食品留样}、原料和加工工具，等待卫生监督部门检验
    \item \imp{保护现场}——保留呕吐物、排泄物等现场证据
    \item \imp{报告}——同时报告疾控中心、卫生监督部门和教育行政部门
    \item \imp{配合调查}——协助卫生部门开展流行病学调查和食品卫生学调查
    \item \imp{追踪病例}——对所有进食可疑食物者进行追踪和医学观察
    \item \imp{信息沟通}——及时向家长通报情况，做好解释安抚工作
\end{enumerate}

\textbf{学校食堂的预防措施：}
\begin{itemize}[leftmargin=*]
    \item 严格食品采购索证索票制度
    \item 规范食品加工操作流程（生熟分开、充分加热）
    \item 落实食品留样制度（每餐每种食品留样$\geq$125g，保存48小时以上）
    \item 加强从业人员健康管理和食品安全培训
    \item 定期消毒餐具和加工场所
\end{itemize}
\end{keybox}

% ----- 12.4 群体性心因性反应 -----
\section{群体性心因性反应事件的应对}

\begin{defbox}
\textbf{群体性心因性反应（Mass Psychogenic Illness）}：在某一群体中，由心理社会因素诱发，多人出现相似主观症状（头痛、头晕、恶心、呼吸困难、昏厥等），但无器质性病变依据的群体现象。在学校中时有发生。

\textbf{特点：}
\begin{itemize}[leftmargin=*]
    \item 多见于女性、青春期学生
    \item 通常由某一诱发事件触发（如异味、疫苗接种、考试压力等）
    \item 症状通过暗示和模仿在群体中迅速传播
    \item 医学检查无阳性发现
    \item 排除食物中毒、传染病等器质性疾病后诊断
\end{itemize}

\textbf{处置原则：}
\begin{enumerate}[leftmargin=*]
    \item 排除器质性疾病（做好基础医学检查）
    \item 隔离疏散——将出现症状者与未受影响者分离
    \item 减少关注——避免过度关注和强化症状
    \item 心理疏导——专业人员进行心理干预
    \item 信息透明但不过度渲染
\end{enumerate}
\end{defbox}

\subsection*{参考答案要点}

\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item \textbf{学校突发公共卫生事件}：在学校发生的，对师生健康和生命安全造成或可能造成严重损害的突发公共卫生事件。\textbf{群体性心因性反应}：由心理社会因素诱发，多人出现相似主观症状但无器质性病变依据的群体现象。

    \item 分类：重大传染病疫情、群体性食物中毒、群体性不明原因疾病、职业中毒、其他。特点：突发性和不确定性、群体性、社会影响大、处置复杂性（多部门协作）、低龄化（学生自我保护能力弱）。

    \item 六步流程：(1)第一时间报告（疾控中心+教育行政部门，不得瞒报）；(2)现场初步处置（隔离患者、通风消毒、排查密接、必要时停课）；(3)配合流行病学调查；(4)落实防控措施（应急接种、预防性服药、加强监测）；(5)信息发布与风险沟通；(6)恢复与评估。

    \item (1)立即送医；(2)停止供餐；(3)封存48小时食品留样、原料和加工工具；(4)保护现场（保留呕吐物、排泄物）；(5)报告疾控、卫生监督和教育部门；(6)配合调查；(7)追踪所有进食可疑食物者；(8)信息沟通（向家长通报、解释安抚）。

    \item (1)建立应急预案；(2)完善组织管理（校长负责）；(3)落实日常监测（晨检、因病缺课追踪、传染病报告）；(4)加强健康教育；(5)保障物资储备；(6)定期应急演练；(7)改善学校卫生条件（食堂、饮用水、环境卫生）。

    \item (1)排除器质性疾病（做好基础医学检查）；(2)隔离疏散（将出现症状者与未受影响者分离）；(3)减少过度关注和症状强化；(4)专业心理疏导；(5)信息透明但不过度渲染。
\end{enumerate}
