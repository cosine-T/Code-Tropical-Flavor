\chapter{健康社会决定因素}
\setcounter{chapter}{3}
\chapterintro{
掌握健康社会决定因素的概念和理论模型（\textbf{教学难点}）；理解社会因素影响健康的规律与特点；了解健康社会决定因素的发展历程、行动框架及"将健康融入所有政策"的理念与实践。
}

% ----- 3.1 概述 -----
\section{社会因素概述}

\begin{defbox}
\textbf{社会因素（Social Factor）}：指社会的各项构成要素，包括\keyword{社会环境、社会经济、社会制度、社会文化、社会行为、卫生服务}等，是影响人群健康的重要因素。

\textbf{社会因素的内容：}
\begin{itemize}[leftmargin=*]
    \item 社会经济因素——经济发展水平、收入、贫困等
    \item 社会环境因素——人口、营养、城市化、社会支持等
    \item 社会文化因素——教育、文化、宗教、风俗习惯等
    \item 行为心理因素——生活方式、心理压力、人格等
    \item 卫生服务因素——卫生资源配置、服务可及性等
    \item 社会制度因素——政治体制、社会保障制度等
\end{itemize}
\end{defbox}

% ----- 3.2 健康社会决定因素概念 -----
\section{健康社会决定因素的概念}

\begin{defbox}
\textbf{健康社会决定因素（Social Determinants of Health, SDH）}：\imp{（教学难点*）}指在那些直接导致疾病的因素之外，由人们的社会地位和所拥有资源所决定的生活和工作环境条件，及其对健康产生影响的社会性因素。即\keyword{"影响健康影响因素的社会因素"}——决定人们健康水平的"原因的原因"（causes of the causes）。
\end{defbox}

\begin{keybox}
\textbf{理解"原因的原因"：}
\begin{itemize}[leftmargin=*]
    \item 直接病因：如吸烟→肺癌（直接因果关系）
    \item SDH：如低教育水平→吸烟率高→肺癌（社会因素是直接病因背后的深层原因）
    \item SDH关注的是\imp{健康不公平的社会根源}——为什么有些人比其他人更容易生病？
    \item 核心观点：\keyword{社会不平等→健康不平等}，社会结构性因素决定了人群健康状况的系统性差异
\end{itemize}
\end{keybox}

\begin{keybox}
\textbf{CSDH（2008）三大行动原则：}
\begin{enumerate}[leftmargin=*]
    \item \imp{改善日常生活条件}——从出生开始改善儿童早期发育；改善居住环境和工作条件；建立全民社会保障制度
    \item \imp{改变权力、资金和资源分配的不公平}——将健康公平纳入政府各部门政策；建立公平的筹资机制；增强市场行为的规范性
    \item \imp{测量和理解问题并评估行动效果}——建立健全的健康公平监测体系；培训研究人员；扩大公众对SDH的认知
\end{enumerate}
\end{keybox}

% ----- 3.4 SDH理论模型 -----
\section{健康社会决定因素的理论模型}

\subsection{Dahlgren和Whitehead模型（1991）}

\begin{keybox}
Dahlgren和Whitehead于1991年提出的\keyword{彩虹模型}是最经典的健康社会决定因素理论模型，将影响健康的因素分为五个层次：

\begin{enumerate}[leftmargin=*]
    \item \imp{核心层——个体因素}：年龄、性别、遗传因素（不可改变的个人特征）
    \item \imp{第二层——个人生活方式}：吸烟、饮酒、饮食、体力活动等健康相关行为
    \item \imp{第三层——社会与社区网络}：家庭支持、朋友网络、社区参与等社会关系因素
    \item \imp{第四层——生活与工作条件}：住房、工作环境、教育、医疗服务、水和卫生设施、农业和食物生产、失业等
    \item \imp{最外层——宏观社会经济、文化和环境条件}：国家经济政策、社会文化价值观、政治制度、全球化等
\end{enumerate}

该模型强调：\imp{从内到外的影响路径}——外层因素通过影响内层因素最终影响个体健康；\imp{从外到内的决定路径}——越外层的因素影响力越广泛深远。
\end{keybox}

\subsection{WHO的SDH行动框架}

\begin{defbox}
\textbf{WHO健康社会决定因素委员会（CSDH）行动框架}将影响健康的社会因素分为两大类别：

\begin{enumerate}
    \item \imp{结构性决定因素（Structural Determinants）}：社会经济和政治环境（社会治理、经济政策、社会政策、公共政策、文化和社会价值观）与社会经济地位（收入、教育、职业、社会阶层、性别、种族/民族）
    \item \imp{中介决定因素（Intermediary Determinants）}：物质环境（住房、居住环境、工作环境等）、社会心理环境（社会支持、压力等）、行为和生物因素、卫生系统
\end{enumerate}
\end{defbox}

% ----- 3.5 影响健康的主要社会因素 -----
\section{影响健康的主要社会因素}

\subsection{社会因素影响健康的规律与特点}

\begin{keybox}
\textbf{社会因素影响健康的五大特点：}
\begin{enumerate}[leftmargin=*]
    \item \imp{非特异性}：社会因素对健康的影响往往是广泛而非特异性的。同一社会因素可影响多种健康结局（如贫困可增加多种疾病的发病率和死亡率）
    \item \imp{交互作用}：各种社会因素之间相互影响、相互制约，共同作用于人群健康。社会经济、文化、行为因素之间存在复杂的交互效应
    \item \imp{累积效应}：社会因素对健康的影响往往具有长期累积效应。如生命历程中不同阶段的社会经济劣势可累积为成年后的健康损害
    \item \imp{广泛性和持久性}：社会因素影响整个人群，其作用持久而深远，往往跨越一代人
    \item \imp{双向性}：社会因素与健康之间存在双向关系——不良社会条件损害健康，而健康损害又会加剧社会劣势，形成恶性循环
\end{enumerate}
\end{keybox}

\subsection{主要社会因素}

\begin{tcolorbox}[colback=light, colframe=main, boxrule=0.5pt, arc=4pt, title=\textbf{影响健康的主要社会因素}, breakable]
\begin{enumerate}[leftmargin=*]
    \item \imp{社会经济因素}：收入水平、贫富差距、经济发展阶段、就业状况、职业类型等
    \item \imp{社会制度因素}：政治体制、法律制度、社会保障制度、医疗卫生制度等
    \item \imp{社会文化因素}：教育水平、文化传统、价值观念、道德规范、宗教信仰等
    \item \imp{社会人口因素}：人口数量、结构、素质、流动与迁移等
    \item \imp{社会关系因素}：家庭关系、社会支持网络、社区参与、社会资本等
    \item \imp{社会行为因素}：健康相关行为、生活方式、吸烟、饮酒、饮食、运动等
    \item \imp{卫生服务因素}：卫生资源的配置、质量、可及性和利用等
    \item \imp{环境因素}：自然环境（空气、水、土壤）、建成环境（住房、交通、绿地）等
\end{enumerate}
\end{tcolorbox}

% ----- 3.6 将健康融入所有政策 -----
\section{将健康融入所有政策}

\begin{defbox}
\textbf{将健康融入所有政策（Health in All Policies, HiAP）}：由WHO倡导，指在制定所有公共政策和决策时，系统性地、明确地考虑其对人群健康的影响，明确跨部门责任，加强部门协作，以实现改善人群健康和健康公平的目标。
\end{defbox}

\begin{keybox}
\textbf{HiAP的核心理念：}
\begin{enumerate}[leftmargin=*]
    \item 健康不仅是卫生部门的职责，政府部门的所有决策都可能影响健康
    \item 需要跨部门合作和政府高层领导的承诺
    \item 政策制定应考虑对健康的长远影响和在不同人群中的分布
    \item 实现健康公平是社会正义的体现
\end{enumerate}

\textbf{HiAP的实践领域（WHO, 2014年提出6个关键组成部分）：}
\begin{enumerate}[leftmargin=*]
    \item 确定HiAP的需求和优先领域
    \item 建立必要的组织框架和行动计划
    \item 建立支持性的机制和流程
    \item 增强跨部门协作的能力
    \item 确保监测、评估和报告
    \item 加强培训和能力建设
\end{enumerate}
\end{keybox}

\begin{tipbox}
\textbf{中国HiAP实践：}
\begin{itemize}[leftmargin=*]
    \item 2013年国家卫生计生委发布《"健康中国"战略规划》
    \item 2016年《"健康中国2030"规划纲要》将"将健康融入所有政策"确立为核心理念
    \item 爱国卫生运动是HiAP在中国的早期实践——多部门参与、全社会动员
    \item 健康影响评价制度正在逐步建立和完善
\end{itemize}
\end{tipbox}

\subsubsection*{二、名词解释}
\begin{enumerate}[leftmargin=*]
    \item \textbf{SDH}：由人们社会地位和资源所决定的生活与工作环境条件及其对健康产生影响的社会性因素，即"原因的原因"。
    \item \textbf{HiAP}：在制定所有公共政策时系统性地考虑其对健康的影响，加强跨部门协作，改善人群健康和健康公平。
    \item \textbf{结构性决定因素}：社会经济政治环境和社会经济地位（治理、政策、文化价值观与收入、教育、职业等）。\textbf{中介决定因素}：物质环境、社会心理环境、行为和生物因素、卫生系统。
\end{enumerate}

\subsubsection*{三、简答题}
\begin{enumerate}[leftmargin=*]
    \item 五大特点：非特异性、交互作用、累积效应、广泛性和持久性、双向性。社会因素影响力广泛深远，需要从多维度系统分析其健康效应。
    \item 五层结构：①个体因素（年龄、性别、遗传）；②个人生活方式；③社会与社区网络；④生活与工作条件；⑤宏观社会经济文化环境。外层通过影响内层最终影响个体健康。
    \item ①改善日常生活条件（从出生开始改善儿童早期发育、改善居住工作条件、建立全民社保）；②改变权力、资金和资源分配不公（健康公平融入各部门政策、公平筹资机制）；③测量理解问题并评估行动（建立监测体系、扩大公众认知）。
    \item 核心理念：健康不仅是卫生部门职责，需要跨部门合作；政策应考虑对健康长远影响和人群分布差异；实现健康公平是社会正义体现。
\end{enumerate}
