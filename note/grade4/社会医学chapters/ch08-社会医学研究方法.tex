% ----- 8.1 社会医学研究概述 -----
\section{社会医学研究概述}

\subsection{社会医学的相关研究方法}

社会医学研究综合运用多种研究方法：
\begin{enumerate}[leftmargin=*]
    \item \imp{调查研究法}——最常用的方法，包括问卷法、访谈法、观察法等
    \item \imp{文献研究法}——系统查阅和分析已有文献资料
    \item \imp{实验研究法}——社会干预实验、健康教育效果评价等
    \item \imp{比较研究法}——不同人群、地区、时期之间的比较
    \item \imp{综合评价法}——健康危险评价、生命质量评价、卫生服务综合评价等
\end{enumerate}

\subsection{社会医学研究的步骤}

\begin{tcolorbox}[colback=light, colframe=main, boxrule=0.5pt, arc=4pt, title=\textbf{社会医学研究的五个步骤}, breakable]
\begin{enumerate}[leftmargin=*]
    \item \imp{选题与确定研究问题}——根据实际需求、资料可及性和研究条件提出明确的研究问题
    \item \imp{制定研究设计/方案}——包括研究目的、对象、方法、资料收集和分析方案、时间进度、经费预算等
    \item \imp{收集资料}——通过问卷、访谈、观察、文献等方式收集原始资料和二手资料
    \item \imp{整理和分析资料}——定性资料和定量资料的整理、编码、统计分析或内容分析
    \item \imp{撰写研究报告}——研究报告的规范性格式包括前言（问题背景）、研究设计和实施、结果、讨论与建议
\end{enumerate}
\end{tcolorbox}

\subsection{课题评价三原则}

\begin{keybox}
\textbf{课题评价的三个基本原则：}
\begin{enumerate}[leftmargin=*]
    \item \imp{需要性原则}——研究问题应满足社会卫生实践的需要或理论发展的需要，解决实际问题
    \item \imp{创新性原则}——研究问题应有一定的新颖性，在前人研究基础上有所突破或创新
    \item \imp{科学性原则}——研究问题应具备科学依据和理论支撑，方法科学、设计合理
\end{enumerate}

此外，还应考虑\textbf{可行性原则}——研究目标在现有条件下（人力、物力、财力、时间等）是否可以实现。
\end{keybox}

\subsection{抽样方法概述}

\begin{tcolorbox}[colback=light, colframe=main, boxrule=0.5pt, arc=4pt, title=\textbf{抽样方法的两种类型}, breakable]
\textbf{概率抽样（Probability Sampling）}——按概率均等原则抽取样本，代表性好，可用于统计推断：
\begin{itemize}[leftmargin=*]
    \item 单纯随机抽样（Simple Random Sampling）
    \item 系统抽样（Systematic Sampling）
    \item 分层抽样（Stratified Sampling）
    \item 整群抽样（Cluster Sampling）
    \item 多阶段抽样（Multi-stage Sampling）
\end{itemize}

\textbf{非概率抽样（Non-probability Sampling）}——不按概率均等原则抽取样本，在定性研究中广泛应用，详见下表。
\end{tcolorbox}

\subsection{非概率抽样}

\begin{defbox}
\textbf{非概率抽样（Non-probability Sampling）}：不按概率均等原则抽取样本，而是根据研究者的主观判断或便利条件选择研究对象。在社会医学的定性研究中广泛应用。

\textbf{四种主要类型：}
\end{defbox}

\begin{table}[htbp]
\centering
\caption{非概率抽样方法比较}
\begin{tabularx}{\textwidth}{lXX}
\hline
\textbf{方法} & \textbf{定义} & \textbf{特点与适用} \\
\hline
\imp{方便抽样（Convenience/Accidental Sampling）} & 选择最方便接触到的研究对象 & 简单易行但代表性差；适用于初步探索和预调查 \\
\imp{立意抽样（Purposive/Judgmental Sampling）} & 根据研究者判断和目的选择有代表性的对象 & 灵活但主观性强；适用于了解典型个案 \\
\imp{滚雪球抽样（Snowball Sampling）} & 从已知研究对象开始，由他们推荐符合条件的新研究对象 & 适用于难以找到的"隐藏人群"（如静脉吸毒者、性工作者等） \\
\imp{定额抽样（Quota Sampling）} & 按某些特征将总体分层后非随机抽取各层样本 & 在各层按比例抽取，但各层内非随机；可用于快速调查 \\
\hline
\end{tabularx}
\end{table}

% ----- 8.2 定性研究 -----
\section{定性研究}

\begin{defbox}
\textbf{定性研究（Qualitative Research）}：通过与研究对象的深入互动来理解人们的行为、观点、经验和感受，获取深层次信息的研究方法。研究结果以文字描述形式呈现。

\textbf{特点：}
\begin{enumerate}[leftmargin=*]
    \item 注重对事物的深层理解和"意义"的挖掘
    \item 研究设计灵活，可在研究过程中调整
    \item 样本量较小，采用非概率抽样
    \item 资料收集以开放式交流为主
    \item 结果难以推广到大范围人群，但具有深刻性
\end{enumerate}
\end{defbox}

\subsection{常用定性研究方法}

\begin{keybox}
\textbf{四种主要定性研究方法：}

\textbf{1. 深入访谈法（In-depth Interview）：}
\begin{itemize}[leftmargin=*]
    \item 访谈者与被访者一对一深入交谈
    \item 按事先准备的访谈提纲进行
    \item 开放式问题，鼓励被访者自由表达
    \item 时间：通常30分钟-2小时
    \item 适用于：敏感问题、复杂问题的深入探讨
\end{itemize}

\textbf{2. 焦点组讨论（Focus Group Discussion, FGD）：}
\begin{itemize}[leftmargin=*]
    \item 8-12人为一组，在主持人引导下围绕特定主题进行讨论
    \item 利用群体互动激发观点和见解
    \item 时间：通常1-2小时
    \item 优势：高效、经济，群体互动产生丰富信息
    \item 劣势：个别个体可能主导讨论，敏感问题不适合
    \item 案例：一项研究进行了18场共10小时的焦点组讨论
\end{itemize}

\textbf{3. 提名小组讨论（Nominal Group Discussion）：}
\begin{itemize}[leftmargin=*]
    \item 结构化较强的定性方法
    \item 成员先独立书面回答问题，再依次分享、讨论、排序
    \item 减少了群体压力和个别主导问题
\end{itemize}

\textbf{4. 观察法（Observation）：}
\begin{itemize}[leftmargin=*]
    \item 参与式观察——研究者融入被研究群体
    \item 非参与式观察——研究者保持客观旁观
    \item 适用于研究实际行为（而非口头报告的行为）
\end{itemize}
\end{keybox}

\subsection{定性研究资料的处理与分析}

\begin{enumerate}[leftmargin=*]
    \item 将录音/录像资料转录为文字
    \item 反复阅读资料，熟悉内容
    \item 编码——对资料进行主题分类编码
    \item 提取主题——归纳核心主题和模式
    \item 解释和分析——结合理论框架进行深度解读
    \item 撰写报告——用引语（受访者原话）支撑分析结果
\end{enumerate}

% ----- 8.3 定量研究 -----
\section{定量研究}

\begin{defbox}
\textbf{定量研究（Quantitative Research）}：通过收集和分析数量化资料来检验假设、描述事物特征和因果关系的研究方法。研究结果以数值形式呈现，可进行统计分析。

\textbf{特点：}
\begin{enumerate}[leftmargin=*]
    \item 研究设计结构化、标准化程度高
    \item 样本量大，采用概率抽样
    \item 资料以数字形式收集和分析
    \item 结果可推广到目标总体
    \item 强调客观性和可重复性
\end{enumerate}
\end{defbox}

% ----- 8.4 问卷设计 -----
\section{问卷设计}

\subsection{问卷的概念、类型和结构}

\begin{defbox}
\textbf{问卷（Questionnaire）}：由一系列与研究目标相关的问题组成的调查工具，用于系统化地收集研究对象的特征、行为、态度、知识等信息。

\textbf{问卷类型：}
\begin{itemize}[leftmargin=*]
    \item 按填写方式：自填式问卷、访谈式问卷
    \item 按发放方式：现场问卷、邮寄问卷、网络问卷、电话问卷
\end{itemize}

\textbf{问卷结构：}
\begin{enumerate}[leftmargin=*]
    \item \imp{封面信/卷首语}——说明调查目的、意义、保密承诺、填写方法、致谢
    \item \imp{指导语}——具体说明如何填写和回答问题
    \item \imp{问题与答案}——问卷的主体部分
    \item \imp{编码}——便于计算机录入和分析
\end{enumerate}
\end{defbox}

\subsection{问卷设计的原则和步骤}

\begin{keybox}
\textbf{问卷设计六原则：}
\begin{enumerate}[leftmargin=*]
    \item \imp{目的性原则}——所有问题应紧密围绕研究目的
    \item \imp{可接受性原则}——问题易于理解，应答者愿意回答
    \item \imp{逻辑性原则}——问题编排有逻辑顺序，由浅入深
    \item \imp{简明性原则}——问题简洁明了，避免冗长复杂
    \item \imp{可接受性原则}——考虑文化和道德接受度
    \item \imp{科学性原则}——问题设计科学，能有效测量研究变量
\end{enumerate}

\textbf{问卷设计七步骤：}
确定所需信息→选择问卷类型→设计问题→确定问题顺序→预调查→修改完善→定稿印刷
\end{keybox}

\subsection{封闭式问题与开放式问题}

\begin{table}[htbp]
\centering
\caption{封闭式与开放式问题的比较}
\begin{tabularx}{\textwidth}{lXX}
\hline
\textbf{比较} & \textbf{封闭式问题} & \textbf{开放式问题} \\
\hline
\imp{定义} & 提供固定选项，应答者从中选择 & 不提供选项，应答者自由回答 \\
\imp{优点} & 易于编码和统计分析；回答率高；节省时间 & 可获取丰富详细的信息；不受选项限制；可能发现新问题 \\
\imp{缺点} & 选项可能不全；不能深入了解；限制应答者的表达 & 编码困难；应答率低；分析工作量大；受表达能力影响 \\
\hline
\end{tabularx}
\end{table}

\subsection{问卷设计常见错误及注意事项}

\begin{tipbox}
\textbf{问卷设计常见的错误：}
\begin{enumerate}[leftmargin=*]
    \item 双重问题（Double-barreled Question）——一个问题包含两个以上询问内容
    \item 诱导性问题（Leading Question）——问题措辞暗示了期望的答案
    \item 专业术语过多——受访者难以理解
    \item 问题含糊不清——不同应答者理解不一致
    \item 选项不互斥或不穷尽——答案选项设计缺陷
    \item 回忆性偏倚——要求应答者回忆太久远的事情
    \item 敏感问题放置不当——引起应答者防御心理
\end{enumerate}

\textbf{问题排列顺序原则：}
\begin{itemize}[leftmargin=*]
    \item 从简单到复杂，从一般到具体
    \item 从非敏感到敏感问题（敏感问题放在后面）
    \item 同类问题集中排列
    \item 漏斗式编排——先宽后窄
\end{itemize}
\end{tipbox}

% ----- 8.5 效度与信度 -----
\section{效度与信度}

\subsection{效度的概念与类型（教学难点*）}

\begin{defbox}
\textbf{效度（Validity）}：\imp{（教学难点*）}指测量工具或手段能够准确测出所需测量的事物的程度，即\keyword{"测的是否正确"（准确性）}。

\textbf{效度的四种类型：}
\begin{enumerate}[leftmargin=*]
    \item \imp{表面效度（Face Validity）}：从表面上主观判断测量工具是否看起来测量了所要测的内容。是最基本层次的效度，非正式评估
    \item \imp{内容效度（Content Validity）}：测量工具的内容是否全面、均衡地涵盖了所要测量的概念领域。通过专家评定来判断
    \item \imp{效标效度（Criterion Validity）}：用测量工具的测量结果与某个效标（金标准）进行比较的一致性程度。分为：
    \begin{itemize}
        \item 同时效度（Concurrent Validity）——与同时测量的效标比较
        \item 预测效度（Predictive Validity）——与将来测量的效标比较
    \end{itemize}
    \item \imp{结构效度（Construct Validity）}：测量工具实际测量到的理论结构和特质与假设的理论结构相符合的程度。是最深层次的效度
\end{enumerate}
\end{defbox}

\subsection{信度的概念与类型（教学难点*）}

\begin{defbox}
\textbf{信度（Reliability）}：\imp{（教学难点*）}指采用同一方法对同一对象重复测量时所得结果的一致性程度，即\keyword{"测量结果的一致性或稳定性"（可靠性）}。

\textbf{信度的四种类型：}
\begin{enumerate}[leftmargin=*]
    \item \imp{重测信度（Test-Retest Reliability）}：同一量表对相同对象在不同时间重复测量，两次结果的相关性。间隔时间不宜过短（记忆效应）或过长（真实变化）
    \item \imp{复本信度（Parallel-Forms Reliability）}：两个等价的测量工具版本对同批对象测量的结果一致性
    \item \imp{折半信度（Split-Half Reliability）}：将量表问题分为两半（如奇数题和偶数题），计算两部分得分的一致性
    \item \imp{内部一致性信度（Internal Consistency Reliability）}：量表各条目测量同一概念的程度，常用\keyword{Cronbach's $\alpha$系数}衡量，$\alpha \geq 0.70$为可接受，$\alpha \geq 0.80$为良好
\end{enumerate}
\end{defbox}

\subsection{信度与效度的关系（教学难点*）}

\begin{keybox}
\imp{（教学难点*）}\textbf{信度与效度的关系：}

\textbf{核心关系：信度是效度的必要条件，但不是充分条件。}

\begin{enumerate}[leftmargin=*]
    \item \imp{信度高，效度不一定高}——测量结果可能稳定但测的不是想要测量的内容（如用体重秤来"测量"身高——每次都得到一致的结果，但数值是错的）
    \item \imp{效度高，信度一定高}——如果测量工具确实测量了想测的内容，那么结果必定是可重复和稳定的
    \item \imp{信度低，效度一定低}——测量结果本身不稳定、不可靠，就不可能有准确的测量
    \item \imp{效度低，信度不一定低}——可能稳定地测到了别的内容
\end{enumerate}

\textbf{形象的比喻：打靶}
\begin{itemize}[leftmargin=*]
    \item \imp{高效度高信度}：弹孔集中（信度高）在靶心（效度高）
    \item \imp{高信度低效度}：弹孔集中（信度高）但偏离靶心（效度低）
    \item \imp{无信度无效度}：弹孔分散且偏离靶心——测量无价值
\end{itemize}
\end{keybox}

\begin{tipbox}
\textbf{如何在研究中保证信度和效度：}
\begin{itemize}[leftmargin=*]
    \item 严格的研究设计和标准化操作流程
    \item 预调查和预测试，检验测量工具的信度和效度
    \item 访谈员培训和质量控制
    \item 使用广泛验证过的标准化量表
    \item 三角验证（Triangulation）——多种方法和来源的数据相互验证
\end{itemize}
\end{tipbox}

\subsection*{二、名词解释}
\begin{enumerate}[leftmargin=*, itemsep=3pt]

\subsubsection*{二、名词解释}
\begin{enumerate}[leftmargin=*]
    \item \textbf{定性研究}：通过与研究对象深入互动了解其行为、观点和体验的研究方法，结果以文字形式呈现，注重深层理解。
    \item \textbf{效度}：测量工具准确测出所需测量事物的程度，即"测的是否正确"。
    \item \textbf{信度}：同一方法对同一对象重复测量结果的一致性程度，即"测量结果的稳定性"。
    \item \textbf{非概率抽样}：不按概率均等原则抽取样本，根据主观判断或便利条件选择对象，包括方便抽样、立意抽样、滚雪球抽样和定额抽样。
\end{enumerate}

\subsubsection*{三、简答题}
\begin{enumerate}[leftmargin=*]
    \item 定性：深入理解意义、设计灵活、样本小、非概率抽样、开放式交流、文字描述。定量：检验假设、设计结构化、样本大、概率抽样、标准量表、数值分析、可推广。
    \item 四种类型：①表面效度（主观判断"看起来是否测对了"）；②内容效度（是否全面涵盖测量领域）；③效标效度（与金标准比较，分同时效度和预测效度）；④结构效度（与理论结构的吻合程度，最深层次）。
    \item 四种类型：①重测信度（不同时间重复测量的稳定性）；②复本信度（两个等价版本的一致性）；③折半信度（量表分半的一致性）；④内部一致性信度（Cronbach's $\alpha$系数，各条目内在一致性）。
    \item 核心：信度是效度的必要条件而非充分条件。信度高，效度不一定高；效度高，信度一定高；信度低，效度一定低；效度低，信度不一定低。
    \item 六原则：目的性、可接受性、逻辑性、简明性、可接受性、科学性。常见错误：双重问题、诱导性问题、专业术语过多、问题含糊、选项不互斥/穷尽、回忆偏倚、敏感问题放置不当。
\end{enumerate}
