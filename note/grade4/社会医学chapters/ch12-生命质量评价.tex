% ----- 12.1 生命质量概述 -----
\section{生命质量的概念与构成}

\subsection{生命质量的概念}

\begin{defbox}
\textbf{生命质量（Quality of Life, QOL）}：由美国经济学家Galbraith于20世纪50年代提出。在社会医学领域，生命质量指不同文化和价值体系中的个体对与其目标、期望、标准以及所关心事情有关的生活状态的体验。

\textbf{健康相关生命质量（Health-related Quality of Life, HRQOL）}：在疾病、意外损伤及医疗干预的影响下，测定与个人生活事件相联系的主观健康状态和个体满意度。
\end{defbox}

\begin{keybox}
\textbf{WHO对生命质量的定义：}\\
不同文化和价值体系中的个体对与他们的目标、期望、标准以及所关心事情有关的生活状态的体验。包含以下核心要素：
\begin{itemize}[leftmargin=*]
    \item 主观性——以个体自身体验为核心
    \item 多维度性——包含身体、心理、社会等多维度
    \item 文化依赖性——受文化背景和价值观念的影响
\end{itemize}
\end{keybox}

\subsection{生命质量的构成}

\begin{tcolorbox}[colback=light, colframe=main, boxrule=0.5pt, arc=4pt, title=\textbf{生命质量的构成维度}, breakable]
\begin{enumerate}[leftmargin=*]
    \item \imp{生理状态}——身体功能（活动能力、自理能力）、感官功能（视力、听力）、疾病症状和体征、体力
    \item \imp{心理状态}——情绪状态（焦虑、抑郁等）、认知功能（记忆、注意力、思维）、自我评价（自尊、自我效能感、生活满意度）
    \item \imp{社会功能状态}——社会角色功能（工作、学习能力）、社会交往与人际关系、社会参与和社会支持
    \item \imp{角色功能（Role Function）}——能否完成日常社会角色要求。角色功能障碍表现为工作能力下降、家务能力下降等
    \item \imp{总体健康感受和满意度}——对自身健康和生活状况的总体主观评价
\end{enumerate}
\end{tcolorbox}

% ----- 12.2 生命质量评价内容与方法 -----
\section{生命质量的评价内容与方法}

\subsection{评价内容}

\begin{enumerate}[leftmargin=*]
    \item \imp{生理功能评价}：活动受限（自理能力、躯体活动、体力活动等）、角色受限（工作、学习、家务等受限天数）
    \item \imp{心理功能评价}：情绪反应（焦虑、抑郁等心理困扰）、认知功能（注意力、记忆力、思维清晰度）
    \item \imp{社会功能评价}：社会交往的频率和质量、社会资源的利用
    \item \imp{总体健康评价}：健康自评（Self-rated Health）——个体对自己健康状况的总体主观评价；生活满意度
\end{enumerate}

\subsection{评价方法（教学难点*）}

\begin{keybox}
\imp{（教学难点*）}\textbf{生命质量评价的主要方法：}

\textbf{1. 量表评定法（最常用）：}
\begin{itemize}[leftmargin=*]
    \item 通过标准化问卷获取个体对各个维度主观感受的评分
    \item 分为\imp{普适性量表}（通用）和\imp{特异性量表}（针对特定疾病或人群）
    \item 普适性量表如SF-36、WHOQOL、EQ-5D，适用于各类人群比较
    \item 特异性量表如癌症患者QOL量表（FACT-G）、关节炎影响测量量表（AIMS）等
\end{itemize}

\textbf{2. 访谈法：}
\begin{itemize}[leftmargin=*]
    \item 通过深入访谈了解受访者对生活质量的真实感受和体验
    \item 可获取量化量表无法捕捉的个性化信息
\end{itemize}

\textbf{3. 健康效用值法：}
\begin{itemize}[leftmargin=*]
    \item 将健康状态量化为一个0（死亡）到1（完全健康）之间的效用值
    \item 用于质量调整寿命年（QALY）的计算
    \item 方法：标准博弈法（SG）、时间权衡法（TTO）、等级评定法（RS）
\end{itemize}
\end{keybox}

% ----- 12.4 生命质量评价的应用 -----
\section{生命质量评价的应用}

\begin{enumerate}[leftmargin=*]
    \item \imp{临床疗效评价}——不仅评价是否"治好病"，还要评价"活得怎么样"。肿瘤治疗中从"生存率"到"生存质量"的评价转变
    \item \imp{卫生服务效果评价}——比较不同干预方案和治疗措施对患者生命质量的综合影响
    \item \imp{药物经济学评价}——结合QOL效用值计算QALY（质量调整寿命年），为卫生经济决策提供依据：QALY = 生命年 $\times$ 健康效用值
    \item \imp{人群健康监测}——定期评估人群生命质量，发现高危人群和弱势群体
    \item \imp{卫生政策制定}——为资源配置和政策决策提供人群主观健康状态的数据支持
    \item \imp{老龄化研究}——评估老年人群的生命质量和功能状态，指导长期照护政策
\end{enumerate}

\subsection*{参考答案要点}

\subsubsection*{一、选择题}
1. (B)；2. (B)；3. (B)；4. (C)

\subsubsection*{二、名词解释}
\begin{enumerate}[leftmargin=*]
    \item \textbf{HRQOL}：在疾病、损伤及医疗干预影响下，与个人生活事件相联系的主观健康状态和个体满意度。
    \item \textbf{QALY}：将生存年数乘以该年数中的健康效用值（0-1），综合反映生命的"数量"和"质量"，用于卫生经济学评价。
    \item \textbf{健康自评}：个体对自己健康状况的总体主观评价，是生命质量评价中最简单有效的方法之一，也是死亡和患病的重要预测因子。
\end{enumerate}

\subsubsection*{三、简答题}
\begin{enumerate}[leftmargin=*]
    \item 五个维度：生理状态（身体功能、疾病症状）、心理状态（情绪、认知、自我评价）、社会功能状态（社会角色、人际交往、社会参与）、角色功能（工作学习能力）、总体健康感受和满意度。
    \item 普适性量表（如SF-36、WHOQOL）：适用于各类人群间的比较，范围广但敏感度稍低。特异性量表（如FACT、AIMS）：针对特定疾病或人群，敏感度高、变化反映更精确，但不能跨疾病比较。
    \item 临床：评价疗效不仅看"是否治好"还看"活得怎样"；比较不同治疗方案对QOL的影响。卫生决策：结合效用值计算QALY进行药物经济学评价；监测人群QOL发现弱势群体；辅助资源配置和政策决策。
\end{enumerate}
