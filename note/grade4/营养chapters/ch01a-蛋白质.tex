\chapter{营养学基础（一）：蛋白质}
\setcounter{chapter}{1}
\chapterintro{
掌握蛋白质的生理功能、必需氨基酸概念及中英文名称、氨基酸模式与限制氨基酸、氮平衡*的分类及意义、食物蛋白质营养学评价（AAS、PDCAAS*、BV、NPU、PER）、蛋白质-能量营养不良（PEM）。熟悉必要氮损失（ONL）、蛋白质消化率及影响因素。了解蛋白质分类和食物来源。

% ----- 第一节 氨基酸与蛋白质概述 -----
\section{氨基酸与蛋白质概述}

\begin{defbox}
\textbf{蛋白质（Protein）}是机体细胞、组织和器官的重要组成结构，是功能因子和调控因子的重要组成成分，是一切生命的物质基础。基本构成单位为氨基酸，以肽键连接。组成元素：C、H、O、N，部分含S。
\end{defbox}

\begin{keybox}
\textbf{必需氨基酸（Essential Amino Acid, EAA）：}人体内不能合成或合成速度不能满足机体需要，必须从食物中直接获得的氨基酸。

人体必需氨基酸（8种+组氨酸）：
\begin{enumerate}[leftmargin=*]
    \item 异亮氨酸（Isoleucine, Ile）
    \item 亮氨酸（Leucine, Leu）
    \item 赖氨酸（Lysine, Lys）
    \item 蛋氨酸（Methionine, Met）
    \item 苯丙氨酸（Phenylalanine, Phe）
    \item 苏氨酸（Threonine, Thr）
    \item 色氨酸（Tryptophan, Trp）
    \item 缬氨酸（Valine, Val）
    \item 组氨酸（Histidine, His）——婴儿必需氨基酸
\end{enumerate}
\end{keybox}

\begin{tipbox}
\textbf{条件必需氨基酸（半必需氨基酸）：}半胱氨酸（Cys）可替代约30\%的蛋氨酸需要量；酪氨酸（Tyr）可替代约50\%的苯丙氨酸需要量。
\end{tipbox}

\begin{keybox}
\textbf{氨基酸模式（Amino Acid Pattern）与限制氨基酸（教学难点*）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{氨基酸模式：}蛋白质中各种必需氨基酸的构成比例。
    \item \textbf{参考蛋白质：}全鸡蛋蛋白的氨基酸模式与人体最接近，常被作为参考蛋白质。
    \item \textbf{限制氨基酸（Limiting Amino Acid）：}蛋白质中含量相对较低的必需氨基酸。第一限制氨基酸为含量最低者。
    \item 赖氨酸是谷类食物的第一限制氨基酸；蛋氨酸是豆类食物的第一限制氨基酸。
    \item \textbf{蛋白质互补作用：}不同食物蛋白质混合食用，必需氨基酸相互补充，提高蛋白质利用率。如谷类（缺赖氨酸）+ 豆类（缺蛋氨酸）。
\end{itemize}

\textbf{蛋白质互补三原则：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{食物生物学种属愈远愈好：}动植物搭配优于同类搭配
    \item \textbf{搭配种类愈多愈好：}多种食物混合食用，氨基酸互补更全面
    \item \textbf{食用时间愈近愈好，最好同时食用：}氨基酸只有在同时到达靶器官时才能充分发挥互补效应
\end{enumerate}
\end{keybox}

\begin{keybox}
\textbf{蛋白质分类（按营养价值）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{完全蛋白质（优质蛋白）：}含必需氨基酸种类齐全、数量充足、比例适宜。包括奶类、蛋类、肉类、鱼类和大豆蛋白。植物中仅大豆蛋白属完全蛋白质。
    \item \textbf{半完全蛋白质：}必需氨基酸种类齐全但比例不适宜，只能维持生命但不能促进生长发育。如谷类蛋白（赖氨酸不足）。
    \item \textbf{不完全蛋白质：}必需氨基酸种类不全，不能维持生命。如胶原蛋白（缺乏色氨酸）。
\end{enumerate}
\end{keybox}

% ----- 第二节 氮平衡 -----
\section{氮平衡}

\begin{keybox}
\textbf{氮平衡（Nitrogen Balance，教学难点*）：}摄入蛋白质的量和排出蛋白质的量之间的关系。

公式：$B = I - (U + F + S)$\\
其中B:氮平衡，I:食物摄入氮，U:尿氮，F:粪氮，S:皮肤等途径的氮损失

三种状态：
\begin{enumerate}[leftmargin=*]
    \item \textbf{零氮平衡（B=0）：}摄入氮=排出氮，健康成人正常状态
    \item \textbf{正氮平衡（B>0）：}摄入氮>排出氮，见于婴幼儿、青少年、孕妇、疾病恢复期
    \item \textbf{负氮平衡（B<0）：}摄入氮<排出氮，见于饥饿、疾病、创伤、衰老
\end{enumerate}
\end{keybox}

\begin{defbox}
\textbf{必要氮损失（Obligatory Nitrogen Loss, ONL）：}机体在完全不摄入蛋白质时仍不可避免的氮损失，约54mg/kg体重/天。是确定蛋白质需要量的重要依据。
\end{defbox}

% ----- 第三节 蛋白质的生理功能 -----
\section{蛋白质的生理功能}

\begin{enumerate}[leftmargin=*]
    \item \textbf{人体组织的构成成分：}占人体重量的16\%～19\%
    \item \textbf{构成体内各种重要的生理活性物质：}酶、激素（胰岛素、生长激素）、抗体、转运蛋白（血红蛋白、转铁蛋白）、收缩蛋白（肌动蛋白和肌球蛋白）、凝血因子
    \item \textbf{供给能量：}每克蛋白质提供能量16.7kJ（4kcal）
    \item \textbf{肽类的特殊生理功能：}某些活性肽具有调节免疫、抗氧化、降血压等特殊功能
\end{enumerate}

% ----- 第四节 蛋白质的消化、吸收和代谢 -----
\section{蛋白质的消化、吸收和代谢}

\begin{itemize}[leftmargin=*]
    \item 胃：胃蛋白酶初步消化
    \item 小肠：胰蛋白酶、糜蛋白酶等酶解为氨基酸和小肽（主要消化场所）
    \item 吸收：主要以氨基酸形式通过小肠黏膜吸收，少数以二肽、三肽形式吸收
    \item 氨基酸代谢池：来自食物消化吸收的氨基酸+体内蛋白质分解产生的氨基酸
\end{itemize}

% ----- 第五节 食物蛋白质营养学评价 -----
\section{食物蛋白质营养学评价}

\begin{keybox}
\textbf{评价意义（教学难点*）：}综合评价食物蛋白质的营养价值，包括含量、消化率和利用率三个方面。
\end{keybox}

\subsection{蛋白质含量}

\begin{keybox}
\textbf{蛋白质含量测定方法：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{凯氏定氮法（Kjeldahl法）：}测定食物中的氮含量，再乘以换算系数6.25（蛋白质平均含氮量约16\%，即每克氮相当于6.25g蛋白质）。蛋白质含量 = 氮含量 $\times$ 6.25。\keyword{需注意：不能区分蛋白氮和非蛋白氮，三聚氰胺事件即利用了此漏洞。}
    \item \textbf{双缩脲法（Biuret法）：}基于肽键与铜离子的呈色反应。
    \item \textbf{Folin-酚试剂法（Lowry法）：}灵敏度高于双缩脲法。
    \item \textbf{紫外吸收法：}利用蛋白质在280nm处的特征吸收。
    \item \textbf{考马斯亮蓝法（Bradford法）：}灵敏度高、操作简便。
\end{enumerate}
\end{keybox}

各类食物蛋白质含量：蛋类11\%～14\%，奶类3.0\%～3.5\%，新鲜肌肉15\%～22\%，大豆35\%～40\%。

\subsection{蛋白质消化率}
\begin{itemize}[leftmargin=*]
    \item 真消化率（TD）= [食物氮 - (粪氮 - 肠道代谢氮)] / 食物氮 $\times$ 100\%
    \item 表观消化率（AD）= (食物氮 - 粪氮) / 食物氮 $\times$ 100\%
    \item 常见食物真消化率：鸡蛋97\%，牛奶95\%，肉、鱼94\%，大米87\%，大豆粉86\%，中国混合膳食96\%
\end{itemize}

\begin{keybox}
\textbf{蛋白质利用率评价指标：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{生物价（BV）= 氮潴留量 / 氮吸收量 $\times$ 100\%}：反映蛋白质被吸收后在体内的利用程度。常见：全鸡蛋94、全牛奶87、鱼83、牛肉74、大豆73
    \item \textbf{蛋白质净利用率（NPU）= BV $\times$ TD = 氮潴留量 / 食物氮 $\times$ 100\%}：综合评价消化率和利用率
    \item \textbf{蛋白质功效比值（PER）= 动物体重增加(g) / 摄入蛋白质(g)}：一般选用刚断乳的雄性大鼠，被测蛋白质作为唯一蛋白质来源，占饲料的10\%，喂饲28天。全鸡蛋PER为3.92，全牛奶为3.09。\textbf{校正PER} = 待测蛋白质PER / 酪蛋白PER $\times$ 2.5。
    \item \textbf{氨基酸评分（AAS，教学难点*）：}将待评食物蛋白质的必需氨基酸与人体必需氨基酸模式加以比较，反映蛋白质构成和利用的关系。AAS = [待测蛋白质每克蛋白质（或氮）中某种氨基酸含量(mg) / 理想模式或参考蛋白质每克蛋白质（或氮）中该氨基酸含量(mg)] $\times$ 100。得分最低的氨基酸为限制氨基酸，该分值即为该蛋白质的AAS
    \item \textbf{经消化率修正的氨基酸评分（PDCAAS，教学难点*）：}PDCAAS = AAS $\times$ 真消化率。是FAO/WHO推荐的蛋白质质量评价标准。酪蛋白、鸡蛋蛋白PDCAAS均为1.00
\end{enumerate}
\end{keybox}

% ----- 第六节 蛋白质营养不良 -----
\section{蛋白质营养不良}

\begin{keybox}
\textbf{蛋白质-能量营养不良（Protein-Energy Malnutrition, PEM）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Kwashiorkor（水肿型营养不良）：}能量摄入基本满足但蛋白质严重不足。表现：腹腿部水肿、虚弱、表情淡漠、生长滞缓、头发变色变脆易脱落。多见于以淀粉类为主食的儿童。
    \item \textbf{Marasmus（消瘦型营养不良）：}蛋白质和能量摄入均严重不足。表现：明显消瘦、肌肉萎缩、皮下脂肪消失、无力。
\end{enumerate}
\end{keybox}

% ----- 第七节 蛋白质供给量和食物来源 -----
\section{蛋白质供给量和食物来源}

\begin{keybox}
\textbf{推荐摄入量（RNI）：}
\begin{itemize}[leftmargin=*]
    \item 成人（轻体力）：男75g/d，女65g/d
    \item 中体力：男80g/d，女70g/d；重体力：男90g/d，女80g/d
    \item 孕期中期、晚期：+15g/d；乳母：+20g/d
    \item 60岁以上：男75g/d，女65g/d
    \item 供能比：占总能量10\%～12\%
\end{itemize}
\end{keybox}

\textbf{人体蛋白质营养状况评价：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{临床检查：}身高、体重、发育；上臂围；上臂肌围、上臂肌区。
    \item \textbf{实验室检查：}
    \begin{itemize}[leftmargin=*]
        \item 血清白蛋白（正常值 35～50g/L）——反映较长期的蛋白质营养状况
        \item 血清运铁蛋白（正常值 2.2～4.0g/L）——较白蛋白更敏感
        \item 血清甲状腺结合前白蛋白（正常值 280～350mg/L）——短半衰期，反映近期变化
        \item 视黄醇结合蛋白（正常值 26～76mg/L）——最敏感，反映极近期蛋白质摄入
        \item \textbf{尿3-甲基组氨酸：}营养不良时排出量减少，反映肌肉蛋白分解程度
        \item \textbf{头发的毛干与毛根的形态改变：}慢性蛋白质营养不良时可出现头发变细、变脆、颜色变浅。
    \end{itemize}
\end{enumerate}

\textbf{食物来源：}动物性食物（优质蛋白）：蛋类11\%～14\%，奶类3.0\%～3.5\%，肉类15\%～22\%；植物性食物：大豆35\%～40\%，谷类约8\%～12\%。膳食中优质蛋白质占比建议30\%～50\%。
