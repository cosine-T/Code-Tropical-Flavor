\chapterintro{\keyword{掌握}氮平衡（零/负/正）与必要氮损失（ONL）；掌握必需氨基酸（9种）、氨基酸模式、限制氨基酸、蛋白质互补作用；掌握BV、NPU、PER、AAS、PDCAAS；掌握蛋白质-能量营养不良（PEM）；掌握蛋白质营养状况评价。\keyword{熟悉}蛋白质生理功能、蛋白质分类和食物来源、影响氮平衡的因素；熟悉NPU。\keyword{了解}蛋白质代谢与氨基酸代谢池、蛋白质组成元素与特点、必需氨基酸中英文名称及需要量、食物蛋白质消化率影响因素、实验室检查指标。}

\section{蛋白质代谢与氨基酸代谢池（了解）}

\begin{tipbox}
\textbf{蛋白质代谢：}食物蛋白质经消化吸收后，氨基酸进入血液循环，参与体内蛋白质的合成与分解。蛋白质在体内处于动态平衡状态，不断进行合成与分解代谢。机体蛋白质的合成与分解同时进行，构成蛋白质转换（protein turnover）。

\textbf{氨基酸代谢池（amino acid metabolic pool）：}体内游离氨基酸分布在细胞内外液中，构成氨基酸代谢池。代谢池中的氨基酸来源包括：
\begin{itemize}[leftmargin=*]
    \item 食物蛋白质消化吸收的氨基酸（外源性）
    \item 组织蛋白质分解产生的氨基酸（内源性）
    \item 体内合成的非必需氨基酸
\end{itemize}
代谢池中的氨基酸去路：合成组织蛋白质、合成含氮活性物质（激素、酶、抗体等）、脱氨基后氧化供能或转变为糖和脂肪。

\textbf{食物蛋白质代谢路径：}食物蛋白质 $\rightarrow$ 消化吸收 $\rightarrow$ 氨基酸代谢池 $\rightarrow$ 储留利用（合成机体蛋白质）或未储留（随尿排出）；未吸收部分随粪便排出。（来源：0613营养笔记）
\end{tipbox}

\section{蛋白质的组成元素与特点（了解）}

\begin{tipbox}
\textbf{蛋白质组成元素：}蛋白质主要由碳（C）、氢（H）、氧（O）、氮（N）四种元素组成，多数蛋白质还含有硫（S），部分含有磷（P）、铁（Fe）、锌（Zn）等。其中\imp{氮元素的含量比较恒定，平均约为16\%}，这是凯氏定氮法测定蛋白质含量的基础——蛋白质含量 = 氮含量 $\times$ 6.25（因100/16 = 6.25）。

\textbf{蛋白质的特点：}
\begin{itemize}[leftmargin=*]
    \item 蛋白质是生命的物质基础，是人体氮的唯一来源
    \item 蛋白质的基本构成单位是氨基酸，由肽键连接
    \item 构成人体的氨基酸共20种
    \item 蛋白质是大分子有机化合物，分子量大，结构复杂
    \item 每克蛋白质在体内可产生16.72千焦耳（4kcal）的能量
\end{itemize}
（来源：0613营养笔记）
\end{tipbox}

\section{蛋白质的生理作用与氮平衡}

\begin{defbox}
\textbf{蛋白质的生理作用：}
\begin{enumerate}[leftmargin=*]
    \item 人体内细胞的主要成分
    \item 人体内重要物质的主要原料（激素、酶、抗体、血红蛋白）
    \item 提供热量：每克蛋白质在体内可产生16.72千焦耳（4kcal）的能量，由蛋白质提供的能量占每日人需总能量的\imp{10\%$\sim$20\%}，成年人蛋白质RNI：\imp{男性65g/d，女性55g/d}
\end{enumerate}
\end{defbox}

\begin{keybox}
\textbf{氮平衡（Nitrogen Balance）\difficulty：}反映机体摄入氮的量和排出氮的量的关系。
\textbf{关系式：}
$$B = I - (U + F + S)$$
$B$：氮平衡；$I$：摄入氮；$U$：尿氮；$F$：粪氮；$S$：皮肤等氮损失

\textbf{分类及意义：}
\begin{itemize}[leftmargin=*]
    \item $B=0$ \imp{零氮平衡}：摄入氮和排出氮相等。健康成人应维持零氮平衡并富裕5\%
    \item $B>0$ \imp{正氮平衡}：儿童处于生长发育阶段、妇女怀孕、疾病恢复和运动及劳动需要增加肌肉时应保持适当正氮平衡
    \item $B<0$ \imp{负氮平衡}：老年人，饥饿过度节食，消耗性疾病
\end{itemize}

\textbf{必要氮损失（Obligatory Nitrogen Losses, ONL）：}成人进食无蛋白膳食时，经尿、粪、皮肤等途径所丢失的氮量，约\imp{54mg/kg体重}（成人每日约\imp{20g}蛋白质）。（来源：0613营养笔记）
\end{keybox}

\begin{defbox}
\textbf{影响氮平衡的因素（熟悉）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{膳食蛋白质摄入量：}摄入蛋白质不足 $\rightarrow$ 负氮平衡；摄入充足且优质 $\rightarrow$ 利于维持正氮或零氮平衡
    \item \textbf{能量摄入状况：}能量摄入不足时，蛋白质被氧化供能，导致负氮平衡；能量充足时，蛋白质可节约用于合成
    \item \textbf{生理状态：}生长（正氮平衡）、妊娠（正氮平衡）、衰老（易负氮平衡）、疾病恢复期（需正氮平衡）
    \item \textbf{蛋白质质量：}优质蛋白质（氨基酸模式与人体接近） $\rightarrow$ 利于维持氮平衡；劣质蛋白质 $\rightarrow$ 利用率低，需更多摄入量
    \item \textbf{其他因素：}运动（肌肉合成增加需要正氮平衡）、应激状态（如创伤、感染时分解代谢增强 $\rightarrow$ 负氮平衡）、激素水平等
\end{enumerate}
（来源：0613营养笔记）
\end{defbox}

\section{氨基酸的分类与蛋白质营养价值}

\begin{keybox}
\textbf{氨基酸的分类：}

\textbf{1. 必需氨基酸（Essential Amino Acid, EAA）：}人体内不能合成或合成速度不能满足机体需要，必须从食物中直接获得的氨基酸。构成人体的氨基酸：20种（\imp{9种必需氨基酸}）。

\textbf{9种必需氨基酸（中英文名称）：}
\begin{itemize}[leftmargin=*]
    \item 缬氨酸（Valine, Val）
    \item 异亮氨酸（Isoleucine, Ile）
    \item 亮氨酸（Leucine, Leu）
    \item 苯丙氨酸（Phenylalanine, Phe）
    \item 蛋氨酸/甲硫氨酸（Methionine, Met）
    \item 色氨酸（Tryptophan, Trp）
    \item 组氨酸（Histidine, His）——\imp{婴儿必需氨基酸}
    \item 苏氨酸（Threonine, Thr）
    \item 赖氨酸（Lysine, Lys）
\end{itemize}
记忆口诀：\textbf{``携一两本单色组书来''}（缬/异亮/亮/苯/蛋/色/组/苏/赖）。
\end{keybox}

\begin{tipbox}
\textbf{必需氨基酸需要量（了解）：}人体对必需氨基酸的需要量随年龄、生理状况而异。单位体重EAA需要量随年龄增长而下降。成人必需氨基酸需要量约占蛋白质需要量的20\%（儿童约占30\%$\sim$40\%）。计算食物必需氨基酸组成时，一般将半胱氨酸和蛋氨酸、酪氨酸和苯丙氨酸合并计算。（来源：0613营养笔记）
\end{tipbox}

\begin{keybox}
\textbf{2. 非必需氨基酸（Nonessential Amino Acid）：}人体自身可以合成，不一定需要从食物中直接供给的氨基酸。

\textbf{3. 条件必需氨基酸（Conditionally Essential Amino Acid）：}某些特定条件下，由于合成能力有限或需要量增加，不能满足机体需要，必须从食物中获取。半胱氨酸和酪氨酸在体内分别由蛋氨酸和苯丙氨酸转变而成，如果膳食中能直接提供这两种氨基酸，则人体对蛋氨酸和苯丙氨酸的需要可\imp{分别减少30\%和50\%}。

\textbf{氨基酸模式（Pattern）：}蛋白质中各种必需氨基酸的构成比例。其计算方法是将该种蛋白质中的色氨酸含量定为1，分别计算出其他必需氨基酸的相应比值。

\textbf{参考蛋白质（Reference Protein）：}食物蛋白质氨基酸模式与人体蛋白质氨基酸模式越接近，必需氨基酸被机体利用的程度越高，食物蛋白质的营养价值就越高。

\textbf{限制氨基酸（Limiting Amino Acid）\difficulty：}当食物蛋白质中一种或几种必需氨基酸相对含量较低或缺乏，限制了食物蛋白质中的其它必需氨基酸被机体利用的程度，使其营养价值降低，这些含量相对较低的必需氨基酸称为限制氨基酸，含量最低的称为\imp{第一限制氨基酸}。

\textbf{蛋白质互补作用（Protein Complementary Action）：}为了提高植物性蛋白质的营养价值，往往将两种或两种以上的食物混合食用，从而达到以多补少、提高膳食蛋白质营养价值的目的。\imp{三原则：}食物生物学种属愈远愈好；搭配种类愈多愈好；食用时间愈近愈好，最好同时食用。
\end{keybox}

\section{蛋白质分类}

\begin{defbox}
\textbf{蛋白质分类：}
\begin{itemize}[leftmargin=*]
    \item \imp{完全蛋白质（优质蛋白）}：植物仅大豆；动物性蛋白质——蛋、奶、鱼、肉
    \item \imp{半完全蛋白质}：大多数植物蛋白，虽可维持生命，但不能促进生长发育
    \item \imp{不完全蛋白质}：胶原蛋白、玉米胶蛋白
\end{itemize}
\end{defbox}

\section{食物蛋白质营养学评价（难点*）}

\begin{keybox}
\textbf{1. 蛋白质含量：}凯氏定氮法，\imp{蛋白质含量 = 氮含量 × 6.25}。

\textbf{2. 蛋白质消化率（Digestibility）：}反映蛋白质在消化道内被分解的程度，还反映消化后氨基酸和肽被吸收的程度。
\begin{itemize}[leftmargin=*]
    \item \imp{真消化率（TD）(\%) = [食物氮 $-$ (粪氮 $-$ 粪代谢氮)] / 食物氮 × 100}
    \item \imp{表观消化率（AD）(\%) = (食物氮 $-$ 粪氮) / 食物氮 × 100}
\end{itemize}
\end{keybox}

\begin{tipbox}
\textbf{消化率的影响因素（了解）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{食物来源：}一般动物性食物中蛋白质消化率 $>$ 植物性食物
    \item \textbf{加工方式：}大豆加工成豆腐后，消化率提高（加工后去除了纤维素和其他不利于蛋白质吸收的因素）
    \item \textbf{食物成分：}膳食纤维、蛋白酶抑制剂等抗营养因子降低消化率
    \item \textbf{个体因素：}消化功能（如胃酸分泌、消化酶活性）影响消化率
\end{itemize}
（来源：0613营养笔记）
\end{tipbox}

\begin{keybox}
\textbf{3. 蛋白质利用率：}

\textbf{（1）生物价（BV）} = 储留氮 $/$ 吸收氮 $\times$ 100

吸收氮 = 食物氮 $-$ (粪氮 $-$ 粪代谢氮)

储留氮 = 吸收氮 $-$ (尿氮 $-$ 尿内源性氮)

BV高$\rightarrow$氨基酸主要用于合成人体蛋白，减少肝肾负担。

\textbf{（2）蛋白质净利用率（NPU）(\%)} = 消化率 $\times$ 生物价

NPU(\%) = (储留氮量 $\div$ 食物氮) $\times$ 100

\textbf{（3）蛋白质功效比值（PER）：}测定生长发育中的幼小动物摄入1g蛋白质所增加的体重克数。

PER = 动物体重增加(g) $\div$ 摄入蛋白质(g)

校正PER = (待测蛋白质PER $\div$ 酪蛋白PER) $\times$ 2.5

\textbf{（4）氨基酸评分（AAS）/化学评分\difficulty：}

AAS = (待测蛋白质每g氮中氨基酸mg $\div$ 理想模式每g氮中该氨基酸mg) $\times$ 100

\textbf{经消化率修正的AAS（PDCAAS）：}

PDCAAS = AAS $\times$ 真消化率

PDCAAS $\times$ 2.5 = PER
\end{keybox}

\section{蛋白质-能量营养不良与营养状况评价}

\begin{keybox}
\textbf{蛋白质-能量营养不良（PEM）：}
\begin{itemize}[leftmargin=*]
    \item \imp{水肿型（Kwashiorkor）}：能量摄入基本满足，蛋白质严重不足。腹腿部水肿、虚弱、表情淡漠、生长迟缓、头发变色变脆易落
    \item \imp{干瘦型（Marasmus）}：蛋白质和能量摄入均严重不足。小手无力，易感染其他疾病而死亡
\end{itemize}
\end{keybox}

\begin{keybox}
\textbf{蛋白质营养状况评价：}
\begin{itemize}[leftmargin=*]
    \item 血清蛋白质：\imp{血清白蛋白正常值35$\sim$50g/L}；\imp{血清运铁蛋白正常值2.2$\sim$4.0g/L}
    \item 上臂肌围（AMC）和上臂肌区（AMA）：评价总体蛋白质储存较可靠
    \item 血清氨基酸比值（SAAR）：SAAR $<$ 2 → 正常；$>$ 3 → 蛋白质营养不良
    \item \textbf{尿3-甲基组氨酸：}营养不良时排出量减少
    \item \textbf{头发的毛干与毛根形态改变：}蛋白质营养不良时可出现特征性变化
\end{itemize}
\end{keybox}

\begin{tipbox}
\textbf{实验室检查指标（了解）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{血清白蛋白：}正常值35$\sim$50g/L
    \item \textbf{血清运铁蛋白：}正常值2.2$\sim$4.0g/L
    \item \textbf{血清甲状腺结合前蛋白：}正常值280$\sim$350mg/L
    \item \textbf{视黄醇结合蛋白：}正常值26$\sim$76mg/L
    \item \textbf{尿3-甲基组氨酸：}反映肌肉蛋白质分解代谢
    \item \textbf{上臂肌围（AMC）和上臂肌区（AMA）：}评价总体蛋白质储存较可靠。一般测量上臂中点处的围长（AC）和三头肌部皮褶厚（TSF）。男25.3cm，女23.2cm，测定值$>$90\%标准 → 正常
\end{itemize}
（来源：0613营养笔记）
\end{tipbox}

\begin{defbox}
\textbf{蛋白质参考摄入量：}成年人每天摄入约30g蛋白质即可满足零氮平衡。我国成人蛋白质RNI：\imp{男65g/d，女55g/d}。蛋白质供能比：\imp{10\%$\sim$15\%}（AMDR）。
\end{defbox}

\section{蛋白质食物来源（熟悉）}

\begin{defbox}
\textbf{蛋白质食物来源：}
\begin{itemize}[leftmargin=*]
    \item \textbf{动物性蛋白质：}蛋、奶、鱼、肉等，质量好、利用率高，但同时富含饱和脂肪酸和胆固醇
    \item \textbf{植物性蛋白质：}大豆及其制品为优质蛋白质（完全蛋白质），赖氨酸含量较高；谷类（大米、面粉）缺赖氨酸，豆类缺蛋氨酸
    \item \textbf{蛋白质互补搭配：}应适当搭配动植物性食物，注意蛋白质互补。增加牛奶、大豆及其制品的消费
    \item 动物性蛋白质质量好但含饱和脂肪酸和胆固醇，植物性蛋白利用率低，两者搭配可互补
\end{itemize}
（来源：0613营养笔记）
\end{defbox}

\subsection*{参考答案要点}
\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item \textbf{氮平衡}：$B = I - (U + F + S)$，反映摄入氮与排出氮的关系。\textbf{必需氨基酸}：人体不能合成或合成速度不能满足需要，必须从食物中直接获得的氨基酸，共9种（缬/异亮/亮/苯丙/蛋/色/组/苏/赖，组氨酸为婴儿必需）。\textbf{条件必需氨基酸}：特定条件下合成能力有限或需要量增加，须从食物获取（半胱氨酸由蛋氨酸转变、酪氨酸由苯丙氨酸转变，膳食提供可分别减少30\%和50\%需要量）。\textbf{限制氨基酸}：食物蛋白质中相对含量较低或缺乏的必需氨基酸，含量最低者为第一限制氨基酸。\textbf{氨基酸模式}：蛋白质中各种必需氨基酸的构成比例，以色氨酸含量为1计算比值。\textbf{蛋白质互补作用}：两种以上食物混合食用达到以多补少、提高营养价值。\textbf{ONL}：必要氮损失，成人进食无蛋白膳食时经尿、粪、皮肤丢失的氮量，约54mg/kg体重/日。
    \item \textbf{BV} = 储留氮/吸收氮×100，储留氮 = 吸收氮 $-$ (尿氮 $-$ 尿内源性氮)。\textbf{NPU} = 消化率×生物价 = 储留氮/食物氮×100。\textbf{PER} = 动物体重增加(g)/摄入蛋白质(g)；校正PER = 待测PER/酪蛋白PER×2.5。\textbf{AAS} = 待测蛋白质每克蛋白质中某种氨基酸含量(mg)/理想模式中该氨基酸含量(mg)×100。\textbf{PDCAAS} = AAS×真消化率。\textbf{氨基酸代谢池}：体内游离氨基酸分布在细胞内外液中构成的代谢库，来源包括食物蛋白、组织蛋白分解和体内合成。
    \item 零氮平衡（$B=0$）：摄入氮与排出氮相等，健康成人应维持并富裕5\%。正氮平衡（$B>0$）：儿童发育、妇女怀孕、疾病恢复、运动增肌。负氮平衡（$B<0$）：老年人、饥饿过度节食、消耗性疾病。
    \item 影响因素：(1)膳食蛋白质摄入量；(2)能量摄入状况（能量不足→负氮平衡）；(3)生理状态（生长/妊娠→正，衰老/应激→负）；(4)蛋白质质量（优质蛋白利于维持氮平衡）；(5)运动、应激、激素水平等。
    \item 9种EAA：缬氨酸Val、异亮氨酸Ile、亮氨酸Leu、苯丙氨酸Phe、蛋氨酸Met、色氨酸Trp、组氨酸His（婴儿必需）、苏氨酸Thr、赖氨酸Lys。需要量：单位体重EAA需要量随年龄增长而下降；成人EAA需要量约占蛋白质需要量的20\%，儿童约占30\%$\sim$40\%。
    \item 三原则：食物生物学种属愈远愈好；搭配种类愈多愈好；食用时间愈近愈好，最好同时食用。例：谷类缺赖氨酸富蛋氨酸，豆类缺蛋氨酸富赖氨酸，同时食用互补提高营养价值。
    \item 水肿型（Kwashiorkor）：能量摄入基本满足，蛋白质严重不足→腹腿部水肿、虚弱、表情淡漠、生长迟缓、头发变色变脆易落。干瘦型（Marasmus）：蛋白质和能量均严重不足→小手无力，易感染其他疾病而死亡。
    \item 六项指标：(1)蛋白质含量（凯氏定氮法，氮×6.25）；(2)蛋白质消化率（真消化率TD/表观消化率AD）；(3)生物价BV（储留氮/吸收氮×100）；(4)蛋白质净利用率NPU（消化率×BV）；(5)蛋白质功效比值PER（幼小动物增重/摄入蛋白）；(6)氨基酸评分AAS及经消化率修正的PDCAAS（PDCAAS = AAS×真消化率，PDCAAS×2.5 = PER）。
    \item 动物性蛋白质：蛋、奶、鱼、肉——质量好、利用率高，含饱和脂肪酸和胆固醇。植物性蛋白质：大豆（优质蛋白/完全蛋白），谷类缺赖氨酸，豆类缺蛋氨酸。应适当搭配动植物性食物，发挥蛋白质互补作用。
\end{enumerate}
