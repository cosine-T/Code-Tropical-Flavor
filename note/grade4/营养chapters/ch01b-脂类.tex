\chapter{脂类}
\chapterintro{掌握脂肪、磷脂、胆固醇的生理功能；掌握脂肪酸命名（Cx:y）和分类；掌握必需脂肪酸（亚油酸/$\alpha$-亚麻酸）概念、功能和缺乏表现；掌握膳食脂肪营养价值评价指标；熟悉参考摄入量。}

\section{脂类的生理功能}

\begin{keybox}
\textbf{脂肪的生理功能：}提供热量；构成机体细胞；构成皮下脂肪；滋润皮肤；御寒保暖；保护内脏；支撑内脏；构成体脂；调节生理机能；溶解脂溶性维生素。

\textbf{磷脂的生理功能（除供能外）：}(1)细胞膜和亚细胞器膜的重要成分；(2)促进细胞内外物质交流，有利于脂肪的吸收、转运和代谢；(3)磷脂缺乏会造成细胞膜结构受损，出现毛细血管脆性增加和通透性增加，皮肤细胞对水的通透性增高引起水代谢紊乱，产生皮疹等。

\textbf{胆固醇的生理功能：}(1)所有体细胞，特别是神经组织的组成成分；(2)胆酸、7-脱氢胆固醇、维生素D$_2$、性激素、黄体酮、前列腺素、肾上腺皮质激素等重要生理活性物质和激素的前体物；(3)研究认为胆固醇具有一定的抗癌作用。
\end{keybox}

\section{脂肪酸}

\begin{defbox}
\textbf{分类：}按碳链长短（长链/中链/短链）；按饱和度（饱和/不饱和）；按双键数目；按空间结构。

\textbf{脂肪酸命名 Cx:y：}C为碳原子总数，x为碳数，y为双键个数。Cx:0为饱和脂肪酸（无双键）；Cx:1为一个双键（单不饱和）；Cx:2及以上为多不饱和。$\omega$n（Omega-n）：从甲基端往羧基端数，第一个双键位置。

\textbf{具有重要营养学意义的两类不饱和脂肪酸：}n-3（$\omega$-3）系列和n-6（$\omega$-6）系列。

\textbf{必需脂肪酸（Essential Fatty Acid, EFA）：}人体不可缺少而自身又不能合成，必须通过食物供给的多不饱和脂肪酸。\imp{n-6系列中的亚油酸（C18:2, n-6）和n-3系列中的$\alpha$-亚麻酸（C18:3, n-3）是人体必需的两种脂肪酸。}

\textbf{EFA功能：}(1)磷脂的重要组成成分；(2)合成体内某些重要物质的原料；(3)与胆固醇代谢有关；(4)促进发育，保护皮肤。\imp{缺乏：婴儿湿疹（对称），涂抹橄榄油。}
\end{defbox}

\section{膳食脂肪评价与参考摄入量}

\begin{defbox}
\textbf{膳食脂肪营养价值评价：}消化率；必需脂肪酸含量；脂溶性维生素含量；脂类稳定性。

\textbf{参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成人脂肪摄入占总能量\imp{20\%$\sim$30\%}
    \item 饱和脂肪酸 $<$ 10\%
    \item n-6系脂肪酸 2.5\%$\sim$9\%，n-3系 0.5\%$\sim$2.0\%
    \item EPA+DHA 250$\sim$2000mg/d
\end{itemize}
脂肪摄入过多→肥胖、心血管疾病、高血压、癌症（乳腺癌/大肠癌）发病率增高。
\end{defbox}

\section{复习题}
\begin{enumerate}[leftmargin=*, itemsep=8pt]
    \item \textbf{【名词解释】}必需脂肪酸（EFA）；脂肪酸命名Cx:y；$\omega$-3和$\omega$-6系列脂肪酸
    \item \textbf{【简答题】}简述脂肪、磷脂、胆固醇的生理功能。
    \item \textbf{【简答题】}简述必需脂肪酸的定义、种类和功能及缺乏表现。
    \item \textbf{【简答题】}简述膳食脂肪营养价值评价的四个方面及成人脂肪参考摄入量。
\end{enumerate}

\subsection*{参考答案要点}
\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item \textbf{EFA}：人体不可缺少而自身不能合成、必须通过食物供给的多不饱和脂肪酸（亚油酸C18:2 n-6和$\alpha$-亚麻酸C18:3 n-3）。\textbf{Cx:y}：x碳数，y双键数，$\omega$n从甲基端数第一个双键位置。
    \item 脂肪：供能/构成细胞/皮下脂肪/御寒保暖/保护支撑内脏/溶解脂溶性维生素。磷脂：细胞膜成分/促进物质交流/缺乏致毛细血管脆性增加。胆固醇：神经组织成分/激素和VD2前体/可能抗癌。
    \item 定义和种类见第1题。功能：磷脂组成成分/合成重要物质原料/与胆固醇代谢有关/促发育保护皮肤。缺乏：婴儿湿疹（对称）。
    \item 四方面：消化率/EFA含量/脂溶性维生素含量/脂类稳定性。摄入量：20-30\%E，SFA<10\%，n-6 2.5-9\%/n-3 0.5-2.0\%，EPA+DHA 250-2000mg/d。
\end{enumerate}
