\chapter{碳水化合物}
\chapterintro{\keyword{掌握}膳食纤维的概念、分类及功能；血糖指数的概念及其意义*。\keyword{熟悉}碳水化合物的分类、生理功能及食物来源。\keyword{了解}乳糖不耐受。}

\section{碳水化合物的分类（熟悉）}

\begin{tipbox}
\textbf{碳水化合物的分类：}

碳水化合物按分子结构可分为单糖、双糖、寡糖和多糖四大类。

\begin{itemize}[leftmargin=*]
    \item \imp{单糖：}不能被水解的最简单碳水化合物。
    \begin{itemize}
        \item 葡萄糖、果糖、半乳糖
        \item 糖醇类：糖的还原产物，广泛存在于自然界中（特别是植物）。如山梨糖醇、甘露糖醇、木糖醇等
        \item 其他单糖：核糖、脱氧核糖等
    \end{itemize}
    \item \imp{双糖：}
    \begin{itemize}
        \item 蔗糖：最有商业意义的糖，由1分子葡萄糖+1分子果糖组成，主要来源甘蔗、甜菜
        \item 麦芽糖：由2分子葡萄糖组成，淀粉在酶的作用下分解产生
        \item 乳糖：仅存在于奶制品中的碳水化合物，由1分子葡萄糖+1分子半乳糖组成
        \item 海藻糖
    \end{itemize}
    \item \imp{寡糖（低聚糖）：}3$\sim$10个单糖分子通过糖苷键聚合而成的聚合物。
    \begin{itemize}
        \item 功能性低聚糖：存在于水果、蔬菜中。如棉子糖（三糖）、水苏糖（四糖）
        \item 多数低聚糖不能或仅部分被消化吸收，在结肠被肠道益生菌发酵利用，产生短链脂肪酸
    \end{itemize}
    \item \imp{多糖：}$>$10个糖单位的聚合物，一般不溶于水，无甜味，不形成结晶，无还原性。
    \begin{itemize}
        \item 淀粉：人类的主要能源物质，存在于谷类、根茎类植物中
        \item 膳食纤维：包括纤维素、木质素（不属于碳水化合物，但其与碳水化合物紧密共存于植物细胞壁中，营养学上归入膳食纤维）、抗性低聚糖、果胶、抗性淀粉和其他不被消化吸收的碳水化合物
    \end{itemize}
\end{itemize}

\textbf{按人体能否消化吸收分类：}
\begin{itemize}[leftmargin=*]
    \item \imp{可消化吸收：}淀粉、糊精、糖原
    \item \imp{不可消化吸收：}抗性淀粉、膳食纤维
\end{itemize}
\end{tipbox}

\section{碳水化合物的功能（熟悉）}

\begin{tipbox}
\textbf{碳水化合物的主要功能：}

\imp{1. 提供能量：}
\begin{itemize}[leftmargin=*]
    \item 可消化的碳水化合物：1g = 16.7kJ（4kcal）
    \item 不可消化的碳水化合物：0$\sim$3kcal/g，平均2kcal/g
\end{itemize}

\imp{2. 构成组织结构及生理活性物质：}糖脂、糖蛋白、蛋白聚糖等是细胞膜、结缔组织、神经组织等的重要组成成分；核糖和脱氧核糖是核酸的组成成分。

\imp{3. 血糖调节作用：}碳水化合物是血糖的主要来源，维持血糖在正常范围内。

\imp{4. 节约蛋白质作用（sparing protein action）：}如果碳水化合物充足，人体首先利用碳水化合物供给能量，从而减少蛋白质单纯作为供给能量的分解代谢，有利于改善和提高蛋白质的利用，并增强人体内氮的贮存量，使蛋白质生理功能更有效地发挥。

\imp{5. 抗生酮作用（antiketogenesis）：}脂肪酸在体内分解代谢时产生的乙酰基需与碳水化合物代谢产生的草酰乙酸结合，才能进入三羧酸循环进而最终被彻底氧化。当碳水化合物不足时，因草酰乙酸不足使脂肪酸不能彻底氧化分解而产生过多酮体，会发生酮症酸中毒；而当碳水化合物充足时可防止酮症酸中毒的发生。

\imp{6. 膳食纤维促进肠道健康：}
\begin{itemize}[leftmargin=*]
    \item 增加饱腹感，减少进食量
    \item 促进排便：增加粪便含水量、减少粪便硬度；增加粪便体积、机械刺激肠道蠕动。谷类纤维比蔬菜水果类更有效地增加粪便体积和促进排便
    \item 降低血糖和血胆固醇
    \item 改变肠道菌群，诱导益生菌大量繁殖
\end{itemize}
\end{tipbox}

\section{膳食纤维（掌握）}

\begin{keybox}
\textbf{[概念] 膳食纤维（Dietary Fiber）：}膳食纤维是指植物性食物中不能被人体消化酶水解的多糖类物质和木质素的总称。主要包括纤维素、半纤维素、果胶、木质素、抗性淀粉、抗性低聚糖等。木质素虽为苯丙烷聚合物，不属于碳水化合物，但在营养学上归入膳食纤维。

\textbf{膳食纤维的分类：}
\begin{itemize}[leftmargin=*]
    \item \imp{可溶性膳食纤维：}果胶、树胶、黏胶、部分半纤维素等。溶于水，吸水膨胀，在结肠被肠道微生物发酵利用。
    \begin{itemize}
        \item 功能：增加粪便含水量、减少粪便硬度、降低血糖和血胆固醇、延缓胃排空
    \end{itemize}
    \item \imp{不溶性膳食纤维：}纤维素、木质素、部分半纤维素等。不溶于水。
    \begin{itemize}
        \item 功能：增加粪便体积、机械刺激肠道蠕动、促进排便。谷类纤维比蔬菜水果类更有效地增加粪便体积和促进排便
    \end{itemize}
\end{itemize}

\textbf{膳食纤维的生理功能：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{增加饱腹感：}吸水膨胀，增加胃内容物体积，减少进食量
    \item \textbf{促进排便：}可溶性膳食纤维增加粪便含水量、减少粪便硬度；不溶性膳食纤维增加粪便体积、机械刺激肠道蠕动。谷类纤维比蔬菜水果类更有效地增加粪便体积和促进排便
    \item \textbf{降低血糖和血胆固醇：}延缓碳水化合物消化吸收，降低餐后血糖；可结合胆酸促进胆固醇排出
    \item \textbf{改变肠道菌群：}在结肠被肠道微生物发酵，产生短链脂肪酸（醋酸、丙酸、丁酸），诱导益生菌大量繁殖，改善肠道微生态环境
    \item \textbf{对结肠癌的可能预防作用：}增加粪便体积，稀释肠道致癌物浓度；缩短肠道通过时间，减少致癌物与肠黏膜接触时间
\end{enumerate}
\end{keybox}

% [补充自往届笔记]
\begin{tipbox}
\textbf{抗性淀粉（Resistant Starch, RS）：}抗性淀粉属于膳食纤维的一类，指在小肠内不被消化吸收、在大肠内被发酵的淀粉及其降解产物。

\textbf{膳食纤维的组成（往届补充）：}膳食纤维包括纤维素、木质素、抗性低聚糖、果胶、抗性淀粉和其他不被消化吸收的碳水化合物。其中木质素虽非碳水化合物，但因与碳水化合物紧密共存于植物细胞壁中，营养学上归入膳食纤维。
\end{tipbox}

\section{乳糖不耐受（了解）}

\begin{tipbox}
\textbf{乳糖不耐受（Lactose Intolerance）：}由于体内乳糖酶活性降低或缺乏，不能完全消化乳糖，摄入乳糖后出现腹胀、腹泻、腹部不适等症状。乳糖酶活性随年龄增长而下降，这是乳糖不耐受的主要原因。

\textbf{应对方法（针对学龄前儿童的解决措施）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{少量多次饮奶或吃酸奶：}酸奶中乳糖部分已被发酵分解
    \item \textbf{饮奶前进食一定量主食：}避免空腹饮奶，减缓乳糖进入肠道速度
    \item \textbf{改吃无乳糖奶或饮奶时加用乳糖酶：}用乳糖酶预处理乳制品
\end{enumerate}
\end{tipbox}

\section{血糖指数（GI）与血糖负荷（GL）（掌握）*}

\begin{keybox}
\textbf{[概念] 血糖指数（Glycemic Index, GI）\difficulty：}反映食物引起人体血糖升高程度的指标，是人体进食后机体血糖生成的应答状态。以含50g碳水化合物的试验食物餐后2h血糖应答曲线下面积与等量碳水化合物标准食物（通常为葡萄糖或白面包）餐后2h血糖应答曲线下面积之比表示。

\[\text{GI} = \frac{\text{含50g CHO试验食物餐后2h血糖曲线下面积}}{\text{等量CHO标准食物餐后2h血糖曲线下面积}} \times 100\]

\begin{itemize}[leftmargin=*]
    \item \imp{低GI食物：GI $\leq$ 55}
    \item 中GI食物：55 $<$ GI $<$ 75
    \item \imp{高GI食物：GI $\geq$ 75}
\end{itemize}

\textbf{GI的生理意义：}GI反映食物被消化吸收后血糖升高的速度和程度。GI高的食物，进入胃肠道后消化快、吸收完全，葡萄糖迅速进入血液；反之，在胃肠道停留时间长、释放缓慢，葡萄糖进入血液后峰值低、下降速度慢。

\textbf{GI的影响因素：}
\begin{itemize}[leftmargin=*]
    \item 食物淀粉的性状
    \item 食物中非淀粉多糖（膳食纤维）的种类和数量
    \item 食物蛋白质含量
    \item 烹调方法：煮熟的淀粉GI高；加工越精细，GI越高
    \item 消化速度：越快被消化吸收，GI越高；脂肪和蛋白质延缓消化速度，因此脂肪和蛋白质含量高的食物GI低
\end{itemize}

\textbf{GI的应用：}
\begin{itemize}[leftmargin=*]
    \item 指导糖尿病患者选择食物：低GI有利于改善糖代谢，有效控制糖尿病患者血糖水平
    \item 控制体重
    \item 指导运动员饮食
    \item 防治慢性病
\end{itemize}

\textbf{[概念] 血糖负荷（Glycemic Load, GL）：}评价某种食物摄入量对人体血糖影响的幅度。GL将食物的GI值与实际摄入的碳水化合物量结合起来，更全面地反映食物对血糖的实际影响。

\[\text{GL} = \text{食物GI} \times \frac{\text{摄入该食物的实际可利用CHO含量(g)}}{100}\]

\begin{itemize}[leftmargin=*]
    \item GL $<$ 10 为低GL食物
    \item 10 $\sim$ 20 为中GL食物
    \item GL $>$ 20 为高GL食物
\end{itemize}

\textbf{GI与GL的关系：}GL越高，餐后血糖总体水平越高。GI相同的食物，其GL可能不同（取决于实际摄入的碳水化合物量）。因此，结合GI和GL可以更科学地指导饮食选择。
\end{keybox}

% [补充自往届笔记]
\begin{tipbox}
\textbf{GI的拓展说明（往届补充）：}
\begin{itemize}[leftmargin=*]
    \item GI的另一种表述：GI = 某食物食入后2h血糖升高面积 / 相等量葡萄糖食入后2h血糖升高面积
    \item GI是衡量某种食物或膳食组成对血糖浓度影响的指标，以百分比含量表示，通常以葡萄糖或白面包为标准
    \item GI的分级标准（往届版本略有不同）：GI $>$ 70为高血糖指数食物，70$\sim$55为中等血糖指数食物，$\leq$55为低血糖指数食物

\end{itemize}
\end{tipbox}

\section{碳水化合物的食物来源与参考摄入量（熟悉）}

\begin{tipbox}
\textbf{碳水化合物的食物来源：}
\begin{itemize}[leftmargin=*]
    \item 主要：粮谷类及其制品
    \item 根茎块类蔬菜（土豆、红薯、芋头等）
    \item 杂豆类及制品（红豆、绿豆、豌豆、蚕豆）
    \item 水果及制品
    \item 纯糖类及其制品（蜂蜜、糖果、白糖等）
\end{itemize}

\textbf{碳水化合物的参考摄入量：}我国建议2岁以上人群的碳水化合物供能比应为\imp{55\%$\sim$65\%}（蛋白质供能比10\%$\sim$15\%，脂肪20\%$\sim$30\%）。
\end{tipbox}

\subsection*{参考答案要点}
\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item \textbf{单糖：}不能被水解的最简单碳水化合物，包括葡萄糖、果糖、半乳糖、糖醇类等。\textbf{寡糖（低聚糖）：}3$\sim$10个单糖分子通过糖苷键聚合而成的聚合物，如棉子糖、水苏糖。\textbf{抗性淀粉(RS)：}不被健康人体小肠消化吸收的淀粉及其降解产物，属于膳食纤维的一类。\textbf{膳食纤维：}植物性食物中不能被人体消化酶水解的多糖类物质和木质素的总称，包括可溶性（果胶等）和不溶性（纤维素等）。\textbf{GI：}GI = 含50g碳水化合物试验食物餐后2h血糖曲线下面积 / 等量碳水化合物标准食物餐后2h血糖曲线下面积 $\times$ 100\%。\textbf{GL：}GL = 食物GI $\times$ 摄入该食物的实际可利用碳水化合物含量(g) / 100。

    \item \textbf{节约蛋白质作用：}如果碳水化合物充足，人体首先利用碳水化合物供给能量，从而减少蛋白质单纯作为供给能量的分解代谢，有利于改善和提高蛋白质的利用，并增强人体内氮的贮存量，使蛋白质生理功能更有效地发挥。\textbf{抗生酮作用：}脂肪酸在体内分解代谢时产生的乙酰基需与碳水化合物代谢产生的草酰乙酸结合，才能进入三羧酸循环进而最终被彻底氧化；当碳水化合物不足时，因草酰乙酸不足使脂肪酸不能彻底氧化分解而产生过多酮体，会发生酮症酸中毒；而当碳水化合物充足时可防止酮症酸中毒的发生。\textbf{乳糖不耐受：}因乳糖酶活性降低或缺乏，不能完全消化乳糖，摄入乳糖后出现腹胀、腹泻等。应对：(1)少量多次饮奶或吃酸奶；(2)饮奶前进食主食；(3)改吃无乳糖奶或加用乳糖酶。

    \item 按分子结构分类：单糖（葡萄糖、果糖、半乳糖、糖醇类）；双糖（蔗糖、麦芽糖、乳糖、海藻糖）；寡糖/低聚糖（棉子糖、水苏糖等，3$\sim$10个单糖分子）；多糖（$>$10个糖单位，包括淀粉和膳食纤维）。按人体可消化吸收：可消化吸收（淀粉、糊精、糖原）；不可消化吸收（抗性淀粉、膳食纤维）。

    \item 碳水化合物五大功能：(1)提供能量——可消化CHO 1g=4kcal，不可消化CHO平均2kcal/g；(2)构成组织结构及生理活性物质——糖脂、糖蛋白、蛋白聚糖等；(3)血糖调节作用；(4)节约蛋白质作用——CHO充足时首先利用CHO供能，减少蛋白质分解；(5)抗生酮作用——CHO代谢产生的草酰乙酸与脂肪酸分解产生的乙酰基结合进入TCA循环，防止酮体过多。

    \item 节约蛋白质作用机制：碳水化合物充足 $\rightarrow$ 人体首先利用碳水化合物供给能量 $\rightarrow$ 减少蛋白质单纯作为供给能量的分解代谢 $\rightarrow$ 改善和提高蛋白质的利用 $\rightarrow$ 增强体内氮贮存量。抗生酮作用机制：脂肪酸分解 $\rightarrow$ 产生乙酰基 $\rightarrow$ 需与草酰乙酸（来自碳水化合物代谢）结合 $\rightarrow$ 进入三羧酸循环彻底氧化；碳水化合物不足 $\rightarrow$ 草酰乙酸不足 $\rightarrow$ 脂肪酸不能彻底氧化 $\rightarrow$ 酮体过多 $\rightarrow$ 酮症酸中毒。

    \item 概念：植物性食物中不能被人体消化酶水解的多糖类物质和木质素的总称。分类：可溶性膳食纤维（果胶、树胶等，吸水膨胀，降低血糖胆固醇）和不溶性膳食纤维（纤维素、木质素等，增加粪便体积，促进排便）。功能：(1)增加饱腹感；(2)促进排便；(3)降低血糖和血胆固醇；(4)改变肠道菌群，产生短链脂肪酸；(5)可能预防结肠癌。

    \item GI定义：GI = 含50g CHO试验食物餐后2h血糖曲线下面积 / 等量CHO标准食物餐后2h血糖曲线下面积 $\times$ 100。分级：低GI $\leq$ 55，中GI 55$\sim$75，高GI $\geq$ 75。影响因素：淀粉性状、非淀粉多糖种类和数量、蛋白质含量、烹调方法；越快被消化$\rightarrow$GI越高，脂肪和蛋白质延缓消化。应用：指导糖尿病饮食、控制体重、运动员饮食、防治慢性病。GL = 食物GI $\times$ 摄入CHO量(g) / 100；GL $<$ 10低，10$\sim$20中，$>$ 20高；GL越高血糖总体水平越高，GI相同的食物GL可能不同。

    \item 主要食物来源：粮谷类及其制品；根茎块类蔬菜（土豆、红薯、芋头等）；杂豆类及制品；水果及制品；纯糖类及其制品（蜂蜜、糖果、白糖等）。参考摄入量：我国建议2岁以上人群碳水化合物供能比为55\%$\sim$65\%（蛋白质10\%$\sim$15\%，脂肪20\%$\sim$30\%）。

    \item 乳糖不耐受因乳糖酶活性降低或缺乏所致，摄入乳糖后出现腹胀、腹泻等。应对：(1)少量多次饮奶或吃酸奶；(2)饮奶前进食主食，避免空腹饮奶；(3)改吃无乳糖奶或加用乳糖酶。
\end{enumerate}
