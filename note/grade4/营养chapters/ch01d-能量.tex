\chapter{能量}
\chapterintro{掌握能量单位和产能系数（CHO4/脂肪9/蛋白质4）；掌握人体能量消耗的组成部分（基础代谢/身体活动/食物热效应/生长发育）；掌握基础代谢（BEE/BMR）及其影响因素；掌握食物特殊动力作用（TEF/SDA）；掌握体重评价方法（Broca改良公式/BMI/腰臀围比/腰围）；熟悉身体活动能量消耗及人体能量需要量测定方法。}

\section{能量单位与产能系数}

\begin{keybox}
\textbf{能量单位：}营养学领域常使用的能量单位是卡（cal）、千卡（kcal）。\imp{1kcal $\rightarrow$ 1个标准大气压下，1kg纯水从15℃上升到16℃所需的能量。}

\textbf{换算关系：}\imp{1kcal = 4.174kJ。}

\textbf{产能系数/能量系数/食物热价：}每g碳水化合物、脂肪、蛋白质在体内氧化产生的能量值。
\begin{itemize}[leftmargin=*]
    \item \imp{1g蛋白质 = 4kcal}
    \item \imp{1g脂肪 = 9kcal}
    \item \imp{1g碳水化合物 = 4kcal}
\end{itemize}
\end{keybox}

\section{人体的能量消耗}

\begin{keybox}
\textbf{成人的能量消耗：}维持\imp{基础代谢}、\imp{身体活动}、\imp{食物热效应（食物特殊动力作用）}。

\textbf{孕妇及乳母的能量消耗：}上述三种 + 胎儿生长发育、母体子宫、胎盘等组织增长、合成分泌乳汁、体脂储备等。

\textbf{婴幼儿、儿童少年的能量消耗：}上述三种 + 生长发育。
\end{keybox}

\section{基础代谢（BEE与BMR）}

\begin{keybox}
\textbf{基础代谢（basal metabolism）/基础能量消耗（BEE）：}维持机体最基本生命活动所需要的能量消耗，占人体总能量消耗的\imp{60\%$\sim$70\%}。此时能耗仅用于基本生理功能需要（维持体温、呼吸、血液循环、脉搏、心脏搏动及其他器官组织的活动）。

\textbf{测定条件（WHO/FAO定义）：}人体经过\imp{10$\sim$12h空腹}和良好的睡眠、清醒仰卧、恒温条件下（一般\imp{22$\sim$26℃}），无任何身体活动和紧张的思维活动，全身肌肉放松时的能量消耗。

\textbf{基础代谢率（BMR）：}人处于基础代谢状态下，每小时每平方米体表面积所消耗的能量，常用单位为\imp{kcal/(h$\cdot$m$^2$)}。

\textbf{人一日基础代谢的能量消耗：}\imp{BEE = BMR × 24h × 体表面积。}
\end{keybox}

\section{影响人体BMR的因素}

\begin{keybox}
\textbf{影响人体BMR的因素：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{体型和体质：}同等体重情况下，\imp{瘦体质的BMR > 矮胖者}
    \item \textbf{生理与病理状况：}内分泌和应激、年龄性别、营养与机能
    \item \textbf{生活和作业环境}
    \item \textbf{其他：}咖啡因、尼古丁等
\end{enumerate}
\end{keybox}

\section{身体活动能量消耗}

\begin{defbox}
\textbf{身体活动：}任何由骨骼肌收缩引起能量消耗的身体运动，约占人体总能量消耗的\imp{15\%$\sim$30\%}，随人体运动量增加而增加，是人体能耗中变动最大，控制能量消耗、保持能量平衡和维持健康最重要的部分。

\textbf{影响因素：}
\begin{itemize}[leftmargin=*]
    \item 肌肉越发达的 $\rightarrow$ 活动时能量消耗越多
    \item 体重越重的 $\rightarrow$ 做相同的运动的能量消耗越多
    \item 工作越不熟练的 $\rightarrow$ 消耗能量越多
\end{itemize}
\end{defbox}

\section{食物热效应（TEF/SDA）}

\begin{keybox}
\textbf{食物热效应（TEF）/食物特殊动力作用（SDA）：}人体在摄食过程中所引起的额外能量消耗，是摄食后发生的一系列消化、吸收利用以及营养素及其代谢产物之间相互转化过程中所消耗的能量。

\textbf{不同营养素TEF的差异：}
\begin{itemize}[leftmargin=*]
    \item \imp{蛋白质的TEF最大 = 30\%$\sim$40\%}（占自身产能比例）
    \item \imp{碳水次之 = 5\%$\sim$10\%}
    \item \imp{脂肪最低 = 0\%$\sim$5\%}
\end{itemize}

\textbf{原因：}
\begin{itemize}[leftmargin=*]
    \item 产能营养素ATP最高转化率不同
    \item 产能营养素在体内的代谢形式不同引起能量消耗不同
\end{itemize}

\textbf{TEF的影响因素：}
\begin{itemize}[leftmargin=*]
    \item 摄食量$\uparrow$ $\rightarrow$ 能量消耗$\uparrow$
    \item 进食快$\uparrow$ $\rightarrow$ TEF$\uparrow$（进食快时，CNS活跃，激素和酶分泌速度快且多，吸收和储存的速率高，能量消耗也相对较多）
\end{itemize}
\end{keybox}

\section{体重评价}

\begin{keybox}
\textbf{Broca改良公式：}\imp{标准体重(kg) = 身高(cm) $-$ 105}

\textbf{判定标准：}
\begin{itemize}[leftmargin=*]
    \item 正常体重 = 标准体重 $\pm$ 10\%
    \item 超重：体重 $>$ 标准体重的10\%
    \item 肥胖：体重 $>$ 标准体重的20\%
\end{itemize}

\textbf{体质指数（BMI）：}\imp{BMI = 体重(kg) / [身高(m)]$^2$}

\textbf{我国BMI判定：}
\begin{itemize}[leftmargin=*]
    \item BMI $<$ 18.5：消瘦
    \item 18.5 $\sim$ 23.9：正常
    \item 24.0 $\sim$ 27.9：超重
    \item $\geq$ 28：肥胖
\end{itemize}

\textbf{腰臀围比：}\imp{男 $>$ 0.9 或 女 $>$ 0.85 $\rightarrow$ 中心性肥胖。}

\textbf{腰围：}\imp{男 $\geq$ 85cm 或 女 $\geq$ 80cm $\rightarrow$ 中心性肥胖。}
\end{keybox}

\section{人体能量需要量测定方法}

\begin{defbox}
\textbf{人体能量需要量常用测定方法：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{体重评价法：}Broca改良公式、BMI、腰臀围比、腰围
    \item \textbf{能量消耗测定法：}直接测热法、间接测热法（气体代谢法）、双标水法、心率监测法
\end{enumerate}
\end{defbox}

\section{能量供给}

\begin{keybox}
\textbf{三大产能营养素供能比：}
\begin{itemize}[leftmargin=*]
    \item \imp{碳水化合物：50\%$\sim$65\%}
    \item \imp{脂肪：20\%$\sim$30\%}
    \item \imp{蛋白质：10\%$\sim$15\%}
\end{itemize}
\end{keybox}

\subsection*{参考答案要点}
\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item \textbf{BEE}：维持机体最基本生命活动所需要的能量消耗，占人体总能量消耗的60\%$\sim$70\%，此时能耗仅用于基本生理功能需要（维持体温、呼吸、血液循环、脉搏、心脏搏动及其他器官组织的活动）。\textbf{BMR}：人处于基础代谢状态下，每小时每平方米体表面积所消耗的能量，常用单位kcal/(h$\cdot$m$^2$)。\textbf{TEF/SDA}：人体在摄食过程中所引起的额外能量消耗，是摄食后发生的一系列消化、吸收利用以及营养素及其代谢产物之间相互转化过程中所消耗的能量。\textbf{能量系数}：每g产能营养素在体内氧化产生的能量值（1g蛋白质=4kcal，1g脂肪=9kcal，1g碳水化合物=4kcal）。\textbf{BMI} = 体重(kg)/[身高(m)]$^2$。
    \item 成人能量消耗：基础代谢+身体活动+食物热效应（食物特殊动力作用）。孕妇及乳母额外消耗：胎儿生长发育+母体子宫、胎盘等组织增长+合成分泌乳汁+体脂储备等。婴幼儿/儿童少年额外消耗：生长发育。
    \item 测定条件（WHO/FAO）：人体经过10$\sim$12h空腹和良好的睡眠、清醒仰卧、恒温条件下（一般22$\sim$26℃），无任何身体活动和紧张的思维活动，全身肌肉放松时的能量消耗。计算公式：BEE = BMR × 24h × 体表面积。
    \item 影响因素：体型和体质（同等体重情况下，瘦体质的BMR>矮胖者）；生理与病理状况（内分泌和应激、年龄性别、营养与机能）；生活和作业环境；其他（咖啡因、尼古丁等）。
    \item TEF是人体在摄食过程中所引起的额外能量消耗。不同营养素TEF差异：蛋白质的TEF最大=30\%$\sim$40\%（占自身产能比例），碳水次之=5\%$\sim$10\%，脂肪最低=0\%$\sim$5\%。原因：产能营养素ATP最高转化率不同；产能营养素在体内的代谢形式不同引起能量消耗不同。TEF影响因素：摄食量↑→能量消耗↑；进食快↑→TEF↑（CNS活跃，激素和酶分泌速度快且多，吸收和储存的速率高，能量消耗也相对较多）。
    \item Broca改良公式：标准体重(kg)=身高(cm)-105。判定标准：正常体重=标准体重±10\%；超重>标准体重的10\%；肥胖>标准体重的20\%。我国BMI判定：BMI$<$18.5消瘦，18.5$\sim$23.9正常，24.0$\sim$27.9超重，$\ge$28肥胖。腰臀围比：男$>$0.9或女$>$0.85$\rightarrow$中心性肥胖。腰围：男$\ge$85cm或女$\ge$80cm$\rightarrow$中心性肥胖。
    \item 碳水化合物供能比50\%$\sim$65\%，脂肪20\%$\sim$30\%，蛋白质10\%$\sim$15\%。
\end{enumerate}
