\chapter{能量}
\chapterintro{掌握能量单位和产能系数（CHO4/脂肪9/蛋白质4）；掌握人体能量消耗的四个组成部分（基础代谢/体力活动/食物热效应/生长发育）；掌握BMR的影响因素；掌握体重评价方法（Broca改良公式/BMI）；掌握减重原则。}

\section{能量单位与产能系数}

\begin{defbox}
\textbf{能量单位：}常用单位——卡（calorie, cal）/千卡（kcal）；国际制单位——焦耳（Joule, J）/千焦（kJ）。\imp{1kcal = 4.184kJ，1kJ = 0.239kcal}。

\textbf{能量系数/产热系数：}每g碳水化合物、脂肪、蛋白质在体内氧化产生的能量值。
\begin{itemize}[leftmargin=*]
    \item \imp{碳水化合物：4.1 × 98\% = 4kcal/g}
    \item \imp{脂肪：9.45 × 95\% = 9kcal/g}
    \item \imp{蛋白质：(5.65 $-$ 1.3) × 92\% = 4kcal/g}
\end{itemize}
\end{defbox}

\section{人体能量消耗}

\begin{keybox}
\textbf{人体的能量消耗：}\imp{基础代谢 + 体力活动 + 食物热效应（食物特殊动力作用）+ 生长发育}

\textbf{1. 基础代谢：}基础代谢率（Basal Metabolic Rate, BMR）指单位时间内人体每平方米体表面积所消耗的基础代谢能量。单位：kJ/m$^2\cdot$h或kcal/m$^2\cdot$h。

\textbf{影响BMR的因素：}体型和瘦体质（lean body mass）；内分泌和应激状态；年龄；性别；营养与机能状态；环境因素；其他（咖啡因/尼古丁）。

\textbf{2. 体力活动：}约占人体总能量消耗的15\%$\sim$30\%，是人体能量消耗变化最大，也是控制能量平衡/维持健康最重要的部分。影响因素：肌肉发达程度/体重/活动时间与强度/熟练程度。

\textbf{3. 食物热效应（Thermic Effect of Food, TEF）：}因摄食而引起的能量额外消耗，与消化/吸收/代谢转化等有关，同时引起体温升高和散发热能。

\textbf{4. 生长发育：}婴幼儿/儿童/青少年；孕妇；乳母；疾病恢复期病人。
\end{keybox}

\section{能量平衡与体重评价}

\begin{defbox}
\textbf{能量平衡的判定——体重：}
\begin{itemize}[leftmargin=*]
    \item Broca公式：标准体重(kg) = 身高(cm) $-$ 100
    \item \imp{Broca改良公式：标准体重(kg) = 身高(cm) $-$ 105}
    \item 平田公式：标准体重(kg) = [身高(cm) $-$ 100] × 0.9
\end{itemize}
\textbf{判定标准：}正常=标准体重±10\%；超重>标准体重10\%；肥胖>标准体重20\%。\imp{BMI = 体重(kg) / 身高(m)$^2$}。

\textbf{减重原则：}减的是脂肪而非瘦体质；每周减重不超过1kg；能量摄入每天不应低于1200kcal，最低不少于1000kcal；每天至少三顿饭；供能比蛋白质15\%$\sim$20\%，脂肪20\%。
\end{defbox}

\section{复习题}
\begin{enumerate}[leftmargin=*, itemsep=8pt]
    \item \textbf{【名词解释】}基础代谢率（BMR）；食物热效应（TEF）；能量系数；BMI
    \item \textbf{【简答题】}简述人体能量消耗的四个组成部分。
    \item \textbf{【简答题】}影响基础代谢率的因素有哪些？
    \item \textbf{【简答题】}简述标准体重评价方法和减重原则。
\end{enumerate}

\subsection*{参考答案要点}
\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item \textbf{BMR}：单位时间每m$^2$体表面积消耗的基础代谢能量。\textbf{TEF}：因摄食引起消化/吸收/代谢转化等的额外能量消耗。\textbf{能量系数}：每g产能营养素在体内氧化产生的能量（CHO4/脂肪9/蛋白质4 kcal/g）。\textbf{BMI}=体重(kg)/身高(m)$^2$。
    \item BMR(基础)+体力活动(15-30\%)+TEF(食物热效应)+生长发育。
    \item 体型和瘦体质/内分泌和应激/年龄/性别/营养机能状态/环境/咖啡因尼古丁。
    \item Broca改良=身高-105，正常=±10\%，超重>10\%，肥胖>20\%。减重原则：减脂肪非瘦体质/每周≤1kg/能量≥1200kcal(最低1000)/每天≥3餐/蛋白质15-20\%。
\end{enumerate}
