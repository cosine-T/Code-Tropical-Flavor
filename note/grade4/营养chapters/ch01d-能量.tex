\chapter{能量}
\chapterintro{\keyword{掌握}能量单位和产能系数（碳水化合物4/脂肪9/蛋白质4）；掌握人体能量消耗的组成部分（基础代谢/身体活动/食物热效应）；掌握基础代谢（BEE/BMR）及其影响因素；掌握食物特殊动力作用（TEF/SDA）*；掌握体力活动分级。\keyword{熟悉}人体能量需要量测定方法*、呼吸商（RQ）的概念。\keyword{了解}影响体力活动能量消耗的因素、食物的氧热价。}

\section{能量单位与产能系数}

\begin{keybox}
\textbf{能量单位：}营养学领域常使用的能量单位是卡（cal）、千卡（kcal）。\imp{1kcal $\rightarrow$ 1个标准大气压下，1kg纯水从15℃上升到16℃所需的能量。}

\textbf{换算关系：}\imp{1kcal = 4.174kJ。}% [补充自往届笔记] 反向换算：1kJ = 0.239kcal，1kcal = 4.184kJ。

\textbf{产能系数/能量系数/食物热价：}每g碳水化合物、脂肪、蛋白质在体内氧化产生的能量值。
\begin{itemize}[leftmargin=*]
    \item \imp{1g蛋白质 = 4kcal}
    \item \imp{1g脂肪 = 9kcal}
    \item \imp{1g碳水化合物 = 4kcal}
\end{itemize}
\end{keybox}

\section{人体的能量消耗}

\begin{keybox}
\textbf{成人的能量消耗：}维持\imp{基础代谢}、\imp{身体活动}、\imp{食物热效应（食物特殊动力作用）}。

\textbf{孕妇及乳母的能量消耗：}上述三种 + 胎儿生长发育、母体子宫、胎盘等组织增长、合成分泌乳汁、体脂储备等。

\textbf{婴幼儿、儿童少年的能量消耗：}上述三种 + 生长发育。
\end{keybox}

\section{基础代谢（BEE与BMR）}

\begin{keybox}
\textbf{基础代谢（basal metabolism）/基础能量消耗（BEE）：}维持机体最基本生命活动所需要的能量消耗，占人体总能量消耗的\imp{60\%$\sim$70\%}。此时能耗仅用于基本生理功能需要（维持体温、呼吸、血液循环、脉搏、心脏搏动及其他器官组织的活动）。

\textbf{测定条件（WHO/FAO定义）：}人体经过\imp{10$\sim$12h空腹}和良好的睡眠、清醒仰卧、恒温条件下（一般\imp{22$\sim$26℃}），无任何身体活动和紧张的思维活动，全身肌肉放松时的能量消耗。

\textbf{基础代谢率（BMR）：}人处于基础代谢状态下，每小时每平方米体表面积所消耗的能量，常用单位为\imp{kcal/(h$\cdot$m$^2$)}。

\textbf{人一日基础代谢的能量消耗：}\imp{BEE = BMR $\times$ 24h $\times$ 体表面积。}
\end{keybox}

\section{影响人体BMR的因素}

\begin{tipbox}
\textbf{影响人体BMR的因素（熟悉）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{体型和体质：}同等体重情况下，\imp{瘦体质的BMR > 矮胖者}
    \item \textbf{生理与病理状况：}内分泌和应激（内分泌异常、应激状态）、年龄性别、营养与机能（营养状况、机能状态）
    \item \textbf{生活和作业环境}
    \item \textbf{其他：}咖啡因、尼古丁等
\end{enumerate}
\end{tipbox}

\section{身体活动能量消耗}

\begin{defbox}
\textbf{身体活动（physical activity）：}任何由骨骼肌收缩引起能量消耗的身体运动，约占人体总能量消耗的\imp{15\%$\sim$30\%}，随人体运动量增加而增加，是人体能耗中变动最大，控制能量消耗、保持能量平衡和维持健康最重要的部分。
\end{defbox}

\begin{tipbox}
\textbf{影响体力活动能量消耗的因素（了解）：}
\begin{itemize}[leftmargin=*]
    \item 肌肉越发达的 $\rightarrow$ 活动时能量消耗越多
    \item 体重越重的 $\rightarrow$ 做相同的运动的能量消耗越多
    \item 工作越不熟练的 $\rightarrow$ 消耗能量越多
\end{itemize}
\end{tipbox}

\section{体力活动分级}

\begin{keybox}
\textbf{体力活动分级（掌握）：}
中国营养学会将体力活动分为三级：
\begin{enumerate}[leftmargin=*]
    \item \imp{轻体力活动：}75\%时间坐或站立，25\%时间站立活动。如办公室工作、售货员、教师等。
    \item \imp{中体力活动：}40\%时间坐或站立，60\%时间特殊职业活动。如学生日常活动、机动车驾驶、车床操作等。
    \item \imp{重体力活动：}25\%时间坐或站立，75\%时间特殊职业活动。如非机械化农业劳动、炼钢、舞蹈、竞技体育等。
\end{enumerate}
\end{keybox}

\section{食物热效应（TEF/SDA）}

\begin{keybox}
\textbf{食物热效应（TEF）/食物特殊动力作用（SDA）\difficulty：}人体在摄食过程中所引起的额外能量消耗，是摄食后发生的一系列消化、吸收利用以及营养素及其代谢产物之间相互转化过程中所消耗的能量。

\textbf{不同营养素TEF的差异：}
\begin{itemize}[leftmargin=*]
    \item \imp{蛋白质的TEF最大 = 30\%$\sim$40\%}（占自身产能比例）
    \item \imp{碳水次之 = 5\%$\sim$10\%}
    \item \imp{脂肪最低 = 0\%$\sim$5\%}
\end{itemize}

\textbf{原因：}
\begin{itemize}[leftmargin=*]
    \item 产能营养素ATP最高转化率不同
    \item 产能营养素在体内的代谢形式不同引起能量消耗不同
\end{itemize}

\textbf{TEF的影响因素：}
\begin{itemize}[leftmargin=*]
    \item 摄食量$\uparrow$ $\rightarrow$ 能量消耗$\uparrow$
    \item 进食快$\uparrow$ $\rightarrow$ TEF$\uparrow$（进食快时，CNS活跃，激素和酶分泌速度快且多，吸收和储存的速率高，能量消耗也相对较多）
\end{itemize}
\end{keybox}

\section{呼吸商与食物的氧热价}

\begin{defbox}
\textbf{呼吸商（Respiratory Quotient, RQ）：}一定时间内机体呼出的CO$_2$量与吸入O$_2$量的比值（CO$_2$/O$_2$）。RQ反映体内各种供能物质氧化分解的比例。
\begin{itemize}[leftmargin=*]
    \item 碳水化合物完全氧化：RQ = 1.0
    \item 脂肪完全氧化：RQ $\approx$ 0.71
    \item 蛋白质完全氧化：RQ $\approx$ 0.80
    \item 混合膳食：RQ $\approx$ 0.85
\end{itemize}
RQ > 1表示体内有脂肪合成（碳水化合物转化为脂肪时产生额外CO$_2$）。测定RQ可了解体内各营养素的氧化利用情况。
\end{defbox}

\begin{tipbox}
\textbf{食物的氧热价（了解）：}某种食物氧化时消耗1L O$_2$所产生的热量。
\begin{itemize}[leftmargin=*]
    \item 碳水化合物：约5.0 kcal/L O$_2$
    \item 脂肪：约4.7 kcal/L O$_2$
    \item 蛋白质：约4.5 kcal/L O$_2$
\end{itemize}
食物的氧热价反映不同供能物质的氧耗-产热关系，是间接测热法（气体代谢法）计算能量消耗的基础参数。通过测定机体一定时间的耗氧量和CO$_2$产生量，结合氧热价和RQ，可计算出机体在该段时间内的能量消耗。
\end{tipbox}

\section{人体能量需要量测定方法*（熟悉）}

\begin{tipbox}
\textbf{人体能量需要量常用测定方法：}

\textbf{一、体重评价法}
\begin{itemize}[leftmargin=*]
    \item \textbf{Broca改良公式：}标准体重(kg) = 身高(cm) $-$ 105
    \item \textbf{判定标准：}正常体重 = 标准体重 $\pm$ 10\%；超重：体重 $>$ 标准体重的10\%；肥胖：体重 $>$ 标准体重的20\%
    % [补充自往届笔记] Broca原公式：标准体重(kg)=身高(cm)-100；Broca改良公式：标准体重(kg)=身高(cm)-105；平田公式：标准体重(kg)=[身高(cm)-100]×0.9
    \item \textbf{体质指数（BMI）：}BMI = 体重(kg) / [身高(m)]$^2$
    \item 我国BMI判定：BMI $<$ 18.5 消瘦；18.5 $\sim$ 23.9 正常；24.0 $\sim$ 27.9 超重；$\geq$ 28 肥胖
    \item 腰臀围比：男 $>$ 0.9 或 女 $>$ 0.85 $\rightarrow$ 中心性肥胖
    \item 腰围：男 $\geq$ 85cm 或 女 $\geq$ 80cm $\rightarrow$ 中心性肥胖
\end{itemize}

\textbf{二、能量消耗测定法}
\begin{enumerate}[leftmargin=*]
    \item \textbf{直接测热法（Direct Calorimetry）：}将被测者置于测热室内，直接测定机体向外散发的热量。原理准确但设备昂贵，不适用于大规模现场调查。
    \item \textbf{间接测热法/气体代谢法（Indirect Calorimetry）：}通过测定机体一定时间内的耗氧量（VO$_2$）和CO$_2$产生量（VCO$_2$），结合呼吸商（RQ）和氧热价，计算出机体能量消耗。临床上常用代谢车进行测定。
    \item \textbf{双标水法（Doubly Labeled Water, DLW）：}受试者饮用含有$^2$H和$^{18}$O标记的水，通过测定体液中两种同位素的消除速率差，计算CO$_2$产生率，进而推算能量消耗。是评价自由生活状态下人体能量消耗的"金标准"，适用于各类人群，但费用昂贵。
    \item \textbf{心率监测法：}通过监测心率与能量消耗的关系（需个体校准），推算全天能量消耗。简便易行，但个体差异大，精确度较低。
\end{enumerate}

\textbf{三、能量需要量确定}
实际应用中，常以体重变化和BMI作为人群能量摄入是否充足的主要评价指标。个体能量需要量可通过膳食调查配合体力活动水平评估来推算。
\end{tipbox}
