\chapter{矿物质}
\chapterintro{掌握矿物质的概念与分类（常量/微量元素）；掌握钙、铁、锌、碘、硒的生理功能、吸收、缺乏；熟悉其他矿物质（磷、镁、钾、钠、钴等）；了解生理作用剂量带和矿物质的相互作用。}

\section{矿物质概述}

\begin{keybox}
\textbf{矿物质的概念}

\textbf{概念：}矿物质（Minerals）指人体组织中除碳、氢、氧和氮主要以有机化合物形式存在外的其余元素，统称为矿物质（无机盐或灰分）。
\end{keybox}

\begin{defbox}
\textbf{矿物质的分类}

\begin{itemize}[leftmargin=*]
    \item \imp{常量元素：}体内含量大于体重的0.01\%的矿物质，包括Ca、P、Na、K、S、Cl、Mg，每日摄入量>100mg。
    \item \imp{微量元素：}体内含量小于体重的0.01\%的矿物质。
    \begin{itemize}[leftmargin=*]
        \item \imp{必需微量元素：}铁、钴、钼、铜、锌、硒、碘、铬。
        \item \imp{可能必需微量元素：}锰、硅、镍、硼、钒。
        \item \imp{潜在毒性微量元素：}氟、铅、镉、汞、砷、铝、锡、锂。
        \item \imp{功能未知微量元素：}锶、铷、稀土元素等。
    \end{itemize}
\end{itemize}

\textbf{矿物质的生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 构成机体组织的重要原料
    \item 维持机体酸碱平衡和渗透压（细胞外液渗透压：Na$^+$、Cl$^-$；细胞内液渗透压：K$^+$、HPO$_4^{2-}$）
    \item 维持神经肌肉兴奋性
    \item 参与构成激素、维生素、蛋白质和多种酶
\end{enumerate}
\end{defbox}

\begin{tipbox}
\textbf{生理作用剂量带（了解）}

矿物质的生理作用与剂量密切相关，从缺乏到中毒的剂量-效应曲线通常呈"U"形或"J"形特征。按照摄入剂量可分为以下剂量带：
\begin{itemize}[leftmargin=*]
    \item \textbf{严重缺乏带：}摄入量远低于需要量，出现典型的临床缺乏症状和体征。
    \item \textbf{边缘缺乏带：}摄入量略低于需要量，无明显临床症状，但可降低生理功能和对疾病的抵抗力。
    \item \textbf{适宜摄入带：}摄入量满足机体需要且无不良反应，维持正常生理功能和机体健康。
    \item \textbf{边缘毒性带：}摄入量略超UL，可能出现亚临床的生理功能异常。
    \item \textbf{严重毒性带：}摄入量远超UL，出现明显的中毒症状和靶器官损害。
\end{itemize}
不同矿物质的"适宜带"宽度差异很大——如钙有较宽的安全范围，而氟的适宜带极窄（生理剂量与中毒剂量相近）。

\textbf{矿物质的相互作用（了解）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{协同作用：}钙与磷共同构成骨骼（钙磷比1:1$\sim$1.5），铁与铜协同参与造血。
    \item \textbf{拮抗作用（竞争吸收）：}钙抑制铁、锌的吸收；锌抑制铜的吸收；铁与锌相互抑制；植酸同时螯合多种二价阳离子（Ca$^{2+}$、Fe$^{2+}$、Zn$^{2+}$、Mg$^{2+}$）。
    \item \textbf{间接影响：}镁缺乏可导致低钾血症和低钙血症；钠摄入增加则尿钙排出增加。
\end{itemize}
\end{tipbox}

\section{钙（Calcium）}

\begin{keybox}
\textbf{钙的生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 构成骨骼和牙齿的成分
    \item 维持神经肌肉活动
    \item 促进细胞信号传递
    \item 维持细胞膜稳定性
    \item 调节机体酶活性
    \item 血液凝固
\end{enumerate}

\textbf{影响钙吸收的因素：}
\begin{itemize}[leftmargin=*]
    \item \textbf{机体因素：}随年龄$\uparrow$吸收率$\downarrow$。
    \item \textbf{膳食因素：}
    \begin{itemize}[leftmargin=*]
        \item 难溶性钙盐（植物性食物、膳食纤维、未被消化的脂肪酸、咖啡因和酒精）不易吸收
        \item 可溶性钙盐（蛋白质和乳糖）易吸收
    \end{itemize}
\end{itemize}

\textbf{影响钙排泄的因素：}
\begin{itemize}[leftmargin=*]
    \item \textbf{机体因素：}血钙浓度低尿钙减少，血钙浓度高尿钙增加；年龄$\uparrow$尿钙$\uparrow$；补液、酸中毒、甲状腺素和肾上腺皮质激素$\rightarrow$尿钙$\uparrow$。
    \item \textbf{膳食因素：}钠、蛋白质、咖啡因$\uparrow \rightarrow$尿钙$\uparrow$。
\end{itemize}

\textbf{缺乏：}
\begin{itemize}[leftmargin=*]
    \item \imp{婴幼儿和儿童：}软骨病（生长发育迟缓、骨软化、骨骼变形）；佝偻病（O形或X形腿、肋骨串珠、鸡胸）；龋齿。
    \item \imp{中老年人：}骨质疏松症。
\end{itemize}

\textbf{过量：}高钙血症、高钙尿、血管和软组织钙化、肾结石等。

\textbf{参考摄入量：}800mg/d。

\textbf{食物来源：}乳制品、豆制品。
\end{keybox}

\section{铁（Iron）}

\begin{keybox}
\textbf{铁的体内分布：}70\%在血红蛋白，30\%为贮存铁。

\textbf{影响铁吸收的因素：}
\begin{itemize}[leftmargin=*]
    \item \textbf{促进因素：}血红素铁（Fe$^{2+}$）直接被吸收；非血红素铁（Fe$^{3+}$）需还原为Fe$^{2+}$；维生素C、含巯基氨基酸；胃内酸度（pH）。
    \item \textbf{抑制因素：}植酸、鞣酸。
\end{itemize}

\textbf{缺乏：}缺铁性贫血（小细胞低色素性贫血）。

\textbf{参考摄入量：}男12mg/d，女20mg/d。
\end{keybox}

\section{锌（Zinc）}

\begin{keybox}
\textbf{锌的生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 酶的组成成分
    \item 促进生长发育
    \item 促进性发育
    \item 维持正常味觉和食欲
\end{enumerate}

\textbf{缺乏：}生长发育迟缓、性发育不全、脑发育受损、食欲不振、异食癖。

\textbf{参考摄入量：}男12.5mg/d，女7.5mg/d。
\end{keybox}

\section{碘（Iodine）}

\begin{keybox}
\textbf{碘的生理功能：}参与甲状腺激素合成。

\textbf{缺乏：}
\begin{itemize}[leftmargin=*]
    \item 地方性甲状腺肿
    \item 克汀病（呆小症）
\end{itemize}

\textbf{参考摄入量：}120$\mu$g/d。
\end{keybox}

\section{硒（Selenium）}

\begin{keybox}
\textbf{硒的生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 谷胱甘肽过氧化物酶的组成成分
    \item 抗氧化
    \item 参与甲状腺激素代谢
\end{enumerate}

\textbf{缺乏：}克山病、大骨节病。

\textbf{参考摄入量：}60$\mu$g/d。
\end{keybox}

\section{常量元素（除钙外）}

\begin{defbox}
\textbf{磷（Phosphorus）：}
\begin{itemize}[leftmargin=*]
    \item 常量元素，与钙共同构成骨骼和牙齿（羟磷灰石）
    \item 参与核酸、磷脂、含磷辅酶（NAD$^+$、NADP$^+$、TPP等）的组成
    \item 参与能量代谢（ATP）；维持酸碱平衡
    \item \textbf{吸收利用：}主要在空肠吸收，活性维生素D促进磷吸收；植酸抑制吸收
    \item 钙磷比（Ca:P）：成人适宜1:1$\sim$1:1.5，婴儿母乳2:1（最适宜）
    \item 缺乏：罕见（食物中含量丰富）；严重缺乏可致低磷血症（肌无力、骨软化）
    \item 过量：高磷血症可抑制钙吸收，加重肾性骨营养不良
    \item 来源：瘦肉、鱼、蛋、奶、豆类、坚果
\end{itemize}

\textbf{硫（Sulfur）：}
\begin{itemize}[leftmargin=*]
    \item 常量元素，主要以含硫氨基酸（蛋氨酸、半胱氨酸）形式存在于蛋白质中
    \item 硫胺素（VitB$_1$）、生物素、辅酶A等均含硫，参与多种代谢过程
    \item 含硫氨基酸代谢产生的硫酸根参与解毒（与酚类、甾体激素结合后从尿排出）
    \item 硫酸软骨素是骨骼、软骨、肌腱和血管壁的重要组成成分
    \item 缺乏：罕见（与蛋白质缺乏共存）；过量：含硫氨基酸摄入过多会加速骨骼中钙流失，增加骨质疏松风险
    \item 来源：富含蛋白质的食物——肉类、蛋类、奶类、豆类、鱼等
\end{itemize}

\textbf{镁（Magnesium）：}
\begin{itemize}[leftmargin=*]
    \item 常量元素，成人含镁约25g，60\%$\sim$65\%在骨骼中，27\%在肌肉
    \item 300多种酶的辅因子；参与蛋白质合成和能量代谢
    \item 对离子通道的作用：镁可封闭不同钾通道的外向型电流，阻止钾外流；镁作为钙阻断剂抑制钙通道
    \item 维持神经肌肉兴奋性；促进骨骼生长；心血管保护作用
    \item 影响胃肠道功能：硫酸镁溶液可促进胆囊排空，碱性镁盐可中和胃酸；镁离子在肠道吸收缓慢，促使水分滞留
    \item 调节激素作用：直接影响甲状旁腺激素的分泌
    \item \textbf{缺乏：}神经肌肉兴奋亢进——肌肉震颤、手足抽搐、反射亢进、共济失调，严重时出现谵妄、精神错乱甚至惊厥和昏迷；可出现低钾血症、低钙血症和心脑血管疾病；增加2型糖尿病、代谢综合征风险
    \item \textbf{过量：}腹泻、恶心、胃肠痉挛；重者可出现嗜睡、肌无力、膝腱反射弱、肌麻痹等
    \item 来源：绿叶蔬菜（叶绿素含镁）、大麦、黑米、荞麦、麸皮、苋菜、口蘑、木耳、香菇。RNI：18-29岁330mg/d，30-64岁320mg/d
\end{itemize}

\textbf{钾（Potassium）：}
\begin{itemize}[leftmargin=*]
    \item 常量元素，细胞内液主要阳离子
    \item 参与糖和蛋白质代谢；维持细胞渗透压和酸碱平衡
    \item 维持神经肌肉应激性和心肌正常功能
    \item 降低血压的正常功能
    \item 缺乏（低钾血症）：当失钾超过10\%时表现为肌肉无力、瘫痪、心律失常、横纹肌溶解综合征、肾脏功能障碍（多尿、夜尿、多饮）
    \item 过量（高钾血症）：极度疲乏和四肢无力（下肢较重），严重可因呼吸麻痹猝死；心率缓慢、心律失常
    \item 来源：紫菜、菠萝、土豆、香蕉、牛油果、哈密瓜、黄豆。成人AI：2000mg/d
\end{itemize}

\textbf{钠（Sodium）：}
\begin{itemize}[leftmargin=*]
    \item 常量元素，细胞外液主要阳离子
    \item 调节细胞外液容量与渗透压；维持酸碱平衡和正常血压
    \item 其他功能：维持神经肌肉应激性
    \item 缺乏（低钠血症）：影响细胞对氨基酸和葡萄糖的吸收
    \item 过量（高钠血症）：水肿、体重增加、血容量增加；急性水肿、血压上升、脂肪清除率下降及胃粘膜上皮细胞破裂
    \item 来源：精盐、味精、调味品、腊肉咸菜。成人AI：1500mg/d
\end{itemize}
\end{defbox}

\section{其他微量元素}

\begin{tipbox}
\textbf{铜（Copper）：}
\begin{itemize}[leftmargin=*]
    \item 必需微量元素
    \item 参与铁代谢（铜蓝蛋白促进铁的吸收和利用）
    \item 参与结缔组织中胶原蛋白和弹性蛋白的交联（赖氨酰氧化酶）
    \item 参与黑色素合成（酪氨酸酶）
    \item 缺乏：小细胞低色素性贫血（与铁代谢障碍有关）、中性粒细胞减少、骨质疏松
    \item 来源：牡蛎、贝类、坚果、全谷物、动物肝脏
\end{itemize}

\textbf{铬（Chromium）：}
\begin{itemize}[leftmargin=*]
    \item 必需微量元素，三价铬是葡萄糖耐量因子（GTF）的组成成分
    \item 增强胰岛素作用，促进葡萄糖利用
    \item 缺乏：葡萄糖耐量降低、胰岛素抵抗
    \item 来源：全谷物、肉类、啤酒酵母。成人AI：30$\mu$g/d
\end{itemize}

\textbf{氟（Fluoride）：}
\begin{itemize}[leftmargin=*]
    \item 潜在毒性微量元素（生理剂量与中毒剂量范围极窄）
    \item 促进骨骼和牙齿的矿化，预防龋齿
    \item 缺乏：龋齿发病率增高
    \item 过量：氟斑牙和氟骨症
    \item 来源：饮用水（主要来源）、海产品、茶叶
\end{itemize}

\textbf{锰（Manganese）：}
\begin{itemize}[leftmargin=*]
    \item 可能必需微量元素，是多种酶的组成成分或激活剂（如锰超氧化物歧化酶Mn-SOD、丙酮酸羧化酶）
    \item 参与骨骼发育和生长发育；参与糖类和脂类代谢
    \item 缺乏：罕见（食物中广泛存在）；可能与生长发育迟缓、骨骼异常、糖耐量降低有关
    \item 过量：主要见于长期职业性锰暴露，可致神经系统损害（类似帕金森综合征）
    \item 来源：坚果、全谷物、茶叶、绿叶蔬菜
\end{itemize}

\textbf{钼（Molybdenum）：}
\begin{itemize}[leftmargin=*]
    \item 必需微量元素，是黄嘌呤氧化酶、醛氧化酶、亚硫酸氧化酶的组成成分
    \item 参与嘌呤代谢和含硫氨基酸代谢
    \item 缺乏：极为罕见（食物中广泛存在）
    \item 来源：奶类、豆类、动物肝脏、全谷物
\end{itemize}

\textbf{硅（Silicon）：}
\begin{itemize}[leftmargin=*]
    \item 可能必需微量元素，参与骨基质和结缔组织的形成（胶原和糖胺聚糖的合成）
    \item 缺乏：动物实验中可见骨骼和结缔组织异常
    \item 来源：全谷物（尤其燕麦、大麦）、啤酒、根茎类蔬菜
\end{itemize}

\textbf{硼（Boron）：}
\begin{itemize}[leftmargin=*]
    \item 可能必需微量元素，可能参与钙镁代谢和雌激素代谢的调节
    \item 缺乏：证据有限，可能与钙代谢紊乱有关
    \item 来源：水果（尤其是苹果、梨、葡萄）、蔬菜、坚果、豆类
\end{itemize}

\textbf{镍（Nickel）：}
\begin{itemize}[leftmargin=*]
    \item 可能必需微量元素，可能参与铁吸收和代谢调节
    \item 缺乏：在动物实验中可见生长发育迟缓和铁代谢异常；人体缺乏极为罕见
    \item 过量：主要问题是皮肤过敏（接触性皮炎）；经口镍盐摄入过多可致胃肠道反应
    \item 来源：坚果、豆类、全谷物、巧克力
\end{itemize}
\end{tipbox}

\begin{defbox}
\textbf{钴（Cobalt）：}
\begin{itemize}[leftmargin=*]
    \item 必需微量元素，维生素B$_{12}$的组成成分（唯一含金属元素的维生素）
    \item 通过VitB$_{12}$发挥生理作用
    \item 缺乏即VitB$_{12}$缺乏——巨幼红细胞性贫血、神经系统损害、高同型半胱氨酸血症
\end{itemize}
\end{defbox}

\subsection*{参考答案要点}
\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item \textbf{矿物质：}人体组织中除碳、氢、氧和氮主要以有机化合物形式存在外的其余元素，统称为矿物质（无机盐/灰分）。\textbf{常量元素：}体内含量大于体重0.01\%的矿物质，包括Ca、P、Na、K、S、Cl、Mg，每日摄入量>100mg。\textbf{微量元素：}体内含量小于体重0.01\%的矿物质。\textbf{必需微量元素：}铁、钴、钼、铜、锌、硒、碘、铬。\textbf{可能必需微量元素：}锰、硅、镍、硼、钒。\textbf{潜在毒性微量元素：}氟、铅、镉、汞、砷、铝、锡、锂。\textbf{生理作用剂量带：}从缺乏到中毒，按剂量分为严重缺乏带、边缘缺乏带、适宜摄入带、边缘毒性带、严重毒性带，呈U形或J形剂量-效应曲线，不同矿物质安全范围宽度不同（如氟极窄、钙较宽）。
    \item \textbf{分类：}常量元素（>0.01\%，Ca/P/Na/K/S/Cl/Mg）+ 微量元素（<0.01\%，必需Fe/Co/Mo/Cu/Zn/Se/I/Cr；可能必需Mn/Si/Ni/B/V；潜在毒性F/Pb/Cd/Hg/As/Al/Sn/Li；功能未知Sr/Rb/稀土元素）。\textbf{生理功能：}(1)构成机体组织的重要原料；(2)维持机体酸碱平衡和渗透压（细胞外液：Na$^+$/Cl$^-$；细胞内液：K$^+$/HPO$_4^{2-}$）；(3)维持神经肌肉兴奋性；(4)参与构成激素、维生素、蛋白质和多种酶。
    \item 五个剂量带：严重缺乏带（出现典型临床症状）、边缘缺乏带（无明显症状但功能降低）、适宜摄入带（满足需要不产生不良反应）、边缘毒性带（亚临床功能异常）、严重毒性带（明显中毒症状）。剂量-效应曲线呈U形或J形，不同矿物质适宜带宽度差异大——钙较宽而氟极窄。
    \item \textbf{协同作用：}钙磷共构骨骼；铁铜协同参与造血。\textbf{拮抗作用：}钙抑制铁锌吸收；锌抑制铜吸收；铁锌互抑；植酸螯合多种二价阳离子。\textbf{间接影响：}镁缺乏可致低钾低钙；钠摄入增加尿钙排出增加。
    \item \textbf{钙的生理功能：}(1)构成骨骼和牙齿的成分；(2)维持神经肌肉活动；(3)促进细胞信号传递；(4)维持细胞膜稳定性；(5)调节机体酶活性；(6)血液凝固。\textbf{影响吸收因素：}机体因素——随年龄$\uparrow$吸收率$\downarrow$。膳食因素——难溶性钙盐（植物性食物、膳食纤维、未被消化的脂肪酸、咖啡因和酒精）不易吸收；可溶性钙盐（蛋白质和乳糖）易吸收。\textbf{影响排泄因素：}机体因素——血钙浓度低尿钙减少，血钙浓度高尿钙增加；年龄$\uparrow$尿钙$\uparrow$；补液、酸中毒、甲状腺素和肾上腺皮质激素$\rightarrow$尿钙$\uparrow$。膳食因素——钠、蛋白质、咖啡因$\uparrow \rightarrow$尿钙$\uparrow$。\textbf{缺乏：}婴幼儿和儿童——软骨病、佝偻病（O形或X形腿、肋骨串珠、鸡胸）、龋齿；中老年人——骨质疏松症。\textbf{过量：}高钙血症、高钙尿、血管和软组织钙化、肾结石等。RNI 800mg/d。来源：乳制品、豆制品。
    \item \textbf{铁的体内分布：}70\%在血红蛋白，30\%为贮存铁。\textbf{促进吸收因素：}血红素铁（Fe$^{2+}$）直接被吸收；非血红素铁（Fe$^{3+}$）需还原为Fe$^{2+}$；维生素C、含巯基氨基酸；胃内酸度（pH）。\textbf{抑制因素：}植酸、鞣酸。\textbf{缺乏：}缺铁性贫血（小细胞低色素性贫血）。RNI：男12mg/d，女20mg/d。
    \item \textbf{锌的生理功能：}(1)酶的组成成分；(2)促进生长发育；(3)促进性发育；(4)维持正常味觉和食欲。\textbf{缺乏：}生长发育迟缓、性发育不全、脑发育受损、食欲不振、异食癖。RNI：男12.5mg/d，女7.5mg/d。
    \item \textbf{碘的生理功能：}参与甲状腺激素合成。\textbf{缺乏：}地方性甲状腺肿；克汀病（呆小症）。RNI：120$\mu$g/d。
    \item \textbf{硒的生理功能：}(1)谷胱甘肽过氧化物酶的组成成分；(2)抗氧化；(3)参与甲状腺激素代谢。\textbf{缺乏：}克山病、大骨节病。RNI：60$\mu$g/d。
    \item \textbf{磷：}常量元素，构成骨骼牙齿（羟磷灰石），参与核酸/磷脂/含磷辅酶/能量代谢（ATP），钙磷比成人1:1$\sim$1:1.5、婴儿母乳2:1。\textbf{镁：}常量元素，60\%$\sim$65\%在骨骼，300多种酶辅因子，缺乏致神经肌肉兴奋亢进/手足抽搐/低钾低钙血症/心律失常。\textbf{钾：}常量元素，细胞内液主要阳离子，参与糖和蛋白质代谢，缺乏致肌无力/瘫痪/心律失常，过量致疲乏无力/呼吸麻痹猝死（AI 2000mg/d）。\textbf{钠：}常量元素，细胞外液主要阳离子，调节容量与渗透压，缺乏影响氨基酸和葡萄糖吸收，过量致水肿/血容量增加/血压上升（AI 1500mg/d）。\textbf{硫：}常量元素，主要以含硫氨基酸形式存在，参与解毒和软骨/肌腱组成成分，过量含硫氨基酸摄入加速骨钙流失。
    \item \textbf{铜：}必需微量元素，参与铁代谢（铜蓝蛋白）/胶原交联/黑色素合成，缺乏致小细胞低色素性贫血。\textbf{铬：}必需微量元素，三价铬为GTF成分增强胰岛素作用（AI 30$\mu$g/d）。\textbf{氟：}潜在毒性微量元素，促进骨骼牙齿矿化预防龋齿，过量致氟斑牙/氟骨症。\textbf{钴：}必需微量元素，VitB$_{12}$组成成分，缺乏致巨幼红细胞性贫血。\textbf{锰：}可能必需微量元素，是Mn-SOD等的组成成分，参与骨骼发育和糖脂代谢。\textbf{钼：}必需微量元素，黄嘌呤氧化酶等组成成分，参与嘌呤和含硫氨基酸代谢。
\end{enumerate}
