% ==================== 第一节 矿物质概述 ====================
\section{矿物质概述}

\begin{defbox}
\textbf{矿物质（Minerals）}是存在于体内除碳、氢、氧、氮以外的其他元素，是构成人体组织和维持正常生理功能所必需的一类无机营养素。人体内不能合成，必须由食物和水供给。
\end{defbox}

\subsection{矿物质的分类}

\begin{keybox}
\textbf{按体内含量分类：}

\textbf{常量元素（Macroelements）：}体内含量大于体重 0.01\%，每日需要量在100mg以上。
\begin{itemize}[leftmargin=*]
    \item \keyword{钙（Ca）}、\keyword{磷（P）}、\keyword{钠（Na）}、\keyword{钾（K）}、\keyword{氯（Cl）}、\keyword{镁（Mg）}、\keyword{硫（S）}
    \item 常量元素占体内矿物质总量的约 99.9\%
\end{itemize}

\textbf{微量元素（Microelements / Trace Elements）：}体内含量小于体重 0.01\%，每日需要量在100mg以下。
\begin{itemize}[leftmargin=*]
    \item \textbf{必需微量元素：}\keyword{铁（Fe）}、铜（Cu）、\keyword{锌（Zn）}、\keyword{硒（Se）}、\keyword{铬（Cr）}、\keyword{碘（I）}、\keyword{钴（Co）}、\keyword{钼（Mo）}
    \item \textbf{可能必需微量元素：}锰（Mn）、硅（Si）、镍（Ni）、硼（B）、钒（V）
    \item \textbf{潜在毒性微量元素（低剂量可能必需）：}氟（F）、铅（Pb）、镉（Cd）、汞（Hg）、砷（As）、铝（Al）、锡（Sn）、锂（Li）
\end{itemize}
\end{keybox}

\subsection{矿物质的特点}

\begin{keybox}
\textbf{矿物质的五大共同特点：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{不能在体内合成，必须由食物和饮水供给}
    \item \textbf{矿物质可以通过水进入人体}，饮水也是矿物质的重要来源之一
    \item \textbf{在体内分布极不均匀}：如99\%的钙在骨骼和牙齿，70\%～80\%的碘在甲状腺
    \item \textbf{矿物质之间存在协同或拮抗作用}：如钙与磷的协同、铁与锌的吸收竞争
    \item \textbf{生理安全范围较窄}：摄入不足导致缺乏，摄入过量造成中毒，尤其是微量元素
\end{enumerate}
\end{keybox}

% ==================== 第二节 钙 ====================
\section{钙（Calcium, Ca）}

\subsection{体内分布}

\begin{keybox}
人体总钙量约1000～1200g，是体内含量最多的矿物质元素。其中：
\begin{itemize}[leftmargin=*]
    \item \imp{99\% 存在于骨骼和牙齿}，以羟磷灰石 $[\text{Ca}_{10}(\text{PO}_4)_6(\text{OH})_2]$ 形式存在
    \item 1\% 存在于软组织、细胞外液和血液中
\end{itemize}

\textbf{血钙存在形式（血液总钙约2.25～2.75 mmol/L）：}
\begin{enumerate}[leftmargin=*]
    \item \keyword{蛋白结合钙：约 46\%}（主要与白蛋白结合，无生理活性）
    \item \keyword{复合钙：约 6.5\%}（与柠檬酸、磷酸等结合的扩散性钙）
    \item \keyword{离子化钙（Ca$^{2+}$）：约 47.5\%}（\imp{具有生理活性的形式}，受PTH、降钙素和VitD的精密调控）
\end{enumerate}
\end{keybox}

\subsection{生理功能}

\begin{keybox}
\textbf{钙的生理功能*：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{构成骨骼和牙齿：}以羟磷灰石形式提供结构支撑
    \item \textbf{维持神经肌肉的正常兴奋性：}Ca$^{2+}$稳定神经细胞膜，降低神经肌肉兴奋性；血清Ca$^{2+}$降低时→神经肌肉兴奋性增高→手足抽搐
    \item \textbf{参与细胞信号转导：}作为第二信使参与细胞内信号传递过程
    \item \textbf{参与血液凝固：}凝血因子IV，参与凝血酶原激活为凝血酶的过程
    \item \textbf{调节多种酶的活性：}激活多种酶（如ATP酶、脂肪酶等）
    \item \textbf{维持细胞膜的稳定性与通透性}
\end{enumerate}
\end{keybox}

\subsection{钙的吸收与影响吸收的因素}

\begin{keybox}
\textbf{钙的吸收特点及影响因素*：}

\textbf{吸收途径：}
\begin{itemize}[leftmargin=*]
    \item 主要在小肠上段（十二指肠和空肠）吸收
    \item \imp{主动转运（活性维生素D依赖性）}是主要吸收途径——受1,25-(OH)$_2$D$_3$调控
    \item 旁细胞途径的被动扩散在高钙摄入时参与吸收
\end{itemize}

\textbf{促进钙吸收的因素：}
\begin{enumerate}[leftmargin=*]
    \item \keyword{维生素D（1,25-(OH)$_2$D$_3$）}：诱导钙结合蛋白（CaBP）的合成，促进主动转运（最重要的促进因素）
    \item 乳糖：在肠道内经细菌发酵产酸，降低pH，增加钙溶解度
    \item 氨基酸（赖氨酸、精氨酸等）：与钙形成可溶性螯合物
    \item 适宜的钙磷比（Ca:P约1:1～1.5:1）
\end{enumerate}

\textbf{抑制钙吸收的因素：}
\begin{enumerate}[leftmargin=*]
    \item \keyword{植酸（phytic acid）}：谷类的外壳中含量高，与钙形成不溶性植酸钙
    \item \keyword{草酸}：菠菜、苋菜、竹笋中含丰富草酸，与钙形成草酸钙沉淀
    \item 膳食纤维：过多摄入干扰钙吸收
    \item 脂肪消化不良：脂肪酸与钙形成钙皂，随粪便排出
    \item \textbf{年龄：}随着年龄增长，钙的吸收率下降
\end{enumerate}
\end{keybox}

\subsection{钙缺乏与过量}

\begin{keybox}
\textbf{钙缺乏*：}
\begin{itemize}[leftmargin=*]
    \item \textbf{儿童：}佝偻病（Rickets）——骨骼钙化不良，骨骼变形（O形腿、X形腿、鸡胸、方颅）
    \item \textbf{成年人：}骨质软化症（Osteomalacia）
    \item \textbf{老年人：}骨质疏松症（Osteoporosis）——骨量减少、骨微结构破坏、骨折风险增加
    \item \textbf{手足抽搐：}低血钙引起神经肌肉兴奋性增高（Chvostek征阳性）
\end{itemize}
\end{keybox}

\subsection{参考摄入量与食物来源}

\begin{keybox}
\textbf{钙的膳食参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成人\imp{AI：800 mg/d}
    \item 50岁以上\imp{AI：1000 mg/d}
    \item 孕妇中期、晚期：1000 mg/d
    \item 乳母：1200 mg/d
    \item UL（可耐受最高摄入量）：2000 mg/d
\end{itemize}
\end{keybox}

\begin{exambox}
\textbf{钙的食物来源（常考含量对比，单位 mg/100g）：}
\begin{itemize}[leftmargin=*]
    \item \imp{虾皮 991}、\imp{黑芝麻 780}、奶酪 590、大豆 191
    \item 牛奶 115（吸收率高，是钙的最佳食物来源）
    \item 蔬菜中苋菜、油菜钙含量较高，但草酸影响吸收
\end{itemize}
注意：虾皮虽钙含量高，但不易消化吸收；\imp{牛奶及其制品是钙理想的膳食来源}，钙磷比例适宜，吸收率高。
\end{exambox}

% ==================== 第三节 铁 ====================
\section{铁（Iron, Fe）}

\subsection{概述与体内分布}

\begin{defbox}
铁是人体内含量最多的必需微量元素，成人体内总铁量约4～5g。
\end{defbox}

\begin{keybox}
\textbf{体内铁的分布与存在形式：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{功能性铁（65\%～70\%）：}
    \begin{itemize}[leftmargin=*]
        \item \keyword{血红蛋白（Hemoglobin）}：约占体内总铁的60\%～75\%（最主要的铁存在形式）
        \item \keyword{肌红蛋白（Myoglobin）}：约占3\%～5\%
        \item 含血红素酶类（细胞色素、过氧化氢酶、过氧化物酶等）：约1\%
    \end{itemize}
    \item \textbf{储存铁（25\%～30\%）：}
    \begin{itemize}[leftmargin=*]
        \item \keyword{铁蛋白（Ferritin）}：水溶性，主要储存于肝、脾、骨髓
        \item \keyword{含铁血黄素（Hemosiderin）}：不溶性，铁负荷过量时显著增加
    \end{itemize}
    \item \textbf{转运铁：}运铁蛋白（Transferrin），约占0.1\%
\end{enumerate}
\end{keybox}

\subsection{生理功能}

\begin{keybox}
\textbf{铁的生理功能*：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{参与体内氧的运输和组织呼吸：}血红蛋白携带O$_2$、CO$_2$；肌红蛋白储存肌肉中的O$_2$；细胞色素参与电子传递链
    \item \textbf{参与造血功能：}维持正常的红细胞生成
    \item \textbf{参与免疫功能：}维持淋巴器官的正常发育、T淋巴细胞的功能和中性粒细胞的杀菌活性
    \item \textbf{促进$\beta$-胡萝卜素转化为维生素A：}$\beta$-胡萝卜素-15,15'-加氧酶为含铁酶
\end{enumerate}
\end{keybox}

\subsection{铁的吸收}

\begin{keybox}
\textbf{铁的吸收特点*：}

\textbf{两种铁的吸收：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{血红素铁（Heme Iron）：}
    \begin{itemize}[leftmargin=*]
        \item 存在形式：Fe$^{2+}$，主要来源于\keyword{动物性食物}（红肉、肝脏、血制品）
        \item 吸收机制：直接被肠上皮细胞以完整血红素形式吸收，不受膳食抑制因素影响
        \item \imp{吸收率高}（约15\%～25\%）
    \end{itemize}
    \item \textbf{非血红素铁（Non-heme Iron）：}
    \begin{itemize}[leftmargin=*]
        \item 存在形式：Fe$^{3+}$（三价铁），主要来源于\keyword{植物性食物}（谷类、蔬菜）
        \item 进入肠上皮细胞前必须先被还原为Fe$^{2+}$
        \item \imp{吸收率低}（约2\%～5\%），受膳食因素显著影响
    \end{itemize}
\end{enumerate}

\textbf{促进非血红素铁吸收的因素：}
\begin{itemize}[leftmargin=*]
    \item \keyword{维生素C（VitC）}：将Fe$^{3+}$还原为Fe$^{2+}$，同时形成可溶螯合物——\imp{最重要的促进因素}
    \item 有机酸（柠檬酸、乳酸）
    \item \keyword{"肉类因子"（Meat Factor）}：肉、禽、鱼中的半胱氨酸等含硫氨基酸促进铁吸收
\end{itemize}

\textbf{抑制非血红素铁吸收的因素：}
\begin{itemize}[leftmargin=*]
    \item 鞣酸（茶多酚）——茶、咖啡
    \item 植酸——谷类、豆类
    \item 草酸——菠菜、甜菜
    \item 多酚类物质
    \item 服用抗酸药物（降低胃酸）
\end{itemize}
\end{keybox}

\subsection{铁缺乏}

\begin{keybox}
\textbf{铁缺乏的三阶段（由轻到重）*：}

\begin{enumerate}[leftmargin=*]
    \item \textbf{铁减少期（Iron Depletion, ID）：}
    \begin{itemize}[leftmargin=*]
        \item \imp{储存铁耗竭}（铁蛋白、含铁血黄素减少）
        \item 血清铁蛋白（SF）下降
        \item 尚无明显临床症状，血红蛋白正常
    \end{itemize}
    \item \textbf{红细胞生成缺铁期（Iron-Deficient Erythropoiesis, DE）：}
    \begin{itemize}[leftmargin=*]
        \item \imp{血清铁降低，运铁蛋白饱和度下降}
        \item 红细胞游离原卟啉（FEP）升高
        \item 血红蛋白可能尚在正常范围
    \end{itemize}
    \item \textbf{缺铁性贫血期（Iron Deficiency Anemia, IDA）：}
    \begin{itemize}[leftmargin=*]
        \item \imp{血红蛋白（Hb）明显降低}：男性 < 130g/L，女性 < 120g/L
        \item 呈小细胞低色素性贫血（MCV$\downarrow$、MCHC$\downarrow$）
        \item 临床表现：面色苍白、乏力、头晕、心悸、抵抗力下降、异食癖
    \end{itemize}
\end{enumerate}

\imp{核心考点：三个阶段依次递进，ID→DE→IDA}
\end{keybox}

\subsection{参考摄入量与食物来源}

\begin{keybox}
\textbf{铁的膳食参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成年男性 AI：\imp{15 mg/d}
    \item 成年女性 AI：\imp{20 mg/d}
    \item 孕妇（中期、晚期）：\imp{25～35 mg/d}
    \item 乳母：25 mg/d
\end{itemize}
\end{keybox}

\textbf{食物来源：}动物肝脏、动物全血、红肉（含血红素铁，吸收率高）；黑木耳、芝麻酱、蛋黄（含非血红素铁）。注意：蛋黄虽铁含量高但吸收率仅3\%（卵黄高磷蛋白干扰）。

% ==================== 第四节 锌 ====================
\section{锌（Zinc, Zn）}

\subsection{生理功能}

\begin{keybox}
\textbf{锌的生理功能*：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{金属酶的组成成分和激活剂：}体内约200多种含锌金属酶，如碱性磷酸酶（ALP）、碳酸酐酶、RNA/DNA聚合酶等
    \item \textbf{促进生长发育和组织再生：}参与蛋白质、核酸的合成，对细胞分裂、生长至关重要
    \item \textbf{维持免疫功能：}维持胸腺和T淋巴细胞功能
    \item \textbf{维持细胞膜结构与功能：}抗氧化保护作用
    \item \textbf{味觉功能：}参与构成\keyword{味觉素（Gustin）}，对味蕾功能有重要作用
    \item \textbf{维持皮肤健康和正常视觉：}参与VitA的代谢与转运（视黄醇结合蛋白含锌）
\end{enumerate}
\end{keybox}

\subsection{锌缺乏}

\begin{keybox}
\textbf{锌缺乏的临床表现*：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{生长发育迟缓：}最突出的表现，尤其影响儿童和青少年（缺锌性侏儒）
    \item \textbf{性成熟障碍、性功能减退}
    \item \textbf{味觉障碍、嗅觉减退：}味觉素（gustin）合成减少
    \item \textbf{伤口愈合不良：}细胞增殖和胶原合成受阻
    \item \textbf{免疫功能下降：}易感染
    \item \textbf{皮肤粗糙、鳞屑样皮炎：}类似于肠病性肢体皮炎（Acrodermatitis Enteropathica）
\end{enumerate}
\end{keybox}

\subsection{参考摄入量与食物来源}

\begin{keybox}
\textbf{锌的膳食参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成年男性 RNI：\imp{15.0 mg/d}
    \item 成年女性 RNI：\imp{11.5 mg/d}
    \item 孕妇：16.5 mg/d
    \item 乳母：21.5 mg/d
\end{itemize}
\end{keybox}

\begin{exambox}
\textbf{锌的食物来源（常考含量对比，单位 mg/100g）：}
\begin{itemize}[leftmargin=*]
    \item \imp{生蚝（牡蛎）：71.20}（含锌量最高的食物！）
    \item \imp{小麦胚粉：23.40}
    \item 山核桃：12.59
    \item 扇贝：11.69
    \item 动物肝脏、瘦红肉也是良好的锌来源
\end{itemize}
动物性食物中的锌吸收率高于植物性食物。
\end{exambox}

% ==================== 第五节 硒 ====================
\section{硒（Selenium, Se）}

\subsection{生理功能}

\begin{keybox}
\textbf{硒的生理功能*：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{抗氧化功能（核心功能）：}硒是\imp{谷胱甘肽过氧化物酶（GSH-Px）}的必需组成成分。GSH-Px将有毒的过氧化物（H$_2$O$_2$、脂质过氧化物）还原为无毒的羟基化合物，保护细胞膜和生物大分子免受氧化损伤
    \item \textbf{保护心血管健康：}缺硒地区心血管疾病发病率增高
    \item \textbf{增强免疫功能：}促进淋巴细胞增殖和抗体生成
    \item \textbf{重金属解毒作用：}硒与重金属（Hg、Cd、Pb等）形成不溶性硒化物，降低其毒性
    \item \textbf{甲状腺激素代谢：}碘甲腺原氨酸脱碘酶（ID）为含硒酶，参与T$_4$→T$_3$的转化
\end{enumerate}
\end{keybox}

\subsection{硒缺乏与相关疾病}

\begin{keybox}
\textbf{硒缺乏相关疾病：}

\begin{enumerate}[leftmargin=*]
    \item \textbf{克山病（Keshan Disease）：}
    \begin{itemize}[leftmargin=*]
        \item 一种以心肌多发性坏死为主要病变的\keyword{地方性心肌病}
        \item 好发于低硒地区的儿童和育龄妇女
        \item 补充硒可有效预防克山病的发生
        \item 命名来源：1935年最早在黑龙江省克山县发现
    \end{itemize}
    \item \textbf{大骨节病（Kashin-Beck Disease）：}
    \begin{itemize}[leftmargin=*]
        \item 一种地方性骨关节病，多发于\imp{青少年}
        \item 硒缺乏损伤机体的抗氧化系统，导致软骨细胞坏死和关节变形
        \item 主要表现为关节增粗、短指畸形、身材矮小
    \end{itemize}
\end{enumerate}
\end{keybox}

\subsection{参考摄入量与食物来源}

\begin{keybox}
\textbf{硒的膳食参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成人 RNI：\imp{50 μg/d}
    \item 孕妇：50 μg/d；乳母：65 μg/d
    \item UL（可耐受最高摄入量）：\imp{400 μg/d}（安全范围较窄！）
\end{itemize}
\end{keybox}

\textbf{食物来源：}动物内脏、海产品（鱼、虾、贝类）、肉类、蛋类。植物性食物含硒量受产地土壤硒含量影响极大。

% ==================== 第六节 碘 ====================
\section{碘（Iodine, I）}

\subsection{体内分布与生理功能}

\begin{defbox}
成人体内总碘量约20～50mg，其中\imp{70\%～80\%集中在甲状腺}。
\end{defbox}

\begin{keybox}
\textbf{碘的生理功能*：}

碘的生理功能主要通过甲状腺激素（T$_3$和T$_4$）实现：
\begin{enumerate}[leftmargin=*]
    \item \textbf{调节能量代谢：}促进体内物质分解代谢，增加产热（加速氧化磷酸化）
    \item \textbf{促进蛋白质合成：}调节生长发育期蛋白质的合成
    \item \textbf{促进神经系统发育：}胎儿期和婴幼儿期甲状腺激素对脑发育至关重要
    \item \textbf{调节基础代谢率：}影响脂肪、碳水化合物和蛋白质的代谢
\end{enumerate}
\end{keybox}

\subsection{碘缺乏与过量}

\begin{keybox}
\textbf{碘缺乏*：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{地方性甲状腺肿（Goiter）：}最常见表现，甲状腺代偿性增生肥大
    \item \textbf{\imp{呆小症（克汀病，Cretinism）}：}母孕期严重缺碘导致胎儿脑发育障碍
    \begin{itemize}[leftmargin=*]
        \item 永久性智力低下、精神发育迟滞
        \item 聋哑、语言障碍
        \item 体格发育落后
        \item 是最严重的碘缺乏后果，\imp{不可逆转！}
    \end{itemize}
    \item \textbf{智力低下：}轻度缺碘即可导致智力发育受损，表现为学习能力下降
\end{enumerate}
\end{keybox}

\begin{tipbox}
\textbf{碘过量的危害：}
\begin{itemize}[leftmargin=*]
    \item 高碘性甲状腺肿（摄入过量碘反而抑制甲状腺激素合成——Wolff-Chaikoff效应）
    \item 碘诱发甲状腺功能亢进症（IIH）
    \item 自身免疫性甲状腺疾病风险增加
\end{itemize}
碘的摄入既不可缺乏也不可过量，需维持适宜水平。
\end{tipbox}

\subsection{参考摄入量与碘盐使用}

\begin{keybox}
\textbf{碘的膳食参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成人 RNI：\imp{150 μg/d}
    \item 孕妇、乳母 RNI：\imp{200 μg/d}
    \item UL（可耐受最高摄入量）：1000 μg/d
\end{itemize}
\end{keybox}

\begin{exambox}
\textbf{碘盐（Iodized Salt）的正确使用与保存：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{避光保存：}碘化钾光照易分解
    \item \textbf{密封容器：}防止碘升华逸散
    \item \textbf{出锅前/后加盐：}避免高温烹饪使碘挥发损失（碘酸钾高温分解）
    \item 购买加碘盐时注意查看生产日期和保质期
\end{enumerate}
\end{exambox}

\textbf{食物来源：}海带、紫菜、海鱼等海产品（含碘最丰富）；加碘食盐。

% ==================== 第七节 铬 ====================
\section{铬（Chromium, Cr）}

\begin{defbox}
铬是人体必需微量元素之一，主要以三价铬（Cr$^{3+}$）形式存在于体内。
\end{defbox}

\textbf{生理功能：}铬是\keyword{葡萄糖耐量因子（Glucose Tolerance Factor, GTF）}的重要组成成分。GTF可增强胰岛素的作用，促进葡萄糖进入细胞和利用，改善糖耐量。缺铬可导致血糖升高、血脂异常。

\textbf{参考摄入量与食物来源：}成人AI为\imp{50 μg/d}。食物来源：啤酒酵母、肝脏、肉类、整粒谷物、黑胡椒。

% ==================== 第八节 其他矿物质（简略） ====================
\section{其他矿物质}

\subsection{磷（Phosphorus, P）}
与钙共同构成骨骼和牙齿（羟磷灰石）；参与能量代谢（ATP、磷酸肌酸）、核酸和磷脂的构成。成人AI约700 mg/d。广泛分布于各类食物中，缺乏罕见。

\subsection{镁（Magnesium, Mg）}
约60\%存在于骨骼中。参与300多种酶促反应，维持神经肌肉正常兴奋性，参与蛋白质合成。成人AI约350 mg/d（男）、280 mg/d（女）。来源：绿叶蔬菜（叶绿素含镁）、坚果、豆类。

\subsection{铜（Copper, Cu）}
含铜金属酶：超氧化物歧化酶（Cu-Zn SOD）、铜蓝蛋白、细胞色素c氧化酶。参与铁代谢（铜蓝蛋白促进铁氧化为Fe$^{3+}$以与运铁蛋白结合）、结缔组织形成、黑色素合成。AI约2.0 mg/d。典型缺乏见Menkes病。

\subsection{锰（Manganese, Mn）}
参与骨骼形成、碳水化合物和脂肪代谢。是多种酶的激活剂（糖基转移酶、Mn-SOD等）。AI约4.5 mg/d。来源：坚果、茶叶、谷类。

\subsection{氟（Fluoride, F）}
增强牙齿和骨骼的抗酸能力，预防龋齿。成人AI约1.5 mg/d。饮水是最重要来源。过量导致氟斑牙（斑釉症）和氟骨症。

\subsection{钴（Cobalt, Co）}
钴是\keyword{维生素B$_{12}$（氰钴胺素）}的核心组成元素。钴本身不能直接被人体利用，必须以VitB$_{12}$的形式发挥作用。缺钴即缺乏VitB$_{12}$（导致巨幼红细胞性贫血）。

\subsection{镍（Nickel, Ni）}
参与铁的吸收和红细胞生成，可能对某些酶的活性有影响。成人AI约100～300 μg/d。

\subsection{钼（Molybdenum, Mo）}
含钼酶：黄嘌呤氧化酶、醛氧化酶、亚硫酸盐氧化酶。参与嘌呤代谢和含硫氨基酸代谢。成人AI约60 μg/d。

\end{enumerate}
