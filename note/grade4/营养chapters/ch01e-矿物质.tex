\chapter{矿物质}
\chapterintro{
\keyword{掌握}矿物质的概念与分类（常量元素/微量元素），钙、铁、碘、锌、硒的生理功能、吸收利用及其影响因素、缺乏、过量与毒性；
\keyword{熟悉}矿物质的生理功能，常量元素中磷、镁、钾、钠、硫的生理功能及缺乏，微量元素中钴的生理功能与缺乏；
\keyword{了解}生理作用剂量带，矿物质的相互作用，磷、镁、钾、钠、硫的吸收利用，铜、铬、氟、锰、钼、硅、硼、镍等微量元素。
}

\section{矿物质概述}

\begin{keybox}
\textbf{矿物质的概念}

\textbf{概念：}矿物质指人体组织中除碳、氢、氧和氮主要以有机化合物形式存在外的其余元素，统称为矿物质（无机盐或灰分）。人体组织中的各种元素，除碳、氢、氧和氮主要以有机化合物形式存在外，其余均统称为矿物质。
\end{keybox}

\subsection{矿物质的分类}

\begin{defbox}
\textbf{矿物质的分类}

按体内含量分为两大类：
\begin{itemize}[leftmargin=*]
    \item \imp{常量元素（macroelements）：}体内含量大于体重的0.01\%的矿物质，包括Ca、P、Na、K、S、Cl、Mg，每日摄入量大于100mg。
    \item \imp{微量元素（microelements / trace elements）：}体内含量小于体重的0.01\%的矿物质。按功能可进一步分类：
    \begin{itemize}[leftmargin=*]
        \item \imp{必需微量元素：}铁（Fe）、钴（Co）、钼（Mo）、铜（Cu）、锌（Zn）、硒（Se）、碘（I）、铬（Cr）。
        \item \imp{可能必需微量元素：}锰（Mn）、硅（Si）、镍（Ni）、硼（B）、钒（V）。
        \item \imp{潜在毒性微量元素：}氟（F）、铅（Pb）、镉（Cd）、汞（Hg）、砷（As）、铝（Al）、锡（Sn）、锂（Li）。
        \item \imp{功能未知微量元素：}锶（Sr）、铷（Rb）、稀土元素等。
    \end{itemize}
\end{itemize}
\end{defbox}

% [补充自往届笔记] 矿物质的特点
\begin{tipbox}
\textbf{矿物质的特点（了解）：}
\begin{enumerate}[leftmargin=*]
    \item 体内不能合成，必须从外界摄取。
    \item 是唯一可通过天然水途径获取的营养素。
    \item 体内分布不均匀。
    \item 矿物质之间既有协同又有拮抗作用。
    \item 某些矿物质微量元素在体内的生理作用剂量与中毒剂量范围较窄，过量摄入易产生毒性作用。
\end{enumerate}
\end{tipbox}

\subsection{矿物质的生理功能}

\begin{tipbox}
\textbf{矿物质的生理功能（熟悉）：}
\begin{enumerate}[leftmargin=*]
    \item 构成机体组织的重要原料
    \item 维持机体酸碱平衡和渗透压
    \begin{itemize}[leftmargin=*]
        \item 细胞外液渗透压：Na$^+$、Cl$^-$
        \item 细胞内液渗透压：K$^+$、HPO$_4^{2-}$
    \end{itemize}
    \item 维持神经肌肉兴奋性
    \item 参与构成激素、维生素、蛋白质和多种酶
\end{enumerate}
\end{tipbox}

\subsection{生理作用剂量带}

\begin{tipbox}
\textbf{生理作用剂量带（了解）}

矿物质的生理作用与剂量密切相关，从缺乏到中毒的剂量-效应曲线通常呈"U"形或"J"形特征。按照摄入剂量可分为以下剂量带：
\begin{itemize}[leftmargin=*]
    \item \textbf{严重缺乏带：}摄入量远低于需要量，出现典型的临床缺乏症状和体征。
    \item \textbf{边缘缺乏带：}摄入量略低于需要量，无明显临床症状，但可降低生理功能和对疾病的抵抗力。
    \item \textbf{适宜摄入带：}摄入量满足机体需要且无不良反应，维持正常生理功能和机体健康。
    \item \textbf{边缘毒性带：}摄入量略超UL，可能出现亚临床的生理功能异常。
    \item \textbf{严重毒性带：}摄入量远超UL，出现明显的中毒症状和靶器官损害。
\end{itemize}

不同矿物质的"适宜带"宽度差异很大——如钙有较宽的安全范围，而氟的适宜带极窄（生理剂量与中毒剂量相近）。
\end{tipbox}

\subsection{矿物质的相互作用}

\begin{tipbox}
\textbf{矿物质的相互作用（了解）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{协同作用：}钙与磷共同构成骨骼（钙磷比1:1$\sim$1:1.5），铁与铜协同参与造血。
    \item \textbf{拮抗作用（竞争吸收）：}钙抑制铁、锌的吸收；锌抑制铜的吸收；铁与锌相互抑制；植酸同时螯合多种二价阳离子（Ca$^{2+}$、Fe$^{2+}$、Zn$^{2+}$、Mg$^{2+}$）。
    \item \textbf{间接影响：}镁缺乏可导致低钾血症和低钙血症；钠摄入增加则尿钙排出增加。
\end{itemize}
\end{tipbox}

\section{常量元素}

\subsection{钙（Calcium）}

\begin{keybox}
\textbf{钙的概述}

钙是人体含量最多的矿物质元素，约占体重的1.5\%$\sim$2\%。

\textbf{混溶钙池（miscible calcium pool）：}约99\%的钙集中在骨骼和牙齿中，其余1\%的钙分布在软组织、细胞外液和血液中，这部分钙统称为混溶钙池。

\textbf{钙的生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 构成骨骼和牙齿的成分
    \item 维持神经肌肉活动
    \item 促进细胞信号传递
    \item 维持细胞膜稳定性
    \item 调节机体酶活性
    \item 血液凝固
\end{enumerate}

\difficulty \textbf{影响钙吸收的因素（掌握）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{机体因素：}随年龄$\uparrow$吸收率$\downarrow$。
    \item \textbf{膳食因素：}
    \begin{itemize}[leftmargin=*]
        \item 不利于吸收的（难溶性钙盐）：植物性食物、膳食纤维、未被消化的脂肪酸、咖啡因和酒精
        \item 有利于吸收的（可溶性钙盐）：蛋白质和乳糖
    \end{itemize}
\end{itemize}

\textbf{影响钙排泄的因素（掌握）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{机体因素：}血钙浓度低$\rightarrow$尿钙减少，血钙浓度高$\rightarrow$尿钙增加；年龄$\uparrow \rightarrow$尿钙$\uparrow$；补液、酸中毒、甲状腺素和肾上腺皮质激素$\rightarrow$尿钙$\uparrow$。
    \item \textbf{膳食因素：}钠、蛋白质、咖啡因$\uparrow \rightarrow$尿钙$\uparrow$。
\end{itemize}

\difficulty \textbf{缺乏（掌握）：}
\begin{itemize}[leftmargin=*]
    \item \imp{婴幼儿和儿童：}软骨病（生长发育迟缓、骨软化、骨骼变形）；佝偻病（O形或X形腿、肋骨串珠、鸡胸）；龋齿（影响牙齿质量）。
    \item \imp{中老年人：}骨质疏松症。
\end{itemize}

\difficulty \textbf{过量与毒性（掌握）：}高钙血症、高钙尿、血管和软组织钙化、肾结石等。

\textbf{与其他矿物质的相互作用（了解）：}钙与磷共同构成骨骼（羟磷灰石），钙磷比成人适宜1:1$\sim$1:1.5；钙抑制铁、锌的吸收；钠摄入增加可促进尿钙排出。

\textbf{食物来源（熟悉）：}乳制品、豆制品。

\textbf{参考摄入量：}800mg/d。
\end{keybox}

\subsection{磷（Phosphorus）}

\begin{tipbox}
\textbf{磷的概述（了解）}

磷是常量元素，与钙共同构成骨骼和牙齿（羟磷灰石）。

\textbf{磷的生理功能（熟悉）：}
\begin{itemize}[leftmargin=*]
    \item 与钙共同构成骨骼和牙齿（羟磷灰石）
    \item 参与核酸、磷脂、含磷辅酶（NAD$^+$、NADP$^+$、TPP等）的组成
    \item 参与能量代谢（ATP）；维持酸碱平衡
\end{itemize}

\textbf{吸收利用（了解）：}主要在空肠吸收，活性维生素D促进磷吸收；植酸抑制吸收。钙磷比（Ca:P）：成人适宜1:1$\sim$1:1.5，婴儿母乳2:1（最适宜）。

\textbf{缺乏（熟悉）：}罕见（食物中含量丰富）；严重缺乏可致低磷血症（肌无力、骨软化）。

\textbf{过量与毒性：}高磷血症可抑制钙吸收，加重肾性骨营养不良。

\textbf{食物来源（熟悉）：}瘦肉、鱼、蛋、奶、豆类、坚果。
\end{tipbox}

\subsection{镁（Magnesium）}

\begin{tipbox}
\textbf{镁的概述}

镁是常量元素，成人含镁约25g，60\%$\sim$65\%在骨骼中，27\%在肌肉。

\textbf{镁的生理功能（熟悉）：}
\begin{enumerate}[leftmargin=*]
    \item 多种酶的激活剂——参与300多种酶促反应
    \item 对离子通道的作用——镁可封闭不同钾通道的外向型电流，阻止钾外流；镁作为钙阻断剂抑制钙通道
    \item 维持神经肌肉兴奋性——维持和促进骨骼生长
    \item 影响胃肠道功能——硫酸镁溶液可促进胆囊排空，碱性镁盐可中和胃酸；镁离子在肠道吸收缓慢，促使水分滞留
    \item 调节激素作用——直接影响甲状旁腺激素的分泌
\end{enumerate}

\textbf{缺乏（熟悉）：}
\begin{itemize}[leftmargin=*]
    \item 神经肌肉兴奋亢进——肌肉震颤、手足抽搐、反射亢进、共济失调，严重时出现谵妄、精神错乱甚至惊厥和昏迷
    \item 可出现低钾血症、低钙血症，增加心脑血管疾病风险
    \item 增加2型糖尿病、代谢综合征、冠心病的风险
\end{itemize}

\textbf{过量与毒性：}腹泻、恶心、胃肠痉挛；重者可出现嗜睡、肌无力、膝腱反射减弱、肌麻痹等。

\textbf{食物来源（熟悉）：}绿叶蔬菜（叶绿素含镁）、大麦、黑米、荞麦、麸皮、苋菜、口蘑、木耳、香菇等含镁丰富。

\textbf{参考摄入量：}18-29岁330mg/d，30-64岁320mg/d。
\end{tipbox}

\subsection{钾（Potassium）}

\begin{tipbox}
\textbf{钾的概述}

钾是常量元素，是细胞内液主要阳离子。

\textbf{钾的生理功能（熟悉）：}
\begin{enumerate}[leftmargin=*]
    \item 参与糖和蛋白质代谢
    \item 维持细胞内液渗透压和酸碱平衡
    \item 维持神经肌肉应激性
    \item 维持心肌正常功能
    \item 降低血压的正常功能
\end{enumerate}

\textbf{缺乏（熟悉）：}当失钾超过10\%时表现为肌肉无力、瘫痪、心律失常、横纹肌溶解综合征、肾脏功能障碍（多尿、夜尿、多饮）。

\textbf{过量与毒性：}高钾血症——极度疲乏和四肢无力（下肢较重），严重可因呼吸麻痹猝死；心率缓慢、心律失常。

\textbf{食物来源（熟悉）：}紫菜、菠萝、土豆、香蕉、牛油果、哈密瓜、黄豆。

\textbf{参考摄入量：}成人AI：2000mg/d。
\end{tipbox}

\subsection{钠（Sodium）}

\begin{tipbox}
\textbf{钠的概述}

钠是常量元素，是细胞外液主要阳离子。

\textbf{钠的生理功能（熟悉）：}
\begin{enumerate}[leftmargin=*]
    \item 调节细胞外液容量与渗透压
    \item 维持酸碱平衡
    \item 维持正常血压
    \item 其他功能：维持神经肌肉应激性
\end{enumerate}

\textbf{缺乏（熟悉）：}低钠血症——可影响细胞对氨基酸和葡萄糖的吸收。

\textbf{过量与毒性：}高钠血症——水肿、体重增加、血容量增加；急性水肿、血压上升、脂肪清除率下降及胃粘膜上皮细胞破裂。

\textbf{食物来源（熟悉）：}精盐、味精、调味品、腊肉咸菜。

\textbf{参考摄入量：}成人AI：1500mg/d。
\end{tipbox}

\subsection{硫（Sulfur）}

\begin{tipbox}
\textbf{硫的概述}

硫是常量元素，主要以含硫氨基酸（蛋氨酸、半胱氨酸）形式存在于蛋白质中。

\textbf{硫的生理功能（熟悉）：}
\begin{itemize}[leftmargin=*]
    \item 硫胺素（VitB$_1$）、生物素、辅酶A等均含硫，参与多种代谢过程
    \item 含硫氨基酸代谢产生的硫酸根参与解毒（与酚类、甾体激素结合后从尿排出）
    \item 硫酸软骨素是骨骼、软骨、肌腱和血管壁的重要组成成分
\end{itemize}

\textbf{吸收利用（了解）：}主要以含硫氨基酸形式吸收。

\textbf{缺乏（熟悉）：}罕见（与蛋白质缺乏共存）。

\textbf{过量与毒性：}含硫氨基酸摄入过多会加速骨骼中钙流失，增加骨质疏松风险。

\textbf{食物来源（熟悉）：}富含蛋白质的食物——肉类、蛋类、奶类、豆类、鱼等。
\end{tipbox}

\section{微量元素}

\subsection{铁（Iron）}

\begin{keybox}
\textbf{铁的概述}

\textbf{铁的体内分布：}70\%在血红蛋白，30\%为贮存铁。

\textbf{铁的生理功能（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item 参与体内氧的运输和组织呼吸过程（构成血红蛋白）
    \item 维持正常造血功能
    \item 参与其他重要功能
\end{enumerate}

\difficulty \textbf{影响铁吸收的因素（掌握）：}

\textbf{食物中铁的分类：}
\begin{itemize}[leftmargin=*]
    \item \imp{血红素铁（heme iron）：}主要来自动物性食物，可直接被肠粘膜上皮细胞吸收，吸收率较高，受膳食因素影响较小。
    \item \imp{非血红素铁（nonheme iron）：}主要来自植物性食物，以Fe$^{3+}$形式存在，需经细胞色素B还原为Fe$^{2+}$形式方可被吸收，受膳食因素影响大。
\end{itemize}

\textbf{影响吸收的因素：}
\begin{itemize}[leftmargin=*]
    \item \textbf{促进因素：}血红素铁（Fe$^{2+}$）直接被吸收；非血红素铁（Fe$^{3+}$）需还原为Fe$^{2+}$；维生素C、含巯基氨基酸；胃内酸度（pH）。
    \item \textbf{抑制因素：}植酸、鞣酸。
    \item \textbf{机体因素：}
    \begin{itemize}[leftmargin=*]
        \item 促进吸收：铁需要量增加（缺铁性贫血、孕期、生长发育期）、铁丢失增加（月经量多、献血、寄生虫感染、溶血性疾病）。
        \item 抑制吸收：萎缩性胃炎、胃酸缺乏、服用抗酸药物（胃内pH升高）。
    \end{itemize}
\end{itemize}

\difficulty \textbf{缺乏（掌握）：}

缺铁性贫血（小细胞低色素性贫血），缺铁分为三个阶段：
\begin{enumerate}[leftmargin=*]
    \item \imp{铁减少期（贮存铁减少期）：}体内贮存铁减少，血清铁蛋白浓度下降，无临床症状。
    \item \imp{红细胞生成缺铁期：}血清铁蛋白浓度和血清铁下降，总铁结合力升高，红细胞游离原卟啉浓度升高。
    \item \imp{缺铁性贫血期：}血红蛋白和红细胞容积比下降，出现小细胞低色素性贫血的临床表现。
\end{enumerate}

\difficulty \textbf{过量与毒性（掌握）：}

铁过量主要损害为肝脏——导致肝纤维化和肝细胞损伤，引起DNA损伤，诱发突变，并增加心血管疾病风险。
\begin{itemize}[leftmargin=*]
    \item 原发性：遗传性血色素沉着症。
    \item 继发性：反复大量输血、膳食铁摄入过多等。
\end{itemize}

\textbf{食物来源（掌握）：}动物血、肝脏、瘦肉、鱼类、黑木耳、芝麻酱、红糖。

\textbf{参考摄入量：}男12mg/d，女20mg/d。
\end{keybox}

\subsection{碘（Iodine）}

\begin{keybox}
\textbf{碘的生理功能（掌握）：}参与甲状腺激素合成。

\difficulty \textbf{缺乏（掌握）：}
\begin{itemize}[leftmargin=*]
    \item 缺碘性甲状腺功能减退症状——甲状腺肿大
    \item \imp{地方性甲状腺肿}
    \item \imp{克汀病（呆小症）：}婴幼儿生长发育迟缓、智力低下，影响神经系统发育
\end{itemize}

\difficulty \textbf{过量与毒性（掌握）：}高碘性甲状腺肿。

% [补充自往届笔记] 碘过量也可引起高碘性甲状腺肿

\textbf{食物来源（掌握）：}海产品（海带、紫菜等）、加碘盐。

\textbf{参考摄入量：}120$\mu$g/d。
\end{keybox}

\subsection{锌（Zinc）}

\begin{keybox}
\textbf{锌的生理功能（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item 促进生长发育
    \item 促进免疫功能
    \item 作为酶的组成成分或激活剂
    \item 维持细胞膜结构
\end{enumerate}

% [补充自往届笔记] 锌的详细功能

\subsection{钴（Cobalt）}

\begin{defbox}
\textbf{钴的概述（熟悉）}

\textbf{缺乏（熟悉）：}即VitB$_{12}$缺乏——巨幼红细胞性贫血、神经系统损害、高同型半胱氨酸血症。

\subsection{铜（Copper）}

\begin{tipbox}
\textbf{铜的概述（了解）}

铜是必需微量元素。

\textbf{生理功能（了解）：}
\begin{itemize}[leftmargin=*]
    \item 参与铁代谢（铜蓝蛋白促进铁的吸收和利用）

\subsection*{参考答案要点}
\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item \textbf{矿物质：}人体组织中除碳、氢、氧和氮主要以有机化合物形式存在外的其余元素，统称为矿物质（无机盐/灰分）。\textbf{常量元素：}体内含量大于体重0.01\%的矿物质，包括Ca、P、Na、K、S、Cl、Mg，每日摄入量>100mg。\textbf{微量元素：}体内含量小于体重0.01\%的矿物质。\textbf{必需微量元素：}铁、钴、钼、铜、锌、硒、碘、铬。\textbf{可能必需微量元素：}锰、硅、镍、硼、钒。\textbf{潜在毒性微量元素：}氟、铅、镉、汞、砷、铝、锡、锂。\textbf{生理作用剂量带：}从缺乏到中毒，按剂量分为严重缺乏带、边缘缺乏带、适宜摄入带、边缘毒性带、严重毒性带，呈U形或J形剂量-效应曲线，不同矿物质安全范围宽度不同（如氟极窄、钙较宽）。\textbf{混溶钙池：}约99\%钙集中在骨骼和牙齿中，其余1\%分布在软组织、细胞外液和血液中，统称为混溶钙池。

    \item \textbf{分类：}常量元素（体内含量>0.01\%体重，Ca、P、Na、K、S、Cl、Mg，每日>100mg）+ 微量元素（体内含量<0.01\%体重，分为必需微量元素Fe/Co/Mo/Cu/Zn/Se/I/Cr；可能必需微量元素Mn/Si/Ni/B/V；潜在毒性微量元素F/Pb/Cd/Hg/As/Al/Sn/Li；功能未知微量元素Sr/Rb/稀土元素）。

    \item \textbf{生理功能：}(1)构成机体组织的重要原料；(2)维持机体酸碱平衡和渗透压（细胞外液：Na$^+$/Cl$^-$；细胞内液：K$^+$/HPO$_4^{2-}$）；(3)维持神经肌肉兴奋性；(4)参与构成激素、维生素、蛋白质和多种酶。

    \item 五个剂量带：严重缺乏带（出现典型临床症状）、边缘缺乏带（无明显症状但功能降低）、适宜摄入带（满足需要不产生不良反应）、边缘毒性带（亚临床功能异常）、严重毒性带（明显中毒症状）。剂量-效应曲线呈U形或J形，不同矿物质适宜带宽度差异大——钙较宽而氟极窄。

    \item \textbf{协同作用：}钙磷共构骨骼（钙磷比1:1$\sim$1:1.5）；铁铜协同参与造血。\textbf{拮抗作用：}钙抑制铁锌吸收；锌抑制铜吸收；铁锌互抑；植酸螯合多种二价阳离子（Ca$^{2+}$、Fe$^{2+}$、Zn$^{2+}$、Mg$^{2+}$）。\textbf{间接影响：}镁缺乏可致低钾低钙；钠摄入增加尿钙排出增加。

    \item \textbf{钙的生理功能：}(1)构成骨骼和牙齿的成分；(2)维持神经肌肉活动；(3)促进细胞信号传递；(4)维持细胞膜稳定性；(5)调节机体酶活性；(6)血液凝固。\textbf{影响吸收因素：}机体因素——随年龄$\uparrow$吸收率$\downarrow$。膳食因素——不利于吸收：植物性食物、膳食纤维、未被消化的脂肪酸、咖啡因和酒精（难溶性钙盐）；有利于吸收：蛋白质和乳糖（可溶性钙盐）。\textbf{影响排泄因素：}机体因素——血钙浓度低$\rightarrow$尿钙$\downarrow$，血钙浓度高$\rightarrow$尿钙$\uparrow$；年龄$\uparrow \rightarrow$尿钙$\uparrow$；补液、酸中毒、甲状腺素和肾上腺皮质激素$\rightarrow$尿钙$\uparrow$。膳食因素——钠、蛋白质、咖啡因$\uparrow \rightarrow$尿钙$\uparrow$。\textbf{缺乏：}婴幼儿和儿童——软骨病、佝偻病（O形或X形腿、肋骨串珠、鸡胸）、龋齿；中老年人——骨质疏松症。\textbf{过量：}高钙血症、高钙尿、血管和软组织钙化、肾结石等。RNI 800mg/d。来源：乳制品、豆制品。

    \item \textbf{铁的体内分布：}70\%在血红蛋白，30\%为贮存铁。\textbf{铁的分类：}血红素铁（Fe$^{2+}$，来自动物性食物，直接被吸收，受膳食因素影响小）；非血红素铁（Fe$^{3+}$，来自植物性食物，需还原为Fe$^{2+}$，受膳食因素影响大）。\textbf{促进吸收因素：}维生素C、含巯基氨基酸、胃内酸度（pH）。\textbf{抑制因素：}植酸、鞣酸。\textbf{机体因素：}促进——铁需要量增加（缺铁性贫血、孕期、生长发育期）、铁丢失增加（月经量多、献血等）；抑制——萎缩性胃炎、胃酸缺乏、抗酸药物。\textbf{缺乏的3个阶段：}铁减少期$\rightarrow$红细胞生成缺铁期$\rightarrow$缺铁性贫血期（小细胞低色素性贫血）。\textbf{过量：}主要损害肝脏——肝纤维化、DNA损伤、诱发突变、增加心血管疾病风险。RNI：男12mg/d，女20mg/d。来源：动物血、肝脏、瘦肉、鱼类、黑木耳、芝麻酱、红糖。

    \item \textbf{锌的生理功能：}(1)促进生长发育；(2)促进免疫功能；(3)作为酶的组成成分或激活剂；(4)维持细胞膜结构。\textbf{影响吸收因素：}机体因素——锌营养状态、妊娠期和哺乳期、疾病状态等。膳食因素——促进：组氨酸、蛋氨酸、半胱氨酸、葡萄糖、VitD$_3$；抑制：膳食纤维、植酸、铜、钙、亚铁离子。\textbf{缺乏：}食欲不振、异食癖、生长发育停滞（儿童侏儒症）、免疫功能低下。RNI：男12.5mg/d，女7.5mg/d。来源：动物性食物（牡蛎等）。

    \item \textbf{碘的生理功能：}参与甲状腺激素合成。\textbf{缺乏：}地方性甲状腺肿；克汀病（呆小症）——婴幼儿生长发育迟缓、智力低下。\textbf{过量：}高碘性甲状腺肿。RNI：120$\mu$g/d。来源：海产品、加碘盐。

    \item \textbf{硒的生理功能：}(1)抗氧化（谷胱甘肽过氧化物酶的组成成分）；(2)保护心血管和心肌；(3)增强免疫功能；(4)有毒重金属的解毒作用。\textbf{缺乏：}克山病、大骨节病。\textbf{过量：}地方性硒中毒。RNI：60$\mu$g/d。来源：动物性食品、海产品。

    \item \textbf{钴：}必需微量元素，是VitB$_{12}$的组成成分（唯一含金属元素的维生素），通过VitB$_{12}$发挥生理作用。缺乏即VitB$_{12}$缺乏——巨幼红细胞性贫血、神经系统损害、高同型半胱氨酸血症。来源：动物肝脏、肾脏、鱼类、肉类。

    \item \textbf{磷：}常量元素，与钙共同构成骨骼牙齿（羟磷灰石），参与核酸/磷脂/含磷辅酶/能量代谢（ATP），钙磷比成人1:1$\sim$1:1.5、婴儿母乳2:1，缺乏罕见（低磷血症）。\textbf{镁：}常量元素，60\%$\sim$65\%在骨骼，300多种酶辅因子，镁对离子通道有调节作用，缺乏致神经肌肉兴奋亢进/手足抽搐/低钾低钙血症/心律失常，过量致腹泻/恶心/嗜睡/肌无力（RNI 330mg/d 18-29岁）。\textbf{钾：}常量元素，细胞内液主要阳离子，参与糖和蛋白质代谢，维持心肌正常功能及降低血压，缺乏致肌无力/瘫痪/心律失常（失钾>10\%），过量致疲乏无力/呼吸麻痹猝死（AI 2000mg/d）。\textbf{钠：}常量元素，细胞外液主要阳离子，调节容量与渗透压，缺乏影响氨基酸和葡萄糖吸收（低钠血症），过量致水肿/血容量增加/血压上升（AI 1500mg/d）。\textbf{硫：}常量元素，主要以含硫氨基酸形式存在，参与解毒（硫酸根与酚类等结合）、软骨/肌腱组成成分（硫酸软骨素），过量含硫氨基酸摄入加速骨钙流失。

    \item \textbf{铜：}必需微量元素，参与铁代谢（铜蓝蛋白）/胶原交联/黑色素合成，缺乏致小细胞低色素性贫血。\textbf{铬：}必需微量元素，三价铬为GTF成分增强胰岛素作用（AI 30$\mu$g/d）。\textbf{氟：}潜在毒性微量元素，促进骨骼牙齿矿化预防龋齿，过量致氟斑牙/氟骨症。\textbf{锰：}可能必需微量元素，是Mn-SOD等的组成成分，参与骨骼发育和糖脂代谢，过量致神经系统损害（帕金森样综合征）。\textbf{钼：}必需微量元素，黄嘌呤氧化酶等组成成分，参与嘌呤和含硫氨基酸代谢。\textbf{硅：}可能必需微量元素，参与骨基质和结缔组织形成。\textbf{硼：}可能必需微量元素，可能参与钙镁和雌激素代谢调节。\textbf{镍：}可能必需微量元素，可能参与铁吸收和代谢调节，过量致皮肤过敏（接触性皮炎）。
\end{enumerate}
