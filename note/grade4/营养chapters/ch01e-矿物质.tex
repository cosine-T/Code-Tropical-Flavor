\chapter{矿物质}
\chapterintro{掌握矿物质的概念与分类（常量/微量元素）；掌握钙、铁、锌、碘、硒的生理功能、吸收、缺乏；熟悉其他矿物质。}

\section{矿物质概述}

\begin{keybox}
\textbf{矿物质的概念与分类}

\textbf{概念：}矿物质（Minerals）指人体组织中除碳、氢、氧和氮主要以有机化合物形式存在外的其余元素，统称为矿物质（无机盐或灰分）。

\textbf{分类：}
\begin{itemize}[leftmargin=*]
    \item \imp{常量元素：}体内含量大于体重的0.01\%的矿物质，包括Ca、P、Na、K、S、Cl、Mg，每日摄入量>100mg。
    \item \imp{微量元素：}体内含量小于体重的0.01\%的矿物质。
    \begin{itemize}[leftmargin=*]
        \item \imp{必需微量元素：}铁、钴、钼、铜、锌、硒、碘、铬。
        \item \imp{可能必需微量元素：}锰、硅、镍、硼、钒。
        \item \imp{潜在毒性微量元素：}氟、铅、镉、汞、砷、铝、锡、锂。
    \end{itemize}
\end{itemize}

\textbf{矿物质的生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 构成机体组织的重要原料
    \item 维持机体酸碱平衡和渗透压（细胞外液渗透压：Na$^+$、Cl$^-$；细胞内液渗透压：K$^+$、HPO$_4^{2-}$）
    \item 维持神经肌肉兴奋性
    \item 参与构成激素、维生素、蛋白质和多种酶
\end{enumerate}
\end{keybox}

\section{钙（Calcium）}

\begin{keybox}
\textbf{钙的生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 构成骨骼和牙齿的成分
    \item 维持神经肌肉活动
    \item 促进细胞信号传递
    \item 维持细胞膜稳定性
    \item 调节机体酶活性
    \item 血液凝固
\end{enumerate}

\textbf{影响钙吸收的因素：}
\begin{itemize}[leftmargin=*]
    \item \textbf{机体因素：}随年龄$\uparrow$吸收率$\downarrow$。
    \item \textbf{膳食因素：}
    \begin{itemize}[leftmargin=*]
        \item 难溶性钙盐（植物性食物、膳食纤维、未被消化的脂肪酸、咖啡因和酒精）不易吸收
        \item 可溶性钙盐（蛋白质和乳糖）易吸收
    \end{itemize}
\end{itemize}

\textbf{影响钙排泄的因素：}
\begin{itemize}[leftmargin=*]
    \item \textbf{机体因素：}血钙浓度低尿钙减少，血钙浓度高尿钙增加；年龄$\uparrow$尿钙$\uparrow$；补液、酸中毒、甲状腺素和肾上腺皮质激素$\rightarrow$尿钙$\uparrow$。
    \item \textbf{膳食因素：}钠、蛋白质、咖啡因$\uparrow \rightarrow$尿钙$\uparrow$。
\end{itemize}

\textbf{缺乏：}
\begin{itemize}[leftmargin=*]
    \item \imp{婴幼儿和儿童：}软骨病（生长发育迟缓、骨软化、骨骼变形）；佝偻病（O形或X形腿、肋骨串珠、鸡胸）；龋齿。
    \item \imp{中老年人：}骨质疏松症。
\end{itemize}

\textbf{过量：}高钙血症、高钙尿、血管和软组织钙化、肾结石等。

\textbf{参考摄入量：}800mg/d。

\textbf{食物来源：}乳制品、豆制品。
\end{keybox}

\section{铁（Iron）}

\begin{keybox}
\textbf{铁的体内分布：}70\%在血红蛋白，30\%为贮存铁。

\textbf{影响铁吸收的因素：}
\begin{itemize}[leftmargin=*]
    \item \textbf{促进因素：}血红素铁（Fe$^{2+}$）直接被吸收；非血红素铁（Fe$^{3+}$）需还原为Fe$^{2+}$；维生素C、含巯基氨基酸；胃内酸度（pH）。
    \item \textbf{抑制因素：}植酸、鞣酸。
\end{itemize}

\textbf{缺乏：}缺铁性贫血（小细胞低色素性贫血）。

\textbf{参考摄入量：}男12mg/d，女20mg/d。
\end{keybox}

\section{锌（Zinc）}

\begin{keybox}
\textbf{锌的生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 酶的组成成分
    \item 促进生长发育
    \item 促进性发育
    \item 维持正常味觉和食欲
\end{enumerate}

\textbf{缺乏：}生长发育迟缓、性发育不全、脑发育受损、食欲不振、异食癖。

\textbf{参考摄入量：}男12.5mg/d，女7.5mg/d。
\end{keybox}

\section{碘（Iodine）}

\begin{keybox}
\textbf{碘的生理功能：}参与甲状腺激素合成。

\textbf{缺乏：}
\begin{itemize}[leftmargin=*]
    \item 地方性甲状腺肿
    \item 克汀病（呆小症）
\end{itemize}

\textbf{参考摄入量：}120$\mu$g/d。
\end{keybox}

\section{硒（Selenium）}

\begin{keybox}
\textbf{硒的生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 谷胱甘肽过氧化物酶的组成成分
    \item 抗氧化
    \item 参与甲状腺激素代谢
\end{enumerate}

\textbf{缺乏：}克山病、大骨节病。

\textbf{参考摄入量：}60$\mu$g/d。
\end{keybox}

\section{其他矿物质}

\begin{defbox}
\textbf{磷（Phosphorus）：}
\begin{itemize}[leftmargin=*]
    \item 常量元素，与钙共同构成骨骼和牙齿（羟磷灰石）
    \item 参与核酸、磷脂、含磷辅酶（NAD$^+$、NADP$^+$、TPP等）的组成
    \item 参与能量代谢（ATP）
    \item 维持酸碱平衡
    \item 钙磷比（Ca:P）：成人适宜1:1$\sim$1:1.5，婴儿母乳2:1（最适宜）
\end{itemize}

\textbf{镁（Magnesium）：}
\begin{itemize}[leftmargin=*]
    \item 常量元素，成人含镁约25g，60\%$\sim$65\%在骨骼中，27\%在肌肉
    \item 300多种酶的辅因子；参与蛋白质合成和能量代谢
    \item 维持神经肌肉兴奋性；心血管保护作用
    \item 缺乏：神经肌肉兴奋性增高、心律失常、低钙血症（镁缺乏同时导致低血钙）
    \item 来源：绿叶蔬菜（叶绿素含镁）、坚果、豆类、全谷物
\end{itemize}

\textbf{钾（Potassium）：}
\begin{itemize}[leftmargin=*]
    \item 常量元素，细胞内液主要阳离子
    \item 维持细胞渗透压和酸碱平衡；维持神经肌肉应激性和心肌正常功能
    \item 缺乏（低钾血症）：肌肉无力、瘫痪、心律失常
    \item 过量（高钾血症）：疲乏无力（下肢较重），严重可因呼吸麻痹猝死
    \item 成人AI：2000mg/d
\end{itemize}

\textbf{钠（Sodium）：}
\begin{itemize}[leftmargin=*]
    \item 常量元素，细胞外液主要阳离子
    \item 调节细胞外液容量与渗透压；维持酸碱平衡和正常血压
    \item 缺乏（低钠血症）：影响细胞对氨基酸和葡萄糖的吸收
    \item 过量（高钠血症）：水肿、血容量增加、血压上升
    \item 来源：食盐（NaCl）、味精、调味品、腊肉咸菜。成人AI：1500mg/d
\end{itemize}

\textbf{铜（Copper）：}
\begin{itemize}[leftmargin=*]
    \item 必需微量元素
    \item 参与铁代谢（铜蓝蛋白促进铁的吸收和利用）
    \item 参与结缔组织中胶原蛋白和弹性蛋白的交联（赖氨酰氧化酶）
    \item 参与黑色素合成（酪氨酸酶）
    \item 缺乏：小细胞低色素性贫血（与铁代谢障碍有关）、中性粒细胞减少、骨质疏松
\end{itemize}

\textbf{铬（Chromium）：}
\begin{itemize}[leftmargin=*]
    \item 必需微量元素，三价铬是葡萄糖耐量因子（GTF）的组成成分
    \item 增强胰岛素作用，促进葡萄糖利用
    \item 缺乏：葡萄糖耐量降低、胰岛素抵抗
    \item 来源：全谷物、肉类、啤酒酵母。成人AI：30$\mu$g/d
\end{itemize}

\textbf{氟（Fluoride）：}
\begin{itemize}[leftmargin=*]
    \item 潜在毒性微量元素（生理剂量与中毒剂量范围极窄）
    \item 促进骨骼和牙齿的矿化，预防龋齿
    \item 缺乏：龋齿发病率增高
    \item 过量：氟斑牙和氟骨症
    \item 来源：饮用水（主要来源）、海产品、茶叶
\end{itemize}

\textbf{钴（Cobalt）：}
\begin{itemize}[leftmargin=*]
    \item 必需微量元素，维生素B$_{12}$的组成成分（唯一含金属元素的维生素）
    \item 通过VitB$_{12}$发挥生理作用
    \item 缺乏即VitB$_{12}$缺乏——巨幼红细胞性贫血
\end{itemize}
\end{defbox}

\subsection*{参考答案要点}
\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item \textbf{矿物质：}人体组织中除碳、氢、氧和氮主要以有机化合物形式存在外的其余元素，统称为矿物质（无机盐/灰分）。\textbf{常量元素：}体内含量大于体重0.01\%的矿物质，包括Ca、P、Na、K、S、Cl、Mg，每日摄入量>100mg。\textbf{微量元素：}体内含量小于体重0.01\%的矿物质。\textbf{必需微量元素：}铁、钴、钼、铜、锌、硒、碘、铬。\textbf{可能必需微量元素：}锰、硅、镍、硼、钒。\textbf{潜在毒性微量元素：}氟、铅、镉、汞、砷、铝、锡、锂。
    \item \textbf{分类：}常量元素（>0.01\%，Ca/P/Na/K/S/Cl/Mg）+ 微量元素（<0.01\%，必需Fe/Co/Mo/Cu/Zn/Se/I/Cr；可能必需Mn/Si/Ni/B/V；潜在毒性F/Pb/Cd/Hg/As/Al/Sn/Li）。\textbf{生理功能：}(1)构成机体组织的重要原料；(2)维持机体酸碱平衡和渗透压（细胞外液：Na$^+$/Cl$^-$；细胞内液：K$^+$/HPO$_4^{2-}$）；(3)维持神经肌肉兴奋性；(4)参与构成激素、维生素、蛋白质和多种酶。
    \item \textbf{钙的生理功能：}(1)构成骨骼和牙齿的成分；(2)维持神经肌肉活动；(3)促进细胞信号传递；(4)维持细胞膜稳定性；(5)调节机体酶活性；(6)血液凝固。\textbf{影响吸收因素：}机体因素——随年龄$\uparrow$吸收率$\downarrow$。膳食因素——难溶性钙盐（植物性食物、膳食纤维、未被消化的脂肪酸、咖啡因和酒精）不易吸收；可溶性钙盐（蛋白质和乳糖）易吸收。\textbf{影响排泄因素：}机体因素——血钙浓度低尿钙减少，血钙浓度高尿钙增加；年龄$\uparrow$尿钙$\uparrow$；补液、酸中毒、甲状腺素和肾上腺皮质激素$\rightarrow$尿钙$\uparrow$。膳食因素——钠、蛋白质、咖啡因$\uparrow \rightarrow$尿钙$\uparrow$。\textbf{缺乏：}婴幼儿和儿童——软骨病、佝偻病（O形或X形腿、肋骨串珠、鸡胸）、龋齿；中老年人——骨质疏松症。\textbf{过量：}高钙血症、高钙尿、血管和软组织钙化、肾结石等。RNI 800mg/d。来源：乳制品、豆制品。
    \item \textbf{铁的体内分布：}70\%在血红蛋白，30\%为贮存铁。\textbf{促进吸收因素：}血红素铁（Fe$^{2+}$）直接被吸收；非血红素铁（Fe$^{3+}$）需还原为Fe$^{2+}$；维生素C、含巯基氨基酸；胃内酸度（pH）。\textbf{抑制因素：}植酸、鞣酸。\textbf{缺乏：}缺铁性贫血（小细胞低色素性贫血）。RNI：男12mg/d，女20mg/d。
    \item \textbf{锌的生理功能：}(1)酶的组成成分；(2)促进生长发育；(3)促进性发育；(4)维持正常味觉和食欲。\textbf{缺乏：}生长发育迟缓、性发育不全、脑发育受损、食欲不振、异食癖。RNI：男12.5mg/d，女7.5mg/d。
    \item \textbf{碘的生理功能：}参与甲状腺激素合成。\textbf{缺乏：}地方性甲状腺肿；克汀病（呆小症）。RNI：120$\mu$g/d。
    \item \textbf{硒的生理功能：}(1)谷胱甘肽过氧化物酶的组成成分；(2)抗氧化；(3)参与甲状腺激素代谢。\textbf{缺乏：}克山病、大骨节病。RNI：60$\mu$g/d。
    \item \textbf{磷：}常量元素，构成骨骼牙齿（羟磷灰石），参与核酸/磷脂/含磷辅酶/能量代谢（ATP），钙磷比成人1:1$\sim$1:1.5、婴儿母乳2:1。\textbf{镁：}常量元素，60\%$\sim$65\%在骨骼，300多种酶辅因子，缺乏致神经肌肉兴奋性增高/心律失常/低钙血症。\textbf{钾：}常量元素，细胞内液主要阳离子，缺乏致肌无力/心律失常，过量致疲乏无力/呼吸麻痹猝死（AI 2000mg/d）。\textbf{钠：}常量元素，细胞外液主要阳离子，缺乏影响氨基酸和葡萄糖吸收，过量致水肿/血容量增加/血压上升（AI 1500mg/d）。\textbf{铜：}必需微量元素，参与铁代谢（铜蓝蛋白）/胶原交联/黑色素合成，缺乏致小细胞低色素性贫血。\textbf{铬：}必需微量元素，三价铬为GTF成分增强胰岛素作用（AI 30$\mu$g/d）。\textbf{氟：}潜在毒性微量元素，促进骨骼牙齿矿化预防龋齿，过量致氟斑牙/氟骨症。\textbf{钴：}必需微量元素，VitB$_{12}$组成成分，缺乏致巨幼红细胞性贫血。
\end{enumerate}
