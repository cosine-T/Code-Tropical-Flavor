% ==================== 第一节 维生素概述 ====================
\section{维生素概述}

\begin{defbox}
\textbf{维生素（Vitamins）}是维持人体正常生命活动所必需的一类微量低分子有机化合物。它们既不是构成机体组织的原料，也不为机体提供能量，但在物质代谢中发挥不可或缺的调节作用。
\end{defbox}

\subsection{维生素共同特点}

\begin{keybox}
\textbf{维生素的共同特点：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{均以维生素本身或其前体形式存在于天然食物中}，人体必须从食物中获得
    \item \textbf{既不构成机体组织，也不为机体提供能量}（有别于产能营养素）
    \item \textbf{需要量极微}（mg或$\mu$g水平），但生理调节作用重大
    \item \textbf{常以辅酶或辅基的形式参与物质代谢}
    \item \textbf{调节机体新陈代谢}
\end{enumerate}
\end{keybox}

\subsection{维生素的分类}

\begin{keybox}
\textbf{维生素的分类*（按溶解性）：}

\textbf{脂溶性维生素：}
\begin{itemize}[leftmargin=*]
    \item \keyword{维生素A（视黄醇）}、\keyword{维生素D（钙化醇）}、\keyword{维生素E（生育酚）}、\keyword{维生素K（凝血维生素）}
    \item 吸收需要脂肪和胆汁的存在
    \item 可在体内储存（主要在肝脏和脂肪组织），缺乏症状出现缓慢
    \item 摄入过量可引起中毒（尤其VitA和VitD，VitK除外）
    \item 除VitK外，均有UL限制
\end{itemize}

\textbf{水溶性维生素：}
\begin{itemize}[leftmargin=*]
    \item \keyword{B族维生素}：B$_1$（硫胺素）、B$_2$（核黄素）、B$_3$（烟酸）、B$_5$（泛酸）、B$_6$（吡哆醇）、B$_7$（生物素）、B$_9$（叶酸）、B$_{12}$（钴胺素）
    \item \keyword{维生素C（抗坏血酸）}
\end{itemize}

\textbf{水溶性维生素的特点：}
\begin{itemize}[leftmargin=*]
    \item 体内储存量极少（B$_{12}$除外），多余者以原形或代谢产物经尿排出
    \item 缺乏症状出现较快（尤其B族维生素）
    \item 一般不引起中毒（过量摄入可通过尿液快速排泄）
    \item 饱和剂量以上几乎不增加组织浓度
\end{itemize}
\end{keybox}

\begin{exambox}
\textbf{考点提示：}两类维生素的比较是常考基础知识点。关键记忆点：脂溶性→可储存、慢缺乏、易中毒（VitK除外）；水溶性→储存少、快缺乏、不易中毒（尿液排泄）。
\end{exambox}

% ==================== 第二节 维生素A ====================
\section{维生素A（视黄醇）}

\subsection{结构与分类}

\begin{defbox}
\textbf{维生素A}是一组具有视黄醇生物活性的物质，包括已形成的维生素A（preformed vitamin A）和维生素A原（provitamin A carotenoids）。基本结构为$\beta$-白芷酮环连着不饱和异戊二烯侧链。
\end{defbox}

\begin{keybox}
\textbf{存在形式：}
\begin{itemize}[leftmargin=*]
    \item \textbf{已形成的VitA：}视黄醇（Retinol）、视黄醛（Retinal）、视黄酸（Retinoic acid），主要来源于动物性食品
    \item \textbf{维生素A原：}类胡萝卜素，其中\imp{$\beta$-胡萝卜素（$\beta$-carotene）}最重要，可被裂解转化为视黄醛
\end{itemize}
\end{keybox}

\subsection{生理功能}

\begin{keybox}
\textbf{维生素A的生理功能*：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{维持正常视觉：}
    \begin{itemize}[leftmargin=*]
        \item 视网膜杆状细胞中的\keyword{视紫红质（Rhodopsin）}由视蛋白和11-顺视黄醛组成
        \item 光照→视紫红质分解→神经冲动→视觉
        \item 暗处视紫红质再合成需要VitA持续供给
        \item 缺乏导致暗适应延长，严重者为\imp{夜盲症（Night Blindness）}
    \end{itemize}
    \item \textbf{促进细胞生长分化：}视黄酸通过核受体（RAR/RXR）调控基因表达，维持上皮组织的正常分化与增殖
    \item \textbf{维护上皮组织健康：}维持皮肤和黏膜（呼吸道、消化道、泌尿道）的完整性
    \item \textbf{增强免疫功能：}促进免疫细胞分化和抗体生成
    \item \textbf{抗氧化作用：}$\beta$-胡萝卜素具有抗氧化功能，可淬灭单线态氧和清除自由基
\end{enumerate}
\end{keybox}

\subsection{维生素A缺乏与过量}

\begin{keybox}
\textbf{维生素A缺乏*：}

\textbf{眼部表现（由轻到重）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{暗适应能力下降}→\imp{\keyword{夜盲症（Night Blindness）}}（最早出现的症状！）
    \item \textbf{干眼病（Xerophthalmia）：}结膜干燥、失去光泽→Bitot斑
    \item \textbf{角膜软化→角膜溃疡→穿孔→失明}（儿童失明的主要原因）
\end{enumerate}

\textbf{其他缺乏表现：}
\begin{itemize}[leftmargin=*]
    \item 皮肤干燥、角化、毛囊角化过度（蟾皮症）
    \item 生长迟缓
    \item 免疫功能低下，呼吸道和消化道感染风险增加
    \item 味觉、嗅觉减退
\end{itemize}
\end{keybox}

\begin{tipbox}
\textbf{维生素A过量的危害：}
\begin{itemize}[leftmargin=*]
    \item \textbf{急性中毒：}恶心、呕吐、剧烈头痛、眩晕、视力模糊（一次大量摄入）
    \item \textbf{慢性中毒：}食欲不振、脱发、皮肤干燥/瘙痒、肝肿大、关节痛
    \item \textbf{致畸性（Teratogenicity）：}孕早期过量VitA可导致胎儿畸形
    \item 注意不包括$\beta$-胡萝卜素过量（仅引起皮肤黄染，停药后可消退）
\end{itemize}
\end{tipbox}

\subsection{视黄醇当量与参考摄入量}

\begin{keybox}
\textbf{视黄醇当量（Retinol Equivalent, RE）*：}

不同来源的维生素A折算公式：
\[
\imp{\text{视黄醇当量} (\mu\text{g RE}) = \text{视黄醇} (\mu\text{g}) + \beta\text{-胡萝卜素} (\mu\text{g}) \times 0.167}
\]

即：1 $\mu$g视黄醇 = 1 $\mu$g RE；1 $\mu$g $\beta$-胡萝卜素 = 0.167 $\mu$g RE = 1/6 $\mu$g RE

\textbf{膳食参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成年男性 RNI：\imp{800 μg RE/d}
    \item 成年女性 RNI：\imp{700 μg RE/d}
    \item 孕妇早期：700 μg RE/d；中、晚期：770 μg RE/d
    \item 乳母：1300 μg RE/d
    \item UL（可耐受最高摄入量）：\imp{3000 μg RE/d}
\end{itemize}
\end{keybox}

\textbf{食物来源：}动物肝脏、鱼肝油（最丰富）、蛋黄、奶制品（已形成的VitA）；深色蔬菜（胡萝卜、菠菜、南瓜）、水果（芒果、杏）——$\beta$-胡萝卜素。

% ==================== 第三节 维生素D ====================
\section{维生素D（钙化醇）}

\subsection{来源与活化}

\begin{keybox}
\textbf{维生素D的来源与活化*：}

\textbf{两种形式：}
\begin{itemize}[leftmargin=*]
    \item \keyword{维生素D$_2$（Ergocalciferol，麦角钙化醇）}：来源于植物（酵母、真菌经紫外线照射后麦角固醇转化）
    \item \keyword{维生素D$_3$（Cholecalciferol，胆钙化醇）}：\imp{皮肤中的7-脱氢胆固醇经紫外线（UVB，290～315nm）照射转化生成}（内源性来源，最重要）
\end{itemize}

\textbf{维生素D的活化过程：}
肝 $\xrightarrow{\text{25-羟化酶}}$ 25-(OH)D$_3$（肝内主要储存形式）$\xrightarrow{\text{肾1}\alpha\text{-羟化酶}}$ \imp{1,25-(OH)$_2$D$_3$（骨化三醇}，生物活性最强的形式）

1,25-(OH)$_2$D$_3$被称为"活性维生素D"，本质属于\keyword{类固醇激素}。
\end{keybox}

\subsection{生理功能}

\begin{keybox}
\textbf{维生素D的生理功能*：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{促进肠道钙的吸收：}诱导肠上皮细胞合成钙结合蛋白（CaBP）
    \item \textbf{促进肾小管对钙和磷的重吸收}
    \item \textbf{调节骨骼的钙化与脱钙：}维持血钙和血磷的正常水平
    \item \textbf{调节免疫功能：}1,25-(OH)$_2$D$_3$可作为免疫调节因子
    \item \textbf{调节细胞增殖与分化}
\end{enumerate}
\end{keybox}

\subsection{缺乏与过量}

\begin{keybox}
\textbf{维生素D缺乏*：}
\begin{itemize}[leftmargin=*]
    \item \textbf{儿童——佝偻病（Rickets）：}骨骼钙化不良，O形腿、X形腿、鸡胸、方颅、肋骨串珠、Harrison沟
    \item \textbf{成人——骨软化症（Osteomalacia）：}骨矿物化不良、骨痛、肌无力
    \item \textbf{老年人——骨质疏松症（Osteoporosis）：}骨量减少、骨折风险增高
\end{itemize}
\end{keybox}

\subsection{参考摄入量}

\begin{keybox}
\textbf{维生素D的膳食参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成人 RNI：\imp{5 μg/d（200 IU）}
    \item 婴幼儿、儿童、孕妇、乳母、老年人 RNI：\imp{10 μg/d（400 IU）}
    \item UL（可耐受最高摄入量）：50 μg/d（2000 IU）
    \item 单位换算：1 IU VitD$_3$ = 0.025 $\mu$g VitD$_3$；1 $\mu$g = 40 IU
\end{itemize}
\end{keybox}

\textbf{食物来源：}鱼肝油（最丰富）、海水鱼（鲑鱼、沙丁鱼）、肝脏、蛋黄、维生素D强化食品。\imp{晒太阳是获得维生素D最经济、最主要的方式。}

% ==================== 第四节 维生素E ====================
\section{维生素E（生育酚）}

\begin{defbox}
\textbf{维生素E}是一组具有抗氧化活性的生育酚（Tocopherols）和生育三烯酚的统称，其中\keyword{$\alpha$-生育酚}的生物活性最强。
\end{defbox}

\textbf{生理功能：}最重要的功能是\keyword{抗氧化作用}——保护细胞膜多不饱和脂肪酸免受自由基攻击（链阻断型抗氧化剂），防止脂质过氧化（形成脂褐素即老年斑）；维持生殖功能；抗衰老；保护红细胞膜完整性。

\textbf{特点：}维生素E是\imp{毒性最小的脂溶性维生素}，口服大量耐受良好。

\begin{keybox}
\textbf{膳食参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成人 AI：\imp{14 mg $\alpha$-TE/d}
    \item 乳母：17 mg $\alpha$-TE/d
\end{itemize}
\end{keybox}

\textbf{食物来源：}植物油（小麦胚芽油、葵花籽油、玉米油）、坚果、种子、绿叶蔬菜。

% ==================== 第五节 维生素K ====================
\section{维生素K（凝血维生素）}

\begin{keybox}
\textbf{维生素K的生理功能：}

维生素K是肝脏合成凝血因子所必需的物质，参与\keyword{凝血因子II（凝血酶原）、VII、IX、X}的谷氨酸残基$\gamma$-羧基化修饰，使其获得与Ca$^{2+}$结合的能力从而具有凝血活性。

缺乏导致凝血障碍、出血倾向。新生儿因肠道菌群未建立且母乳中VitK含量低，易出现\keyword{新生儿出血症}，出生后常规注射维生素K预防。
\end{keybox}

\textbf{食物来源：}绿色绿叶蔬菜（菠菜、甘蓝、西蓝花）含量最丰富；肠道细菌也能合成一部分。

% ==================== 第六节 维生素B1 ====================
\section{维生素B$_1$（硫胺素，Thiamin）}

\begin{keybox}
\textbf{维生素B$_1$的生理功能与缺乏*：}

\textbf{活性形式：}\keyword{焦磷酸硫胺素（TPP）}，为丙酮酸脱氢酶、$\alpha$-酮戊二酸脱氢酶和转酮醇酶的辅酶。

\textbf{生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 参与$\alpha$-酮酸的氧化脱羧反应（糖代谢关键环节）
    \item 维持神经组织正常功能（神经系统所需能量主要来自糖代谢）
    \item 促进胃肠蠕动和消化液分泌
\end{enumerate}

\textbf{缺乏病——脚气病（Beriberi）*：}
\begin{itemize}[leftmargin=*]
    \item \textbf{干性脚气病（Dry Beriberi）：}以多发性周围神经炎为主，感觉异常、肌力减退、腱反射消失
    \item \textbf{湿性脚气病（Wet Beriberi）：}以心血管系统损害为主，水肿、心悸、心脏扩大、心力衰竭
    \item \textbf{混合性脚气病：}兼有神经和心血管症状
    \item \textbf{婴儿脚气病：}见于VitB$_1$缺乏的乳母喂养的婴儿
\end{itemize}
\end{keybox}

\begin{keybox}
\textbf{参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成年男性 RNI：\imp{1.4 mg/d}
    \item 成年女性 RNI：\imp{1.3 mg/d}
\end{itemize}
\end{keybox}

\textbf{食物来源：}全谷类（米糠、麦麸）、豆类、瘦猪肉、动物内脏、坚果。谷类加工精度越高（碾磨越白），VB$_1$损失越严重。

% ==================== 第七节 维生素B2 ====================
\section{维生素B$_2$（核黄素，Riboflavin）}

\begin{keybox}
\textbf{维生素B$_2$的生理功能与缺乏*：}

\textbf{特征：}黄色晶体，水溶液呈黄绿色荧光。\imp{对紫外光敏感}（光照易破坏，故牛奶不宜用透明瓶储存）。

\textbf{活性形式：}
\begin{itemize}[leftmargin=*]
    \item \keyword{黄素单核苷酸（FMN）}
    \item \keyword{黄素腺嘌呤二核苷酸（FAD）}
\end{itemize}

FMN和FAD是体内氧化还原酶类（黄素蛋白）的辅基，参与生物氧化中的氢传递，在能量代谢中起核心作用。

\textbf{缺乏病——"口腔-生殖系统综合征"（Oral-Genital Syndrome）*：}
\begin{enumerate}[leftmargin=*]
    \item \keyword{口角炎（Angular Stomatitis）}
    \item \keyword{唇炎（Cheilitis）}
    \item \keyword{舌炎（Glossitis）}：舌头红肿、乳头萎缩、地图舌
    \item \keyword{脂溢性皮炎（Seborrheic Dermatitis）}：鼻唇沟、眉间、耳后
    \item 阴囊炎/外阴炎
    \item 眼部症状：畏光、流泪、角膜血管增生
\end{enumerate}
\end{keybox}

\begin{keybox}
\textbf{参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成年男性 RNI：\imp{1.4 mg/d}
    \item 成年女性 RNI：\imp{1.2 mg/d}
\end{itemize}
\end{keybox}

\textbf{食物来源：}动物内脏（肝、心、肾最丰富）、奶类、蛋类、豆类、绿叶蔬菜。

\begin{tipbox}
\textbf{维生素B$_2$缺乏的膳食因素：}
\begin{itemize}[leftmargin=*]
    \item 谷类加工过精（VB$_2$集中于外层和胚芽）
    \item 牛奶光照储存（VB$_2$光解破坏）
    \item 绿色蔬菜摄入不足
    \item 动物性食物摄入不足
\end{itemize}
维生素B$_2$缺乏常与其他B族维生素缺乏同时发生。
\end{tipbox}

% ==================== 第八节 烟酸 ====================
\section{烟酸（Niacin，维生素B$_3$）}

\begin{keybox}
\textbf{烟酸的生理功能与缺乏*：}

\textbf{活性形式：}
\begin{itemize}[leftmargin=*]
    \item \keyword{烟酰胺腺嘌呤二核苷酸（NAD$^+$）}
    \item \keyword{烟酰胺腺嘌呤二核苷酸磷酸（NADP$^+$）}
\end{itemize}

NAD$^+$和NADP$^+$是体内多种脱氢酶的辅酶，参与氧化还原反应和能量代谢（糖酵解、三羧酸循环、氧化磷酸化）。

\textbf{体内转化：}
色氨酸可在体内合成烟酸——\imp{约60mg色氨酸转化为1mg烟酸}。

\textbf{烟酸当量（Niacin Equivalent, NE）：}
\[
\text{NE (mg)} = \text{烟酸 (mg)} + \frac{\text{色氨酸 (mg)}}{60}
\]

\textbf{缺乏病——癞皮病（Pellagra）*（"3D"症状）：}
\begin{enumerate}[leftmargin=*]
    \item \keyword{Dermatitis（皮炎）：}暴露部位对称性皮炎（日照处红斑、粗糙、色素沉着）——Casal项链
    \item \keyword{Diarrhea（腹泻）：}消化不良、腹泻、舌炎
    \item \keyword{Dementia（痴呆）：}精神症状、记忆减退、焦虑、抑郁（严重时）
\end{enumerate}
\textbf{病因：}以玉米为主食且缺乏优质蛋白质的地区多发（玉米中烟酸为结合型无法吸收，且色氨酸含量低）。
\end{keybox}

\begin{keybox}
\textbf{参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成年男性 RNI：\imp{14 mg NE/d}
    \item 成年女性 RNI：\imp{13 mg NE/d}
\end{itemize}
\end{keybox}

\textbf{食物来源：}动物肝脏、肉类、鱼类、花生、豆类。谷类中的烟酸80\%～90\%存在于麸皮中。

% ==================== 第九节 维生素B6 ====================
\section{维生素B$_6$}

\begin{keybox}
\textbf{维生素B$_6$的生理功能与缺乏*：}

\textbf{三种形式：}吡哆醇（Pyridoxine）、吡哆醛（Pyridoxal）、吡哆胺（Pyridoxamine）。

\textbf{活性形式：}\keyword{磷酸吡哆醛（PLP）}——最重要的辅酶形式。

\textbf{生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 氨基酸代谢：参与转氨作用和脱羧反应
    \item 糖原代谢：参与糖原磷酸化酶作用（PLP使其活化）
    \item 参与不饱和脂肪酸代谢和神经递质（5-HT、GABA、多巴胺）的合成
\end{enumerate}

\textbf{缺乏表现：}脂溢性皮炎、舌炎、口角炎（类似于VB$_2$缺乏）；神经系统症状（易激惹、抑郁、周围神经炎）；小细胞低色素性贫血（参与血红素合成第一步$\delta$-ALA的生成）。
\end{keybox}

\begin{keybox}
\textbf{参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成人 AI：\imp{1.2 mg/d}
\end{itemize}
\end{keybox}

\textbf{食物来源：}广泛分布于动物和植物性食物中，肉类、肝脏、全谷物、豆类、香蕉、土豆含量较高。

% ==================== 第十节 叶酸 ====================
\section{叶酸（Folate，维生素B$_9$）}

\begin{defbox}
\textbf{叶酸}属蝶酰谷氨酸，因首先在菠菜叶中分离而得名。活性形式为\keyword{四氢叶酸（THF）}。
\end{defbox}

\begin{keybox}
\textbf{叶酸的生理功能与缺乏*：}

\textbf{活性形式：}\keyword{四氢叶酸（Tetrahydrofolate, THF）}

\textbf{生理功能：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{"一碳单位"的载体：}THF是一碳单位（-CH$_3$、-CH$_2$、-CHO等）的载体，参与核苷酸（嘌呤、胸腺嘧啶dTMP）的生物合成
    \item 参与氨基酸代谢：同型半胱氨酸→蛋氨酸的甲基化过程需要THF（N$^5$-甲基THF）作为甲基供体
    \item 参与血红蛋白的合成（与VitB$_{12}$共同作用）
\end{enumerate}

\textbf{缺乏*}：
\begin{enumerate}[leftmargin=*]
    \item \textbf{\imp{巨幼红细胞性贫血（Megaloblastic Anemia）}：}因DNA合成受阻，细胞分裂减慢，细胞体积增大但无法正常分裂。为叶酸缺乏最典型的表现
    \item \textbf{\imp{胎儿神经管畸形（Neural Tube Defects, NTD）}：}孕早期叶酸缺乏导致。包括脊柱裂（Spina Bifida）和无脑儿（Anencephaly）。\imp{围孕期（孕前至孕早期）补充400$\mu$g/d叶酸可有效预防NTD}
    \item \textbf{高同型半胱氨酸血症：}同型半胱氨酸→蛋氨酸的甲基化受阻，血Hcy升高，为动脉粥样硬化的独立危险因素
\end{enumerate}
\end{keybox}

\begin{keybox}
\textbf{膳食叶酸当量（Dietary Folate Equivalent, DFE）*：}

食物中的叶酸（天然）与叶酸补充剂（合成）的生物利用率不同：
\[
\imp{\text{DFE} (\mu\text{g}) = \text{天然食物中叶酸} (\mu\text{g}) + 1.7 \times \text{补充剂/强化食品中叶酸} (\mu\text{g})}
\]

\textbf{参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成人 RNI：\imp{400 μg DFE/d}
    \item 孕妇：600 μg DFE/d；乳母：550 μg DFE/d
    \item UL（可耐受最高摄入量）：\imp{1000 μg DFE/d}
\end{itemize}
\end{keybox}

\textbf{食物来源：}深色绿叶蔬菜（菠菜、芥蓝）、动物肝脏、豆类、坚果、酵母。食物中叶酸易在加热和储存过程中损失。

% ==================== 第十一节 维生素B12 ====================
\section{维生素B$_{12}$（钴胺素，Cobalamin）}

\begin{keybox}
\textbf{维生素B$_{12}$的生理功能与缺乏：}

\textbf{特征：}
\begin{itemize}[leftmargin=*]
    \item 含有钴（Co）元素的维生素——唯一含金属元素的维生素
    \item \imp{仅存在于动物性食品中！}植物性食物不含维生素B$_{12}$（\keyword{严格素食者（Vegan）有缺乏风险}）
    \item 吸收需要内因子（Intrinsic Factor, IF）——胃壁细胞分泌
\end{itemize}

\textbf{生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 作为蛋氨酸合成酶的辅酶（与叶酸共同参与同型半胱氨酸→蛋氨酸）
    \item 参与神经系统的维护与髓鞘的形成
\end{enumerate}

\textbf{缺乏表现：}
\begin{enumerate}[leftmargin=*]
    \item \imp{巨幼红细胞性贫血：}与叶酸缺乏的贫血难以区分
    \item \textbf{神经系统损害：}脊髓后索和侧索退行性变（亚急性联合变性），而叶酸缺乏不引起神经系统损害——\imp{是与叶酸缺乏贫血的鉴别要点}！
\end{enumerate}
\end{keybox}

\begin{keybox}
\textbf{参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成人 AI：\imp{2.4 μg/d}
    \item 孕妇：2.6 μg/d；乳母：2.8 μg/d
\end{itemize}
\end{keybox}

\textbf{食物来源：}动物肝脏、肉类、鱼类、蛋类、奶类。植物性食物基本不含VitB$_{12}$。

% ==================== 第十二节 维生素C ====================
\section{维生素C（抗坏血酸，Ascorbic Acid）}

\begin{keybox}
\textbf{维生素C的生理功能*：}

\textbf{特征：}水溶性维生素，L-型具有生物活性。极易被氧化（尤其在中性和碱性环境），是一种强还原剂。

\textbf{生理功能*：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{抗氧化作用：}保护细胞免受自由基的氧化损伤，可还原VitE的氧化形式使其重新具有活性
    \item \textbf{参与胶原蛋白的合成：}维持\keyword{脯氨酸羟化酶}和\keyword{赖氨酸羟化酶}的活性（羟化反应以VitC为辅因子），促进前胶原中脯氨酸和赖氨酸的羟化——是构成结缔组织、血管壁、骨骼基质的基础
    \item \textbf{促进铁的吸收：}将Fe$^{3+}$还原为Fe$^{2+}$，并与之形成可溶螯合物促进其吸收
    \item \textbf{促进叶酸转化为四氢叶酸（THF）：}维持叶酸的活性形式
    \item \textbf{参与胆固醇代谢：}促进胆固醇向胆汁酸的转化
    \item \textbf{增强免疫功能：}促进抗体生成、增强白细胞吞噬功能
\end{enumerate}
\end{keybox}

\subsection{维生素C缺乏}

\begin{keybox}
\textbf{维生素C缺乏——坏血病（Scurvy）*：}

因胶原蛋白合成障碍，导致结缔组织和血管壁损伤。典型表现：
\begin{enumerate}[leftmargin=*]
    \item \textbf{牙龈出血、牙龈炎、牙齿松动}
    \item \textbf{皮下出血（瘀斑、紫癜、毛囊角化伴出血）}
    \item \textbf{伤口愈合不良}
    \item 关节痛、贫血、乏力、免疫功能下降
\end{enumerate}

维生素C缺乏潜伏期约3～4个月（因体内有少量储存）。
\end{keybox}

\subsection{参考摄入量与食物来源}

\begin{keybox}
\textbf{维生素C的膳食参考摄入量：}
\begin{itemize}[leftmargin=*]
    \item 成人 RNI：\imp{100 mg/d}
    \item 孕妇早期：100 mg/d；中、晚期：115 mg/d
    \item 乳母：150 mg/d
    \item UL（可耐受最高摄入量）：1000 mg/d
\end{itemize}
\end{keybox}

\textbf{食物来源：}新鲜水果（猕猴桃、柑橘、草莓、鲜枣等）和蔬菜（西蓝花、青椒、番茄）含量丰富。注意维生素C极易热破坏和氧化损失，因此\imp{新鲜生食为最佳摄取方式}。

% ==================== 第十三节 其他维生素（简略） ====================
\section{其他维生素}

\subsection{泛酸（Pantothenic Acid，维生素B$_5$）}
是\keyword{辅酶A（CoA）}和酰基载体蛋白（ACP）的组成成分，在糖、脂肪和蛋白质代谢中起核心作用。广泛存在于各类食物中（Pan-thenic意为"到处存在"），缺乏罕见。成人AI：5.0 mg/d。

\subsection{生物素（Biotin，维生素B$_7$）}
是多种羧化酶（丙酮酸羧化酶、乙酰辅酶A羧化酶等）的辅酶，参与CO$_2$固定反应。缺乏极为罕见，除非长期摄入大量生鸡蛋清（含\keyword{抗生物素蛋白（Avidin）}，可与生物素紧密结合无法吸收，导致"蛋白伤害"）。成人AI：40 μg/d。

% ==================== 第十四节 类维生素 ====================
\section{类维生素}

\subsection{肉碱（Carnitine）}
在肝和肾中由赖氨酸和蛋氨酸合成。核心功能是作为长链脂肪酸进入线粒体基质进行$\beta$-氧化的载体（"脂肪酸运输穿梭系统"）。成人AI：200 mg/d。食物来源：红肉最丰富。

\subsection{牛磺酸（Taurine）}
含硫$\beta$-氨基酸，游离存在于组织中。功能：结合胆汁酸、视网膜发育、渗透压调节、抗氧化。对早产儿是条件必需营养素。

\subsection{肌醇（Inositol）}
参与细胞信号转导（肌醇磷脂途径），促进脂肪代谢。成人AI：1000 mg/d。来源：水果、豆类、谷类。

% ==================== 第十五节 水 ====================
\section{水}

\subsection{水的分布}

人体总含水量约占体重的\imp{50\%～70\%}，随年龄增大含水量下降、随体脂增加含水量下降。
\begin{itemize}[leftmargin=*]
    \item 细胞内液：约占总体水量的 2/3
    \item 细胞外液：约占总体水量的 1/3（包括血液、组织间液、淋巴液等）
\end{itemize}

\subsection{水的生理功能}

\begin{enumerate}[leftmargin=*]
    \item \textbf{构成人体组织：}是人体含量最多的成分
    \item \textbf{作为溶剂和运载工具：}运送营养物质和代谢产物
    \item \textbf{参与物质代谢：}直接参与水解、水合等化学反应
    \item \textbf{调节体温：}通过排汗散热维持体温恒定
    \item \textbf{润滑作用：}关节液、胸腔液、腹腔液等起缓冲保护作用
\end{enumerate}

\subsection{水的平衡}

成人每日水需要量约\imp{2500 mL}。来源：饮水约1200mL、食物水约1000mL、内生水（代谢水）约300mL。排出：尿约1500mL、皮肤蒸发约500mL、肺呼出约350mL、粪便约150mL。

\section{复习题}

\begin{enumerate}[leftmargin=*, itemsep=8pt]
    \item \textbf{【名词解释】}维生素、视黄醇当量（RE）、$\beta$-胡萝卜素、夜盲症、1,25-(OH)$_2$D$_3$、烟酸当量（NE）、膳食叶酸当量（DFE）、癞皮病（Pellagra）、坏血病（Scurvy）、巨幼红细胞性贫血
    \item \textbf{【简答题】}比较脂溶性维生素和水溶性维生素的特点（至少3条）。
    \item \textbf{【简答题】}简述维生素A的生理功能、缺乏的表现（由轻到重）及主要食物来源。
    \item \textbf{【简答题】}写出视黄醇当量（RE）的计算公式。
    \item \textbf{【简答题】}简述维生素D的来源、活化过程及生理功能。
    \item \textbf{【简答题】}简述维生素B$_1$、B$_2$和烟酸（B$_3$）各自缺乏的特征性表现。
    \item \textbf{【简答题】}简述叶酸的生理功能及缺乏的主要表现（含神经管畸形）。
    \item \textbf{【简答题】}写出膳食叶酸当量（DFE）的计算公式。
    \item \textbf{【简答题】}比较叶酸缺乏与维生素B$_{12}$缺乏引起的贫血有何异同？
    \item \textbf{【简答题】}简述维生素C的主要生理功能及缺乏病（坏血病）的典型表现。
\end{enumerate}

\subsection*{参考答案要点}

\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item 见本章各概念定义框。
    \item 脂溶性：需脂肪和胆汁协助吸收、可体内储存缺乏症状出现慢、摄入过量可引起中毒（VitK除外）。水溶性：直接吸收、体内储存极少、缺乏症状出现快、几乎不中毒（多余从尿排泄）。
    \item 生理功能：①维持正常视觉（视紫红质）；②促进细胞生长分化；③维护上皮组织健康；④增强免疫；⑤$\beta$-胡萝卜素抗氧化。缺乏：暗适应↓→夜盲症→干眼病→角膜软化→失明；皮肤干燥角化、生长迟缓、免疫下降。来源：动物肝脏、鱼肝油、蛋奶及深色蔬菜水果（$\beta$-胡萝卜素）。
    \item 总RE（$\mu$g）= 视黄醇（$\mu$g）+ $\beta$-胡萝卜素（$\mu$g）$\times$ 0.167（即 1 $\mu$g $\beta$-胡萝卜素=0.167 $\mu$g RE）。
    \item 来源：VitD$_2$（植物）、VitD$_3$（皮肤7-脱氢胆固醇+UVB）。活化：肝→25-(OH)D$_3$→肾→1,25-(OH)$_2$D$_3$（活性形式）。功能：促肠钙吸收、肾小管重吸收钙磷、骨钙化/脱钙调节、免疫调节。
    \item VB$_1$：脚气病（干性：周围神经炎；湿性：心衰水肿）；VB$_2$：口腔-生殖系统综合征（口角炎、唇炎、舌炎、脂溢性皮炎）；烟酸：癞皮病-3D（皮炎Dermatitis、腹泻Diarrhea、痴呆Dementia）。
    \item 功能：THF为一碳单位载体，参与核苷酸合成、氨基酸代谢（Hcy→Met）。缺乏：巨幼红细胞性贫血、胎儿神经管畸形（脊柱裂/无脑儿）、高同型半胱氨酸血症。
    \item DFE（$\mu$g）= 天然食物中叶酸（$\mu$g）+ 1.7 $\times$ 补充剂/强化食品中叶酸（$\mu$g）。
    \item 同：均可引起巨幼红细胞性贫血。异：VitB$_{12}$缺乏伴有神经系统损害（脊髓后索、侧索变性），而叶酸缺乏不引起神经病变。VitB$_{12}$仅存在于动物性食品。
    \item 生理功能：抗氧化、促进胶原蛋白合成（脯/赖氨酸羟化辅因子）、促进铁吸收、促进叶酸→THF、免疫增强。缺乏——坏血病：牙龈出血、皮下瘀斑/紫癜、伤口愈合不良、贫血。
\end{enumerate}
