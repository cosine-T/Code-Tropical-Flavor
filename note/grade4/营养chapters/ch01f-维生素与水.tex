% ==================== 第一节 维生素概述 ====================
\section{维生素概述}

\begin{defbox}
\textbf{维生素（Vitamin）}是维持机体正常代谢和生理功能所必需的一类微量低分子有机化合物。
\end{defbox}

\begin{keybox}
\textbf{维生素的共同特点：}
\begin{enumerate}[leftmargin=*]
    \item 以本体或前体形式存在于天然食物
    \item 必须经常由食物提供
    \item 需要量甚微，但在调节机体代谢方面作用重要
    \item 不构成组织、也不供能
    \item 多以辅酶或辅基的形式发挥功能
    \item 有的具有几种结构相近、活性相同的化合物
\end{enumerate}
\end{keybox}

\begin{defbox}
\textbf{维生素的分类：}

\textbf{脂溶性维生素：}VA、VD、VE、VK。

\textbf{水溶性维生素：}VB族、VC。
\end{defbox}

\begin{defbox}
\textbf{维生素缺乏的原因：}
\begin{itemize}[leftmargin=*]
    \item 摄入不足
    \item 吸收利用降低
    \item 维生素需要量相对增高
\end{itemize}

\textbf{按缺乏原因分：}
\begin{itemize}[leftmargin=*]
    \item \textbf{原发性维生素缺乏：}膳食中维生素供给不足或维生素生物利用率过低
    \item \textbf{继发性维生素缺乏：}生理或病理原因妨碍维生素的消化吸收利用，或因需要量增加、排泄或破坏增多而引起的条件性维生素缺乏
\end{itemize}

\textbf{按缺乏程度分：}
\begin{itemize}[leftmargin=*]
    \item \textbf{临床缺乏：}出现临床症状
    \item \textbf{亚临床缺乏/维生素边缘缺乏：}轻度缺乏常不出现临床症状，但一般可降低劳动效率及对疾病的抵抗力
\end{itemize}
\end{defbox}

% ==================== 第二节 维生素A ====================
\section{维生素A（视黄醇，Retinol）}

\begin{defbox}
\textbf{维生素A/视黄醇：}含$\beta$-白芷酮环的多烯基结构，具有视黄醇生物活性的一大类物质，包括已形成的维生素A和维生素A原及其代谢产物。
\end{defbox}

\subsection{理化性质与代谢}

\begin{tipbox}
\textbf{维生素A的理化性质：}
\begin{itemize}[leftmargin=*]
    \item 易被氧化，对紫外线敏感
    \item 对酸和碱稳定，一般烹调和罐头加工不易破坏
    \item 脂肪酸败可引起其严重破坏
\end{itemize}

\textbf{吸收与代谢：}
视黄醇 $\rightarrow$ 视黄醛 $\rightarrow$ 视黄酸。视黄醇和视黄醛可相互转化，视黄醛变为视黄酸不可逆。9/11-顺式视黄醛是体内主要生物活性形式。
\end{tipbox}

\subsection{生理功能}

\begin{keybox}
\textbf{维生素A的生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 视觉，维持视力
    \item 细胞生长和分化，促进生长发育
    \item 维护上皮组织细胞健康
    \item 抗氧化作用
    \item 免疫功能
    \item 抑制肿瘤生长
\end{enumerate}
\end{keybox}

\subsection{缺乏与过量}

\begin{keybox}
\textbf{维生素A缺乏：}
\begin{itemize}[leftmargin=*]
    \item 暗适应能力下降 $\rightarrow$ 夜盲症 $\rightarrow$ 眼干燥症 $\rightarrow$ 失明（儿童产生比奥斑）
    \item 上皮组织干燥、增生、角化
    \item 影响生长发育和生殖功能
\end{itemize}

\textbf{维生素A过量：}
妊娠期特别是孕早期，对胎儿最重要的危害是发生先天畸形。
\end{keybox}

\subsection{参考摄入量与食物来源}

\begin{keybox}
\textbf{参考摄入量：}

单位：视黄醇当量RE。膳食中总视黄醇当量($\mu$gRE) = 视黄醇($\mu$g) + $\beta$-胡萝卜素($\mu$g) $\times$ 0.167 + 其他维生素A原($\mu$g) $\times$ 0.084。

\begin{itemize}[leftmargin=*]
    \item 0.167：1$\mu$g $\beta$-胡萝卜素 = 0.167$\mu$gRE，吸收率为1/3
    \item 0.084：1$\mu$g 其他维生素A原 = 0.084$\mu$gRE
    \item 1IU维生素A = 0.31$\mu$gRE
    \item 男性770$\mu$gRAE/d，女性660$\mu$gRAE/d
\end{itemize}

\textbf{食物来源：}肝、鱼肝油、蛋黄、乳品、深色果蔬。
\end{keybox}

% ==================== 第三节 维生素D ====================
\section{维生素D（钙化醇，Calciferol）}

\begin{defbox}
\textbf{维生素D/钙化醇：}含环戊氢烯菲环结构，具有钙化醇生物活性的一大类物质，以维生素D$_2$和D$_3$最常见。
\end{defbox}

\begin{keybox}
\textbf{维生素D的生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 促进小肠对Ca的吸收
    \item 促进肾小管对Ca、P的重吸收
    \item 对骨细胞有多种作用
    \item 通过维生素D内分泌系统调节血钙平衡
    \item 参与机体多种机能调节
\end{enumerate}

\textbf{评价机体维生素D水平的首选指标：}25-(OH)-D$_3$。
\end{keybox}

\begin{keybox}
\textbf{维生素D缺乏：}
\begin{itemize}[leftmargin=*]
    \item 佝偻病（婴儿）
    \item 骨质软化症
    \item 骨质疏松症
    \item 手足痉挛症
\end{itemize}

\textbf{维生素D过量（维生素D过多症）：}
食欲缺乏、体重减轻、恶心、呕吐、腹泻；多尿、肾小管钙化、肾功能减退；心血管系统异常；皮肤瘙痒；严重者可导致死亡。
\end{keybox}

\begin{keybox}
\textbf{参考摄入量：}10$\mu$g/d。

\textbf{食物来源：}肝、鱼肝油、蛋黄、乳品。
\end{keybox}

% ==================== 第四节 维生素E ====================
\section{维生素E（生育酚，Tocopherol）}

\begin{defbox}
\textbf{维生素E/生育酚：}含苯并二氢吡喃结构，具有$\alpha$-生育酚生物活性的一大类物质。
\end{defbox}

\begin{keybox}
\textbf{维生素E的生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 抗氧化
    \item 预防衰老
    \item 调节血小板黏附力和聚集作用
\end{enumerate}

\textbf{维生素E缺乏：}溶血性贫血。

\textbf{参考摄入量：}14mg/d。

\textbf{食物来源：}植物油、坚果、绿色蔬菜。
\end{keybox}

% ==================== 第五节 维生素K ====================
\section{维生素K（凝血维生素）}

\begin{tipbox}
\textbf{维生素K的生理功能：}
参与凝血因子合成，维持正常凝血功能。

\textbf{维生素K缺乏：}凝血障碍、出血倾向。新生儿因肠道菌群未建立且母乳中VitK含量低，易出现新生儿出血症。

\textbf{食物来源：}绿色绿叶蔬菜；肠道细菌也能合成一部分。
\end{tipbox}

% ==================== 第六节 维生素B1 ====================
\section{维生素B$_1$（硫胺素，Thiamin）}

\begin{keybox}
\textbf{维生素B$_1$/硫胺素：}抗脚气病因子和抗神经炎因子。理化性质：酸性环境下较稳定，中性和碱性环境不稳定，对亚硫酸盐极敏感。体内活性形式为TPP（焦磷酸硫胺素）。

\textbf{生理作用：}碳水化合物和能量代谢。
\end{keybox}

\begin{keybox}
\textbf{维生素B$_1$缺乏——脚气病：}
\begin{itemize}[leftmargin=*]
    \item \textbf{干性脚气病：}周围神经炎
    \item \textbf{湿性脚气病：}水肿和心力衰竭
    \item \textbf{混合型脚气病：}既有神经炎又有水肿和心力衰竭
\end{itemize}

\textbf{参考摄入量：}男性1.4mg/d、女性1.2mg/d。

\textbf{食物来源：}瘦肉、谷类、豆类、酵母。
\end{keybox}

% ==================== 第七节 维生素B2 ====================
\section{维生素B$_2$（核黄素，Riboflavin）}

\begin{keybox}
\textbf{维生素B$_2$/核黄素：}具有一个核糖醇侧链的异咯嗪类衍生物。理化性质：碱性溶液易破坏，游离型VB$_2$对紫外光高度敏感。

\textbf{生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 参与生物氧化和能量代谢（FAD和FMN）
    \item 参与烟酸和维生素B$_6$的代谢
\end{enumerate}
\end{keybox}

\begin{keybox}
\textbf{维生素B$_2$缺乏——口腔生殖系统综合征：}
\begin{itemize}[leftmargin=*]
    \item \textbf{眼：}眼球结膜充血、角膜周围血管增生
    \item \textbf{口腔：}唇炎、口角炎、舌炎
    \item \textbf{皮肤：}脂溢性皮炎、阴囊皮炎
\end{itemize}
还可引起缺铁性贫血。

\textbf{参考摄入量：}男性1.4mg/d、女性1.2mg/d。

\textbf{食物来源：}肝、肉类、蛋黄、豆类。
\end{keybox}

% ==================== 第八节 维生素B6 ====================
\section{维生素B$_6$（吡哆醇，Pyridoxine）}

\begin{defbox}
\textbf{维生素B$_6$缺乏：}
\begin{itemize}[leftmargin=*]
    \item 低色素小细胞性贫血
    \item 地图舌
\end{itemize}
\end{defbox}

% ==================== 第九节 维生素B12 ====================
\section{维生素B$_{12}$（钴胺素，Cobalamin）}

\begin{defbox}
\textbf{维生素B$_{12}$/钴胺素：}

\textbf{生理功能：}以两种辅酶形式发挥作用——甲基B$_{12}$（甲基钴胺素）和辅酶B$_{12}$。

\textbf{缺乏：}
\begin{itemize}[leftmargin=*]
    \item 巨幼红细胞性贫血
    \item 神经系统损害
    \item 高同型半胱氨酸血症
\end{itemize}

\textbf{参考摄入量：}2.4$\mu$g/d。

\textbf{食物来源：}动物内脏、肉类、鱼、蛋类。
\end{defbox}

% ==================== 第十节 维生素C ====================
\section{维生素C（抗坏血酸，Ascorbic Acid）}

\begin{defbox}
\textbf{维生素C/抗坏血酸（Ascorbic Acid）：}含6个碳原子的酸性多羟基化合物。
\end{defbox}

\subsection{理化性质}

\begin{tipbox}
\textbf{维生素C的理化性质：}
\begin{itemize}[leftmargin=*]
    \item 强还原剂
    \item 最不稳定，有氧或酸性环境极易氧化
\end{itemize}
\end{tipbox}

\subsection{生理功能}

\begin{keybox}
\textbf{维生素C的生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 抗氧化
    \item 清除自由基
    \item 改善铁、钙和叶酸的利用
    \item 促进类固醇代谢
    \item 作为羟化过程底物和酶的辅因子
    \item 参与合成神经递质
\end{enumerate}

\textbf{反映维生素C储备水平的生化指标：}WBC维生素C浓度。
\end{keybox}

\subsection{缺乏}

\begin{keybox}
\textbf{维生素C缺乏——坏血病：}
\begin{itemize}[leftmargin=*]
    \item 前驱症状：全身乏力、食欲减退
    \item 出血
    \item 牙龈炎
    \item 骨质疏松
\end{itemize}

\textbf{参考摄入量：}100mg/d。

\textbf{食物来源：}新鲜蔬果。
\end{keybox}

% ==================== 第十一节 叶酸 ====================
\section{叶酸（Folate，维生素B$_9$）}

\begin{keybox}
\textbf{维生素B$_9$/叶酸：}

\textbf{生理功能：}只有四氢叶酸有生理功能，作为一碳单位的载体，携带一碳基团（甲酰基、亚甲基和甲基），参与嘌呤和嘧啶核苷酸的合成。

\textbf{缺乏：}巨幼红细胞性贫血。

\textbf{参考摄入量：}400$\mu$g/d。

\textbf{食物来源：}肝、蔬菜、水果、坚果。
\end{keybox}

% ==================== 第十二节 烟酸 ====================
\section{烟酸（Niacin，维生素B$_3$，维生素PP）}

\begin{defbox}
\textbf{烟酸/B$_3$/维生素PP：}

\textbf{生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 构成NAD和NADP
    \item 参与碳水化合物、脂肪和蛋白质代谢
    \item 维护皮肤健康
\end{enumerate}

\textbf{缺乏——"3D"症状：}
\begin{itemize}[leftmargin=*]
    \item 皮炎（Dermatitis）
    \item 腹泻（Diarrhea）
    \item 痴呆（Dementia）
\end{itemize}

\textbf{参考摄入量：}男性15mgNE/d、女性12mgNE/d。
\end{defbox}

% ==================== 第十三节 泛酸 ====================
\section{泛酸（Pantothenic Acid，维生素B$_5$）}

\begin{defbox}
\textbf{泛酸（维生素B$_5$）：}是辅酶A（CoA）和酰基载体蛋白（ACP）的组成成分。

\textbf{生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 辅酶A参与三大营养素的乙酰基转移反应，是糖类、脂肪和蛋白质代谢所必需
    \item ACP参与脂肪酸合成
\end{enumerate}

\textbf{缺乏：}极为罕见（食物中广泛存在）；"灼热足综合征"（burning foot syndrome）——足部灼热感、感觉异常

\textbf{食物来源：}广泛分布于动植物食物——动物肝脏、肉类、蛋、全谷物、豆类、蘑菇。肠道细菌也能合成一部分。
\end{defbox}

% ==================== 第十四节 胆碱 ====================
\section{胆碱（Choline）}

\begin{defbox}
\textbf{胆碱：}曾归类为B族维生素，现被认为是条件必需营养素（体内可由蛋氨酸合成但量常不足）。

\textbf{生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 构成细胞膜的重要成分——磷脂酰胆碱（卵磷脂）和鞘磷脂的组成成分
    \item 参与神经递质乙酰胆碱的合成
    \item 参与一碳单位代谢（作为甲基供体，与蛋氨酸、叶酸、VitB$_{12}$代谢密切相关）
    \item 参与脂蛋白代谢，预防肝脏脂肪蓄积（脂肪肝）
\end{enumerate}

\textbf{缺乏：}肝脏脂肪浸润（脂肪肝）、肌肉损伤

\textbf{食物来源：}蛋黄、动物肝脏、肉类、鱼类、大豆、花生。成人AI：男性500mg/d，女性400mg/d。
\end{defbox}

% ==================== 第十五节 生物素 ====================
\section{生物素（Biotin，维生素B$_7$，维生素H）}

\begin{defbox}
\textbf{生物素（维生素B$_7$）：}是羧化酶的辅酶，参与CO$_2$的固定和转移反应。

\textbf{生理功能：}参与糖异生、脂肪酸合成和氨基酸（亮氨酸）代谢。

\textbf{缺乏：}极为罕见。可见于长期摄入生蛋清者（生蛋清含抗生物素蛋白avidin，与生物素结合阻止其吸收）——表现为脱发、皮肤干燥脱屑、神经系统症状（抑郁、嗜睡、幻觉）

\textbf{食物来源：}广泛存在——动物肝脏、蛋黄、坚果、豆类。肠道细菌也能合成。
\end{defbox}

% ==================== 第十六节 类维生素 ====================
\section{类维生素（Vitamin-like Compounds）}

\begin{keybox}
\textbf{肉碱（Carnitine，左旋肉碱/L-Carnitine）：}

\textbf{生理功能（掌握）：}脂肪酸$\beta$-氧化的必需转运载体，将长链脂肪酸从细胞质转运至线粒体基质进行$\beta$-氧化供能。

\textbf{缺乏（掌握）：}原发性肉碱缺乏（遗传性转运蛋白缺陷）和继发性缺乏（早产儿、长期TPN、肾透析）——表现为肌无力、低血糖、心肌病、脂肪肝。

\textbf{过量（掌握）：}口服补充（2-4g/d）通常耐受良好，偶尔出现胃肠道不适（恶心、腹泻）、体味异常（鱼腥味）。安全性较高，无UL。

\textbf{食物来源（熟悉）：}红肉（含量最高）、奶制品。体内也可由赖氨酸和蛋氨酸合成。
\end{keybox}

\begin{keybox}
\textbf{牛磺酸（Taurine）：}

\textbf{生理功能（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item 结合胆汁酸——与胆酸结合形成牛磺胆酸，促进脂肪消化吸收
    \item 维持视网膜功能——视网膜中含量丰富，对光感受器细胞的发育和功能维持至关重要
    \item 神经系统发育——促进胎儿和新生儿脑和神经系统发育
    \item 渗透压调节——在中枢神经系统和心肌中作为渗透调节物质
    \item 抗氧化和抗炎——清除自由基，保护细胞膜；调节钙稳态
\end{enumerate}

\textbf{缺乏（掌握）：}成人缺乏罕见（体内可合成部分）。早产儿合成能力不足，缺乏可致视网膜发育异常、胆汁淤积。猫科动物必须从食物获取牛磺酸（缺乏致视网膜变性、扩张型心肌病）。

\textbf{过量（掌握）：}口服补充通常安全，无UL。

\textbf{食物来源（熟悉）：}动物性食物——海产品（贝类、鱼类含量最高）、禽肉、畜肉。植物性食物基本不含。
\end{keybox}

\begin{tipbox}
\textbf{肌醇（Inositol）：}

\textbf{生理功能（了解）：}细胞膜磷脂（磷脂酰肌醇）的组成成分；参与细胞信号转导（作为IP3和DAG的前体）；有利于脂肪代谢，防止肝脏脂肪蓄积。

\textbf{缺乏（了解）：}体内可合成，缺乏罕见。动物实验中可见生长迟缓和脂肪肝。

\textbf{过量（了解）：}大量摄入（>12g/d）可致胃肠道不适（腹泻）。

\textbf{食物来源（了解）：}全谷物、坚果、豆类、柑橘类水果。体内可由葡萄糖合成。

\textbf{柠檬素（Citrin/维生素P，了解）：}曾被认为是维生素（维生素P），现归为多酚类化合物的类黄酮成分。主要生理功能为增强毛细血管壁韧性、降低血管通透性、抗氧化，防止坏血病（与维生素C协同）。缺乏：毛细血管脆性增加，易出血倾向。食物来源：柑橘类水果（果皮）、荞麦、茶叶、红酒等。%
% 注：柠檬素（Citrin/维生素P）在参考文献中仅有条目名称，详细内容为基于营养学常识的合理补充。
\end{tipbox}

\section{水}

\begin{defbox}
\textbf{水的体内分布（了解）：}
\begin{itemize}[leftmargin=*]
    \item 水是人体含量最多的成分，约占体重的50\%$\sim$70\%（随年龄和体脂含量而异——新生儿约80\%，成年男性约60\%，成年女性约50\%\~{}55\%）
    \item 体内水主要分布在两大区室：
    \begin{itemize}[leftmargin=*]
        \item \textbf{细胞内液（ICF）：}约占总水量的2/3（约为体重的40\%），主要阳离子为K$^+$，主要阴离子为HPO$_4^{2-}$和蛋白质
        \item \textbf{细胞外液（ECF）：}约占总水量的1/3（约为体重的20\%），包括血浆（约占体重5\%）和组织间液（约占体重15\%），主要阳离子为Na$^+$，主要阴离子为Cl$^-$和HCO$_3^-$
    \end{itemize}
\end{itemize}
\end{defbox}

\begin{defbox}
\textbf{水的生理功能：}
\begin{enumerate}[leftmargin=*]
    \item 参与生化反应
    \item 物质运输
    \item 调节体温
    \item 润滑作用
\end{enumerate}

\textbf{缺乏与过量：}
\begin{itemize}[leftmargin=*]
    \item 缺水 $\rightarrow$ 口渴、粘膜干燥、消化液分泌减少、食欲减退、精神不振、代谢减慢等
    \item 失水达体重的10\% $\rightarrow$ 影响生理功能
    \item 失水达体重的20\% $\rightarrow$ 生命无法维持
    \item 过量 $\rightarrow$ 稀释消化液，不利于消化，饭前饭后不宜大量饮水
\end{itemize}

\textbf{水的来源：}饮水 + 食物水。提倡饮用白开水和茶水，不喝或少喝含糖饮料。

\textbf{水的需要量：}成年男性1700ml/d，成年女性1500ml/d。
\end{defbox}

\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item 见本章各概念定义框。
    \item \textbf{共同特点：}①以本体或前体形式存在于天然食物；②必须经常由食物提供；③需要量甚微，但在调节机体代谢方面作用重要；④不构成组织、也不供能；⑤多以辅酶或辅基的形式发挥功能；⑥有的具有几种结构相近、活性相同的化合物。\textbf{分类：}脂溶性维生素（VA、VD、VE、VK）和水溶性维生素（VB族、VC）。
    \item \textbf{缺乏原因：}①摄入不足；②吸收利用降低；③维生素需要量相对增高。\textbf{按原因分：}原发性缺乏（膳食中维生素供给不足或生物利用率过低）、继发性缺乏（生理或病理原因妨碍维生素的消化吸收利用，或因需要量增加、排泄或破坏增多）。\textbf{按缺乏程度分：}临床缺乏（出现临床症状）、亚临床缺乏/边缘缺乏（轻度缺乏常不出现临床症状，但一般可降低劳动效率及对疾病的抵抗力）。
