\chapter{食物中的生物活性成分}
\setcounter{chapter}{2}
\chapterintro{
\keyword{掌握}植物化学物的概念、分类和生物学作用；掌握多酚类化合物、有机硫化物、植物雌激素的生物学作用；掌握SPL（特定建议值）的概念及其与营养素DRIs的区别。\keyword{熟悉}类胡萝卜素、植物固醇的作用。\keyword{了解}其他生物活性成分。
}

% ----- 第一节 生物活性成分概述 -----
\section{概述}

\begin{defbox}
\textbf{生物活性成分（Bioactive Food Components）：}食物中除了营养素以外的，对人体健康有益的微量非营养素成分。过去曾被称为"非营养素活性成分"，现已统一称为"食物中的生物活性成分"。这类物质虽非维持机体生长发育所必需，但对健康维护、生理调节和疾病预防具有重要意义。
\end{defbox}

\subsection{分类}

食物中的生物活性成分按来源可分为两大类：

\begin{enumerate}[leftmargin=*]
    \item \textbf{植物来源的生物活性成分（植物化学物，Phytochemicals）：}
    \begin{itemize}[leftmargin=*]
        \item 多酚类（类黄酮、酚酸等）
        \item 有机硫化物（异硫氰酸盐、二硫醇硫酮、蒜素等）
        \item 萜类化合物（类胡萝卜素、皂苷、单萜类等）
        \item 植物雌激素（异黄酮、木酚素、芪类等）
        \item 蛋白酶抑制剂
        \item 植酸
        \item 植物固醇
    \end{itemize}
    \item \textbf{动物来源的生物活性成分：}
    \begin{itemize}[leftmargin=*]
        \item 辅酶Q（泛醌）
        \item $\gamma$-氨基丁酸（GABA）
        \item 褪黑素（Melatonin）
        \item 左旋肉碱（L-肉碱）
        \item 硫辛酸
    \end{itemize}
\end{enumerate}

\subsection{植物化学物的代谢特点}

\begin{keybox}
\textbf{植物化学物在体内的代谢过程：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{吸收率低：}多数植物化学物在肠道的吸收率较低，通常为1\%\~{}10\%。
    \item \textbf{I 相代谢（细胞色素P450酶系）：}在肝脏和小肠上皮细胞中，通过CYP450酶系（CYP1A2、CYP3A4等）进行氧化、还原、水解反应。产物往往水溶性增加但可能产生活性中间体。
    \item \textbf{II 相代谢（结合反应）：}与葡萄糖醛酸、硫酸盐、甘氨酸、谷胱甘肽等结合（UGT、SULT、GST酶系），水溶性显著增加。
    \item \textbf{排泄途径：}主要通过尿液（肾脏）和胆汁（粪便）排出体外。
    \item \textbf{肠道菌群代谢：}结肠菌群对未吸收的植物化学物进行发酵转化，产生小分子活性代谢物，可被重新吸收。
\end{enumerate}
\end{keybox}

\begin{tipbox}
\textbf{植物化学物的"低吸收、高活性"特点：}体内浓度仅在纳摩尔至低微摩尔水平，远低于体外实验的有效浓度。提示其生理作用可能更多通过温和调节细胞信号通路（如Nrf2/ARE通路、NF-$\kappa$B通路）而非直接的化学作用实现。
\end{tipbox}

\subsection{生物活性成分的摄入量指导——SPL}

\begin{keybox}
\textbf{特定建议值（Specific Proposed Levels, SPL）：}

SPL是DRIs（膳食营养素参考摄入量）指标体系的第七项，专门针对\textbf{非营养素生物活性成分}（主要为植物化学物）而设立。其定义为：以降低成年人膳食相关非传染性慢性病（NCD）风险为目标的\textbf{其他膳食成分}每日摄入量。

\textbf{SPL与营养素DRIs的区别：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{营养素（蛋白质、维生素、矿物质等）：}使用EAR/RNI/AI/UL指标体系。EAR满足50\%个体、RNI满足97\%\~{}98\%个体、UL设定安全上限。
    \item \textbf{非营养素活性成分（植物化学物等）：}因其非必需、非缺乏性特点，\textbf{不存在EAR和RNI}，仅以SPL给出以降低慢病风险为目标的建议摄入量。
    \item \textbf{本质区别：}SPL不回答"需要多少"，而是回答"摄入多少有助于降低慢病风险"。
\end{enumerate}

\textbf{SPL的制定依据：}
\begin{itemize}[leftmargin=*]
    \item 系统综述和Meta分析（前瞻性队列研究为主）
    \item 剂量-反应关系分析
    \item 基于食物而非补充剂的健康效应证据（强调食物基质的协同作用）
    \item 安全性评估（虽无UL，但仍需考虑过量摄入的潜在不良效应）
\end{itemize}

\textbf{已提出SPL的植物化学物举例（中国DRIs 2023版）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{大豆异黄酮：}SPL为每天若干毫克（以苷元计），主要基于改善围绝经期症状和预防骨质疏松的证据。
    \item \textbf{植物固醇：}SPL为每天若干克，基于降低血清LDL胆固醇的证据。
    \item \textbf{原花青素、花色苷、番茄红素、叶黄素等：}亦有相应的SPL值被提出或正在论证中。
\end{itemize}

\textbf{意义：}SPL的设立标志着营养科学从"预防缺乏"向"促进健康和预防慢病"的范式转变，承认了植物化学物在膳食营养指导中的独立地位，为膳食指南中"多吃蔬果、全谷物、大豆"的建议提供了剂量化依据。
\end{keybox}

% ----- 第二节 类胡萝卜素 -----
\section{类胡萝卜素（Carotenoids）}

\begin{keybox}
\textbf{定义与分类：}类胡萝卜素是广泛存在于果蔬中的脂溶性植物色素，迄今已发现700余种，其中约50种存在于人类膳食中，10\~{}20种可在人血清中检出。

分为两大类：
\begin{enumerate}[leftmargin=*]
    \item \textbf{胡萝卜素类（Carotenes）：}纯碳氢化合物，不含氧。包括 $\alpha$-胡萝卜素、$\beta$-胡萝卜素、$\gamma$-胡萝卜素、番茄红素（Lycopene）。
    \item \textbf{叶黄素类（Xanthophylls）：}含氧的类胡萝卜素衍生物。包括叶黄素（Lutein）、玉米黄素（Zeaxanthin）、$\beta$-隐黄质（$\beta$-Cryptoxanthin）。
\end{enumerate}
\end{keybox}

\subsection{主要生物学作用}

\begin{enumerate}[leftmargin=*]
    \item \textbf{抗氧化作用：}
    \begin{itemize}[leftmargin=*]
        \item 类胡萝卜素是有效的抗氧化剂，能淬灭单线态氧（$^{1}O_{2}$）和清除自由基。
        \item \keyword{番茄红素}的淬灭单线态氧能力在类胡萝卜素中最强，是$\beta$-胡萝卜素的2倍以上。
        \item $\beta$-胡萝卜素在低氧分压下抗氧化效果更佳。
    \end{itemize}
    \item \textbf{维生素A原活性：}
    \begin{itemize}[leftmargin=*]
        \item $\beta$-胡萝卜素（全部反式异构体）的维生素A原活性最高（1分子$\rightarrow$2分子视黄醇），$\alpha$-胡萝卜素和$\beta$-隐黄质次之（1分子$\rightarrow$1分子视黄醇）。
        \item 番茄红素、叶黄素、玉米黄素\textbf{无}维生素A原活性。
    \end{itemize}
    \item \textbf{抗肿瘤作用：}增加富含类胡萝卜素蔬果的摄入可降低多种癌症（肺癌、乳腺癌、胃癌等）的发病风险。
    \item \textbf{免疫调节作用：}增强T淋巴细胞的增殖和NK细胞的活性。
    \item \textbf{视觉保护作用：}
    \begin{itemize}[leftmargin=*]
        \item \keyword{叶黄素}和\keyword{玉米黄素}是视网膜黄斑区（Macula）的专一构成色素，构成"黄斑色素"，可滤过蓝光，保护视网膜免受光损伤。
        \item 增加叶黄素和玉米黄素的摄入有助于降低年龄相关性黄斑变性（AMD）的风险。
    \end{itemize}
\end{enumerate}

\subsection{食物来源}

\begin{itemize}[leftmargin=*]
    \item $\beta$-胡萝卜素：深绿色和橙黄色蔬菜（胡萝卜、南瓜、菠菜、西兰花）
    \item 番茄红素：番茄及番茄制品（熟制后吸收率更高）、西瓜、粉红色葡萄柚
    \item 叶黄素/玉米黄素：深绿色叶菜（菠菜、羽衣甘蓝）、玉米、蛋黄
    \item $\beta$-隐黄质：橙黄色水果（橘子、木瓜、柿子）
\end{itemize}

\begin{tipbox}
\textbf{番茄红素的吸收特征：}番茄红素为反式结构，顺式异构体更易被吸收。加热和油脂加工（如在橄榄油中加热番茄）可促进反式$\rightarrow$顺式转化及从食物基质中释放，显著提高其生物利用度。
\end{tipbox}

% ----- 第三节 多酚类化合物 -----
\section{多酚类化合物（Polyphenols）}

\begin{keybox}
\textbf{定义与结构特征：}多酚类化合物是植物性食物中含量最丰富、研究最深入的植物化学物类别。基本结构特征为\textit{2-苯基色原酮（2-Phenylchromone）}核心骨架，即一个或多个芳香环连接多个酚羟基。按碳骨架结构可分为七大类别。
\end{keybox}

\subsection{分类}

\begin{enumerate}[leftmargin=*]
    \item \textbf{黄酮及黄酮醇类（Flavones and Flavonols）：}
    \begin{itemize}[leftmargin=*]
        \item 黄酮代表：芹菜素（Apigenin）、木犀草素（Luteolin）
        \item 黄酮醇代表：\keyword{槲皮素（Quercetin）}——膳食中最丰富的类黄酮之一；芦丁（Rutin）——槲皮素的芸香糖苷；山奈酚（Kaempferol）；杨梅素（Myricetin）
        \item 来源：洋葱、西兰花、蓝莓、苹果、红茶
    \end{itemize}
    \item \textbf{黄烷酮类（Flavanones）：}代表为柚皮素（Naringenin）、橙皮素（Hesperetin）。主要来源：柑橘类水果（橙子、葡萄柚）。
    \item \textbf{黄烷醇类（Flavanols/儿茶素类）：}代表为\keyword{儿茶素（Catechin）}、表儿茶素（Epicatechin）、表没食子儿茶素没食子酸酯（EGCG）。主要来源：绿茶（含量最丰富）、巧克力、苹果。
    \item \textbf{异黄酮类（Isoflavones）：}代表为\keyword{大豆苷（Daidzin）}、\keyword{染料木素（Genistein）}、葛根素（Puerarin）。主要来源：大豆及其制品。
    \item \textbf{双黄酮类（Biflavones）：}代表为银杏黄酮（Ginkgetin）。来源：银杏叶。
    \item \textbf{花色素类（Anthocyanins）：}代表为矢车菊素（Cyanidin）、天竺葵素（Pelargonidin）。来源：浆果（蓝莓、黑莓、草莓）、紫薯、紫甘蓝。
    \item \textbf{查尔酮类（Chalcones）：}来源：苹果、甘草。
\end{enumerate}

\subsection{主要生物学作用}

\begin{enumerate}[leftmargin=*]
    \item \textbf{抗氧化作用：}多酚类是膳食中最重要的抗氧化活性成分来源。酚羟基提供电子或氢原子直接清除自由基，并通过激活Nrf2/ARE通路诱导抗氧化酶（SOD、CAT、GPx、HO-1）的表达。
    \item \textbf{抗肿瘤作用：}\keyword{茶多酚}（尤其是EGCG）和\keyword{大豆异黄酮}的抗肿瘤作用证据最为充分。机制包括诱导II相解毒酶、抑制I相致癌物代谢酶、阻滞细胞周期、诱导凋亡、抑制血管生成。
    \item \textbf{心血管保护作用：}改善内皮功能、降低血压、抑制LDL氧化、抗血小板聚集。
    \item \textbf{抗炎作用：}抑制NF-$\kappa$B通路和COX-2、iNOS的表达。
    \item \textbf{调节肠道菌群：}促进有益菌（双歧杆菌、乳酸杆菌）增殖，抑制致病菌。
\end{enumerate}

\subsection{膳食摄入量}

\begin{tipbox}
\textbf{多酚类的日常摄入量参考：}总多酚摄入量约为0.1\~{}1.0 g/d，其中类黄酮为20\~{}70 mg/d。值得注意的是，多酚类的摄入量影响因素很多（食物种类、加工方式、分析方法等），数据的可比性有限。
\end{tipbox}

\textbf{主要食物来源：}绿茶、各种有色水果和蔬菜、大豆及其制品、巧克力、红酒、洋葱。

% ----- 第四节 植物雌激素 -----
\section{植物雌激素（Phytoestrogens）}

\begin{keybox}
\textbf{定义：}植物雌激素是植物来源的、能与哺乳动物雌激素受体（ER）结合并发挥类雌激素或抗雌激素效应的非甾体化合物。其特点是\textbf{双向调节作用}——当内源性雌激素水平较低时发挥雌激素样作用，当较高时发挥抗雌激素作用。
\end{keybox}

\subsection{分类}

\begin{enumerate}[leftmargin=*]
    \item \textbf{异黄酮类（Isoflavones）：}大豆苷（Daidzin）、染料木素（Genistein）、大豆黄素（Daidzein）。\textbf{主要来源：大豆及其制品}（豆粉、豆腐、豆浆），大豆中异黄酮含量约为0.1\%\~{}0.5\%。
    \item \textbf{木酚素类（Lignans）：}开环异落叶松脂素（Secoisolariciresinol）、罗汉松脂素（Matairesinol）。在肠道菌群作用下转化为肠内酯（Enterolactone）和肠二醇（Enterodiol）活性形式。\textbf{\keyword{亚麻籽}是木酚素含量最高的食物}，也含于全谷物、种子类。
    \item \textbf{香豆素类（Coumestans）：}香豆雌酚（Coumestrol）。来源：发芽豆类（苜蓿芽、绿豆芽）。
    \item \textbf{芪类（Stilbenes）：}代表为\keyword{白藜芦醇（Resveratrol）}，来源：葡萄/葡萄酒、花生、虎杖（中药）。
\end{enumerate}

\subsection{主要生物学作用}

\begin{enumerate}[leftmargin=*]
    \item \textbf{预防骨质疏松：}异黄酮通过选择性雌激素受体调节（SERM-like）作用，促进成骨细胞活性、抑制破骨细胞分化，减少骨丢失。
    \item \textbf{改善围绝经期症状：}可缓解潮热、盗汗等更年期症状（证据强度存在一定争议）。
    \item \textbf{抗氧化作用：}多酚羟基结构赋予其抗氧化能力。
    \item \textbf{心血管保护作用：}改善血脂谱、保护血管内皮功能。
    \item \textbf{抗肿瘤作用：}流行病学表明高大豆摄入人群（亚洲）的乳腺癌、前列腺癌发病率较低。机制可能涉及抗雌激素效应和直接抗增殖效应。
\end{enumerate}

\begin{exambox}
\textbf{考点提示：}植物雌激素的"双向调节"机制是考试重点——低雌激素水平时激动ER，高雌激素水平时拮抗ER（竞争性抑制内源性雌二醇）。这一特性使其成为SERM（选择性雌激素受体调节剂）的天然类似物。
\end{exambox}

% ----- 第五节 有机硫化物 -----
\section{有机硫化物（Organosulfur Compounds）}

\begin{keybox}
\textbf{定义与来源：}有机硫化物是一类主要存在于\keyword{葱属植物（Allium）}中的含硫植物化学物，以大蒜（Garlic）含量最为丰富，是其特征风味物质的来源。洋葱、大葱、韭菜中亦有分布。
\end{keybox}

\subsection{种类}

\begin{itemize}[leftmargin=*]
    \item \textbf{蒜素（Allicin，二烯丙基硫代亚磺酸酯）：}完整蒜瓣切碎或压碎后由蒜氨酸（Alliin）经蒜氨酸酶催化生成。在室温下不稳定，可进一步分解为多种含硫化合物（如二烯丙基二硫化物DADS、二烯丙基三硫化物DATS、阿霍烯Ajoene等）。
    \item \textbf{蒜氨酸（Alliin）：}大蒜完整的硫化物前体，无活性。
\end{itemize}

\subsection{主要生物学作用}

\begin{enumerate}[leftmargin=*]
    \item \textbf{抗菌作用：}\keyword{蒜素（Allicin）}具有广谱抗菌活性，其杀菌效力约相当于1\%青霉素。对革兰阳性和阴性细菌、真菌、寄生虫均有抑制作用。细菌不易对蒜素产生耐药性。
    \item \textbf{抗氧化作用：}直接清除自由基，并诱导抗氧化酶的表达。
    \item \textbf{调节血脂：}膳食补充大蒜或蒜素可降低血清总胆固醇（TC）、甘油三酯（TG）和低密度脂蛋白胆固醇（LDL），同时升高高密度脂蛋白胆固醇（HDL）。
    \item \textbf{抗血栓形成：}抑制血小板聚集，有一定的纤溶促进作用（类似阿司匹林但较弱）。
    \item \textbf{抗肿瘤作用：}
    \begin{itemize}[leftmargin=*]
        \item 阻断\keyword{亚硝胺（Nitrosamine）}的内源性合成——这是大蒜防癌的重要机制之一。
        \item 诱导II相解毒酶表达，加速致癌物的灭活和排泄。
    \end{itemize}
\end{enumerate}

\begin{tipbox}
\textbf{大蒜的食用建议：}蒜素在高温下易被破坏。建议大蒜切碎后室温放置10\~{}15分钟（让蒜氨酸酶充分作用生成蒜素），再用于凉拌或低温快炒，以获得最佳生物活性效果。
\end{tipbox}

% ----- 第六节 皂苷类化合物 -----
\section{皂苷类化合物（Saponins）}

\begin{defbox}
\textbf{皂苷：}是一类具有表面活性（在水中振摇可产生持久泡沫，类似肥皂）的糖苷类化合物。由苷元（甾体或三萜类）和糖组成。根据苷元结构的不同可分为\keyword{甾体皂苷（Steroidal Saponins）}和\keyword{三萜皂苷（Triterpenoid Saponins）}。
\end{defbox}

\textbf{主要食物来源：}大豆、鹰嘴豆、花生、菠菜、芦笋、燕麦、人参、甘草。

\textbf{主要生物学作用：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{调节脂质代谢：}部分皂苷（如大豆皂苷）可与胆固醇和胆汁酸结合，降低肠道胆固醇吸收。
    \item \textbf{抗菌/抗病毒作用：}破坏微生物细胞膜结构。
    \item \textbf{抗肿瘤作用：}抑制肿瘤细胞增殖、诱导凋亡和分化。
    \item \textbf{抗血栓形成：}抑制血小板聚集。
    \item \textbf{免疫调节作用：}激活巨噬细胞和NK细胞（人参皂苷研究较多）。
\end{enumerate}

\begin{tipbox}
\textbf{皂苷的安全性：}皂苷具有溶血活性（与红细胞膜胆固醇结合破坏膜结构），但经口摄入后胃肠道吸收极少，通常不造成全身性溶血危害。
\end{tipbox}

% ----- 第七节 芥子油苷 -----
\section{芥子油苷（Glucosinolates）}

\begin{defbox}
\textbf{芥子油苷（硫代葡萄糖苷）：}主要存在于\keyword{十字花科蔬菜}（Cruciferous Vegetables）中的含硫化合物。包括西兰花（Broccoli）、卷心菜（Cabbage）、白菜、甘蓝、萝卜、芥菜等。植物细胞破碎后芥子油苷被黑芥子酶（Myrosinase）水解，生成具有生物活性的\keyword{异硫氰酸盐（Isothiocyanates, ITCs）}（如萝卜硫素Sulforaphane）、吲哚类、硫氰酸盐等。
\end{defbox}

\textbf{主要生物学作用：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{抗肿瘤作用——核心功能：}这是芥子油苷最受关注的生物学作用。
    \begin{itemize}[leftmargin=*]
        \item \keyword{萝卜硫素（Sulforaphane）}——西兰花中含有最丰富的芥子油苷前体（萝卜硫苷），是已知最强的天然Nrf2通路激活剂之一，强力诱导II相解毒酶（GST、UGT、NQO1）表达。
        \item 降低多种癌症风险（肺癌、结肠癌、前列腺癌、乳腺癌）的流行病学证据较一致。
    \end{itemize}
    \item \textbf{调节炎症反应：}抑制NF-$\kappa$B通路。
    \item \textbf{抗菌作用：}ITCs对幽门螺杆菌等有一定抑制作用。
\end{enumerate}

% ----- 第八节 植物固醇 -----
\section{植物固醇（Phytosterols）}

\begin{defbox}
\textbf{植物固醇：}存在于植物细胞膜中的固醇类化合物，结构与胆固醇类似（均为环戊烷多氢菲骨架，C17侧链有差异）。人体不能合成，全部来自膳食。主要种类包括\keyword{$\beta$-谷固醇（$\beta$-Sitosterol）}（含量最丰富，约占50\%\~{}65\%）、\keyword{豆固醇（Stigmasterol）}、菜油固醇（Campesterol）。
\end{defbox}

\subsection{主要生物学作用}

\textbf{降胆固醇作用——最明确的核心功能：}
\begin{itemize}[leftmargin=*]
    \item \textbf{竞争机制：}植物固醇与胆固醇在肠道竞争进入混合微胶粒（Mixed Micelle），减少胆固醇的溶解和吸收。
    \item 日常膳食摄入植物固醇150\~{}450 mg/d即可发挥温和的降胆固醇效果。
    \item 每日摄入2\~{}3 g植物固醇可使血清LDL降低5\%\~{}15\%，是公认的辅助降脂功能性成分。
\end{itemize}

\textbf{其他潜在作用：}抗肿瘤（抑制结肠癌细胞增殖）、抗炎。

\subsection{食物来源}

\begin{itemize}[leftmargin=*]
    \item \textbf{植物油：}玉米油、大豆油、菜籽油中含量丰富。
    \item \textbf{坚果和种子：}花生、芝麻、葵花籽。
    \item \textbf{豆类：}大豆及豆制品。
    \item \textbf{全谷物。}
\end{itemize}

% ----- 第九节 蛋白酶抑制剂 -----
\section{蛋白酶抑制剂（Protease Inhibitors）}

\begin{defbox}
\textbf{蛋白酶抑制剂：}广泛存在于植物（特别是豆类和谷物）中的一类能与蛋白水解酶结合并抑制其活性的蛋白质或多肽。典型代表为大豆胰蛋白酶抑制剂（Bowman-Birk抑制剂，BBI；Kunitz胰蛋白酶抑制剂，KTI）。
\end{defbox}

\textbf{主要食物来源：}大豆、鹰嘴豆、扁豆等豆类，以及谷物、土豆。

\textbf{主要生物学作用：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{抗害虫作用（植物自我保护）：}抑制昆虫消化酶活性。
    \item \textbf{免疫调节作用：}BBI可抑制炎症性蛋白酶（如弹性蛋白酶）。
    \item \textbf{抗肿瘤作用：}主要通过抑制肿瘤相关蛋白酶活性、抑制自由基生成、调节癌基因表达发挥防癌效果。
\end{enumerate}

\begin{tipbox}
\textbf{传统观点与现代认识的转变：}蛋白酶抑制剂历来被视为"抗营养因子"（因其抑制胰蛋白酶、降低蛋白质消化率）。然而，充分的加热（如豆浆煮沸）可使其彻底灭活，且现代研究发现其低浓度下有抗癌防炎的健康效益。因此，"抗营养因子"与"有益生物活性成分"是对同一类物质的不同剂量-效应侧面。
\end{tipbox}

% ----- 第十节 单萜类 -----
\section{单萜类（Monoterpenes）}

\begin{defbox}
\textbf{单萜类：}由两个异戊二烯单元（C$_{10}$骨架）组成的萜类化合物，是植物精油（Essential Oils）的主要成分。代表性物质包括\keyword{紫苏醇（Perillyl Alcohol）}和\keyword{柠檬烯（Limonene）}。
\end{defbox}

\textbf{主要食物来源：}柑橘类水果果皮（柠檬烯）、薄荷油、香菜籽。

\textbf{主要生物学作用：}
\begin{itemize}[leftmargin=*]
    \item \textbf{抗肿瘤（主要关注方向）：}紫苏醇和柠檬烯在动物模型中显示抑制化学致癌物诱导的乳腺、肝脏、胰腺肿瘤。机制涉及抑制Ras蛋白异戊二烯化（Ras为GTP结合癌蛋白，需法尼基化才能定位到细胞膜发挥致癌效应）。
    \item \textbf{抗菌/驱虫作用。}
\end{itemize}

% ----- 第十一节 植酸 -----
\section{植酸（Phytic Acid, IP6）}

\begin{defbox}
\textbf{植酸（肌醇六磷酸, IP6）：}豆类、全谷物和坚果中天然存在的肌醇磷酸酯。因含有六个磷酸基团，对二价矿物离子（Ca$^{2+}$、Fe$^{2+}$/Fe$^{3+}$、Zn$^{2+}$、Mg$^{2+}$）具有强烈的螯合能力。
\end{defbox}

\textbf{主要食物来源：}全谷物麸皮、豆类、坚果、油籽。

\begin{keybox}
\textbf{植酸认识的历史演变：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{传统认识（抗营养因子）：}植酸在肠道螯合矿物质，降低矿物质的生物利用度，是发展中国家以谷物为主食人群矿物质缺乏（尤其是铁、锌）的重要影响因素。
    \item \textbf{现代认识（有益生物活性成分）：}植酸具有抗氧化（铁螯合阻断Fenton反应产生自由基）、抗肿瘤（抑制肿瘤细胞增殖、诱导分化）的生物学作用，被认为是潜在的抗癌活性物质。
\end{enumerate}
\end{keybox}

% ----- 第十二节 动物来源的生物活性成分 -----
\section{动物来源的生物活性成分}

\subsection{辅酶Q（Coenzyme Q, Ubiquinone）}

\begin{itemize}[leftmargin=*]
    \item \textbf{结构：}脂溶性苯醌类化合物，侧链为不同长度的异戊二烯单位（人体以CoQ10为主）。
    \item \textbf{功能：}呼吸链电子传递体（线粒体内膜）——参与ATP合成；脂溶性抗氧化剂（保护细胞膜和LDL免受氧化）；心血管保护（改善心肌能量代谢）。
    \item \textbf{来源：}肉类（心、肝、肾肌）、鱼油、植物油。也可内源性合成。
\end{itemize}

\subsection{硫辛酸（Lipoic Acid, $\alpha$-Lipoic Acid）}

\begin{itemize}[leftmargin=*]
    \item \textbf{特点：}同时具有水溶性和脂溶性的"万能抗氧化剂"，能透过血脑屏障。
    \item \textbf{功能：}抗氧化（清除自由基、再生GSH和维生素C/E）；改善胰岛素敏感性（II型糖尿病的辅助治疗）；神经保护（糖尿病神经病变的临床应用）。
    \item \textbf{来源：}菠菜、西兰花、红肉（肾脏和心脏）、酵母。
\end{itemize}

\subsection{褪黑素（Melatonin）}

\begin{itemize}[leftmargin=*]
    \item \textbf{来源：}主要由松果体合成，少量来自食物（含有色氨酸的食物如牛奶、鸡蛋、鱼、坚果可提供前体）。
    \item \textbf{功能：}调节生物节律（昼夜节律和睡眠周期）；强效抗氧化剂（穿越所有生物屏障直达任何细胞）；延缓衰老、免疫调节。
\end{itemize}

\subsection{其他动物来源活性成分}

\begin{itemize}[leftmargin=*]
    \item \textbf{$\gamma$-氨基丁酸（GABA）：}抑制性神经递质，降血压、镇静安神、改善睡眠。
    \item \textbf{左旋肉碱（L-Carnitine）：}脂肪酸$\beta$-氧化的转运载体（将长链脂肪酸从胞质转运至线粒体基质）。主要来源：红肉、奶制品。
\end{itemize}

% ----- 第九节 植物化学物按功能分类 -----
\section{植物化学物按功能分类}

植物化学物还可按其主要生物功能进行分类，以下为三大重要功能类别：

\begin{keybox}
\textbf{1. 抗氧化作用——清除自由基、螯合金属离子：}
\begin{itemize}[leftmargin=*]
    \item \textbf{天然色素：}叶绿素、类胡萝卜素、花青素、花黄素（黄酮类）等。
    \item \textbf{槲皮素（Quercetin）：}广泛存在于红茶、红酒、洋葱、芦笋等。
    \item \textbf{番茄红素（Lycopene）：}存在于番茄、西瓜、番石榴等红色蔬果中，是极强的单线态氧淬灭剂。
    \item \textbf{多酚类化合物：}绿茶中的儿茶素（EGCG）、银杏中的黄酮类，具有强抗氧化活性。
\end{itemize}
\end{keybox}

\begin{keybox}
\textbf{2. 降胆固醇作用——调节脂质代谢：}
\begin{itemize}[leftmargin=*]
    \item \textbf{皂苷（Saponins）：}大豆等豆类中含量丰富，可与胆固醇和胆汁酸结合，减少肠吸收。
    \item \textbf{植物固醇（Phytosterols）：}大豆异黄酮、$\beta$-谷固醇等，结构与胆固醇相似，竞争性抑制胆固醇吸收。
    \item \textbf{有机硫化物：}葱属植物（大蒜、洋葱）中的二硫化合物，可抑制胆固醇合成酶。
\end{itemize}
\end{keybox}

\begin{keybox}
\textbf{3. 抗肿瘤作用——抑制癌细胞增殖、诱导凋亡：}
\begin{itemize}[leftmargin=*]
    \item \textbf{芥子油苷/异硫氰酸酯：}主要存在于十字花科植物（西兰花、卷心菜、芥菜），可激活II相解毒酶。
    \item \textbf{多酚类：}绿茶多酚（EGCG）、银杏多酚等，具有抗氧化和诱导癌细胞凋亡双重作用。
    \item \textbf{有机硫化物：}大蒜中的二烯丙基二硫（DADS）等，可抑制亚硝胺的致癌作用。
\end{itemize}
\end{keybox}

    \item \textbf{【简答题】}简述类胡萝卜素的主要生物学作用，并指出哪些种类具有维生素A原活性。
    \item \textbf{【简答题】}简述植物雌激素的分类、来源和双向调节机制。
    \item \textbf{【简答题】}简述大蒜中有机硫化物的抗菌和抗肿瘤机制。
    \item \textbf{【简答题】}植物固醇降低血清胆固醇的机制是什么？
    \item \textbf{【简答题】}简述植酸从"抗营养因子"到"潜在抗癌活性物质"的认识转变。
    \item \textbf{【简答题】}简述SPL（特定建议值）的概念，并说明其与营养素DRIs（如RNI、UL）的本质区别。
    \item \textbf{【论述题】}以多酚类化合物和有机硫化物为例，试论植物性食物中生物活性成分对维持人类健康的重要意义。
\end{enumerate}

\subsection*{参考答案要点}

\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item 见本章各概念定义框。
    \item (1) 吸收率低（1\%\~{}10\%）；(2) 经CYP450进行I相代谢；(3) 经UGT/SULT/GST进行II相结合反应；(4) 以尿液和胆汁排泄；(5) 肠道菌群参与代谢转化。
    \item (1) 黄酮及黄酮醇（槲皮素）；(2) 黄烷酮（柚皮素）；(3) 黄烷醇/儿茶素（EGCG）；(4) 异黄酮（染料木素）；(5) 双黄酮（银杏黄酮）；(6) 花色素（矢车菊素）；(7) 查尔酮。
    \item 五大作用：抗氧化（番茄红素最强）、维生素A原活性（$\beta$-胡萝卜素最强，番茄红素/叶黄素/玉米黄素无此活性）、抗肿瘤、免疫调节、视觉保护（叶黄素/玉米黄素构成黄斑色素）。
    \item 分类：异黄酮（大豆）、木酚素（亚麻籽）、香豆素（苜蓿芽）、芪类（白藜芦醇/葡萄）。机制：低雌激素时激激动ER，高雌激素时拮抗ER。
    \item 抗菌：蒜素破坏细菌细胞膜，约占1\%青霉素效力，不易耐药。抗肿瘤：阻断亚硝胺合成、诱导II相解毒酶。
    \item 竞争机制：植物固醇与胆固醇竞争进入肠道混合微胶粒，减少胆固醇的溶解和吸收。每日2\~{}3g可降低LDL 5\%\~{}15\%。
    \item 传统：螯合矿物质，降低其生物利用度。现代：抗氧化（阻断Fenton反应）和抗肿瘤（抑制增殖、诱导分化）。
    \item SPL是DRIs第七项指标，专为非营养素活性成分（植物化学物）设立，以降低成人膳食相关NCD风险为目标的每日摄入量。本质区别：(1) 对象——SPL针对非必需的非营养素，RNI/UL针对必需营养素；(2) 逻辑——SPL回答"摄入多少有助降低慢病风险"，RNI回答"需要多少满足基本生理需求"，UL回答"多少以内安全"；(3) 基础——SPL无EAR（不需要满足50\%个体），仅基于流行病学剂量-反应关系。
    \item 要点：(1) 多酚类——膳食最丰富的抗氧化成分来源，抗肿瘤（EGCG、大豆异黄酮）、心血管保护、抗炎、调节肠道菌群；(2) 有机硫化物——抗菌（蒜素）、降脂、抗血栓、抗肿瘤（阻断亚硝胺合成、诱导II相酶）；(3) 共同特征——通过温和调控细胞信号通路（Nrf2、NF-$\kappa$B）发挥多靶点健康效应。
\end{enumerate}
