\chapter{各类食物的营养价值}
\setcounter{chapter}{3}
\chapterintro{
\keyword{掌握}食物营养价值的定义与评定指标（INQ）和评定方法；各类食物的营养特点（谷类、薯类及杂豆、大豆、蔬果、畜禽水产品、蛋类、乳类、坚果）。\keyword{熟悉}影响食物营养价值的因素。\keyword{了解}食物成分数据库。
}

% ----- 第一节 食物营养价值的评价及意义 -----
\section{食物营养价值的评价及意义}

\begin{keybox}
\textbf{食物的营养价值（Nutritional Value）：}某种食物所含营养素和能量能满足人体营养需要的程度。高低取决于所含营养素种类是否齐全、数量是否足够，各种营养素间的相互比例是否适宜和是否易被人体消化吸收和利用。
\end{keybox}

\begin{defbox}
\textbf{影响食物营养价值的因素（熟悉）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{食物品种与产地：}不同品种和产地间营养素含量存在差异（如土壤硒含量影响作物硒含量）
    \item \textbf{栽培/饲养条件：}施肥、灌溉、饲料等影响植物和动物性食物的营养成分
    \item \textbf{成熟度和新鲜度：}采收时的成熟度和采后新鲜度显著影响营养素含量（如Vc随贮存时间下降）
    \item \textbf{加工和烹调方式：}加工精度、烹调温度、时间、水分等影响营养素保留率（精加工使B族维生素和矿物质大量损失；高温油炸破坏维生素）
    \item \textbf{贮存条件：}温度、湿度、光照、氧气等影响营养素稳定性（如油脂氧化酸败、维生素光解）
\end{enumerate}
\end{defbox}

\begin{keybox}
\textbf{食物营养价值评价的常用指标（掌握）：}

\textbf{1. 营养素的种类、含量及其之间的比例。}

\textbf{2. 营养质量指数（INQ）：}某食物中营养素能满足人体营养需要的程度（营养素密度）与该食物能满足人体能量需要程度（能量密度）的比值。
\[
\text{INQ} = \frac{\text{某营养素密度}}{\text{能量密度}} = \frac{\text{某营养素含量 / 该营养素参考摄入量}}{\text{所产生能量 / 能量参考摄入量}}
\]
\begin{itemize}[leftmargin=*]
    \item INQ = 1：营养需要达到平衡
    \item INQ > 1：营养价值高
    \item INQ < 1：营养价值低
\end{itemize}

\textbf{3. 营养素在加工烹调过程中的变化。}

\textbf{4. 血糖指数/生糖指数（GI）。}

\textbf{5. 食物抗氧化能力。}

\textbf{6. 食物中的抗营养因子。}

\textbf{评定方法（掌握）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{化学分析法：}通过实验室检测测定食物中各类营养素的含量，是评价食物营养价值的基础方法。
    \item \textbf{计算法：}利用食物成分数据库中该食物的代表性数据，结合DRIs计算营养素密度和INQ等指标。
    \item \textbf{生物学方法：}通过动物实验或人体实验评价食物的消化率、生物利用度、蛋白质功效比等（如蛋白质生物学价值、氮平衡法）。
\end{itemize}
\end{keybox}

% ----- 第二节 谷类 -----
\section{谷类}

\begin{defbox}
\textbf{谷类种子结构（由外到内）及各部分营养素分布：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{谷皮：}纤维素 + 半纤维素 + 矿物质。
    \item \textbf{糊粉层：}蛋白质 + 脂肪 + 矿物质 + B族维生素。
    \item \textbf{胚乳：}大量淀粉 + 一定量蛋白质（蛋白质含量最多）。
    \item \textbf{胚：}脂肪 + 蛋白质（赖氨酸含量最多）+ 矿物质 + B族维生素 + 维生素E。
\end{enumerate}
\end{defbox}

\begin{defbox}
\textbf{谷类的营养成分和特点：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{蛋白质：}清蛋白、球蛋白、谷蛋白和醇溶蛋白（主要）。
    \item \textbf{碳水化合物：}含量高，主要为淀粉，存在于胚乳中。
    \item \textbf{脂肪：}在糊粉层和胚芽中。
    \item \textbf{矿物质：}主要为磷和钙，存在于谷皮和糊粉层中，加工易丢失。影响吸收利用的是植酸。
    \item \textbf{维生素：}B族维生素摄入的重要来源（VB$_1$、VB$_2$、VB$_3$（烟酸）、VB$_6$等），主要存在于糊粉层和胚芽中，精加工的谷物维生素大量损失。
\end{enumerate}
\end{defbox}

\begin{defbox}
\textbf{全谷物：}未经精细化加工或虽经碾磨/粉碎/压片等处理保留完整谷粒具备的胚乳、胚芽、谷皮及其天然营养成分的谷物。与精致谷物相比，全谷物及杂豆类可提供更多B族维生素、矿物质、膳食纤维、植物化学物，GI远低于精致米面。
\end{defbox}

% ----- 第三节 薯类及杂豆 -----
\section{薯类及杂豆}

\begin{defbox}
\textbf{薯类的营养特点：}

\textbf{马铃薯（土豆）：}
\begin{itemize}[leftmargin=*]
    \item 水分约79\%，淀粉约17\%。淀粉颗粒大，糊化温度低、黏度大、膨润力大，不易老化。
    \item 含钾和维生素C丰富。
    \item 注意：发芽或皮色变绿时含有有毒的龙葵素（solanine），应彻底去除芽眼和绿皮。
\end{itemize}

\textbf{甘薯（红薯/地瓜）：}
\begin{itemize}[leftmargin=*]
    \item 水分约68\%，淀粉约29\%，含2\%\~{}6\%可溶性糖，因此显甜味。
    \item 含有$\beta$-淀粉酶，贮存或蒸煮时可将部分淀粉转化为麦芽糖、糊精等，使甜味增强。
    \item 胡萝卜素含量比谷类高；膳食纤维含量丰富。
\end{itemize}

\textbf{薯类的共同特点：}水分含量高于谷类，蛋白质含量较低；富含碳水化合物（淀粉）；是膳食纤维、钾、维生素C和B族维生素的良好来源（VC在加热烹煮过程中有一定损失）。

\textbf{杂豆的营养特点（了解）：}
\begin{itemize}[leftmargin=*]
    \item 除大豆外的其他豆类（红豆、绿豆、黑豆、花豆、豌豆、蚕豆等）
    \item 蛋白质：20\%\~{}25\%（不及大豆），多为球蛋白
    \item 脂肪：约1\%，远低于大豆
    \item 碳水化合物：55\%以上，主要为淀粉
    \item 维生素：种皮颜色深者胡萝卜素含量较高；B族维生素丰富
    \item 矿物质量不及大豆，但仍为钙、铁、锌的良好来源
    \item 绿豆：蛋白质21\%\~{}24\%，为近全价蛋白质（赖氨酸含量高，含硫氨基酸为限制氨基酸）；含丰富半纤维素和半乳聚糖；中医认为有消热解毒功效
\end{itemize}

与精致谷物相比，全谷物及杂豆类可提供更多B族维生素、矿物质、膳食纤维和植物化学物，血糖生成指数（GI）远低于精致米面。
\end{defbox}

% ----- 第四节 大豆及其制品 -----
\section{大豆及其制品}

\begin{defbox}
\textbf{大豆的营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{蛋白质：}含量35\%，有球蛋白（主要）、清蛋白、谷蛋白和醇溶蛋白。赖氨酸含量较高，属于优质蛋白质。
    \item \textbf{脂肪：}不饱和脂肪酸含量高（大豆油中亚油酸多），优质食用油（亚油酸51.7\%）。
\end{itemize}

\textbf{大豆的抗营养因素：}蛋白酶抑制剂、大豆凝血素和甲状腺肿素。

\textbf{大豆生物活性物质：}大豆膳食纤维，大豆磷脂，大豆皂苷，大豆异黄酮。

\textbf{豆制品的营养特点：}各营养素的利用率都有所提高。
\end{defbox}

% ----- 第五节 蔬菜、水果类 -----
\section{蔬菜、水果类}

\begin{defbox}
\textbf{蔬菜的营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{碳水化合物：}膳食纤维主要来源，根茎类最多。
    \item \textbf{矿物质：}镁。
    \item \textbf{维生素：}维生素C、B族维生素（VB$_2$和叶酸）、胡萝卜素（绿色、黄色或红色蔬菜）；叶部 $>$ 根茎部，深色菜叶 $>$ 浅色菜叶。
\end{itemize}

\textbf{水果的营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{碳水化合物：}主要为果糖（仁果类，如苹果和梨）、葡萄糖（浆果类，如草莓和葡萄）和蔗糖（核果类，如桃、李和柑橘），还富含纤维素、半纤维素和果胶。
\end{itemize}

\textbf{推荐摄入：}
\begin{itemize}[leftmargin=*]
    \item 餐餐有蔬菜，推荐每日摄入300--500g，深色蔬菜占1/2。
    \item 天天吃水果，推荐每日摄入200--350g，果汁不能替代鲜果。
\end{itemize}
\end{defbox}

% ----- 第六节 畜、禽、水产品 -----
\section{畜、禽、水产品}

\begin{defbox}
\textbf{畜肉的营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{蛋白质：}肌原纤维蛋白质（丝状），肌浆蛋白质（水溶性蛋白质），肉基质蛋白质（不完全蛋白质）。
    \item \textbf{脂肪：}90\%为中性脂肪，饱和脂肪酸（硬脂酸和软脂酸），不饱和脂肪酸（油酸），肌内脂肪是判断牛肉好坏的重要标志。
    \item \textbf{浸出物：}包括核苷酸、嘌呤碱、肌酸、肌酐、氨基酸、肽类，是肉汤味鲜主要成分。
\end{itemize}

\textbf{禽肉的营养特点：}脂肪含量低于畜肉，亚油酸丰富（20\%）。

\textbf{水产品的营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{蛋白质：}优质蛋白质，富含亮氨酸和赖氨酸。
    \item \textbf{脂肪：}含量低，不饱和脂肪酸丰富。
    \item \textbf{矿物质：}磷（含量最高，占40\%）、钙（良好来源），海水鱼类含碘丰富。
\end{itemize}
\end{defbox}

% ----- 第七节 乳及乳制品 -----
\section{乳及乳制品}

\begin{defbox}
\textbf{乳的营养成分：}
\begin{itemize}[leftmargin=*]
    \item \textbf{蛋白质：}(1) 酪蛋白：结合蛋白，与钙磷等结合，形成酪蛋白胶粒；(2) 乳清蛋白：加热时发生凝固并沉淀；(3) 乳球蛋白：机体免疫。
    \item \textbf{脂肪：}主要为甘油三酯，少量磷脂和胆固醇；脂肪酸组成复杂：油酸、亚油酸、亚麻酸，短链脂肪酸（丁酸、己酸、辛酸）含量也较高。
    \item \textbf{碳水化合物：}乳糖，促进钙吸收。
    \item \textbf{矿物质：}钙的良好来源，铁含量很低。
    \item \textbf{维生素：}B族维生素的良好来源，特别是维生素B$_2$。
\end{itemize}
\end{defbox}

% ----- 第八节 蛋类及其制品 -----
\section{蛋类及其制品}

\begin{defbox}
\textbf{蛋类的营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{蛋白质：}主要在蛋黄中。必需氨基酸与人体相近，是蛋白质生物学价值最高的食物，常被用作参考蛋白。
    \item \textbf{脂肪：}主要在蛋黄中，甘油三酯、磷脂（良好来源，卵磷脂、脑磷脂、神经鞘磷脂）和固醇。
    \item \textbf{矿物质：}主要在蛋黄中，铁（含量高，但为非血红素铁，并与卵黄高磷蛋白结合，生物利用率低）。
    \item \textbf{维生素：}主要在蛋黄中，维生素A、E、B$_2$、B$_6$、D、K。
\end{itemize}
\end{defbox}

% ----- 第九节 坚果类 -----
\section{坚果类}

\begin{defbox}
\textbf{坚果的营养特点：}
\begin{itemize}[leftmargin=*]
    \item 脂肪以不饱和脂肪酸为主。
    \item 微量营养素：硒和维生素E有抗氧化作用。
\end{itemize}
\end{defbox}

% ----- 第十节 食用油 -----
\section{食用油}

\begin{defbox}
\textbf{常见食用油的营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{大豆油：}优质蛋白质来源，大豆油是健康的食用油资源。
    \item \textbf{花生油：}老年人健康用油之选，油酸（40.4\%），亚油酸（37.9\%），$\alpha$-亚麻酸（0.4\%）。
    \item \textbf{棕榈油：}氧化稳定性高，食品行业广泛应用。
    \item \textbf{橄榄油：}油酸较多，亚油酸和亚麻酸很少。
    \item \textbf{亚麻籽油：}易被空气氧化变质，需低温保存，开盖后尽快吃完（$\alpha$-亚麻酸49\%）。
\end{itemize}

\textbf{推荐量：}烹调油25--30克/日，食用盐 $<$ 6克/日。
\end{defbox}

% ----- 第十一节 菌藻类食物 -----
\section{菌藻类食物}

\begin{defbox}
\textbf{菌藻类食物的营养特点：}
\begin{itemize}[leftmargin=*]
    \item 蛋白质含量高。
    \item 碳水化合物以多糖为主。
    \item 维生素：含丰富的B族维生素。
    \item 矿物质：钙、镁、铁、锌，海产品含碘。
\end{itemize}
\end{defbox}

% ----- 第十二节 食物成分数据库 -----
\section{食物成分数据库（了解）}

\begin{tipbox}
\textbf{食物成分数据库（Food Composition Database）：}系统地收集、整理和存储各类食物的营养成分含量数据的数据库，是营养学研究、膳食调查与评价、营养配餐和公共卫生决策的基础工具。

\textbf{主要内容：}
\begin{itemize}[leftmargin=*]
    \item 各类食物的能量及宏量营养素（蛋白质、脂肪、碳水化合物）含量
    \item 矿物质和维生素含量
    \item 氨基酸和脂肪酸组成
    \item 膳食纤维、胆固醇及其他功能成分（部分数据库）
\end{itemize}

\textbf{主要用途：}
\begin{itemize}[leftmargin=*]
    \item 膳食调查中，将食物摄入量转换为营养素摄入量（食物频率法、24小时回顾法等）
    \item 营养配餐和食谱编制中的营养素计算
    \item 食物营养价值评价（INQ计算的基础数据来源）
    \item 流行病学研究和膳食暴露评估
    \item 食品标签营养声称的依据（NRV\%计算）
\end{itemize}

\textbf{代表性数据库：}中国食物成分表（Chinese Food Composition Table, CFCT）由中国科学院营养与健康所编制和更新，是我国最权威的食物成分数据来源，至今已出版多个版本。国际上代表性的有美国USDA FoodData Central、欧洲EuroFIR等。
\end{tipbox}

\subsection*{参考答案要点}

\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item \textbf{食物营养价值：}某种食物所含营养素和能量能满足人体营养需要的程度。高低取决于所含营养素种类是否齐全、数量是否足够，各种营养素间的相互比例是否适宜和是否易被人体消化吸收和利用。\textbf{INQ：}某食物中营养素能满足人体营养需要的程度（营养素密度）与该食物能满足人体能量需要程度（能量密度）的比值。INQ = 某营养素密度 / 能量密度 = (某营养素含量/该营养素参考摄入量) / (所产生能量/能量参考摄入量)。INQ = 1，营养需要达到平衡；INQ $>$ 1，营养价值高；INQ $<$ 1，营养价值低。\textbf{全谷物：}未经精细化加工或虽经碾磨/粉碎/压片等处理保留完整谷粒具备的胚乳、胚芽、谷皮及其天然营养成分的谷物。
    \item 五个因素：(1) 食物品种与产地——不同品种和产地间营养素含量存在差异；(2) 栽培/饲养条件——施肥、饲料等影响食物营养成分；(3) 成熟度和新鲜度——采收成熟度和采后新鲜度显著影响营养素含量；(4) 加工和烹调方式——精加工使B族维生素和矿物质大量损失，高温油炸破坏维生素；(5) 贮存条件——温度、湿度、光照、氧气等影响营养素稳定性。
    \item \textbf{六项指标：}(1) 营养素的种类、含量及其比例；(2) 营养质量指数（INQ）；(3) 营养素在加工烹调过程中的变化；(4) 血糖指数/生糖指数（GI）；(5) 食物抗氧化能力；(6) 食物中的抗营养因子。\textbf{评定方法：}化学分析法（实验室检测营养素含量）、计算法（利用食物成分数据库和DRIs计算INQ等）、生物学方法（动物/人体实验评价消化率、生物利用度等）。
    \item 谷类种子结构由外到内：(1) 谷皮：纤维素+半纤维素+矿物质；(2) 糊粉层：蛋白质+脂肪+矿物质+B族维生素；(3) 胚乳：大量淀粉+一定量蛋白质（蛋白质含量最多）；(4) 胚：脂肪+蛋白质（赖氨酸含量最多）+矿物质+B族维生素+维生素E。
    \item \textbf{马铃薯：}水分79\%，淀粉17\%，淀粉颗粒大、糊化温度低；含钾和VC丰富；发芽绿皮含龙葵素。\textbf{甘薯：}水分68\%，淀粉29\%，含2\%$\sim$6\%可溶性糖；胡萝卜素高于谷类；膳食纤维丰富。\textbf{杂豆：}蛋白质20\%$\sim$25\%，脂肪约1\%，碳水化合物55\%以上（淀粉为主）；B族维生素丰富；GI远低于精制米面。
    \item 抗营养因素：蛋白酶抑制剂、大豆凝血素和甲状腺肿素。生物活性物质：大豆膳食纤维，大豆磷脂，大豆皂苷，大豆异黄酮。
    \item 蔬菜碳水化合物：膳食纤维主要来源，根茎类最多。水果碳水化合物：主要为果糖（仁果类，如苹果和梨）、葡萄糖（浆果类，如草莓和葡萄）和蔗糖（核果类，如桃、李和柑橘），还富含纤维素、半纤维素和果胶。
    \item 三种形式：(1) 肌原纤维蛋白质（丝状）——肌肉收缩的物质基础；(2) 肌浆蛋白质（水溶性蛋白质）；(3) 肉基质蛋白质（不完全蛋白质）。浸出物包括核苷酸、嘌呤碱、肌酸、肌酐、氨基酸、肽类，是肉汤味鲜主要成分。
    \item 蛋类各营养素主要集中在蛋黄中。蛋白质：主要在蛋黄中，必需氨基酸与人体相近，是蛋白质生物学价值最高的食物，常被用作参考蛋白。脂肪：主要在蛋黄中，甘油三酯、磷脂（卵磷脂、脑磷脂、神经鞘磷脂）和固醇。矿物质：主要在蛋黄中，铁含量高但为非血红素铁并与卵黄高磷蛋白结合，生物利用率低。维生素：主要在蛋黄中，维生素A、E、B$_2$、B$_6$、D、K。
    \item 乳类蛋白质三种类型：(1) 酪蛋白：结合蛋白，与钙磷等结合，形成酪蛋白胶粒；(2) 乳清蛋白：加热时发生凝固并沉淀；(3) 乳球蛋白：机体免疫。
    \item (1) 谷类：蛋白质7\%--12\%，主要为谷蛋白和醇溶蛋白，第一限制氨基酸为赖氨酸；(2) 大豆类：蛋白质35\%，主要为球蛋白，赖氨酸含量较高，属于优质蛋白质，与谷类互补（谷类缺赖氨酸、大豆缺蛋氨酸）；(3) 畜肉类：肌原纤维蛋白质（丝状，优质）、肌浆蛋白质（水溶性）、肉基质蛋白质（不完全蛋白质）；(4) 水产品：优质蛋白质，富含亮氨酸和赖氨酸；(5) 蛋类：蛋白质生物学价值最高的食物，必需氨基酸与人体相近，常被用作参考蛋白；(6) 乳类：酪蛋白（结合蛋白，形成酪蛋白胶粒）、乳清蛋白（加热凝固沉淀）、乳球蛋白（机体免疫）。
\end{enumerate}
