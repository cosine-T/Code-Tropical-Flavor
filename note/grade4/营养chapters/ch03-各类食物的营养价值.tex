\chapter{各类食物的营养价值}
\setcounter{chapter}{3}
\chapterintro{
\keyword{掌握}各类食物的营养特点（谷类、大豆、蔬果、畜禽水产品、蛋类、乳类、坚果）。\keyword{熟悉}食物营养价值的定义及影响因素。\keyword{了解}食物成分数据库。
}

% ----- 第一节 食物营养价值的评价及意义 -----
\section{食物营养价值的评价及意义}

\begin{defbox}
\textbf{食物的营养价值（Nutritional Value）：}某种食物所含营养素和能量能满足人体营养需要的程度。高低取决于所含营养素种类是否齐全、数量是否足够，各种营养素间的相互比例是否适宜和是否易被人体消化吸收和利用。
\end{defbox}

\begin{keybox}
\textbf{食物营养价值评价的常用指标：}

\textbf{1. 营养素的种类及含量。}

\textbf{2. 营养质量指数（INQ）：}某食物中营养素能满足人体营养需要的程度（营养素密度）与该食物能满足人体能量需要程度（能量密度）的比值。
\[
\text{INQ} = \frac{\text{某营养素密度}}{\text{能量密度}} = \frac{\text{某营养素含量 / 该营养素参考摄入量}}{\text{所产生能量 / 能量参考摄入量}}
\]
\begin{itemize}[leftmargin=*]
    \item INQ = 1：营养需要达到平衡
    \item INQ > 1：营养价值高
    \item INQ < 1：营养价值低
\end{itemize}

\textbf{3. 营养素在加工烹调过程中的变化。}

\textbf{4. 血糖指数/生糖指数（GI）。}

\textbf{5. 食物抗氧化能力。}

\textbf{6. 食物中的抗营养因子。}
\end{keybox}

% ----- 第二节 谷类 -----
\section{谷类}

\begin{keybox}
\textbf{谷类种子结构（由外到内）及各部分营养素分布：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{谷皮：}纤维素 + 半纤维素 + 矿物质。
    \item \textbf{糊粉层：}蛋白质 + 脂肪 + 矿物质 + B族维生素。
    \item \textbf{胚乳：}大量淀粉 + 一定量蛋白质（蛋白质含量最多）。
    \item \textbf{胚：}脂肪 + 蛋白质（赖氨酸含量最多）+ 矿物质 + B族维生素 + 维生素E。
\end{enumerate}
\end{keybox}

\begin{keybox}
\textbf{谷类的营养成分和特点：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{蛋白质：}清蛋白、球蛋白、谷蛋白和醇溶蛋白（主要）。
    \item \textbf{碳水化合物：}含量高，主要为淀粉，存在于胚乳中。
    \item \textbf{脂肪：}在糊粉层和胚芽中。
    \item \textbf{矿物质：}主要为磷和钙，存在于谷皮和糊粉层中，加工易丢失。影响吸收利用的是植酸。
    \item \textbf{维生素：}B族维生素摄入的重要来源（VB$_1$、VB$_2$、VB$_3$（烟酸）、VB$_6$等），主要存在于糊粉层和胚芽中，精加工的谷物维生素大量损失。
\end{enumerate}
\end{keybox}

\begin{keybox}
\textbf{全谷物：}未经精细化加工或虽经碾磨/粉碎/压片等处理保留完整谷粒具备的胚乳、胚芽、谷皮及其天然营养成分的谷物。与精致谷物相比，全谷物及杂豆类可提供更多B族维生素、矿物质、膳食纤维、植物化学物，GI远低于精致米面。
\end{keybox}

% ----- 第三节 大豆及其制品 -----
\section{大豆及其制品}

\begin{keybox}
\textbf{大豆的营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{蛋白质：}含量35\%，有球蛋白（主要）、清蛋白、谷蛋白和醇溶蛋白。赖氨酸含量较高，属于优质蛋白质。
    \item \textbf{脂肪：}不饱和脂肪酸含量高（大豆油中亚油酸多），优质食用油（亚油酸51.7\%）。
\end{itemize}

\textbf{大豆的抗营养因素：}蛋白酶抑制剂、大豆凝血素和甲状腺肿素。

\textbf{大豆生物活性物质：}大豆膳食纤维，大豆磷脂，大豆皂苷，大豆异黄酮。

\textbf{豆制品的营养特点：}各营养素的利用率都有所提高。
\end{keybox}

% ----- 第四节 蔬菜、水果类 -----
\section{蔬菜、水果类}

\begin{keybox}
\textbf{蔬菜的营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{碳水化合物：}膳食纤维主要来源，根茎类最多。
    \item \textbf{矿物质：}镁。
    \item \textbf{维生素：}维生素C、B族维生素（VB$_2$和叶酸）、胡萝卜素（绿色、黄色或红色蔬菜）；叶部 $>$ 根茎部，深色菜叶 $>$ 浅色菜叶。
\end{itemize}

\textbf{水果的营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{碳水化合物：}主要为果糖（仁果类，如苹果和梨）、葡萄糖（浆果类，如草莓和葡萄）和蔗糖（核果类，如桃、李和柑橘），还富含纤维素、半纤维素和果胶。
\end{itemize}

\textbf{推荐摄入：}
\begin{itemize}[leftmargin=*]
    \item 餐餐有蔬菜，推荐每日摄入300--500g，深色蔬菜占1/2。
    \item 天天吃水果，推荐每日摄入200--350g，果汁不能替代鲜果。
\end{itemize}
\end{keybox}

% ----- 第五节 畜、禽、水产品 -----
\section{畜、禽、水产品}

\begin{keybox}
\textbf{畜肉的营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{蛋白质：}肌原纤维蛋白质（丝状），肌浆蛋白质（水溶性蛋白质），肉基质蛋白质（不完全蛋白质）。
    \item \textbf{脂肪：}90\%为中性脂肪，饱和脂肪酸（硬脂酸和软脂酸），不饱和脂肪酸（油酸），肌内脂肪是判断牛肉好坏的重要标志。
    \item \textbf{浸出物：}包括核苷酸、嘌呤碱、肌酸、肌酐、氨基酸、肽类，是肉汤味鲜主要成分。
\end{itemize}

\textbf{禽肉的营养特点：}脂肪含量低于畜肉，亚油酸丰富（20\%）。

\textbf{水产品的营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{蛋白质：}优质蛋白质，富含亮氨酸和赖氨酸。
    \item \textbf{脂肪：}含量低，不饱和脂肪酸丰富。
    \item \textbf{矿物质：}磷（含量最高，占40\%）、钙（良好来源），海水鱼类含碘丰富。
\end{itemize}
\end{keybox}

% ----- 第六节 乳及乳制品 -----
\section{乳及乳制品}

\begin{keybox}
\textbf{乳的营养成分：}
\begin{itemize}[leftmargin=*]
    \item \textbf{蛋白质：}(1) 酪蛋白：结合蛋白，与钙磷等结合，形成酪蛋白胶粒；(2) 乳清蛋白：加热时发生凝固并沉淀；(3) 乳球蛋白：机体免疫。
    \item \textbf{脂肪：}主要为甘油三酯，少量磷脂和胆固醇；脂肪酸组成复杂：油酸、亚油酸、亚麻酸，短链脂肪酸（丁酸、己酸、辛酸）含量也较高。
    \item \textbf{碳水化合物：}乳糖，促进钙吸收。
    \item \textbf{矿物质：}钙的良好来源，铁含量很低。
    \item \textbf{维生素：}B族维生素的良好来源，特别是维生素B$_2$。
\end{itemize}
\end{keybox}

% ----- 第七节 蛋类及其制品 -----
\section{蛋类及其制品}

\begin{keybox}
\textbf{蛋类的营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{蛋白质：}主要在蛋黄中。必需氨基酸与人体相近，是蛋白质生物学价值最高的食物，常被用作参考蛋白。
    \item \textbf{脂肪：}主要在蛋黄中，甘油三酯、磷脂（良好来源，卵磷脂、脑磷脂、神经鞘磷脂）和固醇。
    \item \textbf{矿物质：}主要在蛋黄中，铁（含量高，但为非血红素铁，并与卵黄高磷蛋白结合，生物利用率低）。
    \item \textbf{维生素：}主要在蛋黄中，维生素A、E、B$_2$、B$_6$、D、K。
\end{itemize}
\end{keybox}

% ----- 第八节 坚果类 -----
\section{坚果类}

\begin{keybox}
\textbf{坚果的营养特点：}
\begin{itemize}[leftmargin=*]
    \item 脂肪以不饱和脂肪酸为主。
    \item 微量营养素：硒和维生素E有抗氧化作用。
\end{itemize}
\end{keybox}

% ----- 第九节 食用油 -----
\section{食用油}

\begin{keybox}
\textbf{常见食用油的营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{大豆油：}优质蛋白质来源，大豆油是健康的食用油资源。
    \item \textbf{花生油：}老年人健康用油之选，油酸（40.4\%），亚油酸（37.9\%），$\alpha$-亚麻酸（0.4\%）。
    \item \textbf{棕榈油：}氧化稳定性高，食品行业广泛应用。
    \item \textbf{橄榄油：}油酸较多，亚油酸和亚麻酸很少。
    \item \textbf{亚麻籽油：}易被空气氧化变质，需低温保存，开盖后尽快吃完（$\alpha$-亚麻酸49\%）。
\end{itemize}

\textbf{推荐量：}烹调油25--30克/日，食用盐 $<$ 6克/日。
\end{keybox}

% ----- 第十节 菌藻类食物 -----
\section{菌藻类食物}

\begin{keybox}
\textbf{菌藻类食物的营养特点：}
\begin{itemize}[leftmargin=*]
    \item 蛋白质含量高。
    \item 碳水化合物以多糖为主。
    \item 维生素：含丰富的B族维生素。
    \item 矿物质：钙、镁、铁、锌，海产品含碘。
\end{itemize}
\end{keybox}

\section{复习题}

\begin{enumerate}[leftmargin=*, itemsep=8pt]
    \item \textbf{【名词解释】}食物营养价值；营养质量指数（INQ）；全谷物
    \item \textbf{【简答题】}简述食物营养价值评价的常用指标（6项）。
    \item \textbf{【简答题】}简述谷类种子的结构及各部分的营养素分布特点。
    \item \textbf{【简答题】}简述大豆的抗营养因素和生物活性物质。
    \item \textbf{【简答题】}比较蔬菜和水果在碳水化合物构成上的主要差异。
    \item \textbf{【简答题】}简述畜肉中蛋白质的三种存在形式及其特点。
    \item \textbf{【简答题】}简述蛋类中各营养素在蛋清和蛋黄中的分布特点。
    \item \textbf{【简答题】}简述乳类蛋白质的三种类型及其特点。
    \item \textbf{【论述题】}试比较谷类、大豆类、畜肉类、水产品、蛋类和乳类在蛋白质营养特点上的主要差异。
\end{enumerate}

\subsection*{参考答案要点}

\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item \textbf{食物营养价值：}某种食物所含营养素和能量能满足人体营养需要的程度。高低取决于所含营养素种类是否齐全、数量是否足够，各种营养素间的相互比例是否适宜和是否易被人体消化吸收和利用。\textbf{INQ：}某食物中营养素能满足人体营养需要的程度（营养素密度）与该食物能满足人体能量需要程度（能量密度）的比值。INQ = 某营养素密度 / 能量密度 = (某营养素含量/该营养素参考摄入量) / (所产生能量/能量参考摄入量)。INQ = 1，营养需要达到平衡；INQ $>$ 1，营养价值高；INQ $<$ 1，营养价值低。\textbf{全谷物：}未经精细化加工或虽经碾磨/粉碎/压片等处理保留完整谷粒具备的胚乳、胚芽、谷皮及其天然营养成分的谷物。
    \item 六项指标：(1) 营养素的种类及含量；(2) 营养质量指数（INQ）；(3) 营养素在加工烹调过程中的变化；(4) 血糖指数/生糖指数（GI）；(5) 食物抗氧化能力；(6) 食物中的抗营养因子。
    \item 谷类种子结构由外到内：(1) 谷皮：纤维素+半纤维素+矿物质；(2) 糊粉层：蛋白质+脂肪+矿物质+B族维生素；(3) 胚乳：大量淀粉+一定量蛋白质（蛋白质含量最多）；(4) 胚：脂肪+蛋白质（赖氨酸含量最多）+矿物质+B族维生素+维生素E。
    \item 抗营养因素：蛋白酶抑制剂、大豆凝血素和甲状腺肿素。生物活性物质：大豆膳食纤维，大豆磷脂，大豆皂苷，大豆异黄酮。
    \item 蔬菜碳水化合物：膳食纤维主要来源，根茎类最多。水果碳水化合物：主要为果糖（仁果类，如苹果和梨）、葡萄糖（浆果类，如草莓和葡萄）和蔗糖（核果类，如桃、李和柑橘），还富含纤维素、半纤维素和果胶。
    \item 三种形式：(1) 肌原纤维蛋白质（丝状）——肌肉收缩的物质基础；(2) 肌浆蛋白质（水溶性蛋白质）；(3) 肉基质蛋白质（不完全蛋白质）。浸出物包括核苷酸、嘌呤碱、肌酸、肌酐、氨基酸、肽类，是肉汤味鲜主要成分。
    \item 蛋类各营养素主要集中在蛋黄中。蛋白质：主要在蛋黄中，必需氨基酸与人体相近，是蛋白质生物学价值最高的食物，常被用作参考蛋白。脂肪：主要在蛋黄中，甘油三酯、磷脂（卵磷脂、脑磷脂、神经鞘磷脂）和固醇。矿物质：主要在蛋黄中，铁含量高但为非血红素铁并与卵黄高磷蛋白结合，生物利用率低。维生素：主要在蛋黄中，维生素A、E、B$_2$、B$_6$、D、K。
    \item 乳类蛋白质三种类型：(1) 酪蛋白：结合蛋白，与钙磷等结合，形成酪蛋白胶粒；(2) 乳清蛋白：加热时发生凝固并沉淀；(3) 乳球蛋白：机体免疫。
    \item (1) 谷类：蛋白质7\%--12\%，主要为谷蛋白和醇溶蛋白，第一限制氨基酸为赖氨酸；(2) 大豆类：蛋白质35\%，主要为球蛋白，赖氨酸含量较高，属于优质蛋白质，与谷类互补（谷类缺赖氨酸、大豆缺蛋氨酸）；(3) 畜肉类：肌原纤维蛋白质（丝状，优质）、肌浆蛋白质（水溶性）、肉基质蛋白质（不完全蛋白质）；(4) 水产品：优质蛋白质，富含亮氨酸和赖氨酸；(5) 蛋类：蛋白质生物学价值最高的食物，必需氨基酸与人体相近，常被用作参考蛋白；(6) 乳类：酪蛋白（结合蛋白，形成酪蛋白胶粒）、乳清蛋白（加热凝固沉淀）、乳球蛋白（机体免疫）。
\end{enumerate}
