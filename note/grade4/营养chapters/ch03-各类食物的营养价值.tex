\setcounter{chapter}{3}
\chapterintro{
\keyword{掌握}各类食物的营养特点（谷类、大豆、蔬果、畜禽水产品、蛋类、乳类、坚果）。\keyword{熟悉}食物营养价值的定义及影响因素。\keyword{了解}食物成分数据库。

% ----- 第一节 食物营养价值的定义及影响因素 -----
\section{食物营养价值的定义及影响因素}

\begin{defbox}
\textbf{食物营养价值（Nutritional Value of Food）：}食物中所含营养素和能量满足人体营养需要的程度。营养价值的高低取决于食物中营养素种类是否齐全、数量是否充足、相互比例是否适宜，以及是否易于被消化、吸收和利用。
\end{defbox}

\subsection{食物营养价值评价的维度}

\begin{enumerate}[leftmargin=*]
    \item \textbf{营养素种类与含量：}营养素种类齐全、含量丰富者营养价值较高。
    \item \textbf{营养素质量：}主要看该营养素的构成模式与人体的接近程度（如蛋白质氨基酸模式）。
    \item \textbf{消化吸收利用率：}即使含量丰富，若消化吸收差，实际营养价值也有限。
    \item \textbf{功能成分/生物活性物质：}不仅满足基本需求，更关注维护健康、预防疾病的额外价值。
\end{enumerate}

\subsection{影响食物营养价值的因素}

\begin{keybox}
\textbf{食物营养价值评价指标：}

\textbf{1. 营养素种类、含量及其相互比例：}营养素的种类是否齐全、含量是否充足、相互比例是否适宜。食物中某种营养素的含量需结合其生物利用率综合判断。

\textbf{2. 食物营养质量指数（Index of Nutrition Quality, INQ）：}
\[
\text{INQ} = \frac{\text{营养素密度}}{\text{能量密度}} = \frac{\text{某营养素含量 / 该营养素推荐摄入量}}{\text{食物能量含量 / 能量推荐摄入量}}
\]
\begin{itemize}[leftmargin=*]
    \item INQ = 1：该食物提供营养素与提供能量的能力相当（"合格"食物）
    \item INQ > 1：该食物提供营养素的能力大于提供能量的能力（"优质"食物，营养素密度高）
    \item INQ < 1：该食物提供营养素的能力小于提供能量的能力（"营养质量较低"，能量密度过高而营养不足）
\end{itemize}
INQ是针对某一种营养素来研究的指标，用于评价食物中某种营养素和能量含量与人体供给量的比例关系。

\textbf{3. 营养素在加工烹调过程中的变化：}加工精度、烹调温度、时间、介质（水/油）等均显著影响营养素的保留率和生物利用率。

\textbf{4. 食物血糖生成指数（GI）：}反映食物中碳水化合物引起血糖升高的速度和程度（见第1章第3节）。

\textbf{5. 食物抗氧化能力：}食物中抗氧化物质（维生素C、维生素E、类胡萝卜素、多酚类等）清除自由基的综合能力。

\textbf{6. 抗营养因子（Anti-nutritional Factors）：}食物中天然存在的、干扰营养素消化吸收的物质——如植酸（影响矿物质吸收）、蛋白酶抑制剂（影响蛋白质消化）、草酸（影响钙吸收）、单宁（影响铁吸收）等。加工处理（加热、发酵、浸泡）可降低或消除部分抗营养因子。
\end{keybox}

\begin{keybox}
\textbf{影响食物营养素组成和含量的主要因素：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{品种、产地和季节：}不同品种的同种食物营养素差异大（如不同品种大米蛋白质含量差3\%\~{}4\%）；地理环境和气候条件影响产地；收获季节决定成熟度。
    \item \textbf{栽培和饲养条件：}土壤肥沃程度、施肥种类、饲料质量均显著影响农产品营养素水平。
    \item \textbf{收获成熟度：}未充分成熟的一般营养价值较低，但过度成熟可能导致营养素流失或口感变差。
    \item \textbf{加工和烹调：}加工精度（如碾米精度对B族维生素的影响极大）、热加工的温度和时间、加工方式（水煮损失水溶性维生素，油炸增加脂肪并损失热不稳定维生素）。
    \item \textbf{储藏条件：}温度、湿度、时间、光照、氧气影响营养素（尤其是维生素C、维生素E、$\beta$-胡萝卜素等抗氧化性维生素）的保留。
\end{enumerate}
\end{keybox}

\begin{tipbox}
\textbf{食物成分数据库：}我国《中国食物成分表》由营养与食品安全所主编，是国民营养与健康状况评价、膳食指导、食物营养标签等的基础性工具。包括原料型食物和加工型食物两大板块。
\end{tipbox}

% ----- 第二节 谷类、薯类及杂豆 -----
\section{谷类、薯类及杂豆}

\begin{keybox}
\textbf{谷类（Cereals）：}我国居民膳食的\keyword{主食}，提供约50\%\~{}70\%的能量和蛋白质。主要包括稻米（大米）、小麦（面粉）、玉米、小米、燕麦、荞麦等。
\end{keybox}

\subsection{谷类的主要营养特点}

\begin{enumerate}[leftmargin=*]
    \item \textbf{碳水化合物：}含量最高，主要为淀粉（70\%\~{}80\%），是人体最经济、最主要的功能来源。淀粉分为直链淀粉和支链淀粉。含少量膳食纤维（全谷中较丰富）。
    \item \textbf{蛋白质：}含量8\%\~{}12\%，\textbf{谷类蛋白的第一限制氨基酸为\keyword{赖氨酸}}。大米蛋白质含量最低（7\%\~{}8\%），生物价值在谷类中相对最高——因其赖氨酸含量稍优于其他谷类。
    \item \textbf{脂类：}含量低（1\%\~{}4\%），主要集中在胚芽（Germ）和糊粉层（Aleurone Layer）。谷类脂类主要含不饱和脂肪酸，且含有维生素E——这也是全谷物粉贮存中酸败的主要原因。
    \item \textbf{维生素：}谷类是B族维生素（尤其是硫胺素B1、尼克酸、吡哆醇B6）的重要来源，主要分布在糊粉层和胚芽。\textbf{碾磨精度越高，B族维生素损失越严重。}
    \item \textbf{矿物质：}约1.5\%\~{}3\%，以磷为主（多为植酸磷，吸收率低），钙、铁含量均较低。
\end{enumerate}

\subsection{全谷物与精制谷物的比较}

\begin{keybox}
\textbf{全谷物 vs 精制谷物：}
\begin{itemize}[leftmargin=*]
    \item \textbf{全谷物：}保留麸皮、胚乳和胚芽。富含膳食纤维、B族维生素、维生素E、矿物质和植物化学物（多酚、植物固醇、木酚素）。
    \item \textbf{精制谷物：}仅保留胚乳（淀粉+少量蛋白质），B族维生素、膳食纤维等大量损失。
    \item \textbf{推荐：}《中国居民膳食指南（2022）》建议每日摄入全谷物和杂豆50\~{}150 g。
\end{itemize}
\end{keybox}

\subsection{谷类种子的结构和营养素分布}

\begin{keybox}
\textbf{谷类种子解剖结构（由外到内）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{谷皮（Bran/Husk）：}主要由纤维素、半纤维素组成，含一定量的蛋白质、脂肪、维生素和无机盐，\textbf{不含淀粉}。
    \item \textbf{糊粉层（Aleurone Layer）：}较多的蛋白质、脂肪和\keyword{丰富的B族维生素及无机盐}。是B族维生素在谷类中的主要贮存部位。
    \item \textbf{胚乳（Endosperm）：}\keyword{大量淀粉}（70\%\~{}80\%）和一定量的蛋白质，无机盐和维生素含量很少。
    \item \textbf{胚（Germ）：}富含脂肪、无机盐、维生素B$_1$和维生素E。是谷类脂肪和维生素E的集中部位，也是全谷物粉贮存中易酸败的原因。
\end{enumerate}
碾磨加工精度越高，谷皮和糊粉层损失越多 → B族维生素损失越严重。
\end{keybox}

\subsection{常见谷物的营养特点}

\begin{keybox}
\textbf{燕麦（Oat）：}
\begin{itemize}[leftmargin=*]
    \item 蛋白质含量约15\%，脂肪约8.5\%——显著高于大米、小麦、玉米等常用食粮。
    \item 脂肪酸中亚油酸占38.1\%\~{}52.0\%，营养价值高。
    \item 富含VB$_1$、钙和膳食纤维。
    \item 含有禾谷类作物中独有的\keyword{皂苷}，对降低胆固醇、甘油三酯有一定功效。
\end{itemize}
\end{keybox}

\begin{keybox}
\textbf{荞麦（Buckwheat）：}
\begin{itemize}[leftmargin=*]
    \item 含有宝贵的生理活性成分，被当作\keyword{粮药兼备}食物。有甜荞和苦荞两种，后者维生素P（芦丁）含量比前者高12倍多。
    \item 蛋白质含量和氨基酸评分在所有谷类作物中名列前茅，水溶性蛋白占31\%\~{}47\%，易消化吸收。
    \item 淀粉颗粒小，易糊化，故易消化。
    \item \keyword{芦丁（Rutin）}能影响胰岛细胞功能，有助于降低血糖。
\end{itemize}
\end{keybox}

\begin{tipbox}
\textbf{玉米的营养特点：}
\begin{itemize}[leftmargin=*]
    \item 蛋白质主要是醇溶蛋白和谷蛋白，几乎不含赖氨酸，营养价值不高。
    \item 胚芽中脂类含量高达30\%\~{}40\%，以亚油酸较多，是优质食用油来源。
    \item 维生素：VE、VB$_1$和VB$_6$丰富，烟酸属于\keyword{结合型}（不利于吸收，加碱处理可释放）。黄玉米含有较多的$\beta$-胡萝卜素。
    \item 矿物质：钾最多，其次为磷、镁，含钙特少。
\end{itemize}
\end{tipbox}

\subsection{薯类}

\textbf{马铃薯（Potato）：}水分含量79\%左右，淀粉17\%。淀粉颗粒比其他粮谷淀粉大许多，具有糊化温度低、黏度大、膨润力大等品质。含有丰富的钾和VC。\textbf{注意：}发芽或绿色皮部含有有毒的\keyword{龙葵素（Solanine）}。

\textbf{甘薯（Sweet Potato）：}水分约68\%，淀粉29\%，含2\%\~{}6\%的可溶性糖（显甜味）。含有$\beta$-淀粉酶和$\alpha$-淀粉酶，贮存或蒸煮时可将部分淀粉转化为麦芽糖和糊精。$\beta$-胡萝卜素含量高于一般谷类。膳食纤维丰富。

\subsection{杂豆}

\textbf{杂豆（赤豆、绿豆、芸豆、蚕豆、豌豆等）：}
\begin{itemize}[leftmargin=*]
    \item 蛋白质含量20\%\~{}25\%（低于大豆，高于谷类），脂类约1\%，碳水化合物55\%以上（主要是淀粉）。
    \item 种皮颜色深的豆类胡萝卜素含量较高。B族维生素和矿物质丰富。
    \item 代表性品种：
    \begin{itemize}[leftmargin=*]
        \item \textbf{蚕豆：}蛋白质24\%\~{}30\%（豆类中仅次于大豆），脂肪约1\%，碳水化合物51\%\~{}66\%。
        \item \textbf{豌豆：}必需氨基酸丰富，赖氨酸含量接近肉类。青豌豆含丰富的胡萝卜素和VC。
        \item \textbf{绿豆：}蛋白质为近全价蛋白质，赖氨酸含量高，含硫氨基酸为限制氨基酸。含丰富的半纤维素和半乳聚糖，具有清热、解毒的药理作用。
        \item \textbf{赤豆：}蛋白质21\%\~{}24\%，脂肪2.2\%，淀粉64\%，淀粉粒径较大（适合制豆沙），富含VB$_1$和VB$_2$。
    \end{itemize}
\end{itemize}

\begin{keybox}
\textbf{生物强化技术（Biofortification）：}通过常规育种手段提高农作物中微量营养素和功能成分含量，简便、经济地减少和预防人体营养不良。"中国生物强化项目"——如富硒大米、高铁小麦等。
\end{keybox}

% ----- 第三节 大豆类及其制品 -----
\section{大豆类及其制品}

\begin{keybox}
\textbf{大豆（Soybean）：}含蛋白质35\%\~{}40\%，是植物性食物中蛋白质含量最高者，且为\keyword{优质植物蛋白}。含脂肪15\%\~{}20\%，主要为不饱和脂肪酸（亚油酸占50\%以上）。还含大豆异黄酮、大豆皂苷、大豆磷脂、大豆低聚糖等有益健康成分。
\end{keybox}

\subsection{大豆及制品的蛋白质}

\begin{itemize}[leftmargin=*]
    \item 大豆蛋白质氨基酸模式较好，属于优质蛋白。
    \item \textbf{第一限制氨基酸为\keyword{蛋氨酸（Met）}}——这是与谷类蛋白质（第一限制氨基酸为赖氨酸）互补的基础。
    \item \textbf{谷类+豆类互补作用：}谷类的赖氨酸不足→由大豆的蛋氨酸缺陷结构得到互补，显著提高蛋白质生物价。
    \item 大豆制成的传统豆制品：豆腐（蛋白质6\%\~{}8\%）、豆浆（蛋白质2\%\~{}3\%）、腐竹（蛋白质高达44\%\~{}50\%，脂肪含量也相对较高）。
\end{itemize}

\subsection{豆制品分类}

\begin{enumerate}[leftmargin=*]
    \item \textbf{非发酵豆制品：}豆腐、豆浆、豆腐干、腐竹、豆皮等。
    \item \textbf{发酵豆制品：}酱油、豆豉、豆瓣酱、腐乳、纳豆等。发酵可提高B族维生素含量和蛋白质消化率，部分产生具有生物活性的肽类。
\end{enumerate}

\subsection{大豆中其他有益成分}

\begin{itemize}[leftmargin=*]
    \item \textbf{大豆异黄酮：}具有植物雌激素作用和抗氧化作用（见第2章）。
    \item \textbf{大豆皂苷：}降血脂、抗肿瘤、免疫调节作用。
    \item \textbf{大豆磷脂：}降血脂、保护肝脏、促进神经系统发育。
    \item \textbf{大豆低聚糖（水苏糖、棉子糖）：}促进双歧杆菌增殖（益生元作用）。
\end{itemize}

\begin{tipbox}
\textbf{豆类的抗营养因子及处理：}
\begin{itemize}[leftmargin=*]
    \item \textbf{蛋白酶抑制剂（Trypsin Inhibitor）：}抑制胰蛋白酶活性，引起胰腺肥大和蛋白质消化率降低。
    \item \textbf{大豆凝血素（Hemagglutinin）：}凝集红细胞，影响营养吸收。
    \item \textbf{甲状腺肿素（Goitrogens）：}干扰碘的利用，可能引起甲状腺肿大。
    \item \textbf{脂肪氧化酶（Lipoxygenase）：}引起豆腥味。
    \item \textbf{植酸（Phytic Acid）：}干扰矿物质（钙、铁、锌）吸收。
\end{itemize}
这些因子均可通过充分加热（煮沸\~{}30min以上）彻底灭活。因此，\textbf{豆浆必须煮沸后食用}。

\textbf{大豆中的生物活性物质（补充）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{大豆磷脂：}降血脂、保护肝脏、促进神经系统发育。
    \item \textbf{大豆皂苷：}降血脂、抗肿瘤、免疫调节。
    \item \textbf{大豆异黄酮：}植物雌激素作用和抗氧化作用。
    \item \textbf{大豆膳食纤维：}促进肠道健康。
    \item \textbf{大豆低聚糖（水苏糖、棉子糖）：}促进双歧杆菌增殖（益生元作用）。
\end{itemize}

\begin{tipbox}
\textbf{豆制品的营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{豆腐干：}含水量65\%\~{}78\%，千张水分更低，蛋白质可达20\%\~{}35\%。
    \item \textbf{豆浆：}营养成分接近牛奶。各种营养素的利用率经加热后均有所提高。
    \item \textbf{大豆振兴计划：}大豆是优质的植物蛋白资源，也是健康的食用植物油资源。增加大豆制品摄入对富裕地区和欠发达地区居民均具有重要意义。
\end{itemize}
\end{tipbox}
\end{tipbox}

% ----- 第四节 蔬菜、水果类 -----
\section{蔬菜、水果类}

\subsection{蔬菜的分类体系}

\begin{keybox}
\textbf{蔬菜按食用部位分类：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{根菜类：}肉质直根（萝卜、胡萝卜、芜菁等）；肉质块根（甘薯、葛等）。
    \item \textbf{茎菜类：}地上茎（竹笋、莴苣、榨菜等）；地下茎（马铃薯、山药、藕、姜、荸荠等）。
    \item \textbf{叶菜类：}普通叶菜（白菜、菠菜、苋菜等）；结球叶菜（包心菜、大白菜等）；香辛叶菜（葱、韭菜、芫荽等）；鳞茎状叶菜（洋葱、大蒜、百合等）。
    \item \textbf{花菜类：}黄花菜、花椰菜（西兰花、菜花）等。
    \item \textbf{果菜类：}瓠果（黄瓜、南瓜、冬瓜、苦瓜等）；浆果（茄子、番茄、辣椒等）；荚果（菜豆、扁豆、豌豆、蚕豆等）。
    \item \textbf{孢子植物类：}食用蕨类（蕨、紫萁）、食用地衣（石耳）、食用菌类（香菇、猴头菌、竹荪等）、食用藻类（海带、紫菜、石花菜等）。
\end{enumerate}
\end{keybox}

\subsection{水果的分类}

\begin{keybox}
\textbf{水果按果实结构分类：}
\begin{itemize}[leftmargin=*]
    \item \textbf{浆果：}肉质多汁，如葡萄、猕猴桃、蓝莓、草莓等。
    \item \textbf{核果：}如桃、杏、樱桃、李、梅等。
    \item \textbf{仁果：}如苹果、梨、山楂等。
    \item \textbf{柑果：}如橘、橙、柠檬等柑橘类。
\end{itemize}
\end{keybox}

\begin{keybox}
\textbf{蔬菜和水果的共同营养贡献：}蔬菜和水果是维生素C、类胡萝卜素、叶酸、维生素K、钾、镁和膳食纤维的最重要来源。同时富含多种植物化学物（多酚类、有机硫化物、类胡萝卜素等），对健康维护具有不可替代的重要意义。
\end{keybox}

\subsection{蔬菜的营养特点}

\begin{enumerate}[leftmargin=*]
    \item \textbf{碳水化合物：}含量因种类差异大（2\%\~{}20\%），主要为可溶性糖（果糖、葡萄糖、蔗糖）和不溶性膳食纤维（纤维素、半纤维素、木质素）。
    \item \textbf{蛋白质和脂肪：}含量极低（蛋白质≤2\%，脂肪<0.5\%）。
    \item \textbf{维生素：}
    \begin{itemize}[leftmargin=*]
        \item \keyword{维生素C}：深绿色叶菜、辣椒、西兰花、番茄中含量丰富。
        \item \keyword{类胡萝卜素}（维生素A原）：深绿色和橙黄色蔬菜中丰富（胡萝卜、菠菜、南瓜）。
        \item \keyword{叶酸（Folate）}：深绿色叶菜（菠菜、西兰花）、豆类和坚果中含量丰富。
        \item \keyword{维生素K}：绿叶蔬菜是最主要的膳食来源。
    \end{itemize}
    \item \textbf{矿物质：}富含钾（有助于维持血压正常）、镁。钙在部分蔬菜（如芥菜、苋菜）中含量较高，但草酸等抗营养因子影响其吸收。
    \item \textbf{植物化学物：}有机硫化物（葱属和十字花科）、多酚类（各种有色蔬菜）、皂苷等。
\end{enumerate}

\subsection{水果的营养特点}

\begin{enumerate}[leftmargin=*]
    \item \textbf{碳水化合物：}主要为简单糖（果糖、葡萄糖、蔗糖），含量6\%\~{}25\%。含水量一般80\%以上，能量较低。
    \item \textbf{维生素：}新鲜水果是\keyword{维生素C}的最主要来源（猕猴桃、柑橘、草莓、鲜枣中含量极高）。也含类胡萝卜素（芒果、杏、木瓜）。
    \item \textbf{矿物质：}钾含量丰富（与血压控制相关）。
    \item \textbf{膳食纤维：}果胶含量较高，可降低血清胆固醇。
    \item \textbf{植物化学物：}花色素（浆果）、白藜芦醇（葡萄）、类胡萝卜素、黄酮类等。
\end{enumerate}

\begin{tipbox}
\textbf{蔬菜与水果的摄入建议（膳食指南2022）：}蔬菜摄入量300\~{}500 g/d，其中深色蔬菜应占一半（\~{}250 g/d）；水果200\~{}350 g/d，不能以果汁替代整果。蔬菜和水果在营养素谱系上有互补性，不可相互替代。
\end{tipbox}

\begin{keybox}
\textbf{蔬菜营养损失的关键因素：}
\begin{itemize}[leftmargin=*]
    \item \textbf{水溶性维生素：}维生素C和B族维生素在洗切、浸泡和煮沸中极易损失。
    \item \textbf{热敏性维生素：}维生素C、叶酸和部分B族维生素在高温下大量破坏。
    \item \textbf{氧化损失：}切碎后接触空气中的氧和光照，加速维生素氧化破坏。
    \item \textbf{合理烹调建议：}先洗后切、急火快炒、炒完即食，减少加工损失。
\end{itemize}
\end{keybox}

\subsection{菌藻类食物}

\begin{keybox}
\textbf{菌藻类食物营养特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{菌类：}蘑菇、香菇、银耳、木耳等。蛋白质含量高，发菜、香菇、蘑菇可高达20\%以上。碳水化合物以多糖为主（香菇多糖、银耳多糖等具有免疫调节活性）。含有丰富的B族维生素（VB$_1$、VB$_2$），蘑菇和紫菜中富含胡萝卜素。矿物质丰富（钙、镁、铁、锌等）。
    \item \textbf{藻类：}海带、紫菜、发菜等。富含碘（海产品中尤甚）、钙、铁、锌等多种矿物质。干品碳水化合物可高达50\%以上，以多糖为主。蛋白质含量差异大。
\end{itemize}
\end{keybox}

\subsection{野菜、野果的营养价值}

\begin{tipbox}
\textbf{野菜和野果：}野菜的胡萝卜素、核黄素、维生素C、钙和铁的含量很高；野果的维生素C含量非常高（如刺梨、沙棘、酸枣等）。\textbf{高维生素C蔬菜代表（mg/100g可食部）：}小红辣椒（144）、酸浆草（127）、苜蓿芽（118）、甘蓝（76）、柿子椒（72）、豌豆苗（67）、油菜薹（65）。\textbf{高胡萝卜素蔬菜代表（$\mu$g/100g可食部）：}胡萝卜（4010）、菠菜（2920）、韭菜（1410）、南瓜（890）。\textbf{高胡萝卜素水果代表：}芒果（8050）、柑橘类（800\~{}5140）、杏（450）、柿子（440）。
\end{tipbox}

% ----- 第五节 畜、禽、水产品 -----
\section{畜、禽、水产品}

\begin{keybox}
\textbf{动物性食物的核心营养贡献：}畜、禽和水产品是\keyword{优质蛋白质}、\keyword{血红素铁}和\keyword{维生素B${}_{12}$}的最重要来源。B12是唯一只能从动物性食物中获得的维生素。
\end{keybox}

\subsection{畜肉（猪、牛、羊）}

\begin{keybox}
\textbf{畜肉水分特点：}水分在肉中占绝大部分（肌肉70\%\~{}80\%、皮肤60\%\~{}70\%、骨骼12\%\~{}15\%），分布不均匀。在肌肉中水与蛋白质呈凝胶状态而存在。
\end{keybox}

\begin{itemize}[leftmargin=*]
    \item \textbf{蛋白质：}含量10\%\~{}20\%（猪肉9.5\%、羊肉11\%、牛肉20\%），氨基酸模式与人体接近，为优质蛋白。按存在形式分为：
    \begin{enumerate}[leftmargin=*]
        \item \textbf{肌原纤维蛋白质：}由丝状的蛋白质凝胶构成，是肌肉收缩的物质基础，占肌肉总蛋白的2/3，是\keyword{优质蛋白质}。
        \item \textbf{肌浆蛋白质：}为水溶性蛋白质，如\keyword{肌红蛋白}——与肉的颜色有关。
        \item \textbf{肉基质蛋白质：}包括胶原蛋白和弹性硬蛋白，构成动物支撑结构，是\keyword{不完全蛋白质}。
    \end{enumerate}
    \item \textbf{脂肪：}含量因种类、部位而异（5\%\~{}30\%），以\keyword{饱和脂肪酸}为主（猪肉约59\%、羊肉28\%、牛肉10\%）。90\%为中性脂肪，饱和脂肪酸以硬脂酸和软脂酸居多，不饱和脂肪酸以油酸居多。生长在肌束间或肌纤维间的脂肪称为\keyword{肌内脂肪}，使鲜肉呈现大理石样花纹——是牛肉肉质好坏的重要标志。内脏脂肪含量更高，胆固醇含量高。
    \item \textbf{浸出物：}除蛋白质、盐类、维生素外能溶于水的浸出性物质。畜肉在水中煮时溶于水中的含氮物质称为\keyword{"含氮浸出物"}，主要包括核苷酸、嘌呤碱、肌酸、肌酐、氨基酸、肽类等——它们是\keyword{肉汤鲜味的主要成分}。
    \item \textbf{碳水化合物：}极低（<1\%，主要以糖原形式存在）。
    \item \textbf{矿物质：}\textbf{血红素铁（Heme Iron）}——吸收率高达15\%\~{}25\%，是植物性非血红素铁（吸收率1\%\~{}5\%）的数倍。锌的吸收利用率也高。
    \item \textbf{维生素：}B族维生素（硫胺素B1——猪肉含量最高；吡哆醇B6——鸡肉中丰富；B12——所有肉类中均有）。维生素A在内脏（肝）中极高。
\end{itemize}

\subsection{禽肉（鸡、鸭）}

\begin{itemize}[leftmargin=*]
    \item \textbf{蛋白质：}优质蛋白，较畜肉细嫩易消化。含量：鹅约10\%、鸭约16.5\%、\textbf{鸡约21.5\%}。鸡胸肉蛋白质含量最高、脂肪最少（\~{}21\%蛋白、3\%脂肪），为健身人群的首选动物蛋白来源。
    \item \textbf{脂肪：}含量低于畜肉，集中于肉皮；\keyword{亚油酸丰富（约20\%）}，高于畜肉脂肪。不饱和脂肪酸比例高于畜肉、胆固醇含量较低。
    \item \textbf{维生素：}VB族与畜肉接近，烟酸高；含VE。内脏富含VA、VB$_2$。
    \item \textbf{矿物质：}钙、磷、铁高于畜肉。
\end{itemize}

\subsection{水产品（鱼、虾、贝类）}

\textbf{水产品的四大特性：}①多样性（种类繁多）；②\keyword{易腐性}（含水量高、组织脆弱、酶活性高）；③捕获量的多变性（年际产量波动大）；④鱼体成分的变动（随种类、季节、营养状况等变化）。

\textbf{鱼类一般成分：}水分70\%\~{}85\%、蛋白质15\%\~{}20\%、脂肪1\%\~{}10\%、碳水化合物0.5\%\~{}1.0\%、矿物质1.0\%\~{}1.5\%。

\begin{itemize}[leftmargin=*]
    \item \textbf{蛋白质：}含量15\%\~{}20\%，易被人体消化吸收。鱼肉蛋白中赖氨酸和亮氨酸含量丰富。鱼肌肉中的\keyword{红肌含有丰富的氨基乙磺酸（牛磺酸）}——一种含硫氨基酸，有利于胎儿和新生儿的大脑和眼睛发育。
    \item \textbf{脂肪：}显著低（大部分<5\%），多存在于皮下组织、肠间膜、肝脏及头盖腔部位。以\keyword{不饱和脂肪酸}为主（海鱼脂肪中不饱和脂肪酸高达70\%\~{}80\%），鱼脂多呈液态，消化率极高。软脂酸是鱼脂中主要的饱和脂肪酸。磷脂是组织脂肪的主要成分，多分布于脑、内脏和生殖腺中。深海鱼油含EPA（二十碳五烯酸）和DHA（二十二碳六烯酸），对心血管保护和脑发育有重要作用。
    \item \textbf{矿物质：}鱼肉中矿物质含量高于畜肉，其中磷占40\%，钙、钠次之。钾、镁、锌、硒、碘很丰富。鱼类铁含量比多数肉类低。\keyword{牡蛎是含锌、铜最高的海产品}。带骨的鱼类罐头是钙和磷的优质来源。
    \item \textbf{维生素：}鱼肌肉是B族维生素的良好来源（贝类和甲壳类更高）。鱼脂肪是维生素A和维生素D的优良来源。
\end{itemize}

\begin{exambox}
\textbf{考点提示——动物铁 vs 植物铁：}
\begin{itemize}[leftmargin=*]
    \item 动物食物：血红素铁，直接进入小肠上皮细胞，吸收率15\%\~{}25\%，不受膳食中植酸等抑制因素影响。
    \item 植物食物：非血红素铁，需先转化为Fe$^{2+}$，吸收率1\%\~{}5\%，受植酸、多酚、钙等抑制因素影响极大。
\end{itemize}
这是营养学中判断食物铁营养价值的关键区别。
\end{exambox}

% ----- 第六节 蛋类及蛋制品 -----
\section{蛋类及蛋制品}

\begin{keybox}
\textbf{蛋类的营养地位：}全鸡蛋蛋白质的\keyword{氨基酸模式与人体最接近}——是自然食物中生物价值最高的蛋白质来源。常被作为"参考蛋白质（Reference Protein）"用于食物蛋白质质量评价。
\end{keybox}

\subsection{蛋类的化学成分}

\textbf{全蛋：}
\begin{itemize}[leftmargin=*]
    \item 蛋白质11\%\~{}14\%，大部分为卵清蛋白（蛋清中），8种必需氨基酸齐全，生物价（BV）高达94。
    \item 脂肪8\%\~{}11\%，集中于蛋黄中，主要为甘油三酯和磷脂（卵磷脂为主），不饱和脂肪酸为主。
    \item 胆固醇含量高：蛋黄中约1500 mg/100g（每个鸡蛋约200\~{}300 mg胆固醇）。
    \item 微量营养素：维生素A、D、B2、B12、生物素含量较高；矿物质以磷、钙、铁、锌和硒为主（蛋黄中）。
\end{itemize}

\textbf{蛋清 vs 蛋黄：}
\begin{table}[h!]
\centering
\begin{tabular}{p{2.5cm}p{4cm}p{4cm}}
\hline
\textbf{营养成分} & \textbf{蛋清（约占蛋重60\%）} & \textbf{蛋黄（约占蛋重30\%）} \\
\hline
蛋白质 & 以卵清蛋白为主，约10\% & 卵黄蛋白为主，约16\% \\
脂肪 & 含量极低 & 30\%\~{}33\%，含丰富的卵磷脂 \\
维生素 & 极少 & 富含A、D、E、B族 \\
矿物质 & 极少 & 钙、磷、铁、锌、硒 \\
胆固醇 & 无 & 约1500 mg/100g \\
\hline
\end{tabular}
\end{table}

\subsection{蛋制品的营养变化}

\begin{itemize}[leftmargin=*]
    \item \textbf{咸蛋：}水分减少，钠含量显著增高，其他营养素基本保留。
    \item \textbf{皮蛋：}碱性加工条件下维生素B族大量破坏，维生素A和D可保留。
    \item \textbf{蛋粉：}脱水浓缩，营养素浓缩，但维生素A和D有一定损失。
\end{itemize}

% ----- 第七节 乳及乳制品 -----
\section{乳及乳制品}

\begin{keybox}
\textbf{乳类的核心营养地位：}乳类（特别是牛乳）是\keyword{最佳膳食钙来源}——不仅钙含量高（\~{}104 mg/100g），而且钙的吸收率高达32.1\%，远超其他钙源。乳类还是维生素B2（核黄素）和优质蛋白质的重要来源。
\end{keybox}

\subsection{牛乳的营养成分}

\begin{enumerate}[leftmargin=*]
    \item \textbf{蛋白质：}3.0\%\~{}3.5\%，主要为\keyword{酪蛋白（Casein）}（占80\%）和\keyword{乳清蛋白（Whey）}（占20\%）。乳清蛋白中的$\alpha$-乳白蛋白是母乳中的优势蛋白（见第4章）。生物价87，仅次于全鸡蛋。
    \item \textbf{脂肪：}3\%\~{}4\%，以甘油三酯为主，饱和脂肪酸为主，胆固醇含量低（10\~{}15 mg/100g）。以乳化小颗粒状存在，易于消化。
    \item \textbf{碳水化合物：}主要为\keyword{乳糖（Lactose）}（4.5\%\~{}5.0\%）。
    \begin{itemize}[leftmargin=*]
        \item 乳糖在肠道被乳糖酶水解为葡萄糖+半乳糖。
        \item \textbf{乳糖不耐症（Lactose Intolerance）：}因乳糖酶活性不足，未消化的乳糖进入结肠被细菌发酵产生气体和酸性产物，出现腹胀、腹痛和腹泻。亚洲人群发生率极高（80\%\~{}90\%）。
    \end{itemize}
    \item \textbf{钙：}含量104 mg/100g，吸收率32.1\%——高含量+高吸收率使牛乳成为不可替代的膳食钙源。
    \item \textbf{维生素：}维生素A（全脂乳丰富）、维生素B2（约0.13 mg/100g，是重要的核黄素来源）、维生素D（通常强化）。维生素C含量极低，在加热中大部分损失。
    \item \textbf{矿物质：}钙磷比为1.3:1，利于钙吸收。铁含量极低（<0.05 mg/100g，因此牛乳是缺铁性贫血的危险因素——婴幼儿单纯依靠牛乳可引致缺铁）。
\end{enumerate}

\subsection{常见乳制品}

\begin{itemize}[leftmargin=*]
    \item \textbf{酸奶（Yogurt）：}发酵后乳糖含量降低，可部分缓解乳糖不耐受；益生菌有利于肠道健康。
    \item \textbf{奶酪（Cheese）：}蛋白质和钙浓缩（钙400\~{}800 mg/100g），脂肪浓缩（25\%\~{}35\%），乳糖极低（适合乳糖不耐受者）。
    \item \textbf{奶粉（Milk Powder）：}脱水制成，营养素浓缩，适宜储藏和运输。
\end{itemize}

% ----- 第八节 坚果类 -----
\section{坚果类}

\begin{keybox}
\textbf{坚果的营养特点：}坚果富含不饱和脂肪酸、优质蛋白质、维生素E、B族维生素和多种矿物质（镁、锌、硒），是能量和营养素的浓缩食物。主要分为两类：树坚果（核桃、杏仁、腰果等）和籽类（葵花籽、南瓜籽等）。
\end{keybox}

\subsection{坚果的主要营养价值}

\begin{enumerate}[leftmargin=*]
    \item \textbf{脂肪：}含量极高（\~{}50\%，有的高达60\%+），以不饱和脂肪酸（单不饱和+多不饱和）为主。核桃的$\alpha$-亚麻酸（n-3必需脂肪酸）含量在食物中名列前茅。
    \item \textbf{蛋白质：}15\%\~{}25\%，大豆除外，是植物性食物中的高蛋白来源。
    \item \textbf{维生素：}维生素E含量丰富（重要的脂溶性抗氧化剂）、B族维生素（尤其是叶酸和吡哆醇B6）。
    \item \textbf{矿物质：}镁、钾、铜、锌、硒含量较高。
    \item \textbf{植物化学物：}植物固醇、多酚类、木酚素等。
\end{enumerate}

\subsection{食用注意事项}

\begin{itemize}[leftmargin=*]
    \item \textbf{能量密度极高：}100 g核桃约600\~{}700 kcal，过量摄入易导致能量过剩和体重增加。
    \item \textbf{中国居民膳食指南（2022）：}每日建议摄入量10\~{}15 g（约一小把），不宜贪多。
    \item \textbf{过敏风险：}花生和坚果是最常见的食物过敏原之一，在婴幼儿添加食物中须注意。
\end{itemize}

% ----- 第九节 食用油脂 -----
\section{食用油脂}

\subsection{食用油脂的营养价值}

油脂的营养价值主要体现在\keyword{脂肪酸构成}和\keyword{微量营养素}两个方面。常见食用油的脂肪酸构成：

\begin{keybox}
\textbf{常见植物油的脂肪酸构成与特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{大豆油：}使用范围最广、消费量最大。脂肪酸构成：饱和脂肪酸15.9\%，油酸22.4\%，亚油酸51.7\%，$\alpha$-亚麻酸6.7\%。含磷脂、胡萝卜素、VE、甾醇等微量营养素。适合煎炒烹炸。
    \item \textbf{菜籽油（Rapeseed Oil）：}饱和脂肪酸13.2\%，油酸20.2\%，亚油酸16.3\%，$\alpha$-亚麻酸8.4\%。含VE、菜油甾醇。产量仅次于大豆油。
    \item \textbf{棕榈油（Palm Oil）：}全球产量最大的植物油。饱和脂肪酸43.4\%，油酸44.4\%，亚油酸12.1\%。氧化稳定性高，尤其耐煎炸，常代替动物油用于烘焙和方便面生产。
    \item \textbf{橄榄油（Olive Oil）：}地中海沿岸国家最广泛食用的油脂，属于果肉脂肪。油酸含量高（40.4\%），亚油酸37.9\%，$\alpha$-亚麻酸0.4\%。含白藜芦醇、$\beta$-谷固醇、VE等。
    \item \textbf{花生油：}饱和脂肪酸18.5\%，油酸40.4\%，亚油酸37.9\%，$\alpha$-亚麻酸0.4\%。具有特殊花生香气，是老年人的健康用油之选。易被空气氧化变质，需低温保存。
    \item \textbf{茶籽油（Camellia Oil）：}饱和脂肪酸13\%，油酸22\%，亚油酸14\%，\textbf{$\alpha$-亚麻酸高达49\%}。含VE、甾醇、茶多酚、角鲨烯等。
\end{itemize}
\end{keybox}

\begin{keybox}
\textbf{常见动物油的脂肪酸构成与特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{乳脂（Butter Fat）：}短链脂肪酸较多，亚油酸等PUFA较少。含VA、VD等。胆固醇含量较高。
    \item \textbf{牛脂（Tallow）：}软脂酸（C16）、硬脂酸（C18）和油酸约占90\%。熔点35\~{}50$^\circ$C，口中熔化性不好。
    \item \textbf{猪油（Lard）：}不饱和脂肪酸占一半以上，多为油酸和亚油酸，饱和脂肪酸多为软脂酸。熔点较低（约28\~{}30$^\circ$C），起酥性好。
    \item \textbf{鱼油（Fish Oil）：}主要来自海洋鱼类，富含n-3多不饱和脂肪酸（EPA和DHA），易酸败。常制成胶囊服用。
\end{itemize}
\end{keybox}

\begin{tipbox}
\textbf{食用油选择建议：}
\begin{itemize}[leftmargin=*]
    \item 三高人群：橄榄油、茶籽油、稻米油、花生油、亚麻籽油换着吃。
    \item 健康家庭：大豆油、葵籽油是经济实惠又健康的选择。
    \item 海鲜摄入较少的人群：适当食用亚麻籽油可有效补充n-3脂肪酸。
    \item 成人每天烹调油25\~{}30g。
\end{itemize}
\end{tipbox}

% ----- 第十节 食盐与食糖 -----
\section{食盐与食糖}

\subsection{食盐（Salt）}

盐是使用率最高的调味品，主要成分是氯化钠（NaCl）。钠离子提供最纯正的咸味，氯离子为助味剂。除了提供咸味外，盐还是常用的抑菌剂。食盐每日必用，是营养强化的绝佳载体之一（如加碘盐、低钠高钾盐、富硒盐等）。\textbf{成人每天食盐摄入量应控制在6g以下}（2022膳食指南建议<5g）。

\subsection{食糖（Sugar）}

蔗糖提供纯正愉悦的甜味。白糖为精制糖，纯度高；红糖制作相对粗糙，含有较多的矿物质。\textbf{成人每天摄入糖不超过50g，最好控制在25g以下。}

% ----- 第十一节 烹饪加工对食物营养价值的影响 -----
\section{烹饪加工对食物营养价值的影响}

\subsection{合理烹饪的意义}

\begin{defbox}
\textbf{合理烹饪：}根据不同烹饪原料的营养特点和各种营养素的理化性质，合理地采用烹饪加工方法，使菜肴在色、香、味、形等方面达到烹饪工艺要求的同时，尽量保持营养素、消除有害物质、容易消化吸收。
\end{defbox}

合理烹饪的意义：①杀灭原料中的有害生物；②除去或减少有害化学物质；③尽可能保存原料中的营养素；④改善食物的感官性质，使之易于消化吸收。

\subsection{烹饪加工中营养素变化的特点}

最大的影响因素是\keyword{烹调的温度和时间}（"火候"）。

\textbf{营养素损失的两大途径：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{流失：}淘洗、盐渍、研磨等处理造成营养素流失。如大米淘洗后维生素丢失30\%\~{}40\%，矿物质丢失25\%左右。先切后洗、长时间浸泡、焯水等均增加流失。
    \item \textbf{破坏：}食品在加工过程中发生物理、化学或生化变化，使营养素分解或转化，失去营养功能。如油炸造成硫胺素损失，水果加工褐变降低维生素C含量。
\end{enumerate}

\subsection{蛋白质在烹饪加工中的变化}

\begin{keybox}
\textbf{1. 蛋白质变性（Denaturation）：}
天然蛋白质因环境改变（温度、pH、机械作用力）丧失原有生物功能。热处理是最常用的烹饪手段，水能促进蛋白质的热变性。冷冻条件下蛋白质也会变性——\keyword{冻结速度越快，冰晶越小，挤压作用越小，变性程度越小}（速冻可减少营养成分丧失的原因）。

\textbf{2. 蛋白质水解（Hydrolysis）：}
变性蛋白更易水解——如肉类在贮藏过程中因自身酶作用发生适当水解，有利于肉的成熟。烹饪中加入嫩化剂（如生姜含酶）可增加原料水溶性成分，达到增加嫩度和提高风味的目的。恰当的水解可提高蛋白质消化率并增强食品风味。

\textbf{3. 蛋白质分子交联（Cross-linking）：}
温度高、时间长的烹调（如油炸）会促进蛋白质分子间交联，温度越高凝固越紧，食品质感越老，蛋白质消化率大大降低。

\textbf{4. 羰氨反应/美拉德反应（Maillard Reaction）：}
食品中羰基化合物（来自糖类）与氨基化合物（来自蛋白质）经缩合、聚合生成\keyword{类黑色素}。可增色增香（烹饪中良好感官的来源），但也会降低蛋白质成分，产生的中间产物可能具有慢性毒性。
\end{keybox}

\subsection{碳水化合物在烹饪加工中的变化}

\begin{keybox}
\textbf{1. 淀粉糊化（Gelatinization）：}
淀粉粒在适当温度（60\~{}80$^\circ$C）下在水中溶胀、分裂，形成均匀糊状溶液。糊化后淀粉分子呈零散扩张状态，易受淀粉酶作用，更有利于人体吸收。烹饪中的勾芡、煮饭、蒸馒头、烤面包均利用糊化原理。

\textbf{2. 淀粉老化（Retrogradation）：}
糊化后的淀粉在室温或低温下放置，变为不透明并沉淀，称为老化。老化使食物质地变硬干缩，酶水解受阻，影响消化率。
\begin{itemize}[leftmargin=*]
    \item 直链淀粉比支链淀粉更易老化：玉米$>$小麦$>$甘薯$>$马铃薯 $>$ 粘玉米$>$大米$>$糯米
    \item 含水量30\%\~{}60\%最易老化（面包/馒头/米饭）；$<$10\%\~{}15\%不易老化（饼干）；$>$70\%老化变慢
    \item 最适老化温度2\~{}4$^\circ$C；$>$60$^\circ$C或$<$-20$^\circ$C均不易老化
\end{itemize}

\textbf{3. 焦糖化作用（Caramelization）：}
单糖和低聚糖在无氨基化合物存在下，加热至熔点以上变为黑褐色物质。焙烤、油炸、煎炒中发生，可增加食品风味和色泽，且焦糖色素安全性高（我国允许使用的天然色素之一）。

\textbf{4. 多糖和糖苷的水解：}烹饪加热使部分多糖水解，有利于其营养价值的提高。
\end{keybox}

\subsection{脂肪在烹饪加工中的变化}

\begin{keybox}
\textbf{1. 油脂酸败（Rancidity）——三种类型：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{水解酸败：}含低级脂肪酸较多的油脂，在脂酶或微生物作用下水解成低级脂肪酸和甘油，产生特殊汗臭味和苦涩味。
    \item \textbf{酮基酸败：}油脂水解产生的游离饱和脂肪酸，在酶催化下氧化生成有怪味的酮酸和甲基酮。
    \item \textbf{氧化型酸败（自动氧化）：}不饱和脂肪酸暴露在空气中发生自动氧化，氧化产物进一步分解生成低级脂肪酸、醛和酮，产生恶劣臭味。
\end{enumerate}

\textbf{2. 油脂加热老化：}
高温下反复加热的油脂，色泽变深、粘度变稠、泡沫增加、发烟点下降，称为油脂老化。不仅味感变劣，营养价值降低，还产生许多毒性成分。
\begin{itemize}[leftmargin=*]
    \item 高温氧化反应速度远快于低温自动氧化，不饱和脂肪酸含量高的油脂（豆油、菜籽油）更易发生。
    \item 热分解反应：260$^\circ$C以下不明显，290\~{}300$^\circ$C时明显加快。\textbf{烹饪油温应控制在200$^\circ$C以下，最好在150$^\circ$C左右。}
    \item 缩聚反应：油温300$^\circ$C以上或长期反复加热，发生热聚合，产生有毒物质。
\end{itemize}
\end{keybox}

\subsection{维生素和矿物质在烹饪加工中的变化}

\begin{keybox}
\textbf{维生素的变化：}
\begin{itemize}[leftmargin=*]
    \item 脂溶性维生素相对对热稳定（VA一般烹调加工中不易破坏；VD对热、碱稳定；VE对氧敏感易被破坏）。
    \item 水溶性维生素中，烟酸（VPP）最稳定；核黄素（VB$_2$）对热稳定；硫胺素（VB$_1$）在酸性条件下对热稳定；VC在水溶液中极易氧化，遇空气、热、光、碱及氧化酶存在时更易被破坏。\textbf{蔬菜加工中宜采用焯水、热烫等短时间热处理，以防止VC损失。}
\end{itemize}

\textbf{矿物质的变化：}矿物质性质相对稳定，在烹饪中不易流失。但长时间浸泡、焯水、先切后洗、与空气接触面大等不当加工方式均会造成矿物质流失。
\end{keybox}

\subsection{不同烹饪方法对营养素的影响}

\begin{keybox}
\textbf{主食加工：}蒸和煮对营养素保存率最好，烙、烤次之，油炸、油煎最差。油炸时VB$_1$损失100\%，VB$_2$和烟酸损失45\%。加碱会使维生素破坏殆尽。

\textbf{肉、禽、鱼类烹调：}牛肉肌纤维长而粗糙，宜炖、煮、焖、煨等长时间加热法；鱼肉含水量多、肌纤维短，宜蒸以保持水分和细嫩肉质；鸡肉宜炖汤。

\textbf{蔬菜烹调：}\keyword{旺火急炒}营养素损失率最低，可保持蔬菜色泽鲜艳、质地脆嫩。

\textbf{保护营养素的基本方法：}①先洗后切、减少浸泡；②蔬菜旺火急炒；③米饭焖或煮；④适宜的上浆、挂糊、勾芡保护原料中的水分和营养素；⑤加醋和发酵有利于矿物质吸收；⑥酵母发酵可增加VB含量、破坏植酸。
\end{keybox}

\subsection{食品保藏对营养价值的影响}

\begin{keybox}
\textbf{1. 冷冻保藏：}冷冻过程本身不破坏营养素，温度越低保存的营养素越多。合适的包装可防止冷冻食品表面干燥变硬。通常在10$^\circ$C以下大多数微生物难以繁殖，-10$^\circ$C时几乎停止生长。

\textbf{2. 辐照保藏：}用$^{60}$Co和$^{137}$Cs产生的$\gamma$射线处理食品。
\begin{itemize}[leftmargin=*]
    \item 蛋白质：大剂量辐照时部分氨基酸分解、氧化，产生异味。但辐照后蛋白降解相当于预消化，吸收利用率可能增加。
    \item 脂类：不饱和脂肪酸易发生氧化。饱和脂肪对辐射稳定。
    \item 碳水化合物：比较稳定，灭菌剂量（20\~{}50kGy）对糖的消化率和营养价值几乎无影响。
    \item 维生素：损失最多。VE和VA对辐照最敏感；VC极易破坏（浓度越低破坏越大）；VB$_1$敏感，VB$_2$不敏感。\textbf{辐照对维生素的破坏程度比热处理小。}
    \item 矿物质：总量不变，但可能改变存在状态（如二价铁 $\rightarrow$ 三价铁），降低生物有效性。
\end{itemize}

\textbf{3. 高压处理：}在常温或较低温度下加压（流体静压）杀菌。高压对蛋白质三级及四级结构有破坏作用（$>$300MPa不可逆），可提高淀粉对淀粉酶的敏感性从而提高消化率。对一级结构无影响。
\end{keybox}

\end{enumerate}

\subsection*{参考答案要点}

\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item 见本章各概念定义框。
    \item 品种/产地/季节、栽培和饲养条件、收获成熟度、加工/烹调、储藏条件。
    \item 谷类：碳水化合物70\%\~{}80\%（最经济的热能来源），蛋白质8\%\~{}12\%（赖氨酸为第一限制氨基酸），低脂，B族维生素（精加工损失严重）。全谷物优于精制：多含膳食纤维、B族维生素、维生素E、矿物质和植物化学物。
    \item 大豆蛋白35\%\~{}40\%，优质植物蛋白，第一限制氨基酸为蛋氨酸；谷类第一限制氨基酸为赖氨酸。二者混合，谷类提供蛋氨酸、豆类提供赖氨酸，实现必需氨基酸互补。
    \item 蛋白质均优质。脂肪：畜肉饱和脂肪酸多，禽肉不饱和脂肪酸比例较高，水产品以n-3多不饱和脂肪酸为特色。铁：三者均含血红素铁，吸收率15\%\~{}25\%。
    \item 全鸡蛋氨基酸模式与人体最接近，BV高达94。微量营养素集中在蛋黄中（维生素A、D、E、B12、磷、铁、锌、硒、胆固醇）。
    \item 钙含量高（104 mg/100g），吸收率高（32.1\%）。乳糖不耐受机制：乳糖酶活性不足，未被消化的乳糖在结肠经细菌发酵产酸产气。
    \item 共同贡献：维生素C、类胡萝卜素、叶酸、钾、镁、膳食纤维、植物化学物。差异：蔬菜密度更低、可作为主菜食用量大，水果含糖量（简单糖）高，不宜替代蔬菜。
    \item (1) 蛋白质：均优质，蛋类BV最高——参考蛋白质；(2) 脂肪：水产品最低，n-3不饱和脂肪酸具有心血管保护作用；(3) 矿物质：畜肉等动物肉含高吸收率的血红素铁和锌，乳类的钙最高吸收率；(4) 生物活性成分：水产品含EPA/DHA，乳含益生菌/免疫球蛋白，蛋含卵磷脂。
\end{enumerate}
