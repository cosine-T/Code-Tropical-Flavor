% ----- 第一节 生命早期1000天 -----
\section{生命早期1000天与DOHaD理论}

\begin{keybox}
\textbf{生命早期1000天（The First 1000 Days of Life）：}
指从受精卵形成开始的孕期270天（约9个月）加上出生后至2周岁的730天（约2年），总计约1000天的生命早期窗口期。

\textbf{"机遇窗口期"的概念：}这一阶段是生长发育的关键期和敏感期，此时营养状态对个体终身的体格发育、认知潜能、免疫功能和慢性疾病风险产生深远和不可逆的影响。这是进行营养干预以获得最大健康收益的最佳时机。
\end{keybox}

\begin{keybox}
\textbf{DOHaD理论（发育源性健康和疾病，Developmental Origins of Health and Disease）：}
该理论的核心论点为：\textbf{生命早期的营养环境（宫内和出生后早期）可通过"发育程序化（Developmental Programming）"机制，对个体的远期疾病风险产生持久性影响。}

主要机制包括：
\begin{enumerate}[leftmargin=*]
    \item \textbf{表观遗传调控（Epigenetic Regulation）：}早期营养状态通过DNA甲基化、组蛋白修饰等表观遗传机制，改变基因的表达模式和组织分化方向，且这种变化可持续终生。
    \item \textbf{"节俭表型"假说（Thrifty Phenotype Hypothesis）：}宫内营养不良使胎儿产生"预测性适应"——调整代谢和内分泌系统以应对即将来临的"匮乏"环境。若出生后实际面临的是"营养过剩"环境（婴儿期过度喂养），则这种代谢-环境不匹配将大幅增加成年期代谢综合征风险。
    \item \textbf{远期疾病风险：}早期营养不良（宫内发育受限、低出生体重、婴儿期过快生长加速）显著升高成年期肥胖、2型糖尿病、心血管疾病（高血压、冠心病）、骨质疏松甚至某些癌症的发病风险。
\end{enumerate}
\end{keybox}

\begin{exambox}
\textbf{考点提示：} (1) 1000天的计算（270d+730d）；(2) DOHaD的核心逻辑——早期环境→发育程序化（表观遗传）→远期健康/疾病；(3)"不匹配"是DOHaD解释代谢综合征的核心机制——宫内"预测匮乏" + 出生后"实际营养过剩" → 代谢-环境失匹配 → 代谢综合征高风险。
\end{exambox}

\subsection{孕期生理特点}

\begin{keybox}
\textbf{孕期的主要生理变化：}

\textbf{1. 内分泌变化：}
\begin{itemize}[leftmargin=*]
    \item 人绒毛膜促性腺激素（\keyword{HCG}）：妊娠后10天内可在血中检出，是早孕试验的基础。
    \item 人绒毛膜生长激素（HCS）、雌激素（E2/E3）和孕酮（Progesterone）大量升高。
    \item \textbf{胰岛素敏感性下降（生理性胰岛素抵抗）}，是妊娠期糖尿病（GDM）的生理基础。
\end{itemize}

\textbf{2. 消化系统：}
\begin{itemize}[leftmargin=*]
    \item 贲门括约肌松弛→\keyword{早孕反应}（恶心、呕吐、胃灼热）。
    \item 胃肠道蠕动减慢→有利于营养素的吸收（尤其是钙、铁、叶酸），但易出现便秘。
\end{itemize}

\textbf{3. 循环系统：}血容量显著增加（孕中期起），血浆容量增加幅度大于红细胞增加幅度，出现\keyword{生理性贫血}。通常在孕20\~{}30周最明显。

\textbf{4. 体重增长：}正常体重女性（BMI 18.5\~{}23.9 kg/m$^{2}$）的孕期推荐体重增长范围为8\~{}14 kg，平均约12.5 kg。
\end{keybox}

\textbf{关键微量营养素需求：}

\begin{enumerate}[leftmargin=*]
    \item \textbf{叶酸（Folate）——最重要的孕前/孕期微量营养素：}
    \begin{itemize}[leftmargin=*]
        \item \textbf{推荐摄入量：}孕前至孕早期每日补充\keyword{400 μg/d}叶酸或含等量叶酸的复合维生素补充剂。
        \item \textbf{预防神经管缺陷（Neural Tube Defects, NTDs）：}叶酸缺乏是胎儿神经管未能正常闭合（如脊柱裂、无脑儿）的确定病因，充足叶酸补充可使NTDs发生率降低70\%。建议从怀孕前3个月开始补充。
        \item \textbf{膳食来源：}深绿色叶菜、肝脏、豆类、强化面粉。
    \end{itemize}
    \item \textbf{铁（Iron）：}
    \begin{itemize}[leftmargin=*]
        \item RNI：孕中期25 mg/d，孕晚期29 mg/d（较非孕期增加约10\~{}12 mg/d）。
        \item 孕期对铁的需求大幅增加——用于母体红细胞含量增加、胎儿和胎盘需要、产时失血储备。
        \item \textbf{生理性贫血的判断标准（WHO）：}孕期Hb < 110 g/L（非孕女性 < 120 g/L）。
    \end{itemize}
    \item \textbf{碘（Iodine）：}
    \begin{itemize}[leftmargin=*]
        \item RNI：230 μg/d（较非孕增加约110 μg/d）。
        \item \textbf{严重缺乏可致克汀病（先天性碘缺乏综合征、矮小、智力严重障碍）：}因胎儿甲状腺激素依赖母体碘供给。
    \end{itemize}
    \item \textbf{钙（Calcium）：}
    \begin{itemize}[leftmargin=*]
        \item RNI：800 mg/d（孕期钙吸收率显著提高以补偿钙需求）。
        \item 充足摄入可降低妊娠期高血压疾病（妊高征/子痫前期）的发病风险。
    \end{itemize}
    \item \textbf{维生素D：}RNI 10 μg/d（400 IU/d）。缺乏导致新生儿低钙血症和佝偻病风险增加。
\end{enumerate}

\begin{exambox}
\textbf{考点提示——孕期碳水化合物最低量（130 g/d）：}为预防酮体对胎儿中枢神经系统的毒性，孕妇每日碳水化合物摄入量不得低于130 g。在严重早孕呕吐无法进食时，应静脉补充葡萄糖以确保此最低量。
\end{exambox}

\subsection{孕期膳食核心推荐（2022版）}

\begin{keybox}
\textbf{《中国孕期妇女膳食指南（2022）》核心推荐：}
\begin{enumerate}[leftmargin=*]
    \item 孕前将体重调整至适宜范围。
    \item 补充叶酸（400 μg/d），常吃含铁丰富的食物，选用碘盐。
    \item 孕吐严重者，可少量多餐，保证摄入至少\keyword{130 g/d}的碳水化合物。
    \item 孕中晚期适量增加奶、鱼、禽、蛋、瘦肉的摄入（满足蛋白质+15/+30 g/d）。
    \item 适量身体活动，维持孕期适宜体重。
    \item 禁烟酒，愉快孕育新生命。
\end{enumerate}
\end{keybox}

% ----- 第三节 乳母营养 -----
\section{乳母营养}

\subsection{哺乳对母婴双方的益处}

\begin{keybox}
\textbf{母乳喂养对母体的益处：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{促进子宫复原：}吸吮刺激催产素（Oxytocin）的释放，加速产后子宫收缩和恢复。
    \item \textbf{降低远期疾病风险：}哺乳可降低母体患乳腺癌、卵巢癌和2型糖尿病（T2DM）的风险。
    \item \textbf{促进能量负平衡：}哺乳每日消耗约\~{}650 kcal能量—有助于产后体重恢复。
\end{enumerate}
\end{keybox}

\begin{keybox}
\textbf{泌乳过程的神经内分泌调节：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{产奶反射（Milk Production Reflex）：}婴儿吸吮乳头 $\rightarrow$ 刺激下丘脑分泌\keyword{催乳素（Prolactin）} $\rightarrow$ 催乳素与乳腺分泌细胞结合 $\rightarrow$ 促进RNA和DNA合成 $\rightarrow$ 乳腺合成酪蛋白和$\alpha$-乳清蛋白等乳汁成分。
    \item \textbf{下奶反射（Let-down Reflex）：}婴儿吸吮 $\rightarrow$ 下丘脑分泌\keyword{催产素（Oxytocin）} $\rightarrow$ 催产素刺激乳腺肌上皮细胞收缩 $\rightarrow$ 排乳。
\end{enumerate}
\end{keybox}

\begin{keybox}
\textbf{不同阶段母乳的成分变化：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{初乳（Colostrum，产后$\sim$7天）：}脂肪和乳糖含量少但低聚糖高；蛋白质高，多为乳清蛋白；\keyword{免疫球蛋白和细胞因子丰富（SIgA、IL）}；维生素A、锌、铜含量高。
    \item \textbf{过渡乳（Transitional Milk，7$\sim$15天）：}蛋白质总量逐渐减少，脂类和乳糖含量逐渐增高。
    \item \textbf{成熟乳（Mature Milk，15天$\sim$）：}蛋白质含量相对低，乳糖和脂类含量较高。泌乳量：1个月约650mL/d，3个月约750$\sim$1000mL/d，前6个月平均750mL/d。
\end{enumerate}

\textbf{乳汁生成方式：}①直接进入乳汁——钠、钾、磷、镁、氯、大多水溶性维生素和水、血浆；②在乳腺内合成且受激素调控——乳糖由葡萄糖合成（受孕酮、催乳素调控）、乳脂在乳汁内形成膜包围的脂肪小球；③免疫物质——巨噬细胞分泌溶菌酶和乳铁蛋白，淋巴细胞产生IgA和干扰素。
\end{keybox}

\subsection{母乳的分期与成分特点}

\begin{keybox}
\textbf{母乳三阶段的主要成分特征：}

\begin{enumerate}[leftmargin=*]
    \item \textbf{初乳（Colostrum，$\leq$7天）：}
    \begin{itemize}[leftmargin=*]
        \item 脂肪和乳糖含量较低——新生儿早期的能量主要来自乳清蛋白。
        \item \textbf{乳清蛋白/酪蛋白比值最高}（90:10，成熟乳为60:40）。
        \item \textbf{免疫球蛋白含量最高}（sIgA为主）——为新生儿提供被动免疫。
        \item 维生素A和锌含量高于成熟乳。
    \end{itemize}
    \item \textbf{过渡乳（Transitional Milk，7\~{}15天）：}
    \begin{itemize}[leftmargin=*]
        \item 脂肪和乳糖含量逐渐增加，免疫球蛋白等含量开始下降。
    \end{itemize}
    \item \textbf{成熟乳（Mature Milk，>15天）：}
    \begin{itemize}[leftmargin=*]
        \item 脂肪、乳糖和能量含量稳定。前乳（100 ml喂养开始部分）：较稀（约1\%脂肪），富含蛋白质和乳糖。
        \item 后乳（100 ml喂养末尾部分）：脂肪含量可达4\%\~{}5\%（为婴儿提供主要能量）。
    \end{itemize}
\end{enumerate}
\end{keybox}

% ----- 第四节 婴幼儿营养 -----
\section{婴幼儿营养}

\begin{keybox}
\textbf{婴幼儿营养分段管理：}
\begin{itemize}[leftmargin=*]
    \item \textbf{0\~{}6月龄：}纯母乳喂养（Exclusive Breastfeeding）。如无母乳，应给予婴儿配方食品，不可直接用牛乳替代。
    \item \textbf{7\~{}24月龄：}持续母乳喂养+辅食添加（Complementary Feeding）。
\end{itemize}
\end{keybox}

\subsection{0\~{}6月龄——纯母乳喂养}

\begin{keybox}
\textbf{《婴幼儿喂养全球策略》的核心推荐：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{出生后1小时内开始母乳喂养}（产后早期接触有利于建立喂养关系）。
    \item \textbf{0\~{}6月龄纯母乳喂养：}不添加水、其他液体和辅食，仅母乳。
    \item \textbf{按需哺乳（Responsive Feeding）：}按婴儿饥饿信号适时喂养，而非固定时间刻板喂养。
    \item \textbf{每日补充维生素D 400 IU（10 μg）：}母乳中维生素D含量极低。
    \item \textbf{不推荐6月龄前喂牛乳/羊乳：}因铁含量低、蛋白质和矿物质含量过高对婴儿肾脏负担过重。
    \item \textbf{定期监测生长指标：}体重、身长/头围，以了解喂养是否充足。
\end{enumerate}
\end{keybox}

\subsection{母乳的优势}

\begin{enumerate}[leftmargin=*]
    \item \textbf{蛋白质优势：}以\keyword{$\alpha$-乳白蛋白（$\alpha$-Lactalbumin）}为主（占乳清蛋白的41\%），易于消化且过敏性低。酪蛋白与乳清蛋白比例适宜（40:60）。
    \item \textbf{免疫因子：}
    \begin{itemize}[leftmargin=*]
        \item \keyword{乳铁蛋白（Lactoferrin）}：竞争性结合铁，抑制需铁细菌生长。
        \item \keyword{溶菌酶（Lysozyme）}：直接破坏细菌细胞壁。
        \item \keyword{分泌型免疫球蛋白A（sIgA）}：覆盖在婴儿肠黏膜表面，阻断病原体和过敏原的黏附。
    \end{itemize}
    \item \textbf{钙磷比优化：}母乳Ca:P $\approx$ 2:1，利于钙吸收。
    \item \textbf{低聚糖（Oligosaccharides）：}母乳特有的低聚糖可作为益生元，促进双歧杆菌和乳酸杆菌的定植和增殖，构建健康的婴儿肠道微生物群。
    \item \textbf{DHA（二十二碳六烯酸）：}母乳中含有DHA，对婴儿大脑和视觉的发育至关重要。
\end{enumerate}

\subsection{7\~{}24月龄——辅食添加}

\begin{keybox}
\textbf{辅食添加的基本原则：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{开始时间：}满6月龄（约180天）开始添加辅食。
    \item \textbf{第一口辅食：}首选\keyword{富含铁的泥糊状食物}（如强化铁的米粉、动物肝脏泥、瘦肉泥）。因为6月龄时婴儿体内储存的铁已耗尽——\textbf{7\~{}12月龄99\%的铁必须从辅食中获得}，\textbf{75\%的锌同样来自辅食}。
    \item \textbf{添加原则（循序渐进）：}
    \begin{itemize}[leftmargin=*]
        \item \textbf{每次一种：}每次只添加一种新食物，观察3\~{}5天以排除过敏。
        \item \textbf{由少到多：}从1勺开始，逐渐增加到一餐。
        \item \textbf{由稀到稠：}从泥糊状→碎末状→小块状→条状。
        \item \textbf{由细到粗。}
    \end{itemize}
    \item \textbf{禁止添加盐、糖及调味品：}12月龄前避免添加盐，以保护肾脏和减少日后口味偏好带来的健康风险。
    \item \textbf{持续母乳喂养至2岁或更久。}
\end{enumerate}
\end{keybox}

\begin{tipbox}
\textbf{辅食添加常见误区与纠正：}
\begin{itemize}[leftmargin=*]
    \item 误区1："婴儿需要喝米汤（粥）就够了"——米汤能量和铁含量极低。
    \item 纠正：辅食应是营养密集的食物，尤其是铁、锌、蛋白质和维生素A丰富的食物。
    \item 误区2："加辅食后母乳就不重要了"——母乳仍然是重要的营养来源，尤其是优质蛋白质和多种营养素。
    \item 纠正：7\~{}12月龄仍建议母乳提供约一半的能量和蛋白质。
\end{itemize}
\end{tipbox}

% ----- 第五节 学龄儿童与青少年营养 -----
\section{学龄儿童与青少年营养}

\subsection{生长发育特点}

\begin{itemize}[leftmargin=*]
    \item \textbf{第二生长高峰（青春期，Puberty）：}是除外婴儿期的第二个生长发育飞跃期。青春期前生长缓慢（5\~{}6 cm/年），青春期峰值为8\~{}12 cm/年。
    \item \textbf{身体构成变化：}青春期内男性肌肉含量大幅增加（雌激素vs睾酮作用），女性脂肪占比高于男性。
    \item \textbf{性成熟：}女孩约10\~{}12岁（乳房发育为起始标志），男孩约12\~{}14岁开始。
\end{itemize}

\subsection{主要营养问题}

\begin{enumerate}[leftmargin=*]
    \item \textbf{营养不良和微量营养素缺乏：}
    \begin{itemize}[leftmargin=*]
        \item 铁缺乏/缺铁性贫血（男性需铁增加来自肌肉增长，女性来自月经）
        \item 钙摄入不足（峰值骨量积累的关键期）
        \item 锌缺乏可影响青春期发育迟滞
    \end{itemize}
    \item \textbf{超重和肥胖：}含糖饮料摄入、屏幕时间增加、身体活动不足。
    \item \textbf{饮食行为问题：}不吃早餐、偏食挑食、频繁在外就餐、盲目节食减肥（少女多见）。
\end{enumerate}

\subsection{营养与膳食建议}

\begin{itemize}[leftmargin=*]
    \item 保证充足的钙（奶及制品）和铁（红肉、动物血）摄入。
    \item 三餐规律，尤其是早餐——充足的能量和蛋白质。
    \item 避免含糖饮料，增加身体活动。
    \item 女孩注意节食减肥的危害——月经失调、低骨量。
\end{itemize}

% ----- 第六节 老年人营养 -----
\section{老年人营养}

\subsection{老年人的生理变化}

\begin{keybox}
\textbf{衰老的相关生理改变（不可逆性进程）：}

\begin{enumerate}[leftmargin=*]
    \item \textbf{体成分变化：}
    \begin{itemize}[leftmargin=*]
        \item 皮肤弹性下降，身高每10年下降约1 cm。
        \item \textbf{体内水分和瘦组织减少，体脂含量增加}（尤其向心性脂肪分布）。
        \item \textbf{肌肉衰减综合征（\keyword{少肌症/Sarcopenia}）：}50岁后肌肉量以每年1\%\~{}2\%的速度下降，肌力和功能退化。
    \end{itemize}
    \item \textbf{器官变化：}
    \begin{itemize}[leftmargin=*]
        \item 脑神经元数量减少10\%\~{}30\%，认知功能可下降（但并非必然发展为痴呆）。
        \item 心肌发生褐色萎缩，心输出量减少。
        \item 肾脏功能退化（肾小球滤过率及肾小管重吸收能力降低）。
    \end{itemize}
    \item \textbf{代谢变化：}
    \begin{itemize}[leftmargin=*]
        \item \textbf{基础代谢率（BMR）下降：}约每10年下降2\%，导致能量需求减少。
        \item \textbf{分解代谢>合成代谢：}常出现负氮平衡——骨胶原丢失、肌肉量减少。
    \end{itemize}
    \item \textbf{感觉功能减退：}味觉和嗅觉敏感度下降，可影响食欲和食物选择。
    \item \textbf{消化系统功能减弱：}胃酸分泌减少（影响维生素B12和钙的吸收），肠蠕动减慢（便秘）。
\end{enumerate}
\end{keybox}

\begin{keybox}
\textbf{衰老的自由基学说（Free Radical Theory of Aging）：}
\begin{itemize}[leftmargin=*]
    \item 1956年提出，是目前比较公认的衰老学说之一。
    \item 自由基攻击细胞膜脂类中的不饱和脂肪酸 $\rightarrow$ 细胞间信息传导受损 $\rightarrow$ 进而破坏细胞内的蛋白质、脂类及核酸。
    \item \keyword{有氧呼吸导致氧损伤的累积是引起机体衰老死亡的主要原因。}
    \item 人体内存在的抗氧化防御系统：SOD（超氧化物歧化酶）、GPx（谷胱甘肽过氧化物酶）、金属硫蛋白等。膳食中摄入的维生素E、维生素C、$\beta$-胡萝卜素等也具有抗氧化作用。
    \item \textbf{核心矛盾：}维持生命 $\rightarrow$ 摄取食物获得营养 $\rightarrow$ 产生自由基 $\rightarrow$ 可以修复但损害累积 $\rightarrow$ 重要器官老化或突变。
\end{itemize}
\end{keybox}

\subsection{老年人的营养需求}

\begin{keybox}
\textbf{老年人关键营养素推荐摄入量：}

\begin{center}
\begin{tabular}{lcc}
\hline
\textbf{营养素} & \textbf{50岁以上} & \textbf{65岁以上} \\
\hline
能量（男/女，kcal/d）
  & 1950 / 1600
  & 1900 / 1550 \\
蛋白质（男/女，g/d）
  & 65 / 55
  & \textbf{72 / 62} \\
脂肪（\%能量）
  & 20\%\~{}30\%
  & 20\%\~{}30\% \\
碳水化合物（\%能量）
  & 50\%\~{}65\%
  & 50\%\~{}65\% \\
钙（mg/d）
  & 800
  & 800 \\
\hline
\end{tabular}
\end{center}
\end{keybox}

\textbf{重要说明：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{蛋白质需要量不减反增：}老年人蛋白质RNI为\keyword{1.0\~{}1.5 g/(kg$\cdot$d)}，高于普通成人的0.8 g/(kg$\cdot$d)。65岁以上男性72 g/d、女性62 g/d——均高于50岁组的65 g/d和55 g/d。原因：分解代谢亢进和合成抵抗（\keyword{Anabolic Resistance}）。
    \item \textbf{能量需求降低：}随年龄增长BMR和身体活动减少。
    \item \textbf{脂肪：}供能比建议为20\%\~{}30\%，脂肪酸组成应保持SFA:MUFA:PUFA = 0.8\~{}1:1:0.6\~{}0.8。
    \item \textbf{碳水化合物：}供能比50\%\~{}65\%，强调选择低血糖生成指数（\keyword{低GI}）的食物。
    \item \textbf{钙：}800 mg/d——因钙吸收率随年龄下降，预防骨质疏松。
\end{enumerate}

\subsection{老年人膳食原则}

\begin{keybox}
\textbf{老年人合理膳食的五大原则：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{食物柔软、易消化：}选择软烂的食物和烹调方法（蒸、煮、炖），以适应口腔和消化能力减弱的状况。
    \item \textbf{食物多样化：}确保营养全面，防止单一食物依赖。
    \item \textbf{少量多餐：}建议每日\keyword{三餐+2\~{}3次间餐}（如上午加餐、下午加餐、睡前加餐），以提高总摄入量和维持稳定的血糖水平。
    \item \textbf{鼓励共餐（社交性进餐）：}与家人一起进餐可改善食欲和进食质量，预防营养不良。
    \item \textbf{保持适宜运动：}每日行走\keyword{$\geq$6000步}，理想运动心率 $\approx$ 170 - 年龄。运动可延缓少肌症进展、改善免疫功能、促进骨健康。
\end{enumerate}
\end{keybox}

\subsection{成功衰老的概念}

\begin{defbox}
\textbf{成功衰老（Successful Aging）：}指在老年期尽可能延长健康寿命，维持高水平的生理和认知功能，仅于生命末期极短时间段出现功能下降和失能，实现"活得健康、走得从容"的老龄化目标。这与单纯追求寿命延长完全不同，后者可能伴随着长期的功能不全和生活质量下降。
\end{defbox}

% ----- 第七节 运动员及特殊环境人群营养 -----
\section{运动员及特殊环境人群营养（了解）}

\subsection{运动员的营养特点}

\begin{itemize}[leftmargin=*]
    \item 能量需求因运动强度而高度可变（中等强度运动每日约+500\~{}1500 kcal）。
    \item 碳水化合物对运动表现至关重要——赛前3\~{}4天进行"糖原负荷法"（碳水化合物供能比$\geq 70\%$）可最大化糖原储存。
    \item 蛋白质摄入1.2\~{}2.0 g/(kg$\cdot$d)（耐力运动员偏低、力量运动员偏高）。
    \item 补液重要性——运动中每15\~{}20分钟饮水150\~{}300 ml。
    \item 铁缺乏在运动员（尤其是女性耐力运动员）中较常见。
\end{itemize}

\subsection{高温环境}

\begin{itemize}[leftmargin=*]
    \item 能量需求增加，水和电解质损失显著（出汗可流失3\~{}4 L/h）。
    \item 排汗失钠增加——补液应含电解质（0.5\~{}0.7 g/L钠）。
    \item 钾的需求增加，因汗中含钾。
    \item 注意补充水溶性维生素。
\end{itemize}

\subsection{低温环境}

\begin{itemize}[leftmargin=*]
    \item 能量需求显著增加（体热损失加速，基础代谢率上调）。
    \item 脂肪供能比例可升高——因脂肪的高能量密度和皮下脂肪的保温作用。
\end{itemize}

\subsection*{参考答案要点}

\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item 见本章各概念定义框。
    \item 核心：早期营养环境通过发育程序化（表观遗传机制）影响成年慢性疾病风险。指导意义：孕前和孕期营养干预是预防成人慢性病的根本策略，具有极高效费比。

    \item 初乳：低脂肪/乳糖、高乳清蛋白/免疫球蛋白；过渡乳：成分过渡期；成熟乳：营养成分稳定，前乳以蛋白质和乳糖为主，后乳以高脂肪提供能量。
    \item (1) 促进子宫复原；(2) 降低乳腺癌/卵巢癌/T2DM风险；(3) 哺乳消耗能量，加速产后体重恢复；(4) 提供母婴亲子连接。
    \item 见本章定义框。
    \item (1) 出生后1小时内开始哺乳；(2) 0\~{}6个月纯母乳喂养；(3) 按需哺乳；(4) 每日补充维生素D 400 IU。
    \item 从含铁泥糊状食物开始（强化铁米粉、肝泥、肉泥）。每次1种新食物，观察3\~{}5天；由1勺到1餐；泥糊→碎末→小块→条状；细→粗。
    \item 保护肾脏不成熟、培养不良的口味偏好（低盐饮食降低日后高血压风险）。
    \item 乳铁蛋白：竞争结合铁、抑制需铁菌繁殖；溶菌酶：直接溶解细菌细胞壁；sIgA：覆盖肠黏膜表面，阻断病原体/过敏原粘附侵入。
    \item 见本章概念定义框。
    \item 能量降低原因：BMR下降（每10年\~{}2\%）、体力活动减少、瘦组织减少。蛋白质需增加原因：合成抵抗（Anabolic Resistance），老年人蛋白质合成对摄入蛋白质的敏感性下降，需要更高摄入量来刺激蛋白质合成、减缓肌肉丢失。
    \item 柔软易消化、食物多样化、三餐+2\~{}3次间餐、社交性进餐、维持适宜运动（$\geq$6000步/日，心率\~{}170-年龄）。
    \item 答题思路：(1) 生命早期1000天——DOHaD：孕期270d+0-2岁730d为机遇窗口，早期营养不良经表观遗传导致远期代谢综合征高风险；(2) 成年期：膳食行为和生活方式继续强化或修正早期程序化效应；(3) 老年期：少肌症与合成抵抗——蛋白质需要量不减反增（1.0-1.5 g/(kg$\cdot$d)），能量需求下降但蛋白质需求上升形成"老年人的营养悖论"，需配合运动干预以实现成功衰老。三者贯穿营养生命周期理念——营养影响不是点状而是连续的，早期干预效费比最高。
\end{enumerate}
