\section{生命早期营养与DOHaD理论}
% =====================================================================

\subsection{国民营养计划目标}

\begin{tipbox}
\textbf{国民营养计划（2017-2030年）与（2020-2030）目标：}
\begin{enumerate}[leftmargin=*]
    \item 目标1：5岁以下儿童贫血率控制在10\%以下；孕妇贫血率下降至10\%以下。
    \item 目标2：0-5岁儿童生长迟缓率控制在5\%以下。
    \item 目标3：0-6个月婴儿纯母乳喂养率在2020年的基础上提高10\%（2019年为29\%）。
\end{enumerate}
\end{tipbox}

\subsection{DOHaD理论与生命早期1000天}

\begin{keybox}
\textbf{DOHaD理论（Developmental Origins of Health and Disease，健康与疾病的发育起源）：}
生命早期（胎儿和婴幼儿期）经历不利因素（如营养不良、环境不良等），将会增加其成年后患肥胖、糖尿病、心血管疾病等慢性疾病的几率。其影响机制主要包括：表观遗传改变、组织结构与功能异常、成年疾病风险增加。

\textbf{生命早期1000天关键期——三大阶段：}
\begin{enumerate}[leftmargin=*]
    \item 孕期270天（pregnancy）：母体营养决定胎儿发育。
    \item 0-6月龄180天（year 1）：纯母乳喂养，持续生长发育。
    \item 7-24月龄（year 2）：辅食添加，完成食物转换，形成合理膳食结构，促进健康成长。
\end{enumerate}
\end{keybox}

\section{孕妇营养}
% =====================================================================

\subsection{孕期生理特点}

\begin{keybox}
\textbf{妊娠期——生命早期1000天机遇窗口的起始阶段。}营养是最重要的环境因素，对母体双方的近期和远期健康均产生至关重要的影响。

\textbf{整个孕期}，胎儿生长发育+母体乳腺和子宫等生殖器官发育+为分娩后乳汁分泌进行必要的营养储备，均需要额外营养。

\textbf{孕期分期：}
\begin{itemize}[leftmargin=*]
    \item 备孕期（孕前3-6月）。
    \item 孕早期（孕后1-3月，1-12周）。
    \item 孕中期（孕后4-6月，13-27周）。
    \item 孕晚期（孕后7-9月，28周-分娩）。
\end{itemize}
\end{keybox}

\subsubsection{内分泌系统变化}

\begin{keybox}
\begin{enumerate}[leftmargin=*]
    \item \textbf{卵巢及胎盘激素分泌增加}
    \begin{itemize}[leftmargin=*]
        \item \textbf{人绒毛膜促性腺激素（HCG）：}受精1周开始出现，8-9周达高峰，后逐渐下降。
        \item \textbf{人绒毛膜生长激素（HCS）：}妊娠6周出现，36周达高峰至分娩，保证母体营养素输送到胎儿。
        \item \textbf{雌激素：}分泌增加，增加子宫和胎盘的血流量，促进母体乳房发育。
        \item \textbf{孕激素：}分泌增加，减少子宫收缩，维持妊娠。
    \end{itemize}
    \item \textbf{甲状腺激素水平升高} $\rightarrow$ 基础代谢率（BM）升高。
    \item \textbf{胰岛素敏感性下降} $\rightarrow$ 妊娠期糖尿病（GDM）风险升高。
\end{enumerate}
\end{keybox}

\subsubsection{代谢特点}

\begin{keybox}
\textbf{孕期的合成代谢增强，分解代谢也增强；但总体上合成代谢 $>$ 分解代谢。}
\begin{itemize}[leftmargin=*]
    \item 碳水化合物、脂肪和蛋白质的利用发生改变。
    \item \textbf{葡萄糖：}通过胎盘以糖原形式储存，经易化扩散向胎儿转运，是胎儿主要能源。
    \item \textbf{氨基酸：}通过胎盘主动转运到胎儿。
    \item \textbf{脂肪酸：}通过胎盘简单转运到胎儿。
\end{itemize}
\end{keybox}

\subsubsection{消化系统变化}

\begin{keybox}
\begin{itemize}[leftmargin=*]
    \item 恶心、呕吐（早孕反应）：贲门括约肌松弛所致。
    \item 胃排空时间延长、胃肠蠕动减慢：孕酮分泌增加引起。
    \item 对钙、铁、VitB$_{12}$、叶酸等营养素吸收增强。
    \item 营养素需要量+++。
\end{itemize}
\end{keybox}

\subsubsection{循环系统变化}

\begin{keybox}
\begin{itemize}[leftmargin=*]
    \item 孕早期：血压偏低，随后逐渐回升。
    \item \textbf{血容量增加，血液稀释：}血浆容积增加幅度 $>$ 红细胞增加幅度 $\rightarrow$ \textbf{生理性贫血}（20-30周明显），红细胞数量一直增加至分娩。
    \item \textbf{WHO贫血标准（2024）：}孕早期 $\geq$ \imp{110 g/L}，孕中期 $\geq$ \imp{105 g/L}。
\end{itemize}
\end{keybox}

\subsubsection{泌尿系统变化}

\begin{keybox}
\begin{itemize}[leftmargin=*]
    \item 肾小球滤过率增加约50\%，肾血浆流量增加约35\%，肾脏负担加重。
    \item 代谢产物排出量变化大：蛋白质代谢产物（尿素、尿酸、肌酐和肌酸）排出增多；尿量增加，夜尿 $>$ 日尿。
    \item 一些营养素排出增加：葡萄糖、氨基酸、水溶性维生素排出增加；餐后糖尿。
\end{itemize}
\end{keybox}

\subsubsection{体重变化}

\begin{keybox}
\begin{itemize}[leftmargin=*]
    \item \textbf{孕前体重过重：}巨大儿、难产、剖宫产、心衰、妊娠期糖尿病、妊娠期高血压、产程延长等。
    \item \textbf{孕期体重过重：}巨大儿、妊娠期高血压、增加难产率和剖宫产率、胎儿窘迫和新生儿窒息、新生儿产伤、产后体重滞留。
    \item \textbf{孕期体重过轻：}新生儿低体重、胎儿生长受限、早产儿。
\end{itemize}
\end{keybox}

\subsection{孕期营养需要}

\begin{keybox}
\textbf{总原则：}胎儿生长发育全部来自母体；母亲除自身需要外，还要为分娩和泌乳做准备。

\textbf{（1）能量：}中国孕妇膳食能量推荐摄入量（EER）:
\begin{center}
\begin{tabular}{lcc}
\hline
\textbf{年龄/阶段} & \textbf{能量（kcal/d）} \\
\hline
18岁（轻体力） & 1700（2023版） \\
孕早期 & +0 \\
孕中期 & +250 \\
孕晚期 & +400 \\
\hline
\end{tabular}
\end{center}
以平衡为原则，定期监测体重。

\textbf{（2）蛋白质：}
\begin{itemize}[leftmargin=*]
    \item 用于胎儿组织合成、孕妇子宫/胎盘/乳房发育，并为分娩时失血储备和刺激乳腺分泌、增加产后乳汁分泌做准备。
    \item 孕中期 +15 g/d，孕晚期 +30 g/d。增加的蛋白质应为优质蛋白。
\end{itemize}

\textbf{（3）脂类：}
\begin{itemize}[leftmargin=*]
    \item 脂肪是胎儿神经系统重要组分。孕期平均储脂3-4 kg。
    \item 供能比20\%-30\%不变。亚油酸占总能量\imp{4\%}，$\alpha$-亚麻酸\imp{0.6\%}，EPA+DHA达到\imp{250 mg/d}。
\end{itemize}

\textbf{（4）碳水化合物：}
\begin{itemize}[leftmargin=*]
    \item 孕早期：缺乏碳水化合物时，酮体通过胎盘进入胎儿体内，损伤胎儿中枢神经系统发育。
    \item 早孕反应时，每日摄入不低于\imp{130 g}碳水化合物（+10，孕中期+20g，孕晚期+35g）。
    \item 膳食纤维：+0g（孕早期），+4g（孕中期），+4g（孕晚期）。
\end{itemize}

\textbf{（5）矿物质：}

\begin{center}
\begin{tabular}{lccccc}
\hline
\textbf{营养素} & \textbf{非孕期RNI} & \textbf{孕早期} & \textbf{孕中期} & \textbf{孕晚期} & \textbf{备注} \\
\hline
钙（mg/d） & 800 & 800 & 800 & 800 & 孕晚期胎儿大量储存，吸收率代偿升高 \\
铁（mg/d） & 18 & 18 & 25 & 29 & 预防母体贫血+胎儿储存+产时失血储备 \\
锌（mg/d） & 8.5 & 8.5 & 10.5 & 10.5 & 孕晚期达最高，预防先天缺陷 \\
碘（$\mu$g/d） & 120 & 230 & 230 & 230 & 严重缺乏$\rightarrow$克汀病 \\
\hline
\end{tabular}
\end{center}

\textbf{钙代谢特点：}
\begin{itemize}[leftmargin=*]
    \item 胎儿通过浓度差转运钙；母体孕期钙吸收率上调，血钙水平能准确反映体内钙水平。
    \item 低钙可使母体骨密度降低：同年龄非孕妇女为85\%。
    \item 母体缺钙：小腿抽筋、腰腿痛、手足抽搐；骨质疏松症。
    \item 胎儿/婴儿缺钙：先天性佝偻病。
\end{itemize}

\textbf{铁需求增加原因：}
\begin{itemize}[leftmargin=*]
    \item 孕期贫血+分娩失血+胎儿血液和肌肉/胎盘储存（6个月）。
    \item 缺铁危害：孕妇贫血、早产、胎儿贫血/心脏病/妊娠并发症/抵抗力降低/易感染；胎儿发育迟缓、婴儿期缺铁/缺铁性贫血。
\end{itemize}

\textbf{锌需求：}
\begin{itemize}[leftmargin=*]
    \item 需要量持续增加，孕末期达到最大值。促进胎儿发育，预防先天性缺陷。
    \item 孕妇严重缺锌危害：畸形、低体重儿、细胞免疫功能下降。
\end{itemize}

\textbf{碘需求增加：}
\begin{itemize}[leftmargin=*]
    \item 与甲状腺激素合成和分泌有关，促进胎儿生长发育。
    \item 缺碘危害：胎儿甲状腺功能异常、克汀病（矮小、智力低下、生长发育迟缓）。
\end{itemize}

\textbf{（6）维生素：}

\begin{itemize}[leftmargin=*]
    \item \textbf{维生素A：}胎儿发育需要和肝脏储存；母体需要量为维持自身标准。缺乏危害：早产、胎儿发育不全。过量危害：胎儿畸形。

    \item \textbf{维生素D：}缺乏危害：母体骨软化症、新生儿手足抽搐；过量危害：中毒。补充途径：晒太阳+富含维生素D食物+维生素D制剂。

    \item \textbf{维生素E：}维持细胞膜中多不饱和脂肪酸稳定。缺乏：新生儿溶血性溶血性贫血。

    \item \textbf{维生素B$_{1}$：}孕期需要量升高，维持食欲、肠道蠕动和促进产后乳汁分泌。缺乏危害：孕妇疲倦乏力、头痛，严重者影响胃肠道功能，加重早孕反应。

    \item \textbf{维生素B$_{2}$：}孕期需要量增加。与蛋白质代谢有关。

    \item \textbf{叶酸：}孕期需要量增加。缺乏危害：孕妇巨幼细胞贫血、呕吐腹泻舌炎，胎儿神经管畸形。辅助作用：降低妊娠期血脂吸收、调节免疫。建议：孕前3个月起补充\imp{400 $\mu$g DFE/d}，持续整个妊娠期。

    \item \textbf{维生素C：}孕期需要量增加；胎儿生长发育+造血系统发育需要。缺乏危害：孕妇贫血、早产、免疫力降低、产后出血。
\end{itemize}

\textbf{（7）水和膳食水摄入量（AI）：}
孕期需水量增加（代谢增加+胎儿羊水+体液增加）。孕早、中期：每日总摄入水为\imp{1700 mL}（非孕妇女为1500 mL），妊娠中晚期活动量少者增加200 mL。
\end{keybox}

\subsection{孕期膳食指南}

\begin{keybox}
\textbf{《备孕和孕期妇女膳食指南（2022年）》6条关键推荐（8+6）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{调整孕前体重至正常范围，保证孕期体重适宜增长。}
    \item \textbf{常吃含铁丰富的食物，选用碘盐，合理补充叶酸和维生素D。}
    \item \textbf{孕吐严重者，可少量多餐，保证摄入含必需量碳水化合物的食物。}
    \item \textbf{孕中晚期适量增加奶、鱼、禽、蛋、瘦肉的摄入。}
    \item \textbf{经常户外活动，禁烟酒，保持健康生活方式。}
    \item \textbf{愉快孕育新生命，积极准备母乳喂养。}
\end{enumerate}

\textbf{备孕期膳食指南（3条）：}
\begin{enumerate}[leftmargin=*]
    \item 补充叶酸，常吃含铁丰富的食物，选用碘盐。
    \item 孕吐严重者，可少量多餐，保证摄入含必需量碳水化合物的食物；孕中晚期适量增加奶、鱼、禽、蛋、瘦肉的摄入；适量身体活动，维持孕期适宜体重。
    \item 禁烟酒，愉快孕育新生命，积极准备母乳喂养。
\end{enumerate}
\end{keybox}

\section{乳母营养}

\subsection{乳母生理特点}

\subsubsection{乳汁生成}

\begin{keybox}
\textbf{原料：}母体血液中的营养素和活性物质。

\textbf{分泌方式：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{直接进入乳汁：}钠、钾、磷、镁、氯等，水和水溶性维生素。
    \item \textbf{合成乳汁成分并受激素调控：}乳糖（葡萄糖合成，受孕酮调控）；乳脂（在乳腺内合成，以膜包裹形成脂滴，包含胆固醇、磷脂、脂肪酸和脂溶性维生素）。
    \item \textbf{经由细胞间质成分：}溶菌酶、乳铁蛋白（乳腺细胞分泌）；淋巴细胞产生IgA和干扰素。
\end{enumerate}

\textbf{乳汁分泌：}吸吮反射、排乳反射。

\textbf{不同阶段母乳：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{初乳（第1周）：}淡黄色，质地黏稠，蛋白质多，主要为乳清蛋白和乳球蛋白（sIgA和乳铁蛋白），乳糖和脂肪相对较少。
    \item \textbf{过渡乳（第2周）：}蛋白质和乳糖和脂肪增多。
    \item \textbf{成熟乳（第2周以后）：}乳白色，蛋白质相对较少，乳糖和脂肪多。
\end{enumerate}
\end{keybox}

\subsubsection{乳母营养状况对乳汁的影响}

\begin{keybox}
\textbf{乳母营养状况对乳汁中营养素含量有一定影响但程度不同：}
\begin{itemize}[leftmargin=*]
    \item 乳汁中大部分宏量营养素含量相对稳定，不受乳母膳食影响。
    \item 脂肪酸、水溶性维生素和某些矿物质的含量受乳母膳食影响显著。
    \item 铁不能通过乳腺输送到乳汁。
\end{itemize}
\end{keybox}

\subsection{哺乳对母亲健康的影响}

\begin{keybox}
\begin{enumerate}[leftmargin=*]
    \item \textbf{近期影响：}促进产后子宫恢复；促进母体乳房乳汁排空，避免发生乳房肿胀和乳腺炎；延长恢复排卵的时间。
    \item \textbf{远期影响：}预防肥胖、代谢综合征、乳腺癌和卵巢癌等。
\end{enumerate}
\end{keybox}

\subsection{乳母的营养需要}

\begin{keybox}
\textbf{（1）能量：}维持自身需要+生成乳汁+哺育婴儿。据产奶量每日产奶700-800ml（平均750ml），EER较非孕妇女增加\imp{400 kcal}。

\textbf{（2）蛋白质：}影响泌乳量和乳汁蛋白质含量。RNI较非孕妇女增加\imp{25 g}。

\textbf{（3）脂类：}需要量升高，婴儿中枢神经系统发育和脂溶性维生素吸收的需要。

\textbf{（4）矿物质：}
\begin{itemize}[leftmargin=*]
    \item \textbf{钙：}乳汁中含量较稳定，每日排出\imp{300 mg}。乳母钙供给不足时会动用自身骨骼中的钙满足乳汁钙含量。缺钙可能导致乳母腰腿痛、抽搐和骨质疏松症。
    \item \textbf{铁：}铁不能通过乳腺输送到乳汁。
    \item \textbf{锌：}乳汁中含量受乳母膳食影响，对婴儿神经系统和免疫功能发育非常重要。
\end{itemize}

\textbf{（5）维生素：}
\begin{itemize}[leftmargin=*]
    \item \textbf{脂溶性维生素：}维生素A少量通过乳腺进入乳汁；维生素D少量通过乳腺；维生素E促进乳汁分泌。
    \item \textbf{水溶性维生素：}通过乳腺进入乳汁，乳腺可调节使乳汁中的含量维持在一定水平不再升高。
\end{itemize}

\textbf{（6）水：}与乳汁分泌量有密切关系，较非孕妇女每日增加约\imp{1 L}水。
\end{keybox}

\subsection{哺乳期膳食指南}

\begin{keybox}
\textbf{哺乳期膳食指南关键推荐：}
\begin{enumerate}[leftmargin=*]
    \item 增加富含优质蛋白质和维生素A的动物性食物和海产品，选用碘盐，合理补充维生素D。
    \item 产褥期食物多样不过量，重视整个哺乳期营养均衡。
    \item 家庭支持，愉悦心情，充足睡眠，促进乳汁分泌。
    \item 坚持哺乳，适度运动，逐步恢复适宜体重。
    \item 禁烟酒，避免浓茶和咖啡，保证充足饮水。
\end{enumerate}
\end{keybox}

\section{婴幼儿营养}
% =====================================================================

\subsection{婴儿期生理特点}

\begin{keybox}
\textbf{婴儿期——第一生长高峰期。}

\begin{enumerate}[leftmargin=*]
    \item \textbf{生长发育：}第一生长高峰期。

    \item \textbf{消化系统：}
    \begin{itemize}[leftmargin=*]
        \item 口腔：唾液腺分泌少，唾液淀粉酶活性低；无牙齿；约6-8个月开始出牙，咀嚼能力弱。
        \item 胃：胃容量小，呈水平位，贲门括约肌发育不完善，易发生溢奶和呕吐；肠道蠕动、消化和吸收功能不完善；胰腺少，消化酶活性低，消化和吸收能力弱。
    \end{itemize}

    \item \textbf{神经系统：}脑细胞处于快速增长/高峰期，影响智力发育。
\end{enumerate}
\end{keybox}

\subsection{婴幼儿营养需要}

\begin{keybox}
\textbf{1. 能量：}
\begin{itemize}[leftmargin=*]
    \item 婴幼儿基础代谢率高；生长所需能量多。
    \item 单位体重（/kg）能量需要 = 成人的\imp{2-3倍}。
    \item 1y$\sim$：900-800 Kcal/d；2y$\sim$：1100-1000 Kcal/d；3y$\sim$：1250-1150 Kcal/d；4y$\sim$：1300-1250 Kcal/d；5y$\sim$：1400-1300 Kcal/d。
\end{itemize}

\textbf{2. 能量消耗构成：}
\begin{itemize}[leftmargin=*]
    \item 基础代谢（高于成人）。
    \item 生长（约占10\%，0-3月23\%，1岁6\%，5岁2\%）。
    \item 活动（婴儿少，学龄前儿童多）。
    \item 食物热效应。
    \item 排泄（10\% BM）。
\end{itemize}

\textbf{3. 蛋白质：}
\begin{itemize}[leftmargin=*]
    \item 生长需要+组织合成 $>$ 成人RNI。
    \item 优质蛋白应占总蛋白质\imp{1/2至2/3}。蛋白质不宜过多摄入，最高不宜超过\imp{3g/(kg$\cdot$d)}，以免加重肝肾负担。
    \item 婴儿必需氨基酸（EAA）9种，需要量$>$成人，为成年期的\imp{6-8.5倍}。单位体重EAA需要量随年龄增加而下降。
    \item 4-6y AMDR（供能比）：8-20\%；$>$6y：10-20\%。
\end{itemize}

\textbf{4. 脂类：}
\begin{itemize}[leftmargin=*]
    \item 必需脂肪酸：婴儿期大脑/视网膜发育所需，缺乏$\rightarrow$皮疹/脂溶性维生素缺乏。
    \item 母乳：乳清蛋白氨基酸模式接近婴儿需求；保证优质蛋白占总蛋白1/2。
    \item 3-4y AMDR：35\%；$>$4y：20-30\%。
\end{itemize}

\textbf{5. 碳水化合物：}
\begin{itemize}[leftmargin=*]
    \item 0y$\sim$：乳糖（蔗糖、果糖不宜）。
    \item 淀粉酶：2岁达到成人水平。婴幼儿过早接触大量碳水化合物是婴儿营养不良的主要原因，不应添加任何调味品/糖。
    \item 膳食纤维（DF）：婴儿期易便秘。
    \item 供能比：50-65\%，以谷类食物为主，限制添加糖和高糖食物的过度摄入。
    \item 1y$\sim$：AMDR 50-65\%。不应添加任何调味品/糖。
\end{itemize}

\textbf{6. 矿物质：}
\begin{itemize}[leftmargin=*]
    \item \textbf{钙：}母乳$<$牛乳，钙/磷比值：母乳2.3:1（适宜），牛乳1.4:1。0-6月通常不缺钙。0$\sim$ AI 200mg/d；0.5y$\sim$350mg/d；1-4y 500mg/d；4y$\sim$600mg/d。
    \item \textbf{铁：}足月新生儿有足够铁贮存，供4-6个月需要。婴儿血浆铁含量随牛乳摄入量增加而增加。7-12月龄\imp{10 mg/d}。母乳含铁低但吸收率高。6m-2y易患缺铁性贫血。0-0.5y AI 0.3mg/d；0.5y$\sim$ RNI 10；1y$\sim$ 10。
    \item \textbf{锌：}婴儿期缺锌$\rightarrow$生长发育迟缓、脑发育受损、食欲不振、异食癖。婴儿出生时体内锌的储存较多，需要辅食补充。母乳中锌含量高于牛乳，且为初乳，生物学效价也高于牛乳。0y$\sim$ AI 1.5mg/d；0.5y$\sim$ AI 3.2；1y$\sim$ RNI 4.0；4y$\sim$ 5.5。
    \item \textbf{钠：}7-24月龄婴幼儿摄入量不超过\imp{350-700 mg/d}（相当于0.9-1.8g盐）。7-24月龄婴幼儿的肾脏、肝脏等器官还没有发育成熟。
    \item \textbf{碘：}极低。
\end{itemize}

\textbf{7. 维生素：}
\begin{itemize}[leftmargin=*]
    \item Vit A：农村；6m-1y，4-5y；母乳维生素A丰富，乳母注意额外补充（母乳喂养者）。
    \item Vit D：食物+户外活动，乳类含量微量，需要 $>$ 摄入。
    \item Vit K：母乳中很少，单纯母乳喂养易缺乏。
    \item Vit C：受乳母膳食影响。
    \item Vit B$_{1}$、B$_{2}$：\imp{0.5mg/1000kcal}，受乳母膳食影响。
    \item Vit B$_{12}$、叶酸：加热易破坏。
\end{itemize}

\textbf{8. 水：}
\begin{itemize}[leftmargin=*]
    \item 生长发育快；体表面积大；代谢率高；肾脏浓缩功能差。需水量\imp{100-150 ml/(kg$\cdot$d)}。
\end{itemize}
\end{keybox}

\subsection{母乳喂养的优点和基本要求}

\begin{keybox}
\textbf{母乳喂养的优点：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{营养成分最适合婴儿需要，消化吸收利用率高。}
    \begin{itemize}[leftmargin=*]
        \item \textbf{蛋白质：}乳清蛋白为主，必需氨基酸比例适宜，易消化吸收。
        \item \textbf{脂肪：}脂肪颗粒小+乳脂酶，易消化吸收；含必需脂肪酸、长链多不饱和脂肪酸、卵磷脂和鞘磷脂，促进神经系统发育。
        \item \textbf{糖类：}富含乳糖，促进乳酸杆菌生长、抑制大肠埃希菌生长，利于钙、铁、锌吸收。
        \item \textbf{矿物质：}含量低，未发育成熟的肾脏排泄负担轻，易被婴儿利用；钙磷比2:1，吸收率高；铁和锌吸收率高于牛乳。
    \end{itemize}

    \item \textbf{含有大量免疫活性物质：}增强婴儿抗感染能力，如分泌型IgA（sIgA）、乳铁蛋白。

    \item \textbf{不易发生过敏。}

    \item \textbf{经济、方便、卫生。}

    \item \textbf{促进产后恢复，增进母婴交流。}
\end{enumerate}
\end{keybox}

\subsection{婴幼儿喂养指南}

\begin{keybox}
\textbf{《0-6月龄婴儿母乳喂养指南（2022）》6条准则：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{准则1：母乳是婴儿最理想的食物，坚持6月龄内纯母乳喂养。}
    \item \textbf{准则2：生后1小时内开奶，重视尽早吸吮。}
    \item \textbf{准则3：回应式喂养，建立良好的生活规律。}
    \item \textbf{准则4：适当补充维生素D，母乳喂养无需补钙。}
    \item \textbf{准则5：任何动摇母乳喂养的想法和举动都必须咨询医生或其他专业人员，并由他们帮助做出决定。}
    \item \textbf{准则6：定期监测体格指标，保持健康生长。}
\end{enumerate}

\textbf{《7-24月龄婴幼儿喂养指南（2022）》6条准则：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{准则1：继续母乳喂养，满6月龄起必须添加辅食，从富含铁的泥糊状食物开始。}
    \item \textbf{准则2：及时引入多样化食物，重视动物性食物的添加。}
    \item \textbf{准则3：尽量少加糖、盐，油脂适当，保持食物原味。}
    \item \textbf{准则4：提倡回应式喂养，鼓励但不强迫进食。}
    \item \textbf{准则5：注重饮食卫生和进食安全。}
    \item \textbf{准则6：定期监测体格指标，追求健康生长。}
\end{enumerate}

\textbf{添加辅食的意义和原则：}
\begin{itemize}[leftmargin=*]
    \item 由少到多，由细到粗，由稀到稠，由一种到多种，逐渐增加。
    \item 注意观察适应：一般为1周，再考虑添加新的品种，使婴儿有一个适应过程。
    \item 应在婴儿健康、消化功能正常时添加辅助食品。
    \item 保持原味，不加盐、糖及刺激性调味品。1岁后逐渐尝试淡口味的家庭膳食。
\end{itemize}
\end{keybox}

\section{学龄前儿童营养}
% =====================================================================

\subsection{定义}

\begin{tipbox}
学龄前期：2岁末至6岁入小学前。\\
学龄期：由学龄前儿童过渡到成人的生长发育时期，小学学龄期（6/7岁至青春期前），生长发育平稳，也称童年期（学龄儿童）。\\
青春期：指体格发育过程中，从第二性征出现至生殖器官发育成熟的一段时间，中学学龄期（女生11-12岁至17-18岁；男生13-14岁至18-20岁），即青少年。
\end{tipbox}

\subsection{学龄前儿童生理特点}

\begin{keybox}
\begin{enumerate}[leftmargin=*]
    \item \textbf{身高、体重稳步增长}但较婴儿期生长发育速度相对减缓，无性别差异。
    \begin{itemize}[leftmargin=*]
        \item 每年体重增长约\imp{2 kg}，每年身高增长约\imp{5-7 cm}。
    \end{itemize}

    \item \textbf{神经系统发育完善：}依旧属于脑细胞快速增长时期/高峰期。脑细胞体积增大，神经纤维髓鞘化（6岁-2至3岁完成）；突触发育加速，开始于分娩前后，可能维持到4岁；神经纤维髓鞘突触形成和修剪。

    \item \textbf{咀嚼和消化能力仍有限：}3岁乳牙已出齐，6岁第一恒牙萌出，咀嚼和消化能力仍有限，远低于成人。对固体食物需要较长适应时间，不能给予成人膳食，易发生消化紊乱。

    \item \textbf{心理发育特点：}
    \begin{itemize}[leftmargin=*]
        \item 注意力分散（\imp{15分钟}），无法专心进食。
        \item 食物选择有自我倾向，好奇心强，模仿能力强。
        \item 应特别注意培养儿童良好的饮食行为习惯。
    \end{itemize}
\end{enumerate}

\textbf{总体原则：}
\begin{itemize}[leftmargin=*]
    \item 学龄前儿童：生长发育仍需要足够营养；培养良好的饮食习惯。
    \item 学龄期和青春期是儿童青少年生长发育和终身健康的关键时期。
    \item 对策：平衡膳食+营养教育+健康成长监测。
\end{itemize}
\end{keybox}

\subsection{学龄前儿童膳食指南（2022版）}

\begin{keybox}
\textbf{适用对象：}满2岁以后至未满6岁的学龄前儿童。

7-24月龄期间饮食模式的过渡和转变，学龄前儿童的饮食种类和膳食结构已开始接近成人，是饮食行为和生活方式形成的关键时期。

\textbf{核心推荐（8+5）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{食物多样，规律就餐，自主进食，培养健康饮食行为。}
    \begin{itemize}[leftmargin=*]
        \item Key1：合理安排学龄前儿童膳食。
        \item 早、中、晚三次正餐，+2次加餐。
        \item 一般分别安排在上、下午各一次，加餐时间与正餐间隔1.5-2小时，晚间加餐不宜安排甜食，以预防龋齿。
        \item 加餐以奶类、水果为主，配以少量松软面点。
        \item 加餐份量不宜过多，以免影响正餐。
    \end{itemize}

    \item \textbf{每天饮奶，足量饮水，合理选择零食。}
    \begin{itemize}[leftmargin=*]
        \item 奶类是优质蛋白质和钙的最佳食物来源，每日\imp{300-500 mL}（或相当量奶制品）。
        \item 新陈代谢旺盛、活动量大、出汗多：及时补充水分，每日\imp{600-800 mL}，以白开水为主。
        \item 零食作为学龄前儿童全天营养的补充：不应影响正餐，选择营养密度高的食物。
    \end{itemize}

    \item \textbf{合理烹调，少调料少油炸。}
    \begin{itemize}[leftmargin=*]
        \item 从小培养清淡口味，有助于形成终身的健康饮食习惯。
        \item 不应过咸、油腻和辛辣，尽可能少用或不用味精或鸡精、色素、糖精等调味品。
        \item 可选天然新鲜香料（如葱、蒜、姜、香菜等）和新鲜蔬果汁（如番茄汁、南瓜汁、菠菜汁等）进行调味。
        \item 烹调方式宜采用蒸、煮、炖、煨等，不宜用含油高的烹调方式。
        \item 特别注意：进食安全，完全去除皮、骨、刺、核等（大豆、花生等坚果类食物，应先磨碎，制成泥糊浆等状态进食）。
    \end{itemize}

    \item \textbf{参与食物选择与制作，增进对食物的认知与喜爱。}
\end{enumerate}

\textbf{培养儿童规律就餐：}
\begin{itemize}[leftmargin=*]
    \item Key2：培养儿童规律就餐、专注进食。
    \begin{itemize}
        \item 尽可能给儿童提供固定的就餐座位，定时定量进餐。
        \item 避免追着喂、边吃边玩、边吃边看电视等行为。
        \item 吃饭细嚼慢咽但不拖延，最好在30分钟内吃完。
        \item 让孩子自己使用筷、匙进食，养成自主进餐的习惯，既可增加儿童进食兴趣，又可培养其自信心和独立能力。
    \end{itemize}

    \item \textbf{健康饮食行为习惯：}
    \begin{enumerate}
        \item 饭前不吃糖果、不饮汽水等零食。
        \item 饭前洗手，饭后漱口，饭前不做剧烈运动。
        \item 养成自己吃饭的习惯。
        \item 吃饭时专心，不边看电视或边玩边吃。
        \item 不要一次盛太多，吃完再盛，培养不剩饭、不浪费的习惯。
        \item 不要一口饭一口水或经常吃汤泡饭。
        \item 不挑食、不偏食，在许可范围内允许孩子选择食物。
        \item 不以食物作为奖励，避免诱导孩子对某种食物产生偏好。
        \item 家长和看护人应以身作则、言传身教。
    \end{enumerate}
\end{itemize}

\textbf{纠正儿童挑食偏食：}
\begin{itemize}[leftmargin=*]
    \item Key3：纠正儿童挑食偏食。
    \item 建议儿童自主选择食物。通过增加变换食物、口味搭配使儿童熟悉和习惯接受不同食物。
    \item 家长以身作则。
    \item 鼓励儿童选择多种多样食物，及时纠正儿童挑食偏食和厌食的不健康行为。
    \item 对于不喜欢吃的食物：多次尝试——可变换烹调方式、改变食物形式或质地等加以改善。
    \item 不以食物作为奖励或惩罚措施。
    \item 不强迫或诱导儿童进食。
    \item 家长、幼教人员为儿童提供定时定位的进餐制度和和谐温馨的进餐环境。
\end{itemize}

\textbf{Key1：建立饮奶和巩固儿童饮奶习惯：}家长应以身作则。不要空腹饮奶（乳糖不耐受者可从少量开始，逐步增加饮奶量或改为酸奶、奶酪等发酵乳制品，或在餐后饮奶）。

\textbf{2-5岁学龄前儿童零食推荐：}
\begin{enumerate}
    \item 吃好正餐，适量加餐，少量零食。
    \item 零食优先选择水果、奶类和坚果。
    \item 少吃高盐、高糖、高脂零食。
    \item 不喝或少喝含糖饮料。
    \item 零食应新鲜、多样、易消化、营养卫生。
    \item 安静进食零食，谨防呛堵。
    \item 保持口腔清洁，睡前不吃零食。
\end{enumerate}

\textbf{6-12岁学龄儿童零食推荐：}
\begin{enumerate}
    \item 正餐为主，早餐合理，零食少量。
    \item 课间适量加餐，优先选择水果、奶类和坚果。
    \item 少吃高盐、高糖、高脂零食。
    \item 不喝或少喝含糖饮料，不喝含酒精、含咖啡因饮料。
    \item 零食新鲜、营养卫生。
    \item 保持口腔清洁，睡前不吃零食。
\end{enumerate}
\end{keybox}

\subsection{学龄儿童生理特点}

\begin{keybox}
\textbf{生长发育规律：}
\begin{enumerate}[leftmargin=*]
    \item 连续性：生长发育的各个阶段具有承接关系。
    \item 不同步：不同年龄各组织系统的发育不同，缺乏统一协调。
    \item 波浪式：学龄儿童处于比较稳定的阶段，青少年处于第二生长发育高峰期。
    \item 差异性：个体差异+性别差异。
\end{enumerate}

\textbf{具体特点：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{身高和体重：}较快增长但低于学龄前期和青春期，较稳定。
    \begin{itemize}[leftmargin=*]
        \item 身高(cm) = A$\times$6+80（A$\leq$12）。
        \item 体重(kg) = [A$\times$7-5]/2（A$\leq$10）。
    \end{itemize}

    \item \textbf{骨骼和肌肉：}
    \begin{itemize}
        \item \textbf{骨骼：}骨胶多、钙盐少，弹性大、硬度小，不易骨折而易弯曲变形；骨钙沉积+++，肌肉发育++；保证充足的钙摄入，增加骨骼硬度。
        \item \textbf{肌肉：}恒牙换牙6-7岁；钙质沉积更早。
        \item \textbf{肌肉成分：}含水多，CH、Lip、Pro少，易疲劳但恢复快；肌肉发育不均衡：大肌群比小肌群先发育，精细动作耐力差。
    \end{itemize}

    \item \textbf{脑细胞分化基本完成：}
    \begin{itemize}
        \item 脑细胞体积增大、神经纤维髓鞘化完成；神经传导逐渐迅速、准确。
        \item 兴奋和抑制功能增强：综合分析、联想推理、概括归纳能力提高；有目的有意识的行为、模仿性加强。
        \item 抑制功能仍较弱：精神易紧张、情绪易波动，过多的脑力活动易疲劳。
        \item 学龄期儿童的大脑可塑性较高，是培养良好行为和习惯的重要阶段。
    \end{itemize}

    \item \textbf{呼吸系统：}易受损伤和感染。

    \item \textbf{消化系统：}
    \begin{itemize}
        \item 牙齿：处于换牙阶段，较脆弱易损伤。
        \item 消化液：酶活性减弱。
    \end{itemize}

    \item \textbf{泌尿系统：}
    \begin{itemize}
        \item 肾脏功能较差，不宜吃过咸的食物。
        \item 膀胱：膀胱黏膜柔嫩，肌肉层和弹力纤维发育不完善，肾脏浓缩功能较差，小肠蠕动较多；不易感觉胀满而不能很好控制排尿。
    \end{itemize}

    \item \textbf{循环系统：}心率快；不宜从事剧烈活动和紧张的体力劳动。

    \item \textbf{性发育：}
    \begin{itemize}
        \item 男女在形态、生理、生化等方面出现差异。
        \item 青春期前同性别，体重和瘦体重略＞女。
    \end{itemize}
\end{enumerate}

\textbf{学龄期儿童心理发育特点：}
\begin{enumerate}[leftmargin=*]
    \item 游戏性活动 $\rightarrow$ 学习性活动。
    \item 学校生活、活动范围扩大。
    \item 集体意识逐渐增强：竞争和荣誉意识增强，善于表现自己。
    \item 饮食行为：容易受环境影响（同伴压力/攀比），形成挑食偏食等不合理饮食行为。
\end{enumerate}

\textbf{青少年生长发育特点：}
\begin{itemize}[leftmargin=*]
    \item \textbf{发育阶段：}儿童到成人的过渡；性成熟阶段。
    \item \textbf{身心发展状态：}形态、生理、内分泌以及行为。

    \item \textbf{1. 内分泌：}下丘脑-垂体-性腺轴、甲状腺轴、肾上腺轴、生长激素轴。

    \item \textbf{2. 体格发育：}
    \begin{itemize}
        \item \textbf{身高：}性别差异：男性较女性晚2年；男7-9 cm/y（=28cm），女5-7 cm/y（=25cm）。
        \item \textbf{体重：}4-5 kg/y，时间较长且幅度较大，青春期后仍可继续增长。
        \item \textbf{形态：}
        \begin{itemize}
            \item 女10-14岁、男11-15岁时增长较快。
            \item 四肢比脊柱早；从远到近：脚最先$\rightarrow$下肢$\rightarrow$上肢。
            \item 男：显著出现肌肉和骨骼增长；女：持续积累脂肪（皮下脂肪），体脂升高与月经初潮有关，身高增长趋于平缓。
        \end{itemize}
    \end{itemize}

    \item \textbf{3. 神经系统：}脑细胞的形态结构和功能都不断地朝着复杂化发展。

    \item \textbf{4. 骨量：}青春期是骨量和骨密度增加的最关键阶段，是峰值骨量积累的关键期。

    \item \textbf{5. 心肺器官功能发育：}心肺功能、肌力显著提高，运动能力增强。
\end{itemize}

\textbf{青春期性发育：}
\begin{enumerate}[leftmargin=*]
    \item 性腺：10岁前均处于静止状态，青春期后在激素作用下迅速发育，机能逐渐成熟。
    \item 第二性征。
    \item 月经/遗精。
\end{enumerate}

\textbf{青春期发育特点和问题：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{"追赶性生长"：}身高突增的年龄、体格大小、营养需要、对营养素缺乏的反应等都存在差异，特别是高峰出现年龄。
    \item \textbf{心理发育不一致的矛盾。}
    \item \textbf{肥胖促使儿童性发育提前。}
\end{enumerate}
\end{keybox}

\subsection{青少年心理特点}

\begin{defbox}
\begin{itemize}[leftmargin=*]
    \item 体格发育快，逐渐成熟。
    \item 认知能力：抽象逻辑思维发展成熟，自我控制能力逐渐增强，认识片面和表面，易冲动。

    \item \textbf{心理特点：}
    \begin{itemize}
        \item 易产生不良饮食习惯：偏食或爱吃零食。
        \item 独立意识增强，易受同伴影响盲目节食或模仿不良饮食行为。
        \item 肥胖：性格孤僻、自卑、缺乏自信。
    \end{itemize}

    \item \textbf{心理社会化：}
    \begin{itemize}
        \item 同伴影响力增强，同伴关系开始占主导地位。
        \item 思维的组织性、深刻性、批判性和独立性明显发展。
        \item 心理承受和适应能力相对不足。
    \end{itemize}
\end{itemize}
\end{defbox}

\subsection{学龄儿童与青少年营养需要}

\begin{keybox}
\textbf{一、特点：}

    \item 维持生命活动、学习及身体活动+生长发育需要。
    \item 与性发育有关。青春期前同性别，男女营养需要差别较小；青春期后差异大，14-18岁达高峰。

\subsection{衰老概述}

\begin{defbox}
\textbf{1. 衰老}是随年龄增加而出现的生物学过程，是生命体共有的特征，主要包括发育期、生殖期、衰老期等生命阶段。

\textbf{2. 正常衰老（normal aging）：}
涉及机体组织结构和生理功能的改变，如抗氧化酶活性下降、自由基代谢/脂质代谢/糖代谢/蛋白代谢紊乱、动脉粥样硬化、各器官的缺血性改变、免疫功能失调、内分泌失调等。

机体老化的结果可直接导致疾病和衰老的发生。

衰老是一个自然过程：可分为\textbf{病态衰老、常态衰老和成功衰老}。

\textbf{3. 成功衰老}主要表现为：整体身体健康，寿命延长；认知功能较高，残障或功能丧失只在生命晚期出现且持续时间不长。人群中健康者比例越来越高，人口健康预期寿命（Healthy Life Expectancy）延长。

\textbf{4. 营养与衰老：}
合理的膳食结构和习惯不仅有利于维持健康，也有助于延缓衰老、预防某些疾病的发生。如癌症、心血管疾病、糖尿病、骨质疏松等。

\textbf{5. 衰老的自由基学说（1956年）：}
\begin{itemize}[leftmargin=*]
    \item 自由基攻击细胞膜脂质中的不饱和脂肪酸，引起细胞凋亡和信息传递受阻；进一步破坏细胞内溶酶体、线粒体等细胞器，导致蛋白质、脂质以及核酸损伤。
    \item 只要生命存在，在一定条件下就会产生自由基。
    \item 生命过程中的氧化损伤的累积是导致衰老和退行性疾病的主要原因。
    \item 机体同时存在SOD、GPx、抗氧化蛋白（饮食中维生素E、C及多酚类抗氧化物质）等防御机制。
\end{itemize}

\textbf{6. 生命过程的矛盾对立统一：}
维持生命需要摄取食物，代谢营养物质产生自由基，损伤修复能力不断累积，导致机体老化和疾病的发生。
\end{defbox}

\subsection{衰老的机体结构和功能变化}

\begin{defbox}
\textbf{1. 形体变化：}
\begin{itemize}[leftmargin=*]
    \item 身高下降：每10年身高递减1厘米，椎间盘压缩性，脊柱弯曲度增加。椎骨扁平化，下肢弯曲，臀部变平。
    \item 外貌：皮肤弹性下降、松弛、粗糙干裂、无光泽、光滑度降低。皮下脂肪逐渐消失。
    \item 老年斑、白发。
\end{itemize}

\textbf{2. 机体组成成分的变化：}
\begin{itemize}[leftmargin=*]
    \item 组织中水分占体重的比例逐渐减少，实质成分减少的同时脂肪比例却增加。
    \item 如75岁老人与25岁青年比较，组织成分由17\%下降到12\%，细胞内水分由42\%下降到33\%，细胞外水分不变，体脂却由15\%增加到30\%。
\end{itemize}

\textbf{3. 组织器官的改变：}
\begin{itemize}[leftmargin=*]
    \item 细胞萎缩、死亡，细胞数量减少。
    \item 脑细胞数量减少、脑体积缩小、重量减轻、发生萎缩。
    \item 胰腺：胰泡缩小、胰叶间质增生，胰岛纤维化、脂肪浸润，胰岛B细胞减少。
    \item 肾：肾纤维化增加，肾单位萎缩和数量减少。
    \item 肝：肝脏体积减小，结缔组织结构增生，肝细胞脂褐素增加。
    \item 心：心肌细胞脂褐素增加，退行性变，心肌萎缩、心肌纤维化、脂肪浸润；心内膜增厚、纤维组织增多；冠状动脉不同程度硬化。
    \item 结缔组织：
    \begin{enumerate}
        \item 胶原纤维：胶原纤维形成交联且数目减少，弹性差，溶解度低，新胶原合成减少，周期延迟——主要表现为皮肤、肌腱和韧带；主动脉和肺组织失去弹性。
        \item 基质：透明质酸含量降低而被硫酸软骨素等取代，降低了黏多糖的聚合状态，改变了基质中透明质酸和硫酸软骨素的含量比例，造成动脉粥样硬化。
    \end{enumerate}
\end{itemize}

\textbf{4. 器官的功能改变：}
\begin{itemize}[leftmargin=*]
    \item 储备能力降低，应变能力降低，抵抗力下降，免疫功能下降。
    \item 肾小球滤过和分泌功能丧失。
    \item 体内细胞分裂和增殖能力下降，增殖速度减慢。
    \item 某种组织细胞数量减少，组织细胞萎缩，器官营养障碍，从而导致器官血液灌注量减少，血液供应不足，又促使组织进一步萎缩，影响器官的功能。
\end{itemize}

\textbf{5. 机体代谢：}
\begin{itemize}[leftmargin=*]
    \item 基础代谢：基础代谢率下降。
    \item 脂质代谢：体脂增加和血脂升高。
    \item 糖代谢：糖耐量降低。
    \item 蛋白质代谢：分解$>$合成。
\end{itemize}

\textbf{少肌症（Sarcopenia）：}
\begin{itemize}[leftmargin=*]
    \item 由衰老引起的骨骼肌质量的进行性丧失。
    \item 四肢肌肉量（kg）/身高（m）$^2$值低于年轻成人正常参考值-2SD以上即可诊断为少肌症。
    \item 随着年龄增加而改变，50岁以后人的骨骼肌量平均每年减少\imp{1\%-2\%}。60岁以上慢性肌肉丢失估计约\imp{30\%}，80岁以上为\imp{50\%}。肌肉丢失30\%影响肌肉的正常功能。
\end{itemize}

\textbf{6. 神经系统：}
\begin{itemize}[leftmargin=*]
    \item 脑细胞减少（10\%-30\%），脑体积缩小、重量减轻。
    \item 脂褐素等沉积，导致神经细胞功能减退和认知障碍。
    \item 神经萎缩，传导通路功能停滞。
    \item 周围神经系统老龄化改变：神经纤维退行性变；周围神经结缔组织增生。
    \item 兴奋和抑制功能转换减慢：记忆力和反应能力下降，动作协调性差，注意力下降；听力衰退，视觉、味觉和嗅觉等功能减退。
\end{itemize}

\textbf{7. 正常衰老与认知行为（Normal Aging and Cognitive Behavior）：}
\begin{itemize}[leftmargin=*]
    \item 忘记眼镜、钥匙等常用物品放的地方或忘记熟悉人的名字。
    \item 偶尔忘记约会，记不住最近读过的一本书籍等。
    \item 走进房间却不知道为何要进去。
    \item 不能很好地组织要说的话。
    \item 信息处理和决策速度变慢。
\end{itemize}

\textbf{8. 膳食与衰老的关系：}
\begin{itemize}[leftmargin=*]
    \item 延缓基础代谢率降低。
    \item 控制体脂增加（主要沉积于腹部）。
    \item 蛋白质：维持水分、肌肉组织，增强免疫功能，保证矿物质。
    \item 优质蛋白质+维生素D+适度体力活动。
\end{itemize}
\end{defbox}

\subsection{老年人营养需要}

\begin{keybox}
\textbf{1. 能量：}
\begin{itemize}[leftmargin=*]
    \item 基础代谢降低、体力活动减少、生长发育活动停止 $\rightarrow$ 能量需要量降低。
    \item 30y：男\imp{2050} Kcal/d，女\imp{1700}。
    \item 50y：男\imp{1950} Kcal/d，女\imp{1600}。
    \item 65y：男\imp{1900} Kcal/d，女\imp{1550}。
    \item 75y：男\imp{1800} Kcal/d，女\imp{1500}。
    \item 注意：能量减少须保证其他营养素充足。
\end{itemize}

\textbf{2. 蛋白质：}
\begin{itemize}[leftmargin=*]
    \item 蛋白质利用率差，易出现负氮平衡；老年人肝功能降低，摄入蛋白质过多会增加肝肾负担。
    \item ONL（必要氮损失）：老年人基础代谢率下降，蛋白质合成和周转率降低，维持机体组织功能的蛋白质转化和更新占的比例变小。
    \item RNI/d：65y$\sim$：男\imp{72}、女\imp{62}，$>$成年人；占供能比\imp{15\%-20\%}。
    \item 注意：优质蛋白质占\imp{1/2}，动物性蛋白质不宜过多，多选大豆蛋白。
    \item 膳食摄入量\imp{0.8 g/(kg$\cdot$bw/d)}即可预防负氮平衡。如需预防和治疗少肌症，推荐\imp{1.0-1.5 g/(kg$\cdot$bw/d)}为佳。
\end{itemize}

\textbf{3. 脂类：}
\begin{itemize}[leftmargin=*]
    \item 供能占总量\imp{25\%}，AMDR \imp{20\%-30\%}为宜，亚油酸\imp{4\%}，$\alpha$-亚麻酸\imp{0.6\%}。
    \item 饱和脂肪酸\%E $<$\imp{10}。
    \item P:M:S = (0.8-1):1:(0.6-0.8)。
    \item n3:n6 = \imp{1:4-6}。
    \item 胆固醇$<$\imp{300 mg/d}。
\end{itemize}

\textbf{4. 碳水化合物：}
\begin{itemize}[leftmargin=*]
    \item AMDR：占总能量\imp{50\%-65\%}。
    \item 推荐膳食纤维摄入量\imp{25-30 g/d}。
    \item 糖耐量下降：添加糖和含糖饮料$<$\imp{10\%E}；建议选择血糖生成指数（GI）低的食物，且粗细搭配。
\end{itemize}

\textbf{5. 矿物质：}
\begin{itemize}[leftmargin=*]
    \item \textbf{钙：}50y$\sim$ RNI \imp{800 mg/d}=成人。
    \item \textbf{铁：}老年人贫血发生率高于成人。RNI \imp{12 mg/d}=成人。65y$\sim$：\imp{10 mg/d}$<$成人$\female$（绝经后）；50y$\sim$：\imp{18 mg/d}=成人$\female$（有月经）。
    \item \textbf{锌：}参与维持血糖代谢和加强胰岛素调节功能。
    \item \textbf{钠：}维持心肌应激性。

    \item \textbf{绝经后骨质疏松（Postmenopausal osteoporosis，PMOP）：}
    \begin{itemize}
        \item 危险因素：钙摄入不足 + 雌激素缺乏 + 老年人VitD合成减少 + 降钙素合成减少 + 甲状旁腺激素分泌异常。
        \item 中间环节：肠道钙吸收降低 $\rightarrow$ 血钙降低 $\rightarrow$ PTH分泌增强 $\rightarrow$ 骨吸收增加 $\rightarrow$ 骨质疏松。
        \item 老年人钠摄入：味觉降低，食盐摄入增加，是高血压的危险因素。
    \end{itemize}
\end{itemize}

\textbf{6. 维生素：}
\begin{itemize}[leftmargin=*]
    \item 维生素D：65y$\sim$ RNI=\imp{15 $\mu$g/d} $>$ 成人\imp{10 $\mu$g/d}。
    \item 维生素B$_{6}$：50y$\sim$ RNI=\imp{1.6 mg/d} $>$ 成人\imp{1.4 mg/d}。
    \item 其他维生素均与18岁以上成人相同标准。
    \item 补充维生素C效果不佳。VitC有利于保持血管弹性、降低血胆固醇、预防贫血。
\end{itemize}
\end{keybox}

\textbf{衰老是不可逆转、不可避免；合理膳食 + 适宜劳动。}

\textbf{老年人（65岁以上人群）常见膳食问题：}
\begin{itemize}[leftmargin=*]
    \item 极易出现膳食摄入不足（如饱腹感上升、食物摄入不足或营养缺乏）。
    \item 应注意限制纳、精准营养素。
\end{itemize}

\end{enumerate}
\end{keybox}

\subsection{运动员营养代谢特点}

\begin{defbox}
\textbf{一、各系统功能改变：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{心血管：}血容量增加，心输出量大幅提升，供氧供能需求大幅增加。
    \item \textbf{神经：}超负荷训练 $\rightarrow$ 神经调节紊乱、疲劳、运动能力下降。
    \item \textbf{消化：}运动时胃肠道供血减少，营养素吸收减弱。
    \item \textbf{免疫：}长期高强度训练 $\rightarrow$ 免疫力下降、易呼吸道感染；中低强度有氧运动可提升免疫功能。
    \item \textbf{内分泌：}长期大强度训练 $\rightarrow$ 激素紊乱；女性运动员多见月经不调、闭经。
\end{enumerate}

\textbf{二、能量和物质代谢特点：}
\begin{itemize}[leftmargin=*]
    \item 能量消耗极高，易出现氧债，恢复期能耗仍大。
    \item 蛋白质分解增强、尿氮流失多，易发生运动性贫血和中枢疲劳。
    \item 高强度运动时脂肪氧化受限；糖原消耗快，是耐力短板。
    \item 大量出汗时，钠、钾、钙、镁等矿物质和水溶性维生素流失严重。
    \item 训练刺激微量元素代谢，需求量高于普通人。
\end{itemize}
\end{defbox}

\subsection{运动员各类营养素需要}

\begin{defbox}
\textbf{1. 能量：}
\begin{itemize}[leftmargin=*]
    \item 总能量：\imp{3500-4400 kcal/d}（210-280 kJ/kg）。
    \item 供能配比：蛋白质:脂肪:碳水化合物 = \imp{1:(0.7-0.8):4}。
    \item 影响因素：运动项目、强度、时长、体重、膳食结构。
\end{itemize}

\textbf{2. 蛋白质：}
\begin{itemize}[leftmargin=*]
    \item 适宜摄入：\imp{1.2-2.0 g/(kg$\cdot$d)}，优质蛋白占\imp{1/3}以上。
    \item 作用：修复肌肉、促进恢复、抗疲劳、预防运动性贫血。
    \item 摄入不足$\rightarrow$耐力下降、乏力；过量加重肝肾负担。
\end{itemize}

\textbf{3. 脂类：}
\begin{itemize}[leftmargin=*]
    \item 供能占总量\imp{25\%-30\%}；耐力项目可至\imp{50\%-60\%}，极限可达\imp{70\%}。
    \item 限制饱和脂肪，防止血脂升高、降低耐力。
\end{itemize}

\textbf{4. 碳水化合物（核心供能物质）：}
\begin{itemize}[leftmargin=*]
    \item 高强度短时运动：碳水化合物为主；长时间低强度才动用脂肪和蛋白质。
    \item \textbf{糖原填充法：}赛前1周低碳水耗糖原，赛前3天高碳水储备糖原（占膳食\imp{70\%}）。
    \item 作用：快速供能、无代谢废物，决定耐力上限。
\end{itemize}

\textbf{5. 水和矿物质：}
\begin{itemize}[leftmargin=*]
    \item \textbf{补水：}运动大量出汗，少量多次补水，忌一次性暴饮；补液总量应大于出汗量，同步补电解质。
    \item \textbf{钠：}高温训练流失多，低钠可致乏力恶心，食盐$<$\imp{5 g/d}。
    \item \textbf{钾：}流失可致心律失常，推荐\imp{3-4 g/d}，蔬果补充。
    \item \textbf{镁：}多汗缺镁$\rightarrow$抽筋，推荐\imp{400-500 mg/d}。
    \item \textbf{钙：}高强度+高温易缺钙、骨质疏松，推荐\imp{1000-1500 mg/d}，奶制品补充。
    \item \textbf{铁：}训练流失多，易发运动性贫血；女性重点补铁，动物红肉补铁。注意额外补充。
    \item \textbf{锌：}大训练量锌流失、免疫力下降，推荐\imp{25 mg/d}，海产品补充。
    \item \textbf{硒：}提升免疫，推荐\imp{50-150 $\mu$g/d}。
    \item \textbf{碘：}\imp{150 $\mu$g/d}，海产品补充。
\end{itemize}

\textbf{6. 维生素（需求量高于普通人群）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{水溶性B族（运动所需）：}
    \begin{itemize}
        \item 维生素B$_{1}$：\imp{3-5 mg/d}，缺乏$\rightarrow$丙酮酸堆积、神经疲劳；高碳水训练需增量。
        \item 维生素B$_{2}$：\imp{2-2.5 mg/d}，缺乏$\rightarrow$有氧/无氧运动能力下降。
        \item 烟酸、B$_{6}$、B$_{12}$、叶酸：参与能量代谢、造血，缺乏$\rightarrow$运动耐力下降。长时间缺时，赛前\imp{2-3周}补充制剂。
    \end{itemize}
    \item \textbf{维生素C：}训练推荐\imp{200 mg/d}；促进胶原修复、促铁吸收、抗氧化。
    \item \textbf{维生素A：}视力需求高项目（射击、击剑）\imp{1800 $\mu$g RAE/d}；勿过量中毒。
    \item \textbf{维生素D：}\imp{10-12.5 $\mu$g/d}；保护骨骼，缺乏易运动骨折。
    \item \textbf{维生素E：}高原/耐力项目\imp{15-50 mg $\alpha$-TE/d}；抗氧化、清除乳酸堆积；不可过量（蓄积中毒）。
\end{itemize}
\end{defbox}

\section{特殊环境人员营养}

\begin{tipbox}
\textbf{高温环境人员营养需要和膳食原则（了解）：}

\begin{enumerate}[leftmargin=*]
    \item \textbf{水和矿物质：}
    \begin{itemize}[leftmargin=*]
        \item 高温环境大量出汗 $\rightarrow$ 水和电解质大量丢失（钠、钾、钙、镁）。
        \item 少量多次补水，忌一次性暴饮；同时补充电解质（含钠、钾的运动饮料或淡盐水）。
        \item 食盐摄入比常温环境适当增加（通过膳食或饮水中补充）。
    \end{itemize}

    \item \textbf{能量：}高温环境基础代谢率升高（体温每升高1$^\circ$C，BMR增加约13\%），能量需要较常温增加。
    \item \textbf{蛋白质：}高温下蛋白质分解代谢增强，尿氮排出增加，蛋白质需要量相应提高（优质蛋白占1/2以上）。
    \item \textbf{维生素：}
    \begin{itemize}[leftmargin=*]
        \item 水溶性维生素（B族、C）随汗液流失显著增加。
        \item 维生素C推荐摄入提高至150-200 mg/d。
        \item 维生素B$_{1}$、B$_{2}$按能量比例相应增加（每1000 kcal：B$_{1}$ 0.5-0.6 mg，B$_{2}$ 0.55-0.65 mg）。
    \end{itemize}
    \item \textbf{膳食原则：}清淡可口，增进食欲；多提供汤类和液体食物；保证蔬菜、水果供给充足（补充水和矿物质）；适当增加食盐摄入。
\end{enumerate}

\textbf{低温环境人员营养需要和膳食原则（了解）：}

\begin{enumerate}[leftmargin=*]
    \item \textbf{能量：}低温环境下基础代谢率升高，机体需要额外产热以维持体温 $\rightarrow$ 能量需要显著增加（可比常温增加10\%-15\%以上）。
    \item \textbf{宏量营养素：}
    \begin{itemize}[leftmargin=*]
        \item 脂肪供能比可提高至35\%-40\%（增加能量密度和饱腹感）。
        \item 碳水化合物供能比维持在50\%-55\%（保证快速供能）。
        \item 蛋白质供给充足（优质蛋白占1/2以上），防止负氮平衡。
    \end{itemize}
    \item \textbf{维生素：}
    \begin{itemize}[leftmargin=*]
        \item 维生素C有助增强耐寒能力，推荐摄入量适当增加。
        \item 维生素B族需要量随能量增加而相应增加。
        \item 维生素A、D、E的摄入也应适当增加。
    \end{itemize}
    \item \textbf{矿物质：}寒冷气候下尿钠排出增加，钠需要量可适当提高；钙、铁、锌、碘的供给亦需充足。
    \item \textbf{膳食原则：}提供高能量密度食物，保证热食热饮供应；增加脂肪摄入比例；食物温热可口；保证蔬菜、水果；食盐摄入可适当增加。
\end{enumerate}
\end{tipbox}

        \item \textbf{生命早期1000天 $\rightarrow$ DOHaD：}孕期270d + 0-6月龄180d + 7-24月龄为机遇窗口。早期营养不良经表观遗传（发育程序化）导致远期代谢综合征高风险。
        \item \textbf{成年期：}膳食行为和生活方式继续强化或修正早期程序化效应。膳食指南2022贯穿全生命周期。
        \item \textbf{老年期 $\rightarrow$ 少肌症：}蛋白质需要量不减反增（1.0-1.5 g/(kg$\cdot$bw)/d），能量需求下降但蛋白质需求上升。衰老自由基学说：氧化损伤累积$\rightarrow$退行性病变。
        \item \textbf{核心结论：}营养影响是连续的——从生命早期1000天到老年期，营养贯彻生命周期理念。
    \end{itemize}

    \item 运动员五大系统生理改变：心血管（血容量和心输出量增加）、神经（超负荷训练易疲劳紊乱）、消化（运动时供血减少影响吸收）、免疫（高强度训练致免疫下降，中低强度有氧提升免疫）、内分泌（大强度可致激素紊乱，女性月经不调闭经）。整体代谢特点：能量消耗极高，蛋白分解增强尿氮流失多，糖原消耗快为耐力短板，大量出汗致水和矿物质及水溶性维生素流失，微量元素需求高于常人。
    \item 运动员营养素需要要点——碳水化合物：高强度短时运动核心供能，糖原填充法（赛前1周低碳水耗糖原，赛前3天高碳水储备占膳食70\%）；蛋白质：1.2-2.0 g/(kg$\cdot$d)，优质蛋白1/3以上；水与矿物质：少量多次补液，总补液量大于出汗量同步补电解质；钠钾钙镁铁锌硒碘均需充分补充。
    \item 高温环境人员：大量出汗致水电解质流失，少量多次补水+补电解质；BMR升高能量增加；蛋白分解增强；水溶性维生素随汗液流失显著增加（VC 150-200 mg/d）；清淡可口膳食多汤类多蔬果。低温环境人员：BMR升高能量显著增加（10\%-15\%以上）；脂肪供能比可提高至35\%-40\%；碳水50\%-55\%；蛋白充足；VC、B族增加；食盐可适当增加；保证热食热饮，食物温热可口。
\end{enumerate}
