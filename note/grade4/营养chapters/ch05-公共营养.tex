\section{膳食营养素参考摄入量（DRIs）}
% =====================================================================

\begin{keybox}
\textbf{DRIs（Dietary Reference Intakes）：}一组每日平均膳食营养素摄入量的参考值。评价膳食营养素供给量能否满足人体需要及是否存在过量摄入风险的一组参考值。
\end{keybox}

\begin{keybox}
\textbf{DRIs指标体系（七项）：}

\begin{table}[H]
\centering
\begin{tabularx}{\textwidth}{lXl}
\toprule
\textbf{指标} & \textbf{定义} & \textbf{要点} \\
\midrule
\imp{平均需要量（EAR）} & 某一特定性别、年龄及生理状况群体中对某营养素需要量的平均值。摄入量达到EAR水平时可以满足群体中50\%个体的需要。 & EAR是制订RNI的基础。 \\
\midrule
\imp{推荐摄入量（RNI）} & 相当于RDA，可以满足某一特定性别、年龄及生理状况群体中绝大多数（97\%-98\%）个体需要量的摄入水平。 & RNI = EAR + 2SD，或 RNI = 1.2 × EAR。 \\
\midrule
\imp{适宜摄入量（AI）} & 通过观察或实验获得的健康人群某种营养素的摄入量。 & AI > RNI；AI主要用于个体营养素摄入目标，不能用于人群营养评价。AI的准确性不如RNI。 \\
\midrule
\imp{可耐受最高摄入量（UL）} & 平均每日摄入营养素的最高限量。 & 摄入量 < UL时，对健康人群无任何副作用；摄入量 > UL时，发生不良反应的风险增加。 \\
\midrule
\imp{宏量营养素可接受范围（AMDR）} & 宏量营养素的理想摄入量范围，以占能量摄入量的百分比(\%E)表示。 & 脂肪 20\%-30\%，碳水化合物 50\%-65\%，蛋白质 10\%-15\%（成人）。 \\
\midrule
\imp{预防非传染性慢性病的建议摄入量（PI-NCD / PI）} & 以非传染性慢性病的一级预防为目标。 & PI > RNI/AI时，以PI为推荐量；PI < RNI/AI时，以RNI/AI为基本推荐摄入量。 \\
\midrule
\imp{特定建议值（SPL）} & 专用于预防非传染性慢性病。某些膳食成分与NCD的关系已有充分证据，但尚未达到可制定PI的水平，此时可提出SPL。 & 主要针对植物化学物等膳食成分。 \\
\bottomrule
\end{tabularx}
\end{table}
\end{keybox}

\section{膳食模式与膳食指南}
% =====================================================================

\subsection{膳食模式}

\begin{defbox}
\textbf{膳食模式（Dietary Pattern）/ 膳食结构：}膳食中各类食物的种类、数量及其在膳食中所占的比例。主要分为以下四种类型：

\begin{enumerate}[leftmargin=*]
    \item \textbf{东方膳食模式 / 植物性食物为主型}——发展中国家模式。以植物性食物为主，动物性食物摄入较少，膳食纤维丰富但优质蛋白和脂肪不足。
    \item \textbf{西方膳食模式 / 动物性食物为主型}——欧美经济发达国家模式。高能量、高脂肪、高蛋白、低膳食纤维（"三高一低"），与肥胖、心脑血管疾病、糖尿病、结直肠癌等慢性病风险增加相关。
    \item \textbf{地中海膳食模式}——希腊、西班牙、法国和意大利南部等。以橄榄油为主要脂肪来源，大量蔬菜水果、全谷物、豆类、适量鱼类和红酒，饱和脂肪低，与较低的心血管疾病风险相关。
    \item \textbf{日本膳食模式}——动植物食物平衡型。动植物消费均衡，水产品食量较大，三大宏量营养素供能比合理。
\end{enumerate}

\textbf{健康膳食的基本要求：}
\begin{enumerate}[leftmargin=*]
    \item 食物多样，以谷类为主
    \item 吃动平衡，健康体重
    \item 多吃蔬果、奶类、大豆
    \item 适量吃鱼、禽、蛋、瘦肉
    \item 少盐少油，控糖限酒
    \item 杜绝浪费，兴新食尚
\end{enumerate}
\end{defbox}

\subsection{膳食指南}

\begin{keybox}
\textbf{膳食指南（Dietary Guidelines, DG）：}《中国居民膳食指南(2022)》

\textbf{8条核心准则：}
\begin{enumerate}[leftmargin=*]
    \item 食物多样，合理搭配
    \item 吃动平衡，健康体重
    \item 多吃蔬果、奶类、全谷、大豆
    \item 适量吃鱼、禽、蛋、瘦肉
    \item 少盐少油，控糖限酒
    \item 规律进餐，足量饮水
    \item 会烹会选，会看标签
    \item 公筷分餐，杜绝浪费
\end{enumerate}
\end{keybox}

\section{营养调查与评价}
% =====================================================================

\begin{keybox}
\textbf{营养调查（Nutrition Survey）：}

\textbf{目的：}
\begin{enumerate}[leftmargin=*]
    \item 了解不同人群的膳食结构和营养状况
    \item 发现营养相关疾病和问题
    \item 评价营养改善措施效果
    \item 为制定营养政策提供依据
\end{enumerate}

\textbf{营养状况综合评价的四个方面：}
\begin{enumerate}[leftmargin=*]
    \item 膳食调查
    \item 体格测量
    \item 营养缺乏病临床检查
    \item 营养水平生化检验
\end{enumerate}
\end{keybox}

\begin{keybox}
\textbf{常用膳食调查方法：}

\begin{table}[H]
\centering
\begin{tabularx}{\textwidth}{lXX}
\toprule
\textbf{方法} & \textbf{特点} & \textbf{优缺点} \\
\midrule
\imp{称重法（Weighed Food Records）} & 对食物进行准确称量，通常 3-7 天。 & 优点：准确可靠。缺点：费时费力。 \\
\midrule
\imp{24小时膳食回顾法（24h History Recalls）} & 调查对象回顾过去 24 小时内摄入的所有食物。 & 优点：简便快速。缺点：依赖记忆，对调查员要求高，一般 80\% 以上可获得满意结果。 \\
\midrule
\imp{食物频率问卷法（FFQ）} & 通过食物频率问卷评估长期膳食摄入模式。 & 优点：可调查长期膳食模式。缺点：精确度较低。 \\
\midrule
\imp{化学分析法} & 采集食物通过实验室化学分析方法测定营养素含量。 & 优点：最准确的方法，结果精确。缺点：费用高，耗时。 \\
\bottomrule
\end{tabularx}
\end{table}
\end{keybox}

\begin{keybox}
\textbf{人体测量常用指标与评价：}

\textbf{Broca改良公式：}
\[
\text{标准体重(kg)} = \text{身高(cm)} - 105
\]
\begin{itemize}[leftmargin=*]
    \item 正常体重 = 标准体重 ± 10\%
    \item 超重 > 10\%
    \item 肥胖 > 20\%
\end{itemize}

\textbf{体质指数（Body Mass Index, BMI）：}
\[
\text{BMI} = \frac{\text{体重(kg)}}{[\text{身高(m)}]^2}
\]
\begin{table}[H]
\centering
\begin{tabularx}{0.7\textwidth}{lX}
\toprule
\textbf{BMI (kg/m\textsuperscript{2})} & \textbf{评价} \\
\midrule
< 18.5 & 消瘦 \\
18.5 - 23.9 & 正常 \\
24.0 - 27.9 & 超重 \\
≥ 28 & 肥胖 \\
\bottomrule
\end{tabularx}
\end{table}
\end{keybox}

\section{营养教育、营养改善与NRV}
% =====================================================================

\begin{defbox}
\textbf{营养教育（Nutrition Education）：}以改善人民营养状况为目标，通过营养信息交流，帮助个体和群体获得食物与营养知识，形成科学的合理饮食习惯，改善营养健康状况。

\textbf{营养改善措施：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{营养强化（Food Fortification）：}根据营养需要向食品中添加一种或多种营养素或某些天然食品，以提高食品营养价值的过程。
    \item \textbf{营养补充：}以营养补充剂的形式补充特定营养素，适用于特定人群或缺乏状态。
    \item \textbf{膳食多样化：}通过增加食物种类和提高膳食质量，从天然食物中获取全面的营养素。
\end{enumerate}
\end{defbox}

\begin{defbox}
\textbf{营养素参考值（Nutrient Reference Values, NRV）：}用于食品营养标签上比较食品营养素含量的参考值。NRV是专用于食品营养标签的参考标准，便于消费者比较不同食品的营养价值，做出合理的食物选择。
\end{defbox}

\subsection*{参考答案要点}

\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item 见正文DRIs表。
    \begin{itemize}
        \item \textbf{DRIs：}一组每日平均膳食营养素摄入量的参考值，用于评价膳食营养素供给量能否满足人体需要及是否存在过量摄入风险。
        \item \textbf{EAR：}满足群体中50\%个体需要量，是制订RNI的基础。
        \item \textbf{RNI：}满足97\%-98\%个体需要量，RNI = EAR + 2SD 或 RNI = 1.2 × EAR。
        \item \textbf{AI：}资料不足无法计算EAR时，通过观察或实验获得。AI > RNI，AI主要用于个体营养素摄入目标，不能用于人群营养评价；AI的准确性不如RNI。
        \item \textbf{UL：}平均每日摄入营养素的最高限量。摄入量 < UL无副作用，> UL不良反应风险增加。
        \item \textbf{AMDR：}宏量营养素可接受范围——脂肪20\%-30\%，碳水化合物50\%-65\%，蛋白质10\%-15\%。
        \item \textbf{PI-NCD：}预防非传染性慢性病的建议摄入量，以NCD一级预防为目标。
        \item \textbf{SPL：}特定建议值，专用于预防NCD，适用于尚未达到制定PI水平的膳食成分。
        \item \textbf{BMI：}体质指数，体重(kg) / [身高(m)]\textsuperscript{2}。
        \item \textbf{NRV：}营养素参考值，用于食品营养标签上比较食品营养素含量的参考值。
    \end{itemize}

    \item DRIs七项指标体系：
    \begin{itemize}
        \item \textbf{EAR（平均需要量）：}满足50\%个体需要，是制订RNI的基础。
        \item \textbf{RNI（推荐摄入量）：}满足97\%-98\%个体需要，RNI = EAR + 2SD 或 1.2 × EAR。
        \item \textbf{AI（适宜摄入量）：}当EAR无法计算时使用，通过观察或实验获得，准确性不如RNI，不能用于人群营养评价。
        \item \textbf{UL（可耐受最高摄入量）：}安全上限，超过则不良反应风险增加。
        \item \textbf{AMDR（宏量营养素可接受范围）：}脂肪20\%-30\%，碳水化合物50\%-65\%，蛋白质10\%-15\%。
        \item \textbf{PI-NCD（预防慢病建议摄入量）：}以NCD一级预防为目标，与RNI/AI比较取较高者。
        \item \textbf{SPL（特定建议值）：}针对其他膳食成分（如植物化学物）的预防NCD建议量。
    \end{itemize}

    \item 8条核心准则：①食物多样，合理搭配；②吃动平衡，健康体重；③多吃蔬果、奶类、全谷、大豆；④适量吃鱼、禽、蛋、瘦肉；⑤少盐少油，控糖限酒；⑥规律进餐，足量饮水；⑦会烹会选，会看标签；⑧公筷分餐，杜绝浪费。\par
    膳食宝塔五层：第一层谷薯类250-400g/d；第二层蔬菜300-500g/d、水果200-350g/d；第三层鱼禽肉蛋120-200g/d；第四层奶及奶制品300-500g/d、大豆及坚果25-35g/d；第五层盐<5g/d、油25-30g/d。饮水1500-1700ml/d，活动6000步/d。

    \item 四种方法比较：
    \begin{itemize}
        \item \textbf{称重法：}准确可靠，但费时费力，通常3-7天。
        \item \textbf{24h回顾法：}简便快速，但依赖记忆，对调查员要求高，一般80\%以上可获得满意结果。
        \item \textbf{FFQ：}可调查长期膳食模式，但精确度较低。
        \item \textbf{化学分析法：}最准确，结果精确，但费用高、耗时。
    \end{itemize}

    \item BMI分级标准：BMI = 体重(kg) / [身高(m)]\textsuperscript{2}。BMI < 18.5 消瘦，18.5-23.9 正常，24.0-27.9 超重，$\ge$28 肥胖。Broca改良公式：标准体重(kg) = 身高(cm) - 105。正常体重 = 标准体重 ± 10\%，超重 > 10\%，肥胖 > 20\%。

    \item 营养状况综合评价的四个方面：①膳食调查；②体格测量；③营养缺乏病临床检查；④营养水平生化检验。

    \item 四种膳食模式：
    \begin{itemize}
        \item \textbf{东方膳食模式（植物性食物为主型）：}发展中国家模式。
        \item \textbf{西方膳食模式（动物性食物为主型）：}欧美经济发达国家模式，"三高一低"——高能量、高脂肪、高蛋白、低膳食纤维。
        \item \textbf{地中海膳食模式：}希腊、西班牙、法国和意大利南部等，以橄榄油为主。
        \item \textbf{日本膳食模式：}动植物食物平衡型，水产品食量较大。
    \end{itemize}
    健康膳食六项基本要求：①食物多样，以谷类为主；②吃动平衡，健康体重；③多吃蔬果、奶类、大豆；④适量吃鱼、禽、蛋、瘦肉；⑤少盐少油，控糖限酒；⑥杜绝浪费，兴新食尚。
\end{enumerate}
