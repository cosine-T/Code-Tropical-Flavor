\section{膳食营养与慢性病}

% ----- 第一节 超重与肥胖 -----
\subsection{超重与肥胖（掌握）}

\begin{keybox}
\textbf{超重与肥胖的公共卫生意义：}
\begin{itemize}[leftmargin=*]
    \item 超重与肥胖是\imp{多种慢性病的独立危险因素}，包括2型糖尿病、心血管疾病、高血压、脑卒中、某些癌症（如结直肠癌、乳腺癌）。
    \item 中国成人超重率约34.3\%，肥胖率约16.4\%（2020年数据），已成为重要的公共卫生问题。
    \item 肥胖的本质是\imp{能量摄入长期大于能量消耗}导致的体内脂肪过量积累。
\end{itemize}
\end{keybox}

\subsubsection{超重与肥胖的膳食营养防治策略}

\begin{keybox}
\textbf{饮食防治核心策略：}

\textbf{1. 控制总能量摄入：}
\begin{itemize}[leftmargin=*]
    \item 在原有能量摄入基础上，\imp{每天减少500～1000kcal能量摄入}，造成适度能量缺口。
    \item 每月减重1～2kg为宜，不超过4kg。
    \item 每日能量摄入不宜低于1000～1200kcal（女性）/ 1200～1600kcal（男性），保证基本营养需求。
\end{itemize}

\textbf{2. 调整宏量营养素供能比：}
\begin{itemize}[leftmargin=*]
    \item 蛋白质：占总能量\imp{20\%～25\%}，适当提高以保证瘦体重。
    \item 脂肪：占总能量\imp{20\%～30\%}，限制饱和脂肪酸和反式脂肪酸。
    \item 碳水化合物：占总能量\imp{45\%～60\%}，优先选择低GI食物和全谷物。
\end{itemize}

\textbf{3. 增加膳食纤维：}
\begin{itemize}[leftmargin=*]
    \item 每日膳食纤维摄入量\imp{25～30g}。
    \item 富含蔬菜、水果、全谷物和豆类。
    \item 膳食纤维可增加饱腹感，延缓胃排空，减少能量摄入。
\end{itemize}

\textbf{4. 限制添加糖和饱和脂肪：}
\begin{itemize}[leftmargin=*]
    \item 添加糖不超过总能量10\%，最好<5\%。
    \item 限制含糖饮料、甜点、油炸食品。
    \item 饱和脂肪酸不超过总能量10\%。
\end{itemize}

\textbf{5. 体力活动配合：}
\begin{itemize}[leftmargin=*]
    \item 有氧运动与抗阻运动相结合：每周至少150分钟中等强度有氧运动 + 每周2～3次抗阻训练。
    \item 运动不仅消耗能量，还能改善体成分（增加瘦体重，减少体脂）。
    \item 配合行为矫正（自我监测体重、记录饮食日记、规律进餐）。
\end{itemize}
\end{keybox}

\begin{tipbox}
\textbf{减重目标设定：}初始减重目标为减少原体重的5\%～10\%。即使未达到"理想体重"，减轻5\%体重已可显著改善血糖、血脂和血压等代谢指标。
\end{tipbox}

% ----- 第二节 心血管疾病 -----
\subsection{心血管疾病的营养防治（熟悉）}

\begin{defbox}
\textbf{心血管疾病（Cardiovascular Disease, CVD）：}包括冠心病、脑卒中、高血压、外周血管疾病等。不合理膳食是CVD最重要的可改变危险因素之一。
\end{defbox}

\subsubsection{膳食营养预防原则}

\begin{keybox}
\textbf{心血管疾病的营养防治原则：}

\begin{enumerate}[leftmargin=*]
    \item \textbf{控制总能量，维持理想体重：}将BMI维持在18.5～23.9。超重和肥胖是CVD的独立危险因素。
    \item \textbf{限制饱和脂肪酸和反式脂肪酸：}饱和脂肪酸<总能量7\%～10\%，反式脂肪酸摄入量越少越好。减少动物脂肪、黄油、糕点、油炸食品。
    \item \textbf{增加不饱和脂肪酸：}多不饱和脂肪酸（PUFA）占总能量6\%～10\%。单不饱和脂肪酸（MUFA）有益。鱼类（富含EPA/DHA）每周至少2次。
    \item \textbf{限制胆固醇摄入：}膳食胆固醇\imp{<300mg/日}。减少动物内脏、脑、蛋黄等高胆固醇食物。
    \item \textbf{控制钠摄入：}\imp{每日食盐<5g（约2000mg钠）}。减少加工食品、咸菜、调味品。
    \item \textbf{增加钾摄入：}多食蔬菜、水果、豆类和全谷物。钾有助于降低血压。
    \item \textbf{增加膳食纤维：}每天25～30g，可降低血清胆固醇。
    \item \textbf{限制饮酒：}男性<25g/d酒精，女性<15g/d。
\end{enumerate}
\end{keybox}

\subsubsection{DASH饮食}

\begin{defbox}
\textbf{DASH饮食（Dietary Approaches to Stop Hypertension）：}一种为防治高血压设计的膳食模式。强调摄入充足的蔬菜、水果、低脂奶制品、全谷物、鱼、禽肉和坚果，同时限制钠盐、红肉、甜食和含糖饮料的摄入。
\end{defbox}

\begin{exambox}
\textbf{DASH饮食特点总结：}高钾、高钙、高镁、高纤维、高蛋白，低饱和脂肪、低总脂肪、低胆固醇、低钠。临床试验证实可降低收缩压8～14mmHg。
\end{exambox}

% ----- 第三节 糖尿病 -----
\subsection{糖尿病的营养防治（熟悉）}

\begin{keybox}
\textbf{糖尿病（Diabetes Mellitus）医学营养治疗的总原则：}

\begin{enumerate}[leftmargin=*]
    \item \textbf{控制总能量：}以达到或维持理想体重为目标。根据BMI和活动水平确定个体化能量摄入。
    \item \textbf{合理分配宏量营养素：}
    \begin{itemize}[leftmargin=*]
        \item 碳水化合物：\imp{占总能量50\%～60\%}。优先选择低血糖生成指数（GI）食物（全谷物、豆类等）。严格限制单糖和双糖。
        \item 蛋白质：\imp{占总能量15\%～20\%}。优质蛋白占1/3以上。有糖尿病肾病者需控制蛋白质摄入量。
        \item 脂肪：\imp{占总能量20\%～30\%}。饱和脂肪酸<7\%总能量，限制胆固醇<300mg/d。
    \end{itemize}
    \item \textbf{膳食纤维：}\imp{每日25～30g}，可溶性纤维有助于降低餐后血糖。
    \item \textbf{规律进餐：}定时定量，少食多餐，避免血糖大幅波动。三餐能量分配：早1/5、中2/5、晚2/5，或早中晚各1/3。
    \item \textbf{配合运动治疗：}有氧运动与阻力运动结合，运动前后注意血糖监测。
    \item \textbf{综合管理：}营养治疗须与运动、药物治疗和血糖监测有机结合。
\end{enumerate}
\end{keybox}

\begin{tipbox}
\textbf{血糖生成指数（Glycemic Index, GI）：}反映食物摄入后升高血糖的速度和能力。低GI食物（$\leq$55）：全谷物、豆类、牛奶；中等GI（56～69）：全麦面包、糙米；高GI（$\geq$70）：白面包、白米饭、含糖饮料。
\end{tipbox}

% ----- 第四节 癌症 -----
\subsection{癌症的营养防治（了解）}

\subsubsection{世界癌症研究基金会（WCRF）推荐}

\begin{defbox}
\textbf{世界癌症研究基金会（WCRF）}定期发布基于证据的癌症预防建议，从膳食、营养和身体活动角度提出一级预防策略。
\end{defbox}

\textbf{WCRF癌症预防核心建议：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{保持健康体重：}将体重控制在正常BMI范围内，避免成年期体重增长。
    \item \textbf{积极运动：}每周至少150分钟中等强度或75分钟高强度身体活动。
    \item \textbf{摄入丰富的全谷物、蔬菜、水果和豆类：}膳食纤维降低结直肠癌风险。
    \item \textbf{限制"快餐"和其他高脂、高淀粉或高糖的加工食品：}控制能量密度。
    \item \textbf{限制红肉和加工肉类：}红肉每周不超过350～500g，尽量少食加工肉类。加工肉类被IARC列为Ⅰ类致癌物（结直肠癌），红肉被列为IIA类。
    \item \textbf{限制含糖饮料：}多喝白水或不加糖的饮料。
    \item \textbf{限制饮酒：}酒精是多种癌症的确切危险因素（口腔癌、咽喉癌、食管癌、肝癌、结直肠癌、乳腺癌）。
    \item \textbf{不依赖膳食补充剂预防癌症：}满足营养需要应通过膳食，而非补充剂。
    \item \textbf{母乳喂养：}有益于母亲和婴儿，降低母亲乳腺癌风险。
\end{enumerate}

\begin{tipbox}
\textbf{WCRF"第三版专家报告"（2018）核心信息：}在所有人群和国家中推广将癌症预防纳入整体非传染性慢性病预防框架中，因为癌症与肥胖、心血管疾病和糖尿病共享共同的膳食和生活方式危险因素。
\end{tipbox}

% ----- 第五节 痛风 -----
\subsection{痛风的营养防治（了解）}

\begin{defbox}
\textbf{痛风（Gout）：}因嘌呤代谢异常，血尿酸水平升高，尿酸盐结晶沉积于关节和软组织导致的炎症性疾病。高尿酸血症是痛风最重要的生化基础。
\end{defbox}

\textbf{痛风膳食预防原则：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{限制高嘌呤食物：}严格限制动物内脏（肝脏、肾脏）、沙丁鱼、贝类、浓肉汤等高嘌呤食物。痛风急性发作期嘌呤<150mg/d。
    \item \textbf{保持足量饮水：}每日饮水量\imp{>2000mL}，促进尿酸排泄。避免含糖饮料和果汁（果糖促进尿酸生成）。
    \item \textbf{限制酒精摄入：}酒精抑制尿酸排泄并促进生成，啤酒对痛风患者的危害尤为突出。
    \item \textbf{体重管理：}肥胖是痛风的独立危险因素，减重可降低血尿酸水平。
    \item \textbf{合理选择蛋白质来源：}以低脂奶制品、蛋类等低嘌呤蛋白为主。
    \item \textbf{多食蔬菜水果：}富含维生素C的蔬果有助于降低尿酸，但避免花椰菜、菠菜等高嘌呤蔬菜（实际上植物嘌呤对血尿酸影响不及动物嘌呤）。
\end{enumerate}

\section{临床营养}

% ----- 第六节 营养筛查与评价 -----
\subsection{营养筛查与评价（熟悉）}

\begin{defbox}
\textbf{临床营养筛查（Nutrition Screening）：}在短时间内用简单的方法发现营养风险或营养不良的个体，及时采取干预措施。常在入院24小时内完成。
\end{defbox}

\textbf{常用营养筛查与评价方法：}

\begin{keybox}
\textbf{1. NRS-2002（Nutrition Risk Screening 2002）：}
\begin{itemize}[leftmargin=*]
    \item ESPEN推荐，适用于住院成人患者。
    \item 包括营养状况评分（体重减少量、BMI、近期进食量）和疾病严重程度评分（应激代谢程度）。
    \item 年龄 $\geq$ 70岁加1分。
    \item 总分 $\geq$ 3分，判定为有营养风险，需制定营养支持计划。
\end{itemize}

\textbf{2. SGA（Subjective Global Assessment，主观整体评价）：}
\begin{itemize}[leftmargin=*]
    \item 基于病史（体重变化、膳食摄入、胃肠道症状、功能状态）和体格检查（皮下脂肪、肌肉消耗、水肿）。
    \item 将患者分为A级（营养良好）、B级（轻～中度营养不良）、C级（重度营养不良）。
    \item 简单无创，适用于临床快速评价。
\end{itemize}

\textbf{3. MNA（Mini Nutritional Assessment，微型营养评价）：}
\begin{itemize}[leftmargin=*]
    \item 专为\imp{老年人}设计。
    \item 包含18个项目（含人体测量、膳食问卷、全球评估、自我评估）。
    \item 筛查得分 >23.5分正常，17～23.5分有营养不良风险，<17分营养不良。
\end{itemize}
\end{keybox}

% ----- 第七节 医院膳食分类 -----
\subsection{医院膳食分类（熟悉）}

\begin{keybox}
\textbf{医院膳食三大类：}

\textbf{（一）基本膳食（Basic/Routine Diets）：}
\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{类型} & \textbf{适用人群} & \textbf{特点} \\
\midrule
普通饭（Regular） & 无膳食限制者 & 平衡膳食 \\
软饭（Soft） & 轻度消化障碍 & 细软易消化 \\
半流质（Semi-Liquid） & 发热、术后、咀嚼困难 & 少渣半液 \\
流质（Liquid） & 高热、急性感染、术后 & 液体状，营养不足需过渡 \\
\bottomrule
\end{tabular}
\end{center}

\textbf{（二）试验膳食（Test Diets）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{隐血试验膳食：}检查消化道出血前3天，禁食肉类、动物血、绿色蔬菜，食用白色食物（米饭、白面包、牛奶）。
    \item \textbf{胆道造影膳食：}检查前禁食高脂，造影后进食油炸鸡蛋等高脂餐刺激胆囊收缩，观察胆囊排空功能。
    \item \textbf{钙磷代谢试验膳食：}用于甲状旁腺功能诊断。
\end{itemize}

\textbf{（三）治疗膳食（Therapeutic Diets）：}
\begin{itemize}[leftmargin=*]
    \item 低盐膳食（<5g盐/d或<2g严格限盐）
    \item 低脂膳食（脂肪<40～50g/d）
    \item 低蛋白膳食（蛋白质<0.6g/(kg$\cdot$d)）
    \item 高蛋白膳食（蛋白质>1.2g/(kg$\cdot$d)）
    \item 糖尿病膳食（控制总能量与碳水）
    \item 肾脏病膳食（根据肾功能调整蛋白、钠、钾、磷）
    \item 低嘌呤膳食（<150mg/d，适用于痛风及高尿酸血症）
\end{itemize}
\end{keybox}

% ----- 第八节 肠内营养 -----
\subsection{肠内营养（熟悉）}

\begin{defbox}
\textbf{肠内营养（Enteral Nutrition, EN）：}通过胃肠道（口服或管饲）提供营养素，满足机体代谢需要的营养支持方式。
\end{defbox}

\subsubsection{EN的途径}

\begin{itemize}[leftmargin=*]
    \item 口服补充（Oral Nutritional Supplement, ONS）
    \item 鼻胃管（Nasogastric Tube, NGT）：短期使用（<4周）
    \item 鼻肠管（Nasojejunal Tube, NJT）：用于胃排空障碍患者
    \item 经皮内镜下胃造口（Percutaneous Endoscopic Gastrostomy, PEG）：长期使用（>4周）
    \item 经皮内镜下空肠造口（PEJ）
\end{itemize}

\subsubsection{EN的适应证与禁忌证}

\begin{keybox}
\textbf{肠内营养的适应证：}
\begin{itemize}[leftmargin=*]
    \item 不能经口进食者（昏迷、吞咽困难、口腔/食管术后）
    \item 经口进食不足者（严重创伤、烧伤、肿瘤恶病质）
    \item 胃肠道功能存在但口服不能满足需要者
\end{itemize}

\textbf{肠内营养的禁忌证：}
\begin{itemize}[leftmargin=*]
    \item 完全性肠梗阻
    \item 严重胃肠道出血
    \item 严重腹胀/腹泻/呕吐
    \item 弥漫性腹膜炎
    \item 消化道瘘无法建立有效的输注通道
\end{itemize}

\textbf{肠内营养的优点：}
\begin{itemize}[leftmargin=*]
    \item 符合生理途径：维持胃肠道结构和功能的完整性。
    \item 保护肠屏障功能：防止菌群移位和肠源性感染。
    \item 刺激消化液和消化道激素分泌。
    \item 经济、安全、操作简便，并发症较PN少。
\end{itemize}
\end{keybox}

\begin{exambox}
\textbf{EN的"金句"：}有肠道功能就用肠道！（If the gut works, use it!）这是临床营养支持的首要原则。
\end{exambox}

% ----- 第九节 肠外营养 -----
\subsection{肠外营养（熟悉）}

\begin{defbox}
\textbf{肠外营养（Parenteral Nutrition, PN）：}通过静脉途径提供机体所需的全部或部分营养素，用于无法或不宜经胃肠道摄入营养的患者。
\end{defbox}

\subsubsection{PN的途径与分类}

\begin{itemize}[leftmargin=*]
    \item \textbf{外周静脉营养（Peripheral PN）：}经外周静脉输注，渗透压较低（<900mOsm/L），适用于短期（<2周）和部分营养支持。
    \item \textbf{中心静脉营养（Central PN / TPN）：}经中心静脉置管（如PICC、颈内静脉置管）输注，可使用较高渗透压溶液，满足长期、完全营养支持需要。
    \item \textbf{完全肠外营养（Total Parenteral Nutrition, TPN）：}全部营养素由静脉提供的PN。
    \item \textbf{补充性肠外营养（Supplemental PN）：}EN不足时，PN辅助补充。
\end{itemize}

\subsubsection{PN的适应证与并发症}

\begin{keybox}
\textbf{肠外营养的适应证：}
\begin{itemize}[leftmargin=*]
    \item 消化道无功能或功能丧失（短肠综合征、严重放射性肠炎）
    \item EN禁忌或不能满足需要者
    \item 高分解代谢状态（大面积烧伤、严重创伤、多器官功能衰竭）
    \item 胃肠道需要休息时（重症胰腺炎早期、肠道术后过渡期）
\end{itemize}

\textbf{肠外营养常见并发症：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{置管相关并发症：}气胸、血胸、动脉损伤、导管感染、导管血栓。
    \item \textbf{代谢性并发症：}高血糖、低血糖、高三酰甘油血症、电解质紊乱（低钾、低磷、低镁）。
    \item \textbf{肝损害：}胆汁淤积、肝功能异常（长期TPN最主要并发症之一）。
    \item \textbf{肠屏障功能受损：}长期PN缺乏肠内营养刺激，肠黏膜萎缩，菌群移位风险增加。
    \item \textbf{再喂养综合征（Refeeding Syndrome）：}长期饥饿或营养不良后快速再喂养，导致严重低磷、低钾、低镁和水钠潴留，为PN危险并发症。
\end{enumerate}
\end{keybox}

\begin{tipbox}
\textbf{EN vs PN的选择原则：}首选EN（肠道有功能时），EN不能满足需要时联合PN，EN禁忌时用PN。肠屏障功能的维护是ENT的核心理由之一——长期PN会使肠道失去腔内营养刺激，黏膜萎缩，通透性增加，菌群移位风险上升。
\end{tipbox}

% ----- 第十节 特殊医学用途配方食品 -----
\subsection{特殊医学用途配方食品（FSMP）（了解）}

\begin{defbox}
\textbf{特殊医学用途配方食品（Food for Special Medical Purposes, FSMP）：}为了满足进食受限、消化吸收障碍、代谢紊乱或特定疾病状态人群对营养素或膳食的特殊需要，专门加工配制的配方食品。
\end{defbox}

\subsubsection{FSMP的分类}

\begin{enumerate}[leftmargin=*]
    \item \textbf{全营养配方食品：}可作为单一营养来源，满足目标人群全部营养需要。适用于能正常消化吸收各类营养素者。
    \item \textbf{特定全营养配方食品：}可作为单一营养来源，满足目标人群在特定疾病或医学状态下的营养需要。例如：糖尿病专用、肾病专用、肿瘤专用、肝病专用等。
    \item \textbf{非全营养配方食品：}只能满足部分营养需要，不能作为单一营养来源。例如蛋白质组件、脂肪组件、碳水化合物组件、维生素/矿物质组件、电解质组件。
\end{enumerate}

\subsubsection{FSMP与保健食品/药品的区别}

\begin{itemize}[leftmargin=*]
    \item FSMP必须在医生或临床营养师指导下使用，在医疗机构或药店销售。
    \item FSMP没有"蓝帽子"标志，而是标注"国食注字"（注册号格式：TY+年份+序号）。
    \item FSMP需要依据国家标准和管理办法进行注册管理。
    \item 保健食品用于调节特定机体功能，适用于普通人群；FSMP为疾病状态下的营养支持。
\end{itemize}

\subsection*{参考答案要点}

\begin{enumerate}[leftmargin=*, itemsep=5pt]
    \item 见本章各概念定义框。
    \item 控制总能量（每天减少500～1000kcal），调整宏量营养素供能比（蛋白质20\%～25\%，脂肪20\%～30\%，碳水45\%～60\%），增加膳食纤维（25～30g/d），限制添加糖和饱和脂肪，配合有氧+抗阻运动。
    \item 控制能量维持理想体重，限制SFA（<7\%～10\%能量）和反式脂肪，限制胆固醇<300mg/d，控制钠<5g盐/d，增加钾摄入（蔬果），增加膳食纤维，限制饮酒，多食鱼类。
    \item 控制总能量达标体重，碳水50\%～60\%（优先低GI），蛋白质15\%～20\%，脂肪20\%～30\%，膳食纤维25～30g/d，规律进餐少食多餐，配合运动和药物治疗。
    \item 优点：符合生理、保护肠屏障、经济安全。适应证：不能经口进食、经口不足但肠道功能存在。禁忌证：完全肠梗阻、严重胃肠道出血、弥漫性腹膜炎等。
    \item 适应证：肠道无功能或功能丧失，EN禁忌或不足。并发症：置管并发症（气胸、感染、血栓）、代谢紊乱（高/低血糖、电解质紊乱）、肝损害、肠屏障萎缩、再喂养综合征。
    \item 首选EN（肠道有功能时就用肠道），EN不足时联合PN，EN禁忌时单独PN。
    \item 分类：全营养配方、特定全营养配方、非全营养配方。区别于保健食品：FSMP用于疾病状态下营养支持，需医生指导，标注"国食注字"而非"蓝帽子"。
    \item 保持健康体重，积极运动，全谷物/蔬果/豆类为主，限制快餐/加工食品，限制红肉和加工肉类，限制含糖饮料，限制饮酒，不依赖补充剂防癌，母乳喂养。
    \item 基本膳食（普通、软、半流质、流质），试验膳食（隐血试验、胆囊造影），治疗膳食（低盐、低脂、低蛋白、高蛋白、糖尿病膳食等）。
\end{enumerate}
