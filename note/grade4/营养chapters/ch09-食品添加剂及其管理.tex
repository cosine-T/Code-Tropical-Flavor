\section{食品添加剂的定义、分类与使用要求}

\subsection{食品添加剂的定义}

\begin{keybox}
\textbf{食品添加剂（Food Additives，掌握）：}为改善食品品质和色、香、味以及为防腐、保鲜和加工工艺的需要而加入食品中的人工合成或者天然物质。
\end{keybox}

\subsection{食品添加剂的使用要求}

\begin{keybox}
\textbf{食品添加剂的使用要求（掌握）：}

\textbf{1. 基本要求：}
\begin{enumerate}[leftmargin=*]
    \item 不应对人体产生任何健康危害；
    \item 不应掩盖食品腐败变质；
    \item 不应掩盖食品本身或加工过程中的质量缺陷，或以掺杂、掺假、伪造为目的而使用食品添加剂；
    \item 不应降低食品本身的营养价值；
    \item 在达到预期效果的前提下尽可能降低在食品中的使用量。
\end{enumerate}

\textbf{2. 在下列情况下可使用食品添加剂：}
\begin{enumerate}[leftmargin=*]
    \item 保持或提高食品本身的营养价值；
    \item 作为某些特殊膳食用食品的必要配料或成分；
    \item 提高食品的质量和稳定性，改进其感官特性；
    \item 便于食品的生产、加工、包装、运输或者贮藏。
\end{enumerate}
\end{keybox}

\subsection{食品添加剂的分类}

\begin{defbox}
\textbf{食品添加剂分类（按功能，了解）：}酸度调节剂、抗氧化剂（BHA、BHT）、漂白剂、着色剂、护色剂（亚硝酸盐）、酶制剂、增味剂、防腐剂（苯甲酸、山梨酸）、甜味剂等。
\end{defbox}

\section{各类食品添加剂}

\subsection{酸度调节剂}

\begin{tipbox}
\textbf{酸度调节剂（Acidulating Agent，了解）：}用以维持或改变食品酸碱度的物质。

\begin{itemize}[leftmargin=*]
    \item 类别：柠檬酸、乳酸、酒石酸、苹果酸、碳酸钠、碳酸钾。
\end{itemize}
\end{tipbox}

\subsection{抗氧化剂}

\begin{tipbox}
\textbf{抗氧化剂（Antioxidant，了解BHA/BHT）：}能防止或延缓油脂或食品成分氧化分解、变质，提高食品稳定性的物质，可延长食品的保质期和货架期。

\textbf{分类：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{油溶性抗氧化剂：}丁基羟基茴香醚（BHA）、二丁基羟基甲苯（BHT）、没食子酸丙酯（PG）、特丁基对苯二酚（TBHQ）等。
    \item \textbf{过氧化物分解剂：}硫代二丙酸二月桂酯（DLTP）等。
    \item \textbf{水溶性及天然抗氧化剂：}抗坏血酸及植酸、维生素C、维生素E、植酸。
\end{enumerate}
\end{tipbox}

\subsection{漂白剂}

\begin{tipbox}
\textbf{漂白剂（Bleaching Agent，了解）：}能够破坏、抑制食品的发色因素，使其褪色或使食品免于褐变的物质。还原型漂白剂应注意SO$_2$残留量。

\textbf{类别：}亚硫酸钠、焦亚硫酸钾、亚硫酸氢钠、低亚硫酸钠、焦亚硫酸钠、硫磺等。

\textbf{硫磺：}既为漂白剂又是防腐剂，主要用于表面处理、干菜、干果、粉丝、蜜饯、脱水蔬菜、盐渍的蔬菜、蔬菜罐头等。
\end{tipbox}

\subsection{着色剂}

\begin{tipbox}
\textbf{着色剂（Colour / Food Color，了解）：}使食品着色和改善食品色泽的物质。

\textbf{分类：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{天然色素：}植物色素（辣椒红、姜黄素等）、动物色素（紫胶红、胭脂虫红等）、微生物色素（红曲红等）、焦糖色素（焦糖色、酱色、$\beta$-胡萝卜素）。
    \item \textbf{合成色素：}我国允许使用的合成色素有苋菜红、胭脂红、新红、柠檬黄、日落黄、靛蓝、亮蓝以及它们各自的铝色淀和叶绿素铜钠盐、二氧化钛等。
\end{enumerate}
\end{tipbox}

\subsection{护色剂}

\begin{tipbox}
\textbf{护色剂（Colour Fixative，又称发色剂，熟悉亚硝酸盐）：}指与肉及肉制品中呈色物质作用，使之在食品加工、保藏等过程中不致分解、破坏，呈现良好色泽的物质。

\textbf{类别：}亚硝酸钠（钾）、硝酸钠（钾）、硝酸铵等，加上D-异抗坏血酸及其钠盐共7种。
\end{tipbox}

\begin{tipbox}
\textbf{亚硝酸盐的发色机理：}pH 6.5--5.5时，在酸性条件下细菌分解为亚硝酸，亚硝酸再转变为NO，NO与肌红蛋白反应生成亚硝基肌红蛋白，经加热或烟熏则使蛋白质部分变性，转化为一氧化氮亚铁血色原，成为较稳定的粉红色。同时，亚硝酸盐对微生物繁殖有一定的抑制作用，特别是对肉毒梭状芽孢杆菌有特殊的抑制作用。
\end{tipbox}

\begin{exambox}
\textbf{[考点] 常见考点：}
\begin{itemize}[leftmargin=*]
    \item 护色剂=发色剂，亚硝酸盐既是护色剂又是防腐剂（抑制肉毒梭菌）
    \item 亚硝酸盐可与胺类在体内合成$N$-亚硝基化合物，有致癌风险，必须严格控制使用量和残留量（GB 2760）
\end{itemize}
\end{exambox}

\subsection{酶制剂}

\begin{tipbox}
\textbf{酶制剂（Enzyme Preparations，了解）：}由动物或植物的可食或非可食部分直接提取，或由传统或通过基因修饰的微生物（包括细菌、放线菌、真菌菌种）发酵、提取制得的，具有特殊催化功能的生物制品。主要用于加速食品加工过程和提高食品产品质量。

\textbf{常见类别：}谷氨酰胺转氨酶、木瓜蛋白酶、$\alpha$-淀粉酶。
\end{tipbox}

\subsection{增味剂}

\begin{tipbox}
\textbf{增味剂（Flavour Enhancers，了解）：}补充或增强食品原有风味的物质。增味剂可能本身没有鲜味，但却能增加食物的天然鲜味。

\textbf{常见类别：}谷氨酸钠（MSG）、核苷酸类增味剂、麦芽酚、酵母提取物。
\end{tipbox}

\subsection{防腐剂}

\begin{tipbox}
\textbf{防腐剂（Preservative，熟悉苯甲酸和山梨酸）：}防止食品腐败变质、延长食品保质期的物质。

\textbf{分类：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{酸型防腐剂：}苯甲酸、山梨酸、丙酸；
    \item \textbf{酯型防腐剂：}对羟基苯甲酸酯类（尼泊金酯）；
    \item \textbf{生物型防腐剂：}乳酸链球菌素/乳链菌肽。
\end{enumerate}
\end{tipbox}

\begin{tipbox}
\textbf{苯甲酸及其钠盐：}pH越低效果越好，最适pH 2.5--4.0。在酸性条件下以非解离分子形式存在，可穿过微生物细胞膜，干扰细胞代谢。\imp{ADI：0--5 mg/kg bw}。

\textbf{山梨酸及其钾盐：}抑制G$^-$菌效果好，pH 5--6以下使用效果佳。安全性高于苯甲酸（在体内按脂肪酸途径代谢为CO$_2$和水）。\imp{ADI：0--25 mg/kg bw}。
\end{tipbox}

\subsection{甜味剂}

\begin{tipbox}
\textbf{甜味剂（Sweeteners，了解）：}赋予食品甜味的物质。

\textbf{分类：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{天然甜味剂：}木糖醇、山梨糖醇、赤藓糖醇、麦芽糖醇、甘露糖醇、甜菊糖苷、罗汉果甜苷；
    \item \textbf{人工合成甜味剂：}糖精钠、甜蜜素（环己基氨基磺酸钠）、安赛蜜、阿斯巴甜（Aspartame，苯丙酮尿症患者禁用）、三氯蔗糖、阿力甜、纽甜等。
\end{enumerate}

\textbf{理想甜味剂应具备的特点：}安全性好；甜味纯正；稳定性好；水溶性好；价格合理。
\end{tipbox}

% [补充自往届笔记]
\begin{tipbox}
\textbf{抗氧化剂的溶解特点和来源（补充）：}
1. 溶解特点：水溶性（L-抗坏血酸及其钠盐）；脂溶性（BHA、维生素E）。
2. 来源：天然抗氧化剂和合成抗氧化剂。
3. 作用机制：自由基终止剂阻断脂质过氧化物分解。

\textbf{漂白剂的类型（补充）：}
1. 氧化型：使着色物质氧化分解而漂白，主要用于面粉的漂白。
2. 还原型：亚硫酸及其盐类，主要通过亚硫酸产生的还原作用使果蔬褪色，使用时要严格计量和使用后设法消除残留。

\textbf{着色剂的来源特点（补充）：}
1. 天然色素：一般安全，但着色力、稳定性差、色泽单调、成本高。
2. 合成色素：多为偶氮类和非偶氮类，性质稳定、着色力强、色泽鲜艳、成本低、使用方便。
\end{tipbox}

% [补充自往届笔记]
\begin{tipbox}
\textbf{防腐剂的抗菌作用机理（补充）：}
1. 防腐剂的抗微生物作用包括杀菌和抑菌。
2. 按物质性质：
   (1) 酸型防腐剂：抑制霉菌和细菌的酶系统，苯甲酸、山梨酸在酸性pH范围内防腐效果较好。
   (2) 酯型防腐剂：抑制微生物的酶系统和电子传递酶系统活性，对细菌、霉菌和酵母有广泛抑制作用，但对G$^-$杆菌作用较弱。
   (3) 生物型防腐剂（乳酸链球菌素/乳链菌肽）：对肉毒梭状芽孢杆菌、蜡样芽孢杆菌、单核细胞增生李斯特菌、金黄色葡萄球菌等有很强抑菌作用，对霉菌和酵母作用弱。
\end{tipbox}

\section{食品添加剂的卫生管理}

\begin{keybox}
\textbf{食品添加剂的卫生管理（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item 食品添加剂新品种需经安全性毒理学评价和审批；
    \item 实行生产许可制度；
    \item GB 2760《食品安全国家标准 食品添加剂使用标准》规定了食品添加剂的允许品种、使用范围、最大使用量或残留量；
    \item 食品添加剂产品应有明确标签标识；
    \item 食品经营企业应建立食品添加剂使用登记制度。
\end{enumerate}
\end{keybox}

\section{JECFA安全性分类}

\begin{defbox}
\textbf{联合国食品添加剂联合专家委员会（JECFA）安全性分类：}

\begin{enumerate}[leftmargin=*]
    \item \textbf{第一类为GRAS物质：}一般认为是安全的物质，可以按正常需要使用，无需建立ADI值。
    \item \textbf{第二类为A类：}
    \begin{enumerate}[leftmargin=*]
        \item \textbf{A1类：}已经JECFA安全性评价，毒理学资料已经清楚，已制定出正式ADI值者；
        \item \textbf{A2类：}目前毒理学资料不够完善，制定暂时ADI值者。
    \end{enumerate}
    \item \textbf{第三类为B类：}毒理学资料不足，未建立ADI值者。
    \begin{enumerate}[leftmargin=*]
        \item \textbf{B1类：}JECFA曾进行过安全性评价，因毒理学资料不足未制定ADI者；
        \item \textbf{B2类：}JECFA未进行过安全性评价者。
    \end{enumerate}
    \item \textbf{第四类为C类：}原则上禁止使用的食品添加剂。
    \begin{enumerate}[leftmargin=*]
        \item \textbf{C1类：}认为在食品中使用不安全的；
        \item \textbf{C2类：}只限于在某些食品中作特殊用途使用。
    \end{enumerate}
\end{enumerate}
\end{defbox}

\begin{tipbox}
\textbf{GRAS（General Recognized As Safe）：}一般公认安全物质。
\end{tipbox}

\subsection*{一、名词解释}

\begin{enumerate}[leftmargin=*, itemsep=8pt]

    \item \textbf{【熟悉】}简述防腐剂（苯甲酸、山梨酸）的特点和适用范围。
    \item \textbf{【了解】}简述食品添加剂按功能的分类及各类常见品种。
    \item 简述GRAS的含义。
\end{enumerate}

\subsection*{参考答案要点}

\subsubsection*{名词解释（要点）}

\begin{enumerate}[leftmargin=*, itemsep=4pt]
    \item \textbf{食品添加剂：}为改善食品品质和色、香、味以及为防腐、保鲜和加工工艺的需要而加入食品中的人工合成或者天然物质。
    \item \textbf{GRAS：}一般公认安全物质（General Recognized As Safe），可以按正常需要使用，无需建立ADI值。
    \item \textbf{防腐剂：}苯甲酸及其钠盐（最适pH 2.5--4.0）、山梨酸及其钾盐（pH 5--6以下）。\textbf{抗氧化剂：}BHA、BHT、没食子酸丙酯、TBHQ。\textbf{护色剂：}亚硝酸盐，与肌红蛋白结合生成亚硝基肌红蛋白，同时抑制肉毒梭菌。\textbf{漂白剂：}亚硫酸钠、焦亚硫酸钾、SO$_2$。
\end{enumerate}

\subsubsection*{简答题（要点）}

\begin{enumerate}[leftmargin=*, itemsep=6pt]
    \item \textbf{使用原则：}基本要求（五项）：不应对人体产生任何健康危害；不应掩盖食品腐败变质；不应掩盖食品本身或加工过程中的质量缺陷，或以掺杂、掺假、伪造为目的；不应降低食品本身的营养价值；在达到预期效果的前提下尽可能降低使用量。可使用情况（四项）：保持或提高食品本身的营养价值；作为某些特殊膳食用食品的必要配料或成分；提高食品的质量和稳定性，改进其感官特性；便于食品的生产、加工、包装、运输或者贮藏。

    \item \textbf{卫生管理：}新品种需经安全性毒理学评价和审批；实行生产许可制度；GB 2760规定允许品种、使用范围、最大使用量或残留量；产品应有明确标签标识；经营企业应建立使用登记制度。

    \item \textbf{JECFA分类：}第一类GRAS物质（一般认为是安全的，无需ADI）；第二类A类（A1：已制定正式ADI；A2：制定暂时ADI）；第三类B类（B1：已评价但毒理学资料不足，未制定ADI；B2：未进行过安全性评价）；第四类C类（C1：认为在食品中使用不安全，原则上禁止使用；C2：只限于在某些食品中作特殊用途使用）。

    \item \textbf{护色剂（亚硝酸盐）：}作用机理：pH 6.5--5.5时，在酸性条件下细菌分解为亚硝酸，转变为NO，NO与肌红蛋白反应生成亚硝基肌红蛋白，加热后转化为一氧化氮亚铁血色原，呈粉红色。安全性：可与胺类在体内合成$N$-亚硝基化合物，有致癌风险，必须严格控制使用量和残留量（GB 2760）。同时有抑制肉毒梭状芽孢杆菌的作用。

    \item \textbf{防腐剂：}苯甲酸及其钠盐：最适pH 2.5--4.0，ADI 0--5 mg/kg bw；山梨酸及其钾盐：pH 5--6以下效果好，抑制G$^-$菌，安全性高于苯甲酸（体内按脂肪酸途径代谢为CO$_2$和水），ADI 0--25 mg/kg bw。

    \item \textbf{分类（按功能）：}酸度调节剂（柠檬酸、乳酸等）、抗氧化剂（BHA、BHT、PG、TBHQ等）、漂白剂（亚硫酸钠、焦亚硫酸钾等）、着色剂（天然色素和合成色素）、护色剂（亚硝酸盐等）、酶制剂（谷氨酰胺转氨酶、$\alpha$-淀粉酶等）、增味剂（谷氨酸钠等）、防腐剂（苯甲酸、山梨酸等）、甜味剂（天然和人工合成）。

    \item \textbf{GRAS：}一般公认安全物质（General Recognized As Safe）。
\end{enumerate}
