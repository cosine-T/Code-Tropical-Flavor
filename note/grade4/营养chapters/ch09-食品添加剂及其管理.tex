\section{食品添加剂的定义与使用原则}

\subsection{定义}

\begin{keybox}
\textbf{食品添加剂（Food Additives）：}为改善食品品质和色、香、味以及为防腐、保鲜和加工工艺的需要而加入食品中的人工合成或者天然物质。
\end{keybox}

\subsection{使用原则}

\begin{keybox}
\textbf{食品添加剂使用原则（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item 不应对人体产生任何健康危害；
    \item 不应掩盖食品腐败变质；
    \item 不应掩盖食品本身或加工过程中的质量缺陷，或以掺杂、掺假、伪造为目的而使用；
    \item 不应降低食品本身的营养价值；
    \item 在达到预期效果的前提下尽可能降低在食品中的使用量。
\end{enumerate}
\end{keybox}

\section{食品添加剂的分类}

\subsection{按功能分类}

\begin{defbox}
\textbf{食品添加剂分类（按功能）：}酸度调节剂、抗氧化剂、漂白剂、着色剂、护色剂、防腐剂、甜味剂等。
\end{defbox}

\subsection{常见品种}

\begin{defbox}
\textbf{防腐剂（Preservative）：}苯甲酸及其钠盐（抑菌，pH越低效果越好）、山梨酸及其钾盐（抑制G$^-$菌效果好）。

\textbf{抗氧化剂（Antioxidant）：}BHA、BHT、没食子酸丙酯。

\textbf{护色剂（Colour Fixative）：}亚硝酸盐，与肉中肌红蛋白结合生成亚硝基肌红蛋白，使肉呈鲜红色。pH 6.5--5.5时，NO与肌红蛋白结合。

\textbf{漂白剂（Bleaching Agent）：}SO$_2$。

\textbf{甜味剂（Sweeteners）：}天然（甜菊糖苷）和人工合成（阿斯巴甜、糖精钠）。

\textbf{酸度调节剂（Acidulating Agent）：}调节H$^+$和OH$^-$浓度，控制pH值。
\end{defbox}

\section{JECFA安全性分类}

\begin{defbox}
\textbf{JECFA安全性分类：}

\begin{enumerate}[leftmargin=*]
    \item \textbf{A类：}
    \begin{itemize}[leftmargin=*]
        \item \textbf{A1类：}JECFA已制定ADI（每日允许摄入量）
        \item \textbf{A2类：}暂定ADI
    \end{itemize}
    \item \textbf{B类：}
    \begin{itemize}[leftmargin=*]
        \item \textbf{B1类：}JECFA已评价但未制定ADI
        \item \textbf{B2类：}暂定ADI待补充材料
    \end{itemize}
    \item \textbf{C类：}
    \begin{itemize}[leftmargin=*]
        \item \textbf{C1类：}安全性不足，禁止使用
        \item \textbf{C2类：}严格限制使用范围
    \end{itemize}
\end{enumerate}
\end{defbox}

\begin{defbox}
\textbf{GRAS（General Recognized As Safe）：}一般公认安全物质。
\end{defbox}
