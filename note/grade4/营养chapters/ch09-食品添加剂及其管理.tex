\section{食品添加剂的定义与使用原则}

\subsection{定义}

\begin{keybox}
\textbf{食品添加剂（Food Additives）：}为改善食品品质和色、香、味以及为防腐、保鲜和加工工艺的需要而加入食品中的人工合成或者天然物质。
\end{keybox}

\subsection{使用原则}

\begin{keybox}
\textbf{食品添加剂使用原则（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item 不应对人体产生任何健康危害；
    \item 不应掩盖食品腐败变质；
    \item 不应掩盖食品本身或加工过程中的质量缺陷，或以掺杂、掺假、伪造为目的而使用；
    \item 不应降低食品本身的营养价值；
    \item 在达到预期效果的前提下尽可能降低在食品中的使用量。
\end{enumerate}
\end{keybox}

\section{食品添加剂的分类}

\subsection{按功能分类}

\begin{defbox}
\textbf{食品添加剂分类（按功能）：}酸度调节剂、抗氧化剂、漂白剂、着色剂、护色剂、防腐剂、甜味剂等。
\end{defbox}

\subsection{各类食品添加剂详解}

\begin{defbox}
\textbf{1. 酸度调节剂（Acidity Regulator，了解）：}
\begin{itemize}[leftmargin=*]
    \item 功能：调节食品中H$^+$和OH$^-$浓度，控制食品pH值。可改善食品风味、抑制微生物生长、提高防腐效果。
    \item 常见品种：柠檬酸、乳酸、酒石酸、苹果酸、醋酸、磷酸、碳酸钠、碳酸氢钠等。
\end{itemize}

\textbf{2. 抗氧化剂（Antioxidant，了解BHA/BHT）：}
\begin{itemize}[leftmargin=*]
    \item 功能：阻止或延缓食品氧化，提高食品稳定性和延长保质期。
    \item \textbf{BHA（丁基羟基茴香醚）和BHT（二丁基羟基甲苯）：}合成抗氧化剂，对动物油脂和含油脂食品抗氧化效果良好。热稳定性好，适用于油炸和焙烤食品。ADI：BHA为0--0.5 mg/kg bw；BHT为0--0.3 mg/kg bw。BHA对大小鼠前胃有致癌作用（但该部位在人类不存在），IARC Group 2B。
    \item 其他常见品种：没食子酸丙酯（PG）、特丁基对苯二酚（TBHQ）、茶多酚（天然）、维生素E（生育酚）。
\end{itemize}

\textbf{3. 漂白剂（Bleaching Agent，了解）：}
\begin{itemize}[leftmargin=*]
    \item 功能：破坏、抑制食品的发色因素，使食品褪色或免于褐变。
    \item 常见品种：硫磺、二氧化硫（SO$_2$）、亚硫酸钠、焦亚硫酸钠/钾。
    \item 主要应用于：蜜饯、干果、干菜、粉丝、食糖等。
    \item 注意：SO$_2$可破坏维生素B$_1$，亚硫酸盐可诱发哮喘等过敏反应。漂白剂不得用于掩盖食品腐败变质。
\end{itemize}

\textbf{4. 着色剂（Colour / Food Color，了解）：}
\begin{itemize}[leftmargin=*]
    \item 功能：使食品着色，改善食品色泽。
    \item \textbf{天然色素：}红曲米（红曲红）、辣椒红、姜黄素、甜菜红、$\beta$-胡萝卜素、焦糖色等。安全性较高但稳定性较差。
    \item \textbf{合成色素：}胭脂红、柠檬黄、日落黄、亮蓝等。色泽鲜艳、稳定性好、成本低，但安全性受到持续关注。我国GB 2760严格限制使用范围和用量。
\end{itemize}

\textbf{5. 护色剂（Colour Fixative，熟悉亚硝酸盐）：}
\begin{itemize}[leftmargin=*]
    \item 功能：与肉及肉制品中肌红蛋白结合生成亚硝基肌红蛋白，使肉呈鲜红色。同时具有抑制肉毒梭菌的作用。
    \item \textbf{亚硝酸盐（硝酸盐）：}pH 6.5--5.5时，亚硝酸盐还原为NO，与肌红蛋白结合生成亚硝基肌红蛋白。但亚硝酸盐可与胺类在体内合成$N$-亚硝基化合物，有致癌风险。必须严格控制使用量和残留量（GB 2760）。
\end{itemize}

\textbf{6. 酶制剂（Enzyme Preparation，了解）：}
\begin{itemize}[leftmargin=*]
    \item 来源：从动植物可食部分或微生物发酵提取。常见品种：$\alpha$-淀粉酶、糖化酶、果胶酶、蛋白酶、葡萄糖氧化酶、凝乳酶。
    \item 应用于：啤酒酿造、果汁澄清、面包改良、干酪制造等。
\end{itemize}

\textbf{7. 增味剂（Flavor Enhancer，了解）：}
\begin{itemize}[leftmargin=*]
    \item 功能：补充或增强食品原有风味。
    \item 常见品种：味精（谷氨酸钠，MSG）、5$'$-鸟苷酸二钠、5$'$-肌苷酸二钠。谷氨酸钠与核苷酸类增味剂混合使用有协同增效作用。
\end{itemize}

\textbf{8. 防腐剂（Preservative，熟悉苯甲酸/山梨酸）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{苯甲酸及其钠盐：}pH越低效果越好，最适pH 2.5--4.0。在酸性条件下以非解离分子形式存在，可穿过微生物细胞膜，干扰细胞代谢。ADI：0--5 mg/kg bw。
    \item \textbf{山梨酸及其钾盐：}抑制G$^-$菌效果好，pH 5--6以下使用效果佳。安全性高于苯甲酸（在体内按脂肪酸途径代谢为CO$_2$和水）。ADI：0--25 mg/kg bw。
    \item 其他常见品种：丙酸及其盐（防霉，面包糕点专用）、对羟基苯甲酸酯类（尼泊金酯）。
\end{itemize}

\textbf{9. 甜味剂（Sweeteners，了解）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{天然甜味剂：}甜菊糖苷（甜菊叶提取物，甜度约为蔗糖的200--300倍）、甘草酸铵、罗汉果甜苷。
    \item \textbf{人工合成甜味剂：}阿斯巴甜（Aspartame，苯丙酮尿症患者禁用）、糖精钠（Saccharin）、甜蜜素、安赛蜜（Acesulfame-K）、三氯蔗糖（Sucralose）。
    \item 常用于：饮料、糖果、糕点、蜜饯等，适合糖尿病患者和减肥人群。
\end{itemize}
\end{defbox}

\section{食品添加剂的使用要求与卫生管理}

\begin{keybox}
\textbf{食品添加剂的使用要求和卫生管理（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{食品添加剂使用的基本要求（GB 2760）：}
    \begin{enumerate}[leftmargin=*]
        \item 不应对人体产生任何健康危害
        \item 不应掩盖食品腐败变质
        \item 不应掩盖食品本身或加工过程中的质量缺陷，或以掺杂、掺假、伪造为目的
        \item 不应降低食品本身的营养价值
        \item 在达到预期效果的前提下尽可能降低使用量
        \item 食品工业用加工助剂一般应在制成最后成品之前除去
    \end{enumerate}
    \item \textbf{卫生管理：}
    \begin{enumerate}[leftmargin=*]
        \item 食品添加剂新品种需经安全性毒理学评价和审批
        \item 实行生产许可制度
        \item GB 2760《食品安全国家标准 食品添加剂使用标准》规定了食品添加剂的允许品种、使用范围、最大使用量或残留量
        \item 食品添加剂产品应有明确标签标识
        \item 食品经营企业应建立食品添加剂使用登记制度
    \end{enumerate}
\end{enumerate}
\end{keybox}

\section{JECFA安全性分类}

\begin{defbox}
\textbf{JECFA安全性分类：}

\begin{enumerate}[leftmargin=*]
    \item \textbf{A类：}
    \begin{itemize}[leftmargin=*]
        \item \textbf{A1类：}JECFA已制定ADI（每日允许摄入量）
        \item \textbf{A2类：}暂定ADI
    \end{itemize}
    \item \textbf{B类：}
    \begin{itemize}[leftmargin=*]
        \item \textbf{B1类：}JECFA已评价但未制定ADI
        \item \textbf{B2类：}暂定ADI待补充材料
    \end{itemize}
    \item \textbf{C类：}
    \begin{itemize}[leftmargin=*]
        \item \textbf{C1类：}安全性不足，禁止使用
        \item \textbf{C2类：}严格限制使用范围
    \end{itemize}
\end{enumerate}
\end{defbox}

\begin{tipbox}
\textbf{GRAS（General Recognized As Safe）：}一般公认安全物质。
\end{tipbox}
