\section{粮豆类的卫生及管理}

\begin{keybox}
\textbf{粮豆类的主要卫生问题（掌握）：}
\end{keybox}

\subsection{真菌及毒素污染}

\begin{itemize}[leftmargin=*]
    \item \textbf{黄曲霉毒素污染：}玉米、花生、大米等常规谷物中均可发生，为我国粮豆类最关注的霉菌毒素问题。
    \item \textbf{黄变米（Yellow Rice / Yellowed Rice）：}受霉菌（岛青霉、橘青霉、黄绿青霉等）污染而呈黄褐色的变质米粒。含有毒性代谢产物，如岛青霉素（肝毒素）、橘青霉素（肾毒素）和黄绿青霉素（神经毒素）。
    \item \textbf{赤霉病变：}禾谷镰刀菌等污染小麦、大麦，产生DON（呕吐毒素）和ZEN（玉米赤霉烯酮）。
\end{itemize}

\begin{tipbox}
\textbf{历史事件回顾：}
\begin{itemize}[leftmargin=*]
    \item \textbf{日本黄变米事件（1952年）：}日本进口大米被岛青霉严重污染，致大量居民中毒——黄变米毒素中毒的经典案例。
    \item \textbf{英国"十万火鸡事件"（1960年）：}英国某农场10万只火鸡在数月内相继死亡，原因追溯为饲料中被黄曲霉污染的花生粕。这一事件直接导致了黄曲霉毒素的发现和研究！
\end{itemize}
\end{tipbox}

\subsection{农药残留}

\begin{itemize}[leftmargin=*]
    \item 粮豆作物生长期直接施药 $\rightarrow$ 谷物中残留
    \item 储藏期熏蒸杀虫剂（如磷化铝）残留
    \item 历史上有机氯农药（六六六、DDT）在土壤中残留期极长，可被作物吸收
\end{itemize}

\subsection{有害化学物质污染}

\begin{itemize}[leftmargin=*]
    \item \textbf{有害金属：}工业废水灌溉农田 $\rightarrow$ 土壤重金属积累 $\rightarrow$ 水稻等作物吸收富集。\textbf{"镉大米"——水稻富集土壤中镉的典型问题。}
    \item \textbf{其他：}二噁英、多环芳烃等在环境中的沉降物对农作物的污染。
    \item 粮豆加工过程中可能引入的化学物（如溶剂残留于浸出法生产食用油）。
\end{itemize}

\subsection{仓库害虫}

\begin{itemize}[leftmargin=*]
    \item 我国发现的仓库害虫有\textbf{50多种}，包括谷象、米象、谷蠹（甲虫类）、粉螨（螨类）、麦蛾（蛾类）。
    \item 仓库害虫的滋生条件：温度18–21$^\circ$C、相对湿度$>$65\%时最容易孵化繁殖。温度$<$10$^\circ$C则活动受抑制。
    \item 危害：降低粮食营养价值（消耗淀粉、蛋白质），且有虫尸、虫粪、体毛等残留物污染粮食。
\end{itemize}

\subsection{粮食的卫生管理}

\begin{itemize}[leftmargin=*]
    \item 控制水分至安全水分（谷类 $\leq$ 13\%–14\%，豆类 $\leq$ 8\%–9\%）
    \item 低温、通风储藏
    \item 定期检测霉菌毒素、农药残留和重金属含量
    \item 严格控制熏蒸剂的使用和残留
\end{itemize}

\section{蔬菜水果的卫生及管理}

\subsection{主要卫生问题}

\begin{enumerate}[leftmargin=*]
    \item \textbf{微生物污染：}细菌（沙门菌、大肠杆菌等来自粪肥）、霉菌及霉菌毒素（展青霉素等污染水果）。
    \item \textbf{寄生虫及虫卵：}使用未经无害化处理的人畜粪便施肥 $\rightarrow$ 蛔虫卵、钩虫卵等污染蔬菜。
    \item \textbf{有害化学物质：}
    \begin{itemize}[leftmargin=*]
        \item 农药残留
        \item 重金属（工业废水灌溉）
        \item 多环芳烃（大气沉降）
        \item \textbf{硝酸盐 $\rightarrow$ 亚硝酸盐转化：}蔬菜（特别是叶菜类）富含硝酸盐。在不当储存或腌制过程中，硝酸盐可被细菌还原为亚硝酸盐，引起亚硝酸盐中毒（高铁血红蛋白血症）。腌制蔬菜在腌制过程中存在"亚硝峰"现象——亚硝酸盐含量先升后降，约在第7–20天达到峰值。
    \end{itemize}
\end{enumerate}

\subsection{卫生管理}

\begin{itemize}[leftmargin=*]
    \item 食用前充分清洗
    \item 低温、适当湿度储藏
    \item 防止机械性损伤（损伤组织加速腐败和亚硝酸盐生成）
    \item 腌制蔬菜需腌制充分（超过20–30天，亚硝酸盐含量降至安全水平）
    \item 禁止使用未经无害化处理的人畜粪便施肥
\end{itemize}

\section{畜禽肉类的卫生及管理}

\subsection{腐败变质}

\begin{keybox}
\textbf{畜禽肉类腐败变质* —— 最重要、最常见的卫生问题！}

\textbf{原因：}畜禽屠宰后，肌肉经僵直、后熟（自溶作用）阶段后进入自溶后期，组织酶和细菌的共同作用使蛋白质分解、脂肪氧化，肉品失去食用价值。

\textbf{腐败过程：}宰后初期细菌极少（健康动物肌肉内部无菌） $\rightarrow$ 随着存储时间延长，表面细菌向深部侵入 $\rightarrow$ 蛋白质分解 $\rightarrow$ 产生H$_2$S、胺类、吲哚等恶臭物质 $\rightarrow$ 肉色变暗绿色（硫血红蛋白形成）。

\textbf{鉴定指标：}感官（颜色、气味、弹性和黏度）$+$ TVBN（鲜（冻）畜肉 $\leq$ 15 mg/100g）。
\end{keybox}

\subsection{人畜共患传染病}

\begin{keybox}
\textbf{重要人畜共患传染病（重点掌握）：}
\end{keybox}

\subsubsection{炭疽（Anthrax）}

\begin{keybox}
\textbf{炭疽——危害最严重的暴发性人畜共患传染病！}
\begin{itemize}[leftmargin=*]
    \item \textbf{病原体：}炭疽杆菌（Bacillus anthracis），革兰阳性芽孢杆菌。
    \item \textbf{传播途径：}接触病畜或其产品（肉、皮、毛）经皮肤、呼吸道或消化道感染。
    \item \textbf{特征：}在不利环境下形成芽孢——\textbf{芽孢抵抗力极强！}在土壤中可存活数十年！在未形成芽孢前，55–58$^\circ$C 10–15分钟可杀死。
    \item \textbf{处理原则：}一旦发现病畜，\textbf{必须在6小时内}就地隔离消毒、高温化制或深埋（深2米并撒漂白粉）。严禁屠宰、销售、食用病畜肉！
\end{itemize}
\end{keybox}

\subsubsection{口蹄疫（Foot and Mouth Disease, FMD）}

\begin{itemize}[leftmargin=*]
    \item \textbf{病原体：}口蹄疫病毒（FMDV），为RNA病毒。
    \item \textbf{易感动物：}偶蹄类动物（牛、羊、猪等）。马不发病，但可被动携带传播。
    \item \textbf{食品卫生处理：}80–100$^\circ$C加热可杀死病毒。病畜应立即宰杀、高温处理或销毁。同群牲畜亦应全部宰杀。
\end{itemize}

\subsubsection{疯牛病（Bovine Spongiform Encephalopathy, BSE）}

\begin{keybox}
\textbf{疯牛病（BSE）* —— 致病性朊病毒蛋白疾病（教学难点*）：}

\begin{itemize}[leftmargin=*]
    \item \textbf{病原体：}\textbf{致病性朊病毒蛋白（Prion Protein, PrP$^{Sc}$）}——非细菌、非病毒，是一种具有传染性的异常折叠蛋白质！
    \item \textbf{致病机制：}正常朊蛋白（PrP$^C$）在PrP$^{Sc}$的作用下发生构象改变（\(\alpha\)-螺旋 $\rightarrow$ \(\beta\)-折叠），变为抗蛋白酶K的异常蛋白 $\rightarrow$ 在中枢神经系统沉积 $\rightarrow$ 神经细胞空泡化、死亡 $\rightarrow$ 海绵状脑病。
    \item \textbf{人体疾病：}食用含BSE病原体的牛肉或制品 $\rightarrow$ 人感染\textbf{新型变异克雅氏病（nvCJD, new variant Creutzfeldt-Jakob Disease）}。
    \item \textbf{传播途径：}牛肉及其制品（特别是含神经组织的器官和骨髓部位风险最高）。
    \item \textbf{预防关键：}严格禁止将含有神经组织的原料（脑、脊髓等SRM——特定风险物质）进入人类食物链！
\end{itemize}
\end{keybox}

\begin{exambox}
\textbf{疯牛病与其他传染病的本质区别：}
\begin{itemize}[leftmargin=*]
    \item 常规传染病（细菌/病毒/寄生虫）致病因子含核酸，可被常规消毒灭活。
    \item 疯牛病致病因子为蛋白质（不含核酸），\textbf{对常规消毒手段（紫外线、煮沸、甲醛）具有极强的抵抗力}，需134$^\circ$C高压蒸汽消毒1小时或1M NaOH处理才能有效灭活。
    \item 潜伏期极长（可达数年至数十年），目前无有效治疗方法，100\%致死。
\end{itemize}
\end{exambox}

\subsubsection{结核病、布鲁氏菌病}

\begin{itemize}[leftmargin=*]
    \item \textbf{结核病（Tuberculosis）：}牛型结核杆菌（Mycobacterium bovis）可通过病牛乳和肉传播给人。病畜肉尸应高温处理或销毁。
    \item \textbf{布鲁氏菌病（Brucellosis）：}布鲁氏菌（Brucella spp.），主要经牛、羊、猪传播。可引起人的波状热（undulant fever）。对内蒙、西北地区牧区尤为重要。肉尸经高温处理；乳经巴氏消毒。
\end{itemize}

\subsection{寄生虫病}

\begin{enumerate}[leftmargin=*]
    \item \textbf{囊尾蚴病（Cysticercosis）——"米猪肉"：}
    \begin{itemize}[leftmargin=*]
        \item 病原体：猪肉绦虫的幼虫（囊尾蚴）。
        \item 人畜互传：猪为中间宿主摄入虫卵；人为终宿主食用含活囊尾蚴的猪肉。
        \item 最危险的情况：人误食虫卵后囊尾蚴可在人体内发育——称为自体感染 $\rightarrow$ 可侵犯脑、眼、皮下组织，引起脑囊虫病。
        \item 肉品检验：切开咬肌、腰肌等处检查（40 cm$^2$面积内囊尾蚴数）。
        \item 处理：高温处理或冷冻（$-12^\circ$C $>$ 10天可杀死囊尾蚴）。
    \end{itemize}
    \item \textbf{旋毛虫病（Trichinosis）：}
    \begin{itemize}[leftmargin=*]
        \item 病原体：旋毛虫（Trichinella spiralis）。
        \item 感染途径：食用含活旋毛虫包囊的生或未煮熟的猪肉。
        \item 症状：发热、肌痛、眼睑水肿等。
        \item 预防：肉品充分加热（中心温度达71$^\circ$C以上）。
    \end{itemize}
\end{enumerate}

\subsection{兽药残留}

\begin{itemize}[leftmargin=*]
    \item \textbf{盐酸克伦特罗（Clenbuterol）/ 瘦肉精：}$\beta$-肾上腺素受体激动剂，非法添加于饲料中促进瘦肉生长。人体残留可引起心悸、头痛、肌肉震颤、心律失常等。中国严禁使用！
    \item 其他常见残留：抗生素、磺胺类、喹诺酮类等。
\end{itemize}

\section{水产品的卫生及管理}

\subsection{主要卫生问题}

\begin{enumerate}[leftmargin=*]
    \item \textbf{腐败变质（水产品特别容易腐败！）：}
    \begin{itemize}[leftmargin=*]
        \item 鱼肉含水量高（70\%–80\%）、组织脆弱、表面黏液多，细菌极易繁殖。
        \item 鱼死后 $\rightarrow$ 体表黏液中的细菌迅速侵入肌肉 $\rightarrow$ 快速腐败。
        \item \textbf{鲜度指标：}K值（$\leq20\%$新鲜，$>40\%$腐败）。鱼类的K值与含氮物质总量之比是判断鱼类早期鲜度的最灵敏指标。
    \end{itemize}

    \item \textbf{有害金属的富集（生物浓缩）：}
    \begin{itemize}[leftmargin=*]
        \item 鱼贝类对水体中有害金属（Hg、Cd、Pb）具极强的富集能力。
        \item 甲基汞通过食物链在大型捕食性鱼类（金枪鱼、鲨鱼、剑鱼）体内高度富集，浓度可比水体高数万至数十万倍。
        \item 贝类富集镉的能力特别突出（扇贝、牡蛎等）。
    \end{itemize}

    \item \textbf{病原微生物和寄生虫：}
    \begin{itemize}[leftmargin=*]
        \item \textbf{副溶血性弧菌（Vibrio parahaemolyticus）：}海水产品中最常见的食源性致病菌。嗜盐性，3\%–6\% NaCl最适宜生长。引起急性胃肠炎（腹痛、腹泻、呕吐）。
        \item \textbf{华支睾吸虫（Clonorchis sinensis，肝吸虫）：}淡水鱼/虾生食或未煮熟可感染。成虫寄生于人体肝内胆管 $\rightarrow$ 胆管炎、胆囊炎、肝硬化，甚至胆管癌。广东"生食鱼片"文化使其成为高发区。
        \item 沙门菌也可通过污染的水体感染水产品。
    \end{itemize}

    \item \textbf{天然有毒水产品：}
    \begin{itemize}[leftmargin=*]
        \item \textbf{河豚（Pufferfish/Fugu）：}卵巢、肝脏含河豚毒素（Tetrodotoxin, TTX）——强烈神经毒素，阻断钠离子通道，LD$_{50}$约10 $\mu$g/kg。无特效解毒药，死亡率高！
        \item \textbf{有毒贝类：}某些贝类摄食产毒藻类后富集麻痹性贝毒（Paralytic Shellfish Poisoning, PSP）和腹泻性贝毒（DSP）。
    \end{itemize}
\end{enumerate}

\subsection{卫生管理}

\begin{itemize}[leftmargin=*]
    \item 保持冷链：鱼贝类捕获后迅速降温至0$^\circ$C以下
    \item 充分加热：水产品务必煮熟食用（特别是淡水产品）
    \item 严格检验制度：上市水产品检测鲜度指标（K值、TVBN）、有害金属和致病菌
    \item 禁售有毒品种：河豚销售受到严格管制
\end{itemize}

\section{乳及乳制品的卫生及管理}

\subsection{主要卫生问题}

\begin{enumerate}[leftmargin=*]
    \item \textbf{微生物污染：}
    \begin{itemize}[leftmargin=*]
        \item 奶牛沙门菌病——病原体可随乳汁排出。
        \item 挤乳后的粪源性污染——挤乳时牛体、乳房、挤奶员手部和设备的粪便污染。
        \item 金黄色葡萄球菌污染——引起牛乳腺炎（乳房炎）的菌株可产生肠毒素，巴氏消毒可杀灭细菌但毒素耐热不易破坏。
        \item 结核杆菌和布鲁氏菌——可经病牛乳传给人类。
    \end{itemize}

    \item \textbf{化学性污染：}
    \begin{itemize}[leftmargin=*]
        \item 饲料中农药残留——有机氯农药可通过饲料 $\rightarrow$ 牛 $\rightarrow$ 牛乳
        \item 饲料中兽药残留——抗生素（青霉素、四环素等）随乳排出
        \item \textbf{黄曲霉毒素M1——}饲料中AF B1在牛体内经肝脏代谢 $\rightarrow$ AF M1分泌入乳。乳中AF M1是乳品安全的重要指标。
    \end{itemize}

    \item \textbf{掺伪（Adulteration）：}
    \begin{itemize}[leftmargin=*]
        \item 掺水（降低固形物含量）
        \item 掺碱性物质（中和酸度以掩盖变质）
        \item 掺淀粉、米汤、豆浆等增稠
        \item 2008年三聚氰胺事件——在原料乳中非法添加三聚氰胺以提高蛋白质检测值（凯氏定氮法的漏洞），造成大量婴幼儿肾结石。此事件严重打击了中国乳业信誉并直接推动了《食品安全法》的修订。
    \end{itemize}
\end{enumerate}

\subsection{乳的消毒}

\begin{enumerate}[leftmargin=*]
    \item \textbf{巴氏消毒法：}
    \begin{itemize}[leftmargin=*]
        \item LTLT（低温长时间）：62.8$^\circ$C、30 min
        \item HTST（高温短时间）：71.7$^\circ$C、15 s
        \item 保质期短，需冷链配送（2–6$^\circ$C）
    \end{itemize}
    \item \textbf{超高温瞬时灭菌（UHT）：}
    \begin{itemize}[leftmargin=*]
        \item 135–150$^\circ$C、2 s
        \item 达到商业无菌，常温保存数月
    \end{itemize}
\end{enumerate}

\subsection{病畜乳处理原则}

\begin{itemize}[leftmargin=*]
    \item 炭疽、狂犬病等烈性传染病病畜的乳——严禁食用，现场消毒销毁
    \item 结核病牛乳——巴氏消毒后方可食用（严重病例则销毁）
    \item 布鲁氏菌病牛乳——巴氏消毒或煮沸5分钟
    \item 乳房炎病牛乳——不得供人食用
\end{itemize}

\section{蛋与蛋制品的卫生}

\begin{itemize}[leftmargin=*]
    \item \textbf{微生物污染：}细菌可通过蛋壳气孔侵入蛋内容物——沙门菌是禽蛋最主要的食源性致病菌（鸭蛋、鹅蛋沙门菌带菌率30\%–40\%）。
    \item \textbf{化学残留：}饲料中农药（有机氯等）、抗生素、重金属（Cd、Pb）、二噁英等可通过母禽富集于蛋中。
    \item \textbf{鲜蛋变质：}储存时间过长 $\rightarrow$ 蛋黄移位/散黄 $\rightarrow$ 粘壳蛋 $\rightarrow$ 黑腐蛋。
    \item \textbf{卫生管理：}低温保藏（0–1$^\circ$C），避免与有异味的物质同储。禁止用工业级色素制作"红心蛋"。
\end{itemize}

\section{食用油脂的卫生及管理}

\subsection{油脂酸败}

\begin{keybox}
\textbf{油脂酸败（Oil/Fat Rancidity）——掌握*：}
食用油脂在储藏过程中由于光、热、氧、水分和金属离子等的作用，发生一系列化学变化，使其感官性状变劣和产生不良气味的现象。

\textbf{原因：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{生物学原因（水解）：}来自动植物组织残渣和微生物的脂肪酶 $\rightarrow$ 甘油三酯水解为甘油和游离脂肪酸 $\rightarrow$ 酸价（AV）升高。
    \item \textbf{化学原因（自动氧化——最主要原因）：}不饱和脂肪酸（油酸$\rightarrow$亚油酸$\rightarrow$亚麻酸，双键越多越易氧化）在O$_2$作用下经自由基链式反应：
    \begin{itemize}[leftmargin=*]
        \item 引发期（Initiation）：形成自由基 R$\cdot$
        \item 传播期（Propagation）：R$\cdot$ + O$_2$ $\rightarrow$ ROO$\cdot$（过氧自由基）；ROO$\cdot$ + RH $\rightarrow$ ROOH + R$\cdot$（链式反应）
        \item 终止期（Termination）：自由基相互结合
    \end{itemize}
\end{enumerate}

\textbf{特征：}产生"哈喇味"（rancid odor），过氧化值（POV）升高是脂肪酸败的最早期指标！
\end{keybox}

\subsection{油脂酸败的理化指标}

\begin{keybox}
\textbf{食用油脂的理化指标（掌握*——教学难点*）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{酸价（Acid Value, AV）*：}
    \begin{itemize}[leftmargin=*]
        \item 定义：中和1 g油脂中游离脂肪酸所需KOH的毫克数（mg KOH/g）。
        \item 卫生学意义：反映油脂水解程度。AV越高，说明油脂水解越严重，游离脂肪酸越多。
        \item 限值：食用植物油 AV $\leq$ 3 mg KOH/g（GB 2716）。
    \end{itemize}

    \item \textbf{过氧化值（Peroxide Value, POV）*：}
    \begin{itemize}[leftmargin=*]
        \item 定义：1 kg油脂中过氧化物（主要是氢过氧化物ROOH）的毫摩尔数（mmol/kg），通常使用其相当量的碘含量（meq/kg）或毫摩尔数表示。
        \item 卫生学意义：\textbf{反映油脂氧化初期产生的氢过氧化物含量——是脂肪酸败最早期、最灵敏的指标！}
        \item 注意：POV在氧化后期反而可能下降！因为氢过氧化物进一步分解为醛、酮、酸等次级氧化产物。因此判断脂肪酸败程度不能仅凭POV，需结合AV和羰基价。
    \end{itemize}

    \item \textbf{羰基价（Carbonyl Value, CV）*：}
    \begin{itemize}[leftmargin=*]
        \item 定义：油脂氧化分解产生的含羰基化合物（醛、酮等）的量。
        \item 卫生学意义：反映油脂氧化\textbf{分解产物}的总量——与POV形成互补。POV反映初级氧化产物，羰基价反映次级氧化（分解）产物。
    \end{itemize}
\end{enumerate}
\end{keybox}

\begin{exambox}
\textbf{AV、POV、羰基价的关系（考试常见综合题）：}
\begin{itemize}[leftmargin=*]
    \item AV $\uparrow$ = 水解程度高（脂肪酶作用）
    \item POV $\uparrow$ = 氧化初期（最早信号！）
    \item 羰基价 $\uparrow$ = 氧化分解产物积累（后期信号）
    \item 三者联合使用才能全面判断油脂的酸败程度和阶段
\end{itemize}
\end{exambox}

\subsection{油脂酸败的预防}

\begin{enumerate}[leftmargin=*]
    \item 避免光照、低温密封储存
    \item 减少与金属离子（特别是Cu$^{2+}$、Fe$^{3+}$，它们催化自动氧化）的接触
    \item 添加抗氧化剂（BHA、BHT、TBHQ、VE等）
    \item 油脂精炼：脱胶、脱酸、脱色、脱臭——去除促氧化杂质
\end{enumerate}

\subsection{高温加热油脂的卫生问题}

\begin{itemize}[leftmargin=*]
    \item 反复高温加热（油炸用油）产生：
    \begin{itemize}[leftmargin=*]
        \item 聚合物（环状化合物、甘油三酯聚合物）——不易消化、有毒性
        \item 苯并(a)芘、杂环胺等致癌物
        \item 必需脂肪酸破坏、脂溶性维生素损失
    \end{itemize}
    \item 管理要求：煎炸油应定期更换，不得反复过度使用。我国有GB 2716和相应的煎炸油卫生标准。
\end{itemize}

    \item \textbf{锰（Mn）：}高锰酸钾用于除铁除杂后残留
\end{itemize}

    \item $N$-亚硝基化合物（啤酒——麦芽干燥过程）

\section{罐头食品的卫生及管理}

\begin{itemize}[leftmargin=*]
    \item \textbf{主要卫生问题：}
    \begin{enumerate}[leftmargin=*]
        \item 微生物污染——杀菌不彻底（特别是耐热芽孢菌如平酸菌），引起罐头胖听（产气）或平酸变质（不产气但变酸）
        \item 重金属污染——马口铁罐的锡、铅溶出（特别是酸性罐头）
        \item 容器卫生——焊缝铅超标、内壁涂层脱落

\end{itemize}

\section{保健食品、转基因食品、无公害/绿色/有机食品}

\subsection{保健食品（Health Food）}

\begin{defbox}
\textbf{保健食品：}声称具有特定保健功能，或者以补充维生素、矿物质为目的的食品。即适用于特定人群食用，具有调节机体功能，不以治疗疾病为目的，且对人体不产生任何急性、亚急性或者慢性危害的食品。
\end{defbox}

\begin{itemize}[leftmargin=*]
    \item \textbf{特征：}须经国家审批，获得"蓝帽子"标识（国食健字或卫食健字批准文号）。
    \item \textbf{与药品的区别：}保健食品不能声称治疗作用，不能替代药品。
    \item \textbf{适用人群：}特定人群（如中老年、免疫功能低下者），不适用于所有人群。
\end{itemize}

\subsection{转基因食品（Genetically Modified Food, GMO Food）}

\begin{defbox}
\textbf{转基因食品：}利用基因工程技术改变基因组构成的生物所生产的食品（包括转基因植物、动物和微生物食品）。
\end{defbox}

\begin{itemize}[leftmargin=*]
    \item \textbf{安全性评价原则：}
    \begin{itemize}[leftmargin=*]
        \item 实质等同性原则（Substantial Equivalence）——比较GMO食品与其传统对应食品的化学成分、营养和毒性是否实质等同。
        \item 个案评价原则（Case-by-Case）——每种GMO均需单独进行安全性评价。
    \end{itemize}
    \item \textbf{管理要求：}中国实行转基因食品标识制度——列入转基因生物标识管理目录的转基因食品必须标注"转基因"字样。
\end{itemize}

\subsection{无公害食品、绿色食品、有机食品}

\begin{enumerate}[leftmargin=*]
    \item \textbf{无公害食品（Pollution-Free Food）：}
    \begin{itemize}[leftmargin=*]
        \item 最低要求的安全农产品——产地环境、生产过程、产品质量符合国家标准。
        \item 允许使用化学农药和化肥，但残留量需控制在安全限量以下。
    \end{itemize}

    \item \textbf{绿色食品（Green Food）：}
    \begin{itemize}[leftmargin=*]
        \item 中国独创的认证体系，分为A级和AA级。
        \item A级：限量使用化学合成生产资料（化学农药、化肥等）。
        \item AA级：完全按照有机生产方式生产（等同于有机食品）。
    \end{itemize}

    \item \textbf{有机食品（Organic Food）：}
    \begin{itemize}[leftmargin=*]
        \item 国际通行的最高级别认证，禁止使用任何化学合成的农药、化肥、激素、抗生素、转基因技术。
        \item 严格遵循自然规律和生态学原理，维护农业生态系统的可持续性。
    \end{itemize}
\end{enumerate}

\begin{tipbox}
\textbf{无公害 $\rightarrow$ 绿色（A级）$\rightarrow$ 绿色（AA级）/有机：}安全标准逐级升高。有机食品/绿色食品AA级为最高标准。
\end{tipbox}

\subsection*{参考答案要点}

\subsubsection*{名词解释（要点）}

\begin{enumerate}[leftmargin=*, itemsep=4pt]
    \item \textbf{油脂酸败：}油脂在光、热、氧、水分、金属离子等作用下发生的感官变劣和产生不良气味的化学变化。\textbf{AV：}中和1 g油脂游离脂肪酸所需KOH的mg数，反映水解程度。\textbf{POV：}反映氢过氧化物含量，氧化初期最灵敏指标。\textbf{羰基价：}反映氧化分解产物（醛酮）的总量。
    \item \textbf{商业无菌：}杀灭所有在正常储存条件下可生长繁殖的微生物，非绝对无菌。\textbf{疯牛病：}由致病性朊蛋白引起的牛的传染性海绵状脑病。\textbf{PrP$^{Sc}$：}异常折叠的朊蛋白，具传染性、抗蛋白酶K。
    \item \textbf{保健食品：}声称特定保健功能、经国家审批、不能替代药品的食品（蓝帽子）。\textbf{转基因食品：}利用基因工程技术改造基因组生物生产的食品。\textbf{有机食品：}禁止化学合成农药/化肥/激素/转基因的最高级别认证。
    \item \textbf{米猪肉：}含猪囊尾蚴（猪肉绦虫幼虫）的病猪肉。人食用可感染绦虫病或囊虫病。
\end{enumerate}

\subsubsection*{简答题（要点）}

\begin{enumerate}[leftmargin=*, itemsep=6pt]
    \item \textbf{腐败过程：}宰后僵直 $\rightarrow$ 后熟（自溶）$\rightarrow$ 细菌侵入 $\rightarrow$ 蛋白质分解 $\rightarrow$ 硫化氢、胺类、吲哚等恶臭产物 $\rightarrow$ 肉色变暗绿。\textbf{鉴定指标：}感官检查 + TVBN（$\leq$ 15 mg/100g）。
    \item \textbf{炭疽：}危害最严重暴发性人畜共患病，芽孢在土壤存活数十年，病畜须6小时内销毁严禁食用。\textbf{口蹄疫：}病毒性，偶蹄动物易感，80–100$^\circ$C可灭活，病畜须销毁。\textbf{疯牛病：}致病性朊蛋白，不含核酸，常规消毒无效（需134$^\circ$C高压1h），人可患nvCJD。\textbf{本质区别：}病原体为蛋白质（非细菌/病毒/寄生虫），对常规消毒有极强抵抗力。
    \item \textbf{酸败原因：}生物学水解（脂肪酶$\rightarrow$AV升高）和化学自动氧化（自由基链式反应$\rightarrow$POV升高）。\textbf{AV：}反映水解程度；\textbf{POV：}氧化初期最灵敏指标（注意后期可下降）；\textbf{羰基价：}反映次级氧化分解产物总量。三者联合判断：AV高$=$水解为主，POV高$=$氧化初期，羰基价高$=$氧化深度进行。
    \item \textbf{粮豆卫生问题：}真菌及毒素（AF、黄变米、赤霉病变）、农药残留、有害金属（镉大米）、仓库害虫（50余种）。
    \item \textbf{亚硝峰：}蔬菜腌制过程中，在细菌作用下硝酸盐 $\rightarrow$ 亚硝酸盐，约在第7–20天亚硝酸盐含量达峰值（亚硝峰），之后逐渐下降。腌制超过20–30天方可安全食用。
    \item \textbf{水产品特点：}含水量高、组织脆弱、表面黏液多、细菌极易繁殖 $\rightarrow$ 腐败速度快。其他问题：有害金属富集、病原微生物（副溶血性弧菌）、寄生虫（肝吸虫）、天然有毒品种（河豚）。
    \item \textbf{乳品问题：}微生物（沙门菌、结核、布氏菌、金葡菌肠毒素-耐热）、化学（农药/兽药/AF M1残留）、掺伪（掺水/碱/淀粉/三聚氰胺事件）。
    \item \textbf{副溶血性弧菌：}海水产品最常见致病菌，嗜盐，急性胃肠炎。\textbf{肝吸虫：}淡水鱼虾生食感染，寄生于肝内胆管，可致胆管炎/肝硬化/胆管癌。
    \item \textbf{蒸馏酒问题：}甲醇（失明/致死）、醛类、杂醇油、HCN、Pb、Mn。\textbf{发酵酒问题：}展青霉素、SO$_2$残留、$N$-亚硝基化合物（啤酒）。
    \item \textbf{保健食品：}声称保健功能，国家审批（蓝帽子），不以治疗为目的。\textbf{与药品区别：}不能声称治疗作用，不能替代药品。
    \item \textbf{层级递进：}无公害（最低安全基准，允许限量化学生产资料）$\rightarrow$ 绿色A级（限量使用化学合成物）$\rightarrow$ 绿色AA级（等同于有机，禁止化学合成物）$\rightarrow$ 有机（最高标准，全部禁止化学合成物+转基因）。
\end{enumerate}
