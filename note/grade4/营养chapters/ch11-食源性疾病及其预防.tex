\section{食源性疾病概述}

\begin{keybox}
\textbf{食源性疾病（Foodborne Diseases）的定义：}
由摄食进入人体内的各种致病因子引起的、通常具有感染性质或中毒性质的一类疾病。
\end{keybox}

\begin{defbox}
\textbf{WHO定义：}食源性疾病是指通过摄食进入人体内的各种致病因子引起的、通常具有感染性质或中毒性质的一类疾病。食物是疾病传播的媒介，致病因子来源于食物中含有的各种有害因素。
\end{defbox}

\subsection{食源性疾病的三大核心特征}

\begin{keybox}
\textbf{三大核心特征（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{传播媒介（Vehicle）：}食物是携带和传播致病因素的媒介。这是食源性疾病区别于其他疾病的根本特征。
    \item \textbf{致病因子（Pathogenic Agents）：}食物中含有的各种致病因子，包括生物性、化学性、物理性因素。
    \item \textbf{临床综合征（Syndromes）：}临床主要表现为中毒性或感染性综合征。
\end{enumerate}
\end{keybox}

\subsection{食源性疾病的分类}

\begin{keybox}
\textbf{食源性疾病的七大分类（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{食物中毒（Food Poisoning）：}最常见的一类，包括细菌性、真菌毒素、动物性、植物性、化学性食物中毒
    \item \textbf{食源性肠道传染病（Foodborne Enteric Infectious Diseases）：}如霍乱（Vibrio cholerae）、甲肝（HAV）
    \item \textbf{食源性寄生虫病（Foodborne Parasitic Diseases）：}如囊虫病、旋毛虫病、弓形虫病
    \item \textbf{人畜共患传染病（Zoonotic Infectious Diseases）：}如炭疽、疯牛病、口蹄疫、结核病、布鲁氏菌病
    \item \textbf{食物过敏（Food Allergy）：}免疫系统对食物中某些成分的异常反应
    \item \textbf{食物污染物引起的慢性中毒性疾病：}如重金属中毒、农药残留蓄积中毒
    \item \textbf{食源性放射性疾病：}食物受放射性物质污染所致
\end{enumerate}
\end{keybox}

\subsection{致病因子分类}

\begin{itemize}[leftmargin=*]
    \item \textbf{生物性因子：}细菌、病毒、寄生虫、真菌及其毒素——\keyword{最常见}的食源性疾病病因
    \item \textbf{化学性因子：}农药残留、重金属、食品添加剂滥用、环境污染物等
    \item \textbf{放射性因子：}放射性核素污染食品
\end{itemize}

\section{人畜共患传染病}

\subsection{炭疽（Anthrax）}

\begin{keybox}
\textbf{炭疽（Anthrax）——掌握：}
\begin{itemize}[leftmargin=*]
    \item \textbf{病原体：}\keyword{炭疽杆菌（Bacillus anthracis）}，革兰阳性需氧芽胞杆菌，对人和动物危害最大的烈性传染病
    \item \textbf{芽胞形成前：}55～58$^\circ$C加热10～15分钟即可杀灭
    \item \textbf{形成芽胞后：}抵抗力极强！土壤中可存活数十年
    \item \textbf{传播途径：}接触病畜及畜产品（皮毛、肉、骨）→ 皮肤炭疽（最常见）；吸入芽胞 → 肺炭疽；摄入污染食物 → 肠炭疽（最严重）
    \item \textbf{临床类型：}皮肤炭疽（炭疽痈，恶性水肿）、肺炭疽（死亡率极高）、肠炭疽（腹痛、腹泻、呕吐，死亡率25\%～75\%）
    \item \textbf{预防措施：}严格检疫病畜，严禁屠宰销售炭疽病畜肉；病畜肉应化制销毁或焚烧；屠宰加工人员做好个人防护
\end{itemize}
\end{keybox}

\subsection{口蹄疫（Foot and Mouth Disease, FMD）}

\begin{itemize}[leftmargin=*]
    \item \textbf{病原体：}口蹄疫病毒，偶蹄动物（牛、猪、羊等约70种动物易感）。马不感染但可被动携带
    \item \textbf{抵抗力：}80～100$^\circ$C加热可杀灭
    \item \textbf{人感染：}人间感染罕见，主要见于饲养员、屠宰工
\end{itemize}

\subsection{疯牛病（Bovine Spongiform Encephalopathy, BSE）}

\begin{keybox}
\textbf{疯牛病（BSE）——掌握：}
\begin{itemize}[leftmargin=*]
    \item \textbf{病原体：}\keyword{致病性朊病毒蛋白（PrP$^\text{Sc}$）}，与正常PrP$^\text{C}$一级结构相同但空间构象不同（$\alpha$螺旋 $\to$ $\beta$折叠）
    \item \textbf{核心特点：}不能被蛋白酶K降解！耐高温（焚烧才可彻底灭活），无免疫原性
    \item \textbf{发病机制：}PrP$^\text{Sc}$沉积于脑组织 $\to$ 神经元空泡化 $\to$ 神经退行性变 $\to$ 海绵状脑病
    \item \textbf{人类关联疾病：}\keyword{新型变异型克雅氏病（nvCJD）}
    \item \textbf{传播途径：}食用疯牛病牛肉及产品（尤其是神经组织、骨髓等富含PrP$^\text{Sc}$的组织）
    \item \textbf{预防：}严禁将哺乳动物蛋白饲料（肉骨粉）喂养反刍动物；严格监管含神经组织的食品进入食物链；发现病牛立即扑杀销毁
\end{itemize}
\end{keybox}

\begin{exambox}
\textbf{PrP$^\text{Sc}$的特征经常作为考点：}不能被蛋白酶K降解、耐高温、无免疫原性、$\alpha$螺旋 $\to$ $\beta$折叠的构象转变。人类nvCJD与疯牛病BSE的关系。
\end{exambox}

\subsection{其他重要人畜共患病}

\begin{description}[leftmargin=*]
    \item[\textbf{结核病（Tuberculosis）：}]通过病畜乳汁传播，牛型结核杆菌。巴氏消毒可杀灭。
    \item[\textbf{布鲁氏菌病（Brucellosis）：}]布氏杆菌，接触或食用污染畜产品感染。发热呈波浪热型。
    \item[\textbf{霍乱（Cholera）：}]\keyword{霍乱弧菌（Vibrio cholerae）}，O1群和O139群为主要致病群。甲类传染病。通过污染水产品（尤其是甲壳类、贝类）传播。
\end{description}

\section{食源性寄生虫病}

\begin{tipbox}
本节为"熟悉"层次内容，重点掌握主要寄生虫病的寄生虫种类、寄生部位和预防要点。
\end{tipbox}

\subsection{囊虫病（猪囊尾蚴病，Cysticercosis）}

\begin{itemize}[leftmargin=*]
    \item \textbf{病原体：}猪带绦虫（Taenia solium，又称有钩绦虫，其幼虫为猪囊尾蚴Cysticercus cellulosae）和牛带绦虫（T. saginata，无钩绦虫）
    \item \textbf{寄生部位：}囊尾蚴寄生于舌肌、咬肌、臀肌、腰深肌、肩胛外侧肌、颈肌、心肌等
    \item \textbf{"米猪肉"（豆猪肉）：}肌肉内含大量囊尾蚴的猪肉，肉眼可见白色半透明绿豆大小囊泡
    \item \textbf{人体感染方式：}(1)异体感染——误食虫卵；(2)自体体外感染——自身虫卵经口感染；(3)自体体内感染——肠道逆蠕动，虫卵入胃
    \item \textbf{预防：}加强肉品检验检疫，改变不良饮食习惯（生食或半生食猪肉），注意个人卫生
\end{itemize}

\subsection{旋毛虫病（Trichinosis）}

\begin{itemize}[leftmargin=*]
    \item \textbf{病原体：}\keyword{旋毛虫（Trichinella spiralis）}
    \item \textbf{寄生部位：}幼虫寄生于横纹肌，\keyword{膈肌是最常见的寄生部位}（其次是舌肌、心肌）
    \item \textbf{抵抗力：}温度低于70$^\circ$C时，厚度2.5mm的肉制品加热1分钟不能杀死幼虫
    \item \textbf{临床表现：}发热、眼眶周围水肿、肌肉疼痛（以腓肠肌最显著）、嗜酸性粒细胞显著增高（20\%～90\%）
    \item \textbf{预防：}肉类彻底加热至中心温度70$^\circ$C以上；严格执行肉品检验制度
\end{itemize}

\subsection{其他食源性寄生虫}

\begin{itemize}[leftmargin=*]
    \item \textbf{弓形虫病（Toxoplasmosis）：}刚地弓形虫，猫科动物为终宿主。孕妇感染可导致胎儿畸形
    \item \textbf{蛔虫病（Ascariasis）：}蛔虫卵通过污染食物/水传播，儿童多见
    \item \textbf{中华支睾吸虫病（Clonorchiasis）：}\keyword{华支睾吸虫（Clonorchis sinensis）}，即肝吸虫。食用生/半生淡水鱼虾感染。寄生于肝胆管。广东省"鱼生"是重要传播途径
    \item \textbf{绦虫病（Taeniasis）：}食用含囊尾蚴的未熟肉类感染
\end{itemize}

\section{食物过敏}

\begin{defbox}
\textbf{食物过敏（Food Allergy）：}机体免疫系统对进入体内的食物中的某些非感染性物质（过敏原）产生的异常免疫应答，导致生理功能紊乱和/或组织损伤。
\end{defbox}

\subsection{致敏食物}

\begin{keybox}
\textbf{八大类主要致敏食物：}
\begin{enumerate}[leftmargin=*]
    \item 牛奶（Milk）
    \item 鸡蛋（Eggs）
    \item 花生（Peanuts）
    \item 大豆（Soy）
    \item 谷物（小麦等，Grains/Wheat）
    \item 鱼类（Fish）
    \item 甲壳类（Crustaceans）
    \item 坚果类（Tree Nuts）
\end{enumerate}
\end{keybox}

\subsection{食物过敏的特点}

\begin{enumerate}[leftmargin=*]
    \item \textbf{任何食物均可诱发过敏：}但少数食物是主要过敏原
    \item \textbf{仅有部分成分具有变应原性：}食物中的特定蛋白质为主要致敏成分
    \item \textbf{交叉反应性：}不同食物间可发生交叉过敏反应（如花粉与部分果蔬）
    \item \textbf{食物加工可改变致敏性：}加热可使部分过敏原变性降低致敏性，如牛奶中的$\beta$-乳球蛋白属热稳定型过敏原不能消除，需改用脱敏或水解配方
\end{enumerate}

\subsection{卫生假说（Hygiene Hypothesis）}

\begin{tipbox}
\textbf{卫生假说：}免疫系统在进化过程中为应对肠道寄生虫而设计，现代社会中缺乏肠道寄生虫感染，导致免疫系统过度活跃，对无害食物抗原产生异常免疫应答。解释了工业化国家过敏性疾病高发的原因。
\end{tipbox}

\section{食物中毒}

\begin{keybox}
\textbf{食物中毒的定义（掌握）：}
摄入了含有生物性、化学性有毒有害物质的食品，或把有毒有害物质当作食品摄入后出现的非传染性（不属传染病）的急性、亚急性疾病。
\end{keybox}

\begin{exambox}
\textbf{食物中毒定义的三个关键要素：}(1)摄食行为；(2)食品含有/本身即为有毒有害物质；(3)非传染性的急性/亚急性疾病。排除传染病、慢性中毒、寄生虫病。
\end{exambox}

\subsection{食物中毒的特点}

\begin{keybox}
\textbf{食物中毒的临床特点（掌握*）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{潜伏期短、发病急、呈暴发性：}短时间内多人同时发病
    \item \textbf{临床表现相似：}以急性胃肠炎症状（恶心、呕吐、腹痛、腹泻）为主
    \item \textbf{人与人之间一般无直接传染：}停止进食致病食品后发病立即下降
    \item \textbf{发病与进食某种特定食物有关：}发病者均进食同样食物，未食者不发病
\end{enumerate}
\end{keybox}

\begin{keybox}
\textbf{食物中毒的流行病学特点（掌握*）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{与特定食品相关性强：}发病者均进食过同一食品，停止供应该食品后发病立即停止
    \item \textbf{细菌性食物中毒季节性明显：}\keyword{夏秋季（5～10月）高发}，此时气温高、湿度大，细菌易繁殖
    \item \textbf{与地区食品种类和饮食习惯相关：}沿海地区副溶血性弧菌中毒多见；新疆地区肉毒梭菌中毒多见
    \item \textbf{中毒发生场所：}家庭（最常见）、集体食堂、餐饮单位
\end{enumerate}
\end{keybox}

\subsection{食物中毒的分类}

\begin{keybox}
\textbf{食物中毒的五大分类（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{细菌性食物中毒：}\keyword{发病率最高，中毒人数最多！}
    \item \textbf{真菌毒素和霉变食品中毒：}霉菌毒素污染引起
    \item \textbf{动物性食物中毒：}河豚鱼、高组胺鱼类、有毒动物脏器
    \item \textbf{有毒植物食物中毒：}毒蕈、含氰甙植物、发芽马铃薯等
    \item \textbf{化学性食物中毒：}亚硝酸盐、有机磷农药、重金属等
\end{enumerate}
\end{keybox}

% ----- 细菌性食物中毒 -----
\subsection{细菌性食物中毒}

\subsubsection{发病机制}

\begin{keybox}
\textbf{细菌性食物中毒的三大发病机制（掌握*——教学难点）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{感染型（Infection Type）：}
    \begin{itemize}
        \item 致病机制：活的细菌随食物进入肠道 $\to$ 在肠道内大量繁殖 $\to$ 侵入肠黏膜上皮细胞 $\to$ 引起炎症反应 + 内毒素释放
        \item 临床表现：\keyword{发热 + 胃肠道症状}（腹痛、腹泻、恶心、呕吐）
        \item 潜伏期：\keyword{$\ge$4小时}，一般较长
        \item 典型病原体：沙门菌、变形杆菌、副溶血性弧菌（混合型）
    \end{itemize}
    \item \textbf{毒素型（Toxin Type）：}
    \begin{itemize}
        \item 致病机制：食物中细菌\keyword{预先产生肠毒素} $\to$ 毒素随食物进入体内 $\to$ 作用于小肠上皮细胞膜腺苷酸环化酶系统（cAMP系统） $\to$ Na$^+$/Cl$^-$吸收抑制、水分滞留 $\to$ 腹泻
        \item 临床表现：以呕吐、腹泻为主，\keyword{患者一般不发热}
        \item 潜伏期：\keyword{较短（$<$12小时）}，最短仅1小时（葡萄球菌）
        \item 典型病原体：金黄色葡萄球菌肠毒素、肉毒梭菌毒素
    \end{itemize}
    \item \textbf{混合型（Mixed Type）：}
    \begin{itemize}
        \item 致病机制：活菌侵入肠道 + 同时产生肠毒素
        \item 兼有感染型和毒素型的特点
        \item 典型病原体：\keyword{副溶血性弧菌}
    \end{itemize}
\end{enumerate}
\end{keybox}

\begin{tipbox}
\textbf{感染型 vs 毒素型的鉴别要点：}潜伏期长短（$\ge$4h vs 短）和有无发热是重要的临床区分依据。感染型患者通常有发热，毒素型一般不发热。
\end{tipbox}

\subsubsection{流行病学特点}

\begin{keybox}
\textbf{细菌性食物中毒的流行病学特点（掌握*）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{发病率高、病死率低}（\keyword{肉毒梭菌中毒除外}！肉毒梭菌病死率极高）
    \item \textbf{夏秋季（5～10月）高峰}
    \item \textbf{动物性食品为主要中毒食品：}其中以肉类及其制品居首位，其次为变质禽肉、病死畜肉，鱼类、奶类、蛋类
\end{enumerate}
\end{keybox}

\subsubsection{发生的三大环节}

\begin{keybox}
\textbf{细菌性食物中毒发生的三大环节（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{食品被致病菌污染：}原料污染（生前感染或宰后污染）、加工过程污染、从业人员携带菌
    \item \textbf{食品在较高温度下存放：}使致病菌大量生长繁殖和/或产生毒素
    \item \textbf{食用前未彻底加热：}或生熟交叉污染，使食品中心温度未达到灭菌要求
\end{enumerate}
\end{keybox}

\subsubsection{预防原则}

\begin{keybox}
\textbf{细菌性食物中毒的预防——"三管齐下"（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{防止污染（Prevent Contamination）：}加强原料检验、规范加工过程、从业人员健康管理
    \item \textbf{杀灭细菌（Kill Bacteria）：}食品食用前彻底加热，保证中心温度达标
    \item \textbf{控制细菌繁殖（Control Bacterial Growth）：}低温贮存、缩短存放时间、控制水分活度
\end{enumerate}
\end{keybox}

\subsubsection{各类细菌性食物中毒}

% 1. 沙门菌
\paragraph{一、沙门菌食物中毒（Salmonella Food Poisoning）}

\begin{keybox}
\textbf{沙门菌食物中毒（掌握*）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{病原特点：}\keyword{沙门菌属（Salmonella）}，革兰阴性杆菌。\keyword{不分解蛋白质、不产生靛基质}（吲哚）$\to$ \textbf{污染食品后无感官性状改变！}这是最危险的特性
    \item \textbf{污染食品：}畜肉类及其制品居首位，其次为禽肉、蛋类、乳类
    \item \textbf{生前感染}是肉类中沙门菌的主要来源
    \item \textbf{流行病学：}全年发生，夏秋季（5～10月）占80\%。发病率40\%～60\%，最高可达90\%。\keyword{猪霍乱沙门氏菌致病力最强，其次为鼠伤寒沙门氏菌}；年老体弱者易发生食物中毒。青壮年多发，以农民、工人为主。
    \item \textbf{发病机制：}感染型
    \item \textbf{潜伏期：}6～8小时（一般12～14小时）
    \item \textbf{临床表现：}发热、恶心、呕吐、腹痛、\keyword{腹泻（水样便或黄绿色稀便）}
    \item \textbf{诊断标准：}WS/T 13--1996
    \item \textbf{灭菌标准：}肉块中心温度$\ge$80$^\circ$C持续12分钟；鸡蛋煮沸8～10分钟
\end{itemize}
\end{keybox}

% 2. 副溶血性弧菌
\paragraph{二、副溶血性弧菌食物中毒（Vibrio parahaemolyticus）}

\begin{keybox}
\textbf{副溶血性弧菌食物中毒（掌握*）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{病原特点：}\keyword{副溶血性弧菌（Vibrio parahaemolyticus）}，\keyword{嗜盐菌！}最适生长盐浓度3.5\% NaCl。在含盐培养基上生长良好，淡水环境中迅速死亡
    \item \textbf{抵抗力：}56$^\circ$C加热5分钟或90$^\circ$C加热1分钟即可杀灭；\keyword{含1\%食醋处理5分钟即可杀灭}
    \item \textbf{神奈川试验（Kanagawa Test）：}神奈川现象阳性——90\%的致病性菌株为阳性
    \item \textbf{污染食品：}海产品（鱼类、贝类、虾蟹，墨鱼的带菌率可达93\%）、盐渍食品。7～9月是副溶血性弧菌食物中毒的高发季节。
    \item \textbf{发病机制：}混合型（活菌侵入 + 产生肠毒素）
    \item \textbf{潜伏期：}6～10小时
    \item \textbf{临床表现：}\keyword{上腹部阵发性绞痛}（最突出的特点）、\keyword{血水样便}、\keyword{发热较明显（37.5～39.5$^\circ$C）}
\end{itemize}
\end{keybox}

\begin{exambox}
\textbf{副溶血性弧菌的三大特征：}嗜盐性（3.5\% NaCl最适）、神奈川现象（90\%致病株阳性）、血水样便。1\%食醋5分钟可杀灭——食用海鲜蘸醋对预防有意义！
\end{exambox}

% 3. 葡萄球菌肠毒素
\paragraph{三、葡萄球菌肠毒素食物中毒（Staphylococcus aureus Enterotoxin）}

\begin{keybox}
\textbf{葡萄球菌肠毒素食物中毒（掌握*）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{病原体：}\keyword{金黄色葡萄球菌（Staphylococcus aureus）}产生肠毒素（Enterotoxin）
    \item \textbf{肠毒素分型：}A、B、C1、C2、C3、D、E、F共8个血清型
    \item \textbf{肠毒素耐热性：}\keyword{B型耐热性最强！100$^\circ$C煮沸30分钟仍具有部分活性！}普通烹调方法难以彻底破坏
    \item \textbf{流行病学：}全年皆可发生，夏秋季较多
    \item \textbf{污染来源：}\keyword{带菌者污染食品是最重要的污染来源！}人体鼻咽部带菌率20\%～50\%，皮肤化脓性病灶含大量致病性葡萄球菌
    \item \textbf{影响肠毒素形成的因素：}污染程度越重越易形成；37$^\circ$C范围内温度越高产毒越快；含蛋白质丰富、水分多且含一定淀粉的食品（如奶、肉、糕点）最易形成
    \item \textbf{发病机理：}\keyword{毒素型}
    \item \textbf{潜伏期：}\keyword{2～4小时（极短！最短仅1小时）}
    \item \textbf{临床表现：}\keyword{剧烈而频繁的呕吐（最突出！）}、水样便，\keyword{体温一般正常}
    \item \textbf{预防：}\keyword{有局部化脓性感染或上呼吸道感染者应暂时调离食品工作岗位！}
\end{itemize}
\end{keybox}

% 4. 肉毒梭菌毒素
\paragraph{四、肉毒梭菌毒素食物中毒（Clostridium botulinum）}

\begin{keybox}
\textbf{肉毒梭菌毒素食物中毒（掌握*——教学难点）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{病原体：}\keyword{肉毒梭菌（Clostridium botulinum）}产生的外毒素——\keyword{肉毒毒素（Botulinum Toxin）}
    \item \textbf{毒力：}\keyword{已知毒力最强的神经毒素！}毒性比KCN强10000倍！人的致死剂量约10$^{-9}$mg/kg体重！
    \item \textbf{分型：}8个血清型（A～G），其中A、B、E、F型引起人类中毒，\keyword{A、B型最常见}
    \item \textbf{芽胞耐热性极强：}\keyword{干热180$^\circ$C 5～15分钟！湿热100$^\circ$C 5小时！高压蒸汽灭菌121$^\circ$C 30分钟！}
    \item \textbf{地区分布：}中国以\keyword{新疆最高}
    \item \textbf{中毒食品：}中国\keyword{91.48\%由植物性食品引起}（家庭自制发酵豆制品为主！如臭豆腐、豆豉、豆酱等）
    \item \textbf{发病机理：}毒素在小肠被激活 $\to$ 释放神经毒素 $\to$ 作用于神经肌肉接头 $\to$ 阻止乙酰胆碱释放 $\to$ \keyword{肌肉麻痹}
    \item \textbf{临床表现：}\keyword{以运动神经麻痹为主}（对称性颅神经损害为特征），\keyword{胃肠道症状极少见}。主要症状：视物模糊、复视、眼睑下垂、吞咽困难、发音困难、呼吸困难（致死主要原因）
    \item \textbf{婴儿肉毒中毒：}梭菌芽胞在婴儿肠道萌芽繁殖产毒。\keyword{蜂蜜是婴儿中毒的嫌疑食品！}主要E型
    \item \textbf{治疗：}\keyword{尽早使用多价肉毒抗毒素血清}（A、B、E型），同时对症支持治疗
    \item \textbf{预防：}\keyword{家庭自制密封发酵食品 = 高危风险！}罐头食品鼓盖/漏气绝对不要购买食用！
\end{itemize}
\end{keybox}

\begin{exambox}
\textbf{肉毒梭菌与其他细菌性食物中毒的根本区别：}胃肠道症状极少见（不以呕吐腹泻为主），主要表现为运动神经麻痹（对称性颅神经损害）。芽胞耐热性极强——湿热100$^\circ$C需5小时！中国以发酵豆制品为主要中毒食品（91.48\%植物性来源）。新疆高发！蜂蜜是1岁以下婴儿肉毒中毒的嫌疑食品。
\end{exambox}

% 5. 变形杆菌
\paragraph{五、变形杆菌食物中毒（Proteus）}

\begin{itemize}[leftmargin=*]
    \item \textbf{发病机制：}感染型。变形杆菌属中的普通变形杆菌、奇异变形杆菌等。
    \item \textbf{中毒食品：}熟肉制品、动物内脏制品，尤其是凉拌菜
    \item \textbf{潜伏期：}2～30小时，平均10～12小时
    \item \textbf{临床表现：}上腹部绞痛、水样便，发热
\end{itemize}

% 6. 蜡样芽胞杆菌
\paragraph{六、蜡样芽胞杆菌食物中毒（Bacillus cereus）}

\begin{itemize}[leftmargin=*]
    \item \textbf{中毒食品：}米饭为主（尤其剩米饭、炒饭）。25～37$^\circ$C放置过久最危险
    \item \textbf{分型：}
    \begin{enumerate}[leftmargin=*]
        \item \textbf{腹泻型（Diarrheal Type）：}产生\keyword{腹泻毒素}（不耐热肠毒素），对胰蛋白酶敏感。毒性作用类似大肠埃希菌和霍乱弧菌产生的毒素，激活腺苷酸环化酶。潜伏期8～16小时。
        \item \textbf{呕吐型（Emetic Type）：}产生\keyword{呕吐毒素}（Cereulide），\textbf{耐热}（126$^\circ$C 90 min不失活），对胰蛋白酶\underline{不敏感}。潜伏期0.5～5小时。
    \end{enumerate}
    \item \textbf{季节性：}夏、秋季，尤其是6～10月为多见。引起中毒的食品种类繁多：米粉、米饭、乳及乳制品、肉类制品、蔬菜等。
    \item \textbf{预防：}剩米饭及其他熟食品在食用前须彻底加热（100$^\circ$C 20 min）。
\end{itemize}

% 7. 椰毒假单胞菌
\paragraph{七、椰毒假单胞菌酵米面亚种食物中毒}

\begin{itemize}[leftmargin=*]
    \item \textbf{毒素：}\keyword{米酵菌酸（Bongkrekic Acid）}和毒黄素。米酵菌酸作用于线粒体，抑制氧化磷酸化
    \item \textbf{中毒食品：}酵米面（发酵玉米面）、变质银耳
    \item \textbf{特点：}病死率极高（40\%～100\%），无特效解毒药
\end{itemize}

% 8. 李斯特菌
\paragraph{八、李斯特菌食物中毒（Listeria monocytogenes）}

\begin{itemize}[leftmargin=*]
    \item \textbf{特点：}\keyword{能在5$^\circ$C低温条件下生长！}（冰箱冷藏温度不能抑制其生长）
    \item 侵袭型病死率20\%～50\%。孕妇、新生儿、老年人、免疫低下者高危
    \item 中毒食品：乳制品（软奶酪为主）、熟肉制品、即食食品
\end{itemize}

% 9. 大肠埃希菌O157
\paragraph{九、大肠埃希菌O157:H7食物中毒}

    \item 主要症状：出血性结肠炎、剧烈腹痛、血便
    \item \textbf{主要致病血清型：}引起食物中毒的致病性大肠埃希菌血清型主要有O157:H7（最常见）、O111:B4、O55:B5等。
\end{itemize}

    \item \textbf{中毒食品：}霉变甘蔗（尤其越冬后2～3月份销售的甘蔗，截面呈褐色/红色，有酒糟味）

\subsection{动物性食物中毒}

\subsubsection{河豚鱼中毒}

\begin{keybox}
\textbf{河豚鱼中毒（掌握*）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{有毒成分：}\keyword{河豚毒素（Tetrodotoxin, TTX）}，\keyword{耐热！煮沸、盐腌、日晒均不能破坏！}
    \item \textbf{毒性分布：}\keyword{卵巢毒性最强}，肝脏次之。血液、眼球、皮肤亦含毒素。新鲜洗净的肌肉基本无毒。每年\keyword{春季（2～5月）}为卵巢发育期，毒性最强
    \item \textbf{中毒机理：}TTX阻断神经细胞膜钠离子通道 $\to$ 神经传导阻滞 $\to$ 先麻痹神经末梢，后麻痹中枢神经系统
    \item \textbf{临床表现：}发病急而剧烈。手指、口唇、舌尖麻木刺痛，随后出现恶心、呕吐、四肢无力、发冷、口唇四肢末端青紫。严重者全身麻痹瘫痪，呼吸衰竭死亡。\keyword{通常死于发病后4～6小时，最快1.5小时}
    \item \textbf{治疗：}\keyword{无特效解毒药！}尽早洗胃 + 导泻 + 对症支持（维持呼吸、升压等）
    \item \textbf{预防：}严禁餐饮单位经营加工河豚鱼；加强宣传教育
\end{itemize}
\end{keybox}

\subsubsection{鱼类引起的组胺中毒}

\begin{keybox}
\textbf{鱼类组胺中毒（掌握）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{中毒原因：}青皮红肉鱼（鲐鱼、金枪鱼、沙丁鱼、竹荚鱼等）含丰富组氨酸 $\to$ 在\keyword{摩根变形杆菌（Morganella morganii）}的脱羧酶作用下 $\to$ 组氨酸 $\to$ \keyword{组胺（Histamine）}
    \item \textbf{诱导条件：}鱼不新鲜/腐败变质时组胺大量生成
    \item \textbf{特点：}发病快（数分钟至数小时）、症状轻、恢复快
    \item \textbf{临床表现：}面部和全身皮肤潮红、头晕、心跳加速、胸闷、呼吸急促
\end{itemize}
\end{keybox}

% ----- 有毒植物食物中毒 -----
\subsection{有毒植物食物中毒}

\subsubsection{毒蕈中毒}

\begin{keybox}
\textbf{毒蕈中毒（掌握*——教学难点）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{基本概况：}我国已知毒蕈约80余种，含剧毒可致死的不超过10种
    \item \textbf{主要毒素类型：}
    \begin{itemize}
        \item 肝脏毒素（毒肽Phallotoxins、毒伞肽Amanitins）
        \item 肾脏毒素
        \item 神经精神毒素（毒蝇碱Muscarine、裸盖菇素Psilocybin）
        \item 溶血毒素
        \item 胃肠毒素
    \end{itemize}
    \item \textbf{临床分型及治疗：}
    \begin{enumerate}[leftmargin=*]
        \item \textbf{肝肾损害型（最严重！）：}以毒肽和毒伞肽为主。潜伏期长（6～48h），病死率高。\keyword{巯基解毒剂（二巯基丙磺酸钠、二巯基丁二酸钠等）是抢救关键！}
        \begin{itemize}[leftmargin=*]
            \item 病情发展可分为\keyword{6期}：①潜伏期（10～24h）；②胃肠炎期；③\keyword{假愈期}（胃肠炎症状缓解但肝脏损害已开始——最易被忽视！）；④内脏损害期（肝、肾、脑、心损害）；⑤精神症状期（肝性昏迷、中毒性脑病）；⑥恢复期。
        \end{itemize}
        \item \textbf{神经精神型：}毒蝇碱等。出汗、流涎、瞳孔缩小、幻视。\keyword{阿托品（Atropine）}拮抗治疗
        \item \textbf{溶血型：}鹿花蕈毒素。\keyword{肾上腺皮质激素}治疗
        \item \textbf{胃肠炎型：}对症支持治疗
        \item \textbf{类光过敏型：}类似日光性皮炎。在身体暴露部位出现明显的肿胀、疼痛，特别是嘴唇肿胀外翻，指尖疼痛、指甲根部出血等。
    \end{enumerate}
    \item \textbf{鹅膏菌中毒：}食用后\keyword{10小时以内彻底洗胃}是关键！
\end{itemize}
\end{keybox}

\subsubsection{含氰甙植物中毒}

\begin{keybox}
\textbf{含氰甙植物中毒（掌握）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{中毒机制：}含氰甙（如苦杏仁甙Amygdalin）在酶或酸作用下释放\keyword{氢氰酸（HCN）} $\to$ 与细胞色素氧化酶中的铁结合 $\to$ 组织窒息（细胞内窒息）
    \item \textbf{主要中毒食品：}\keyword{苦杏仁（最常见）}、桃仁、李子仁、木薯等
    \item \textbf{特殊解毒剂：}亚硝酸异戊酯 + 亚硝酸钠 + 硫代硫酸钠（形成高铁血红蛋白结合CN$^-$，再加硫代硫酸钠生成硫氰酸盐排出）
\end{itemize}
\end{keybox}

\subsubsection{亚硝酸盐食物中毒}

\begin{keybox}
\textbf{亚硝酸盐食物中毒（掌握*）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{中毒原因：}
    \begin{enumerate}[leftmargin=*]
        \item 误将亚硝酸盐当作食盐使用（最危险！建筑工地常见）
        \item 进食大量含硝酸盐/亚硝酸盐较高的蔬菜（新鲜腌制的蔬菜——"暴腌菜"，\keyword{第4～8天亚硝酸盐达高峰}）
        \item 肠源性青紫症：肠道功能紊乱时，肠道细菌将硝酸盐还原为亚硝酸盐
        \item 某些井水含硝酸盐过高（苦井水）
    \end{enumerate}
    \item \textbf{中毒机制：}亚硝酸盐使血红蛋白氧化为\keyword{高铁血红蛋白（MetHb）} $\to$ 失去携氧能力 $\to$ 组织缺氧 $\to$ \keyword{肠源性青紫症（发绀）}
    \item \textbf{解毒：}\keyword{美蓝（亚甲蓝，Methylene Blue） + 维生素C}
    \item \textbf{腌菜安全：}腌制第4～8天 = 亚硝酸盐峰值期，\keyword{20天以后基本消失}
\end{itemize}
\end{keybox}

\begin{tipbox}
\textbf{暴腌菜：}腌制时间短的泡菜/腌菜，第4～8天亚硝酸盐含量达峰值，此时食用中毒风险最大！腌制20天以上才安全。误将亚硝酸盐当食盐是化学性食物中毒的最常见原因。
\end{tipbox}

\subsubsection{食源性急性有机磷农药中毒}

\begin{keybox}
\textbf{有机磷农药中毒（Acute Organophosphorus Pesticide Poisoning）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{中毒原因：}误食农药拌过的种子；误把有机磷农药当作食用油或酱油食用；食用喷洒农药后未经安全间隔期即采摘的蔬菜水果；误食被农药毒杀的家禽家畜。
    \item \textbf{中毒机制：}有机磷农药\keyword{抑制胆碱酯酶（ChE）活性}，使其失去催化水解乙酰胆碱（ACh）的能力 $\to$ 大量ACh在体内蓄积 $\to$ 胆碱能神经过度兴奋 $\to$ 出现毒蕈碱样、烟碱样和中枢神经系统症状。
    \item \textbf{临床表现（三度分级）：}
    \begin{enumerate}[leftmargin=*]
        \item \textbf{轻度中毒：}全血ChE活性降至正常值的50\%～70\%，头晕、恶心、呕吐、多汗、视力模糊。
        \item \textbf{中度中毒：}ChE活性降至30\%～50\%，除上述症状外出现肌束震颤、瞳孔缩小、呼吸困难。
        \item \textbf{重度中毒：}ChE活性降至30\%以下，肺水肿、昏迷、呼吸麻痹，可因呼吸衰竭死亡。
    \end{enumerate}
    \item \textbf{特效解毒剂：}\keyword{阿托品（Atropine）} + \keyword{氯解磷定（Pralidoxime Chloride）/碘解磷定（解磷定）}。阿托品拮抗毒蕈碱样症状，解磷定复活胆碱酯酶。
    \item \textbf{预防：}严格执行农药安全使用规定，遵守安全间隔期，加强农药保管，严禁食用被农药毒死的动物。
\end{itemize}
\end{keybox}

\section{食物中毒调查处理}

\subsection{调查目的}

\begin{keybox}
\textbf{食物中毒调查处理的目的（掌握*）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{确定中毒食品和致病因素：}查明食品中毒的来源和病因
    \item \textbf{确定病因和发病经过：}为患者救治和现场措施提供依据
    \item \textbf{防止中毒扩散：}采取有效控制措施
    \item \textbf{积累资料以利预防：}为制定预防措施提供科学依据
\end{enumerate}
\end{keybox}

\subsection{现场调查内容}

\begin{keybox}
\textbf{三大调查内容（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{流行病学调查（Epidemiological Investigation）：}
    \begin{itemize}
        \item 确定可疑中毒食品和中毒致病因素
        \item 调查发病人数、时间分布、空间分布
        \item 计算罹患率（Attack Rate）、潜伏期
    \end{itemize}
    \item \textbf{食品卫生学调查（Food Hygiene Investigation）：}
    \begin{itemize}
        \item 中毒食品的来源、加工过程、储存条件、销售环节
        \item 从业人员健康状况
        \item 厨房、餐厅卫生状况
    \end{itemize}
    \item \textbf{实验室检验（Laboratory Testing）：}
    \begin{itemize}
        \item 食品和患者标本（粪便、呕吐物、血液等）
        \item 细菌学检验、毒素测定、血清学试验
    \end{itemize}
\end{enumerate}
\end{keybox}

\subsection{处理措施}

\begin{keybox}
\textbf{食物中毒的应急处理（掌握）：}
\begin{itemize}[leftmargin=*]
    \item \textbf{对患者：}催吐、洗胃（尽早！尤其是化学性中毒）、导泻、灌肠；对症支持；严重者护送至医院救治
    \item \textbf{对爆发：}轻度中毒者就地治疗；重症住院治疗；\keyword{立即收集资料进行流行病学和细菌学检验！}
    \item \textbf{现场处理：}封存可疑食品和原料；彻底清洗消毒加工场所和工具；销毁确定的有毒食品
\end{itemize}
\end{keybox}

\subsection{食品安全事故}

\begin{defbox}
\textbf{食品安全事故（Food Safety Incident）：}食物中毒、食源性疾病、食品污染等源于食品，对人体健康有危害或可能有危害的事故。
\end{defbox}
