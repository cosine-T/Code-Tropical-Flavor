\section{食品安全性毒理学评价}

\subsection{现代食品安全性评价概述}

\begin{defbox}
\textbf{食品安全性毒理学评价：}对食品中的化学、生物和物理因素进行系统的毒理学安全性评估，为食品安全标准的制定和监督管理提供科学依据。
\end{defbox}

现代食品安全性评价不仅仅是传统毒理学的内容，还应考虑：
\begin{enumerate}[leftmargin=*]
    \item 食品的组分、生产工艺流程、卫生学状况、质量标准
    \item 人体研究资料
    \item 残留量研究
    \item 接触量（暴露量）研究
    \item 消费量（人群摄入量）研究
    \item 摄入量风险评估
\end{enumerate}

\begin{keybox}
\textbf{基本执行标准：GB 15193.1--2014《食品安全性毒理学评价程序》}
\end{keybox}

\subsection{适用范围与检验对象}

\textbf{适用范围：}
食品生产、加工、保藏、运输和销售过程中所涉及的可能对健康造成危害的化学、生物和物理因素。

\textbf{检验对象：}
\begin{itemize}[leftmargin=*]
    \item 食品及其原料
    \item 食品添加剂
    \item 新食品原料
    \item 辐照食品
    \item 食品相关产品（包装材料、容器、洗涤剂、消毒剂等）
    \item 食品污染物
\end{itemize}

\subsection{毒性试验的四个阶段}

\begin{keybox}
\textbf{食品安全性毒理学评价的四个阶段（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{第一阶段：急性毒性试验}
    \begin{itemize}
        \item 目的：了解受试物的急性毒性强度和性质
        \item 主要内容：LD$_{50}$（半数致死量）测定、联合急性毒性试验
        \item 观察指标：中毒症状、死亡时间、靶器官损害
    \end{itemize}
    \item \textbf{第二阶段：亚急性毒性试验}
    \begin{itemize}
        \item 目的：了解多次重复染毒条件下受试物的毒性
        \item 主要内容：遗传毒性试验、致畸试验、30天喂养试验
        \item 意义：提供毒性作用的靶器官、剂量-反应关系信息
    \end{itemize}
    \item \textbf{第三阶段：亚慢性毒性试验}
    \begin{itemize}
        \item 目的：观察较长时间重复染毒的毒性效应
        \item 主要内容：90天喂养试验、繁殖试验、代谢试验
        \item 意义：确定NOAEL（未观察到有害作用水平）
    \end{itemize}
    \item \textbf{第四阶段：慢性毒性试验（包括致癌试验）}
    \begin{itemize}
        \item 目的：观察长期（终生）染毒的慢性毒性和致癌性
        \item 主要内容：慢性毒性试验、致癌试验
        \item 时间：一般持续试验动物寿命的大部分时间（大鼠约2年）
    \end{itemize}
\end{enumerate}
\end{keybox}

\subsection{不同受试物的选择原则}

\begin{keybox}
\textbf{不同受试物的试验阶段选择原则（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{创新性物质（全新化学结构）：}一般要求进行\keyword{全部四个阶段}的试验
    \item \textbf{与已知安全性物质化学结构类似的物质：}根据情况提供\keyword{第一、二、三阶段}试验资料
    \item \textbf{多国食用史且规格一致的已知安全物质：}至少进行急性毒性、遗传毒性试验和\keyword{28天喂养}试验
\end{enumerate}
\end{keybox}

\subsection{评价时应考虑的因素}

\begin{enumerate}[leftmargin=*]
    \item \textbf{统计学意义、生物学意义和毒理学意义：}综合判断，不应仅依赖统计学显著性
    \item \textbf{对营养素的可能影响：}对摄入量较大的物质，需关注是否影响营养素的吸收利用
    \item \textbf{时间-毒性关系：}毒效应的发生、发展及恢复的时间规律
    \item \textbf{特殊和易感人群：}孕妇、乳母、儿童等特殊人群的安全性
    \item \textbf{人群资料：}有流行病学调查资料时更优先采用
    \item \textbf{不确定系数（安全系数）：}常规采用\keyword{100倍}（种间差异10 $\times$ 个体差异10）
\end{enumerate}

\begin{tipbox}
\textbf{绝对食品安全不存在！}食品安全是相对的，基于现有科学认知水平的风险可接受性判断。随着科学进步和新的毒理学方法发展，已批准的食品需要不断重新评价。
\end{tipbox}

\section{营养毒理学}

\begin{defbox}
\textbf{营养毒理学（Nutritional Toxicology）：}研究营养素和营养过程与异生物质（Xenobiotics）之间的相互作用，以及营养物质过量摄入对机体造成不良影响及其机制的学科。
\end{defbox}

\subsection{三大研究方向}

\begin{enumerate}[leftmargin=*]
    \item \textbf{营养素过量摄入的不良反应及UL的制定：}
    \begin{itemize}
        \item 营养素并非越多越好！维生素A过量导致肝脏损害，维生素D过量导致高钙血症
        \item 可耐受最高摄入量（UL）的制定依据
    \end{itemize}
    \item \textbf{营养素对外源化学物代谢和毒性的影响：}
    \begin{itemize}
        \item 营养素可影响外来化学物的代谢酶活性
        \item 如蛋白质缺乏可降低细胞色素P450活性，影响药物/毒物代谢
    \end{itemize}
    \item \textbf{膳食中有毒物质对营养素代谢和营养过程的影响：}
    \begin{itemize}
        \item 如植酸影响矿物质吸收，胰蛋白酶抑制剂影响蛋白质消化
    \end{itemize}
\end{enumerate}

\section{食品安全风险分析}

\subsection{风险分析框架}

\begin{keybox}
\textbf{FAO/WHO食品安全风险分析框架——三大模块（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{风险评估（Risk Assessment）——科学基础}
    \begin{itemize}
        \item 基于科学证据，对食品中危害因素可能造成的人体健康风险进行系统评估
        \item 完全以科学为基础，不受政治、经济等非科学因素影响
    \end{itemize}
    \item \textbf{风险管理（Risk Management）——政策基础}
    \begin{itemize}
        \item 在风险评估结果基础上，结合法律、经济、社会等因素
        \item 权衡和选择适当的预防和控制措施的过程
        \item 与各利益相关方磋商，选择最优方案
    \end{itemize}
    \item \textbf{风险交流（Risk Communication）——信息/意见交换}
    \begin{itemize}
        \item 风险评估者、风险管理者、消费者、学术界及其他利益相关方之间
        \item 围绕风险进行的互动式信息和意见交流
    \end{itemize}
\end{enumerate}
\end{keybox}

% ----- 风险评估 -----
\subsection{风险评估（Risk Assessment）}

\begin{keybox}
\textbf{风险评估的四大步骤（掌握*——教学难点）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{危害识别（Hazard Identification）：}
    \begin{itemize}
        \item 确定人体摄入某种食品中的危害因素与健康不良效应之间是否存在\keyword{因果关系}
        \item 信息来源：流行病学研究、动物毒理学试验、体外试验、构效关系分析
        \item 判断依据：证据权重（Weight of Evidence）分析
    \end{itemize}

    \item \textbf{危害特征描述（Hazard Characterization）：}
    \begin{itemize}
        \item 对危害因素可能引起的不良健康效应进行\keyword{定性和/或定量评价}
        \item 核心任务：建立\keyword{剂量-反应关系（Dose-Response Relationship）}
        \item 关键参数：
        \begin{itemize}
            \item \keyword{NOAEL（未观察到有害作用水平）}：No Observed Adverse Effect Level
            \item \keyword{BMDL（基准剂量下限）}：Benchmark Dose Lower Confidence Limit
        \end{itemize}
        \item 对有阈值的有害物质，\keyword{制定健康指导值（HBGV）}
    \end{itemize}

    \item \textbf{暴露评估（Exposure Assessment）：}
    \begin{itemize}
        \item 对通过食品或其他途径摄入人体的生物、化学和物理因素的\keyword{量}进行评估
        \item \keyword{膳食暴露量 = 通过食品途径摄入的化学物量}
        \item 公式：膳食暴露量 = $\sum$(食品中化学物浓度 $\times$ 食品消费量) / 体重
        \item 评估方法：总膳食研究（TDS）、单一食品研究、双份饭法
    \end{itemize}

    \item \textbf{风险描述（Risk Characterization）：}
    \begin{itemize}
        \item 综合前三步的结果，分析和评估危害因素对人群健康产生不良效应的\keyword{风险和严重程度}
        \item 与健康指导值比较：暴露量 $<$ HBGV $\to$ 风险可接受
        \item 暴露边界（MOE, Margin of Exposure）：对遗传毒性致癌物，暴露量与BMDL的比值
        \item 不确定性和变异性的说明
    \end{itemize}
\end{enumerate}
\end{keybox}

% ----- 健康指导值 -----
\subsection{健康指导值（HBGV）}

\begin{keybox}
\textbf{健康指导值（Health-Based Guidance Value, HBGV）——掌握：}
\end{keybox}

\begin{defbox}
\textbf{ADI（Acceptable Daily Intake，每日允许摄入量）：}
人类终生每日摄入某种化学物质，对健康不产生可察觉有害作用的剂量。单位：mg/(kg bw$\cdot$d)。
\end{defbox}

\begin{defbox}
\textbf{ARfD（Acute Reference Dose，急性参考剂量）：}
24小时内或一餐内摄入、对健康不产生有害作用的剂量。
\end{defbox}

\begin{keybox}
\textbf{耐受摄入量（Tolerable Intake, TI）——三种形式（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{TDI（Tolerable Daily Intake，日耐受摄入量）：}
    用于非有意添加到食品中的污染物（如农药残留、环境污染物）
    \item \textbf{PTWI（Provisional Tolerable Weekly Intake，暂定每周耐受摄入量）：}
    用于具有\keyword{蓄积性}的污染物（如重金属：镉、汞、铅）。以每周为时间单位计算
    \item \textbf{PTMI（Provisional Tolerable Monthly Intake，暂定每月耐受摄入量）：}
    用于具有\keyword{强蓄积性且半衰期很长}的物质（如镉）。以每月为时间单位计算
\end{enumerate}
\end{keybox}

\begin{keybox}
\textbf{HBGV推导公式（掌握）：}
\[
\text{HBGV} = \frac{\text{POD}}{\text{UFs}}
\]

其中：
\begin{itemize}[leftmargin=*]
    \item \textbf{POD（Point of Departure，出发点）} = NOAEL 或 BMDL
    \item \textbf{UFs（Uncertainty Factors，不确定系数）}：
    \begin{itemize}
        \item 种间外推差异：10倍（动物 $\to$ 人）
        \item 种内个体差异：10倍（人群中的敏感个体）
        \item 常规合计：\keyword{100倍}
        \item 特殊情况下可增加：数据不充分（$\times$10）、从亚慢性 $\to$ 慢性外推（$\times$10）
    \end{itemize}
\end{itemize}
\end{keybox}

% ----- 风险管理 -----
\subsection{风险管理（Risk Management）}

\begin{keybox}
\textbf{风险管理的常见形式（掌握*）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{食品标准（Food Standards）：}
    \begin{itemize}
        \item 食品中化学物的最大残留限量（MRLs）

        \item 食品标签警示信息要求
    \end{itemize}
    \item \textbf{指南（Guidelines）：}
    \begin{itemize}
        \item 食品污染物控制指南
        \item 微生物控制指南
        \item 良好卫生规范指南
    \end{itemize}
    \item \textbf{操作规程（Operational Procedures）：}
    \begin{itemize}
        \item 良好卫生规范（GHP）
        \item 良好操作规范（GMP）
        \item 营养要求规范
    \end{itemize}
    \item \textbf{培训计划和材料（Training Programs and Materials）：}
    \begin{itemize}
        \item 从业人员食品安全培训
        \item 消费者食品安全教育
    \end{itemize}
\end{enumerate}
\end{keybox}

% ----- 风险交流 -----
\subsection{风险交流（Risk Communication）}

\begin{keybox}
\textbf{风险交流——掌握：}
\end{keybox}

\textbf{风险交流的核心要素：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{风险的性质（Nature of Risk）：}危害因素的类型、涉及的健康效应、受影响的人群范围
    \item \textbf{风险评估的不确定性（Assessment Uncertainty）：}数据来源的局限性、外推的不确定性
    \item \textbf{风险管理的选择（Risk Management Options）：}各项管理措施的利弊、预期效果
    \item \textbf{控制措施如何在管理中发挥作用}
\end{enumerate}

\subsection{Hazard 与 Risk 的概念区别}

\begin{keybox}
\textbf{Hazard（危害） vs Risk（风险/危险性）——核心概念区分（掌握*）：}
\end{keybox}

\begin{defbox}
\textbf{Hazard（危害）：}食品中存在的、可能对健康产生不良影响的生物、化学或物理因素，或食品本身的不安全状态。

危害强调的是\keyword{固有的损害能力（内在属性）}，描述的是"什么东西可能有害"。
\end{defbox}

\begin{defbox}
\textbf{Risk（风险/危险性）：}因接触危害因素而产生不良健康效应的\keyword{概率（Probability）}及该效应的\keyword{严重程度（Severity）}。

风险强调的是\keyword{发生的可能性和后果}，描述的是"有多大可能出问题、出了问题多严重"。
\end{defbox}

\begin{exambox}
\textbf{经典举例说明 Hazard vs Risk：}
\begin{itemize}
    \item 肉毒杆菌毒素本身是Hazard（具有毒性的内在属性）
    \item 但食用充分加热的罐头的Risk很低（概率低，因加热可破坏毒素）
    \item 鲨鱼肉含甲基汞（Hazard），如果每周吃一次且量小（Risk低），如果每天大量吃（Risk高）
\end{itemize}
\end{exambox}

\begin{tipbox}
\textbf{通俗理解：}Hazard = "有毒吗"（本身属性）；Risk = "吃了会中毒吗"（取决于暴露量/概率 $\times$ 后果严重程度）。同一种Hazard在不同暴露场景下的Risk可能完全不同。
\end{tipbox}

% ----- IARC致癌分级 -----
\subsection{IARC人类致癌性分级}

\begin{keybox}
\textbf{IARC（国际癌症研究中心）人类致癌性分级（熟悉）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{1类：人类致癌物（Carcinogenic to Humans）}
    \begin{itemize}
        \item \keyword{有充分证据证明对人类致癌}
        \item 举例：黄曲霉毒素B1、酒精饮料、加工肉类、烟草、苯并(a)芘、砷
    \end{itemize}
    \item \textbf{2A类：对人类很可能致癌（Probably Carcinogenic to Humans）}
    \begin{itemize}
        \item 动物实验充分证据 + 人类证据有限
        \item 举例：红肉、丙烯酰胺、高温油炸排放物、草甘膦
    \end{itemize}
    \item \textbf{2B类：对人类可能致癌（Possibly Carcinogenic to Humans）}
    \begin{itemize}
        \item 人类证据有限 + 动物证据不充分
        \item 举例：腌制蔬菜、咖啡（2016年已降级为3类）、DDT
    \end{itemize}
    \item \textbf{3类：不能分级（Not Classifiable）}
    \begin{itemize}
        \item 人类和动物致癌性证据均不足
        \item 举例：咖啡因、茶、印刷油墨
    \end{itemize}
    \item \textbf{4类：对人类很可能不致癌（Probably Not Carcinogenic）}
    \begin{itemize}
        \item 仅己内酰胺一种物质属于此级
    \end{itemize}
\end{enumerate}
\end{keybox}

% ----- 三级预防 -----
\subsection{三级预防与食品安全风险分析}

\begin{keybox}
\textbf{三级预防（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{一级预防（病因预防）：}疾病发生前，针对致病因素采取措施。如食品安全标准制定、食品污染源头控制
    \item \textbf{二级预防（"三早"预防）：}\keyword{早期发现、早期诊断、早期治疗}。\keyword{食品安全性风险分析 = 二级预防！}
    \item \textbf{三级预防（临床预防）：}疾病发生后采取的对症治疗和康复治疗措施
\end{enumerate}
\end{keybox}

% ----- 毒物学三定律 -----
\begin{tipbox}
\textbf{毒物学三定律：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{剂量决定毒性（Dose Determines Toxicity）：}任何物质在足够剂量下都可能产生毒性（Paracelsus："剂量决定毒物"）
    \item \textbf{化学物的特异性作用：}不同化学物具有各自特定的作用机制和靶器官
    \item \textbf{人类是动物：}动物实验的结果可以在一定条件下外推至人类
\end{enumerate}
\end{tipbox}
