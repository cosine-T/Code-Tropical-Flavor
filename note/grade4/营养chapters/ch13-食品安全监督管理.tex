\section{概述}

\subsection{食品安全监督与食品安全管理}

\begin{defbox}
\textbf{食品安全监督（掌握）：}
国家行政管理部门依法对食品生产、流通和餐饮服务企业的食品安全行使的强制性检查、评价和监督管理活动。
\end{defbox}

\begin{keybox}
\textbf{食品安全监督的特点：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{国家行政监督性质：}属于国家行政管理职能
    \item \textbf{行政监督与行政处罚兼具：}可依法行使监督权和处罚权
    \item \textbf{强制性：}依法强制执行，企业必须配合
    \item \textbf{技术性强：}依靠检测技术和风险评估手段进行科学判断
\end{enumerate}
\end{keybox}

\begin{defbox}
\textbf{食品安全管理（掌握）：}
政府、行业协会、食品企业等各方面对食品生产、流通、消费全过程进行的管理和协调活动。强调\keyword{行业/企业自身管理}，实施方式多样化。
\end{defbox}

\subsection{监督管理范围}

根据《食品安全法》，食品安全监督管理范围包括：
\begin{enumerate}[leftmargin=*]
    \item 食品生产、加工、流通和餐饮服务
    \item 食品添加剂的生产经营
    \item 用于食品的包装材料、容器、洗涤剂、消毒剂、工具、设备等生产经营
    \item 食品生产经营者使用食品添加剂、食品相关产品
    \item 对食品、食品添加剂和食品相关产品的安全管理
\end{enumerate}

\section{食品安全监督管理的原则}

\begin{keybox}
\textbf{《食品安全法》第三条确立的四大原则（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{预防为主（Prevention-oriented）：}
    \begin{itemize}
        \item 在证据和研究尚未完全清楚前即采取预防性措施
        \item 具体体现：
        \begin{itemize}
            \item 食品生产经营许可制度
            \item 食品生产企业强制检验制度
            \item 食品安全标准体系
            \item 食品标签标识制度
        \end{itemize}
    \end{itemize}

    \item \textbf{风险管理（Risk Management）：}
    \begin{itemize}
        \item 将风险评估结果与法律因素结合
        \item 与各利益相关方磋商
        \item 选择最优的预防政策和控制措施
        \item 在食品安全监管全程中贯穿风险管理的理念
    \end{itemize}

    \item \textbf{全程控制（Process Monitoring）：}
    \begin{itemize}
        \item 对\keyword{源头生产 $\to$ 中间流通 $\to$ 终端消费}的全过程进行监督
        \item 从农田到餐桌（From Farm to Table）的全程管理理念
        \item 包括：原料采购、生产加工、储存运输、销售消费全环节
    \end{itemize}

    \item \textbf{社会共治（Social Co-governance）：}
    \begin{itemize}
        \item 全社会共同参与的食品安全治理机制
        \item 主要内容：
        \begin{itemize}
            \item 行业自律（行业协会发挥自律作用）
            \item 社会宣传（普及食品安全知识）
            \item 公众举报权和知情权
            \item 鼓励食品安全科学研究
        \end{itemize}
    \end{itemize}
\end{enumerate}
\end{keybox}

\begin{tipbox}
\textbf{记忆口诀："预防风险全社会"——}预防为主、风险管理、全程控制、社会共治。四大原则是《食品安全法》的基本精神和制度框架。
\end{tipbox}

\section{食品安全法律法规体系}

\subsection{核心法律}

\begin{keybox}
\textbf{《中华人民共和国食品安全法》——掌握：}
\begin{itemize}[leftmargin=*]

    \item \keyword{2015年10月1日：修订版正式实施}——被称为"史上最严"
    \item 2018年12月29日：完成修正
    \item 地位：\keyword{食品安全领域的基本法律}
\end{itemize}
\end{keybox}

\subsection{其他相关法律}

\begin{itemize}[leftmargin=*]
    \item 《农产品质量安全法》（2006年）
    \item 《进出境动植物检疫法》（1992年）
    \item 《标准化法》（2017年修订版）
    \item 《产品质量法》
    \item 《广告法》
\end{itemize}

\subsection{法规层级体系}

\begin{keybox}
\textbf{食品安全法律法规层级（从上至下）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{食品安全法（法律）}——全国人大常委会制定，基本法律
    \item \textbf{食品安全法规}——行政法规（国务院制定）和地方性法规（地方人大制定）
    \item \textbf{食品安全规章}——部门规章（国务院各部委制定）和地方政府规章
    \item \textbf{食品安全标准}——国家标准（强制性）、行业标准、地方标准、团体标准、企业标准
\end{enumerate}
\end{keybox}

\subsection{监管体制演变}

\begin{exambox}
\textbf{中国食品安全监管体制的演变历程：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{建国初期：}卫生部管理，各地卫生防疫站承担具体工作
    \item \textbf{2003年：}国家食品药品监督管理局（SFDA）成立
    \item \textbf{2013年：}组建国家食品药品监督管理总局（CFDA），统一监管
    \item \textbf{2018年：}机构改革，组建\keyword{国家市场监督管理总局（SAMR）}，承担食品安全监管主要职能
\end{enumerate}
\end{exambox}

\subsection{现行食品安全监管架构}

\begin{itemize}[leftmargin=*]
    \item \textbf{国务院食品安全委员会：}食品安全工作的最高协调机构
    \item \textbf{国家市场监督管理总局：}负责食品生产、流通、特殊食品安全监管
    \item \textbf{农业农村部门：}负责食用农产品从种植养殖到进入批发零售市场前的质量安全监管
    \item \textbf{卫生健康部门：}负责食品安全风险监测与评估、国家食品安全标准制定
    \item \textbf{海关部门：}负责进出口食品安全监管
\end{itemize}

\section{食品安全标准}

\subsection{定义}

\begin{defbox}
\textbf{食品安全标准（掌握）：}
对食品中与人类健康相关的质量要素、技术要求、检验方法和评价程序所作的统一规定。
\end{defbox}

\subsection{性质}

\begin{keybox}
\textbf{食品安全标准的三大性质（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{政策性/法规性：}
    \begin{itemize}
        \item 以《食品安全法》为法律依据，具有法定管理属性
        \item 违反标准的食品不得上市销售
    \end{itemize}

    \item \textbf{科学性/技术性：}
    \begin{itemize}
        \item 标准的本质属性
        \item 以毒理学评价和风险评估为基础
        \item 是食品安全监管的核心技术支撑
    \end{itemize}

    \item \textbf{强制性：}
    \begin{itemize}
        \item \keyword{国家食品安全标准是强制执行标准！}
        \item 不得制定其他强制性食品标准（唯一性）
        \item 其他食品标准（行业标准、地方标准等）不具有强制执行力
    \end{itemize}
\end{enumerate}
\end{keybox}

\subsection{内容}

\begin{keybox}
\textbf{食品安全标准的六大内容（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{食品中致病微生物、农药残留、兽药残留、重金属、生物毒素、污染物质的限量规定}
    \item \textbf{食品添加剂的品种、使用范围、用量}
    \item \textbf{专供特定人群（婴幼儿等）食品的营养成分要求}
    \item \textbf{与食品安全和营养有关的标签、标识、说明书要求}
    \item \textbf{食品生产经营过程的卫生要求（与食品安全和营养有关的质量要求）}
    \item \textbf{食品检验方法与规程}
\end{enumerate}
\end{keybox}

\subsection{分类}

\begin{itemize}[leftmargin=*]
    \item \textbf{按适用范围：}
    \begin{itemize}
        \item 基础/通用标准
        \item 产品标准
        \item 生产企业卫生规范标准
        \item 检验方法标准
    \end{itemize}
    \item \textbf{按层级：}
    \begin{itemize}
        \item 国家强制性标准（GB）
        \item 国家推荐性标准（GB/T）
        \item 行业标准
        \item 地方标准
        \item 团体标准
        \item 企业标准
    \end{itemize}
\end{itemize}

\subsection{有害物质限量标准制定步骤}

\begin{keybox}
\textbf{食品中有害物质限量标准的制定程序：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{动物毒性试验} $\to$ 确定NOAEL/L$_{50}$等
    \item \textbf{确定最大无作用剂量（MNL）}
    \item \textbf{应用安全系数（100～1000倍）}
    \item \textbf{计算ADI/TDI（每人每日允许摄入量）}
    \item \textbf{确定每人每日总膳食中允许摄入量}
    \item \textbf{考虑食品种类和人群摄入量，确定单一食品中的最大允许量}
    \item \textbf{结合行业实际情况}
    \item \textbf{最终确定限量标准}
\end{enumerate}
\end{keybox}

\subsection{食品安全标准的主要技术指标}

\begin{keybox}
\textbf{食品安全标准的主要技术指标分类（5类）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{严重危害人体健康的指标：}
    \begin{itemize}[leftmargin=*]
        \item 致病性微生物与毒素：如沙门菌、金黄色葡萄球菌及其肠毒素、真菌毒素等。
        \item 有毒有害化学物质：如砷、铅、汞、镉、多环芳烃类化合物等。
        \item 放射性污染物。
    \end{itemize}
    \item \textbf{反映食品可能被污染及污染程度的指标：}如菌落总数、大肠菌群等——不直接反映危害程度，但指示食品卫生质量。
    \item \textbf{间接反映食品安全质量发生变化的指标：}包括水分含量、含氮化合物、挥发性盐基氮（TVBN）等——食品腐败变质的间接指示物。
    \item \textbf{营养指标：}包括碳水化合物、脂肪、蛋白质、矿物质、维生素等营养素和能量、膳食纤维等。\keyword{专供婴幼儿和其他特定人群的主辅食品的营养成分要求尤其重要。}
    \item \textbf{商品质量指标：}如食品的感官要求（色泽、气味、组织状态）、规格等级等——不直接涉及食品安全，但影响商品价值和消费者接受度。
\end{enumerate}
\end{keybox}

\subsection{国际对比}

\begin{itemize}[leftmargin=*]
    \item \textbf{国际食品安全标准体系：}主要包括\keyword{国际食品法典委员会（CAC）}制定的Codex标准、ISO食品安全管理体系标准（ISO 22000系列）、世界动物卫生组织（OIE）标准等。
    \item \textbf{中国标准与国际主流基本一致：}大部分限量指标与CAC、欧盟、美国标准在同一水平
    \item \textbf{部分品类达国际先进水平}
    \item \textbf{根本区别：}中国食品安全标准是\keyword{强制性}的，CAC标准是\keyword{推荐性}的
\end{itemize}

\section{食品生产的监督管理}

% ----- GMP -----
\subsection{GMP（食品良好生产规范）}

\begin{keybox}
\textbf{GMP（Good Manufacturing Practice，食品良好生产规范）——掌握：}
\end{keybox}

\begin{defbox}
\textbf{GMP定义：}为保障食品安全和质量而制定的贯穿食品生产过程全过程的一系列技术要求、技术措施和技术管理方法。是注重生产过程中产品质量与卫生安全的\keyword{自主性管理制度}。
\end{defbox}

\begin{keybox}
\textbf{GMP的四大管理要素（4M）——掌握*：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Man（人员）：}配备合适且能胜任的从业人员
    \begin{itemize}
        \item 健康检查合格
        \item 经过食品安全知识培训
        \item 良好的个人卫生习惯
    \end{itemize}

    \item \textbf{Material（物料）：}选用良好的原料
    \begin{itemize}
        \item 原料验收标准
        \item 供应商管理
        \item 原料储存条件控制
    \end{itemize}

    \item \textbf{Machine（设备）：}采用标准规范的厂房设施和机械设备
    \begin{itemize}
        \item 厂房布局合理（人流物流分离，避免交叉污染）
        \item 设备易于清洗消毒
        \item 定期维护保养
    \end{itemize}

    \item \textbf{Method（方法）：}遵照既定的最优方法进行生产
    \begin{itemize}
        \item 标准操作规程（SOP）
        \item 工艺参数控制
        \item 关键控制点管理
    \end{itemize}
\end{enumerate}
\end{keybox}

\begin{keybox}
\textbf{GMP的基本精神（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{降低生产过程中的人为错误}
    \item \textbf{防止食品在生产过程中遭到污染和品质劣变}
    \item \textbf{建立企业自主性质量保证体系}
\end{enumerate}
\end{keybox}

\textbf{CAC《食品卫生总则》五大模块（参考）：}
\begin{enumerate}[leftmargin=*]
    \item 初级生产（Primary Production）
    \item 生产控制（Production Control）
    \item 养护和卫生（Maintenance and Hygiene）
    \item 个人卫生（Personal Hygiene）
    \item 卫生与食品安全
\end{enumerate}

% ----- SSOP -----
\subsection{SSOP（卫生标准操作程序）}

\begin{defbox}
\textbf{SSOP（Sanitation Standard Operating Procedure，卫生标准操作程序）——掌握：}
为达到GMP要求而制定的\keyword{书面化}清洁、消毒、卫生维护操作指导文件。
\end{defbox}

\begin{keybox}
\textbf{SSOP的主要特征：}
\begin{itemize}[leftmargin=*]
    \item \textbf{必须是书面文件：}将卫生管理要求标准化、文件化
    \item \textbf{GMP提供总体规范 $\to$ SSOP负责具体落实}
    \item \textbf{内容涵盖：}水（冰）安全、食品接触面清洁、交叉污染预防、洗手消毒设施、食品保护、有毒化合物管理、员工健康管理、虫鼠害防治
\end{itemize}
\end{keybox}

% ----- HACCP -----
\subsection{HACCP（危害分析及关键控制点）}

\begin{keybox}
\textbf{HACCP（Hazard Analysis and Critical Control Point）——掌握*（教学难点）：}
\end{keybox}

\begin{defbox}
\textbf{HACCP定义：}对食品安全/卫生进行系统管理的\keyword{预防性控制体系}。通过识别食品生产过程中的潜在危害，确定关键控制点并采取纠正措施，将危害降低到可接受水平。

\keyword{注意：HACCP不是零风险系统！}目的是将危害减少到可接受水平，并持续改进。
\end{defbox}

\subsubsection{HACCP建立的前提条件}

\begin{keybox}
\textbf{HACCP建立的前提条件（掌握*）：}
HACCP体系\keyword{必须建立在GMP和SSOP的基础之上}！

\begin{itemize}[leftmargin=*]
    \item 无GMP $\to$ 生产环境不能保证基本卫生 $\to$ HACCP难以有效运行
    \item 无SSOP $\to$ 没有具体的卫生操作标准 $\to$ HACCP缺乏实施细节
    \item GMP + SSOP = \keyword{HACCP体系的先决条件和基础}
\end{itemize}
\end{keybox}

\subsubsection{HACCP七大原理}

\begin{keybox}
\textbf{HACCP的七大原理（掌握*——核心内容）：}

\begin{enumerate}[leftmargin=*]
    \item \textbf{原理一：进行危害分析（Hazard Analysis, HA）}
    \begin{itemize}
        \item 对食品生产全过程（原料、生产、加工、储存、运输、消费）进行系统的危害分析
        \item 识别所有潜在的\keyword{生物性、化学性和物理性危害}
        \item 对每个危害进行风险评估（发生概率 $\times$ 危害严重程度）
        \item 确定需要控制的显著危害
    \end{itemize}

    \item \textbf{原理二：确定关键控制点（Critical Control Point, CCP）}
    \begin{itemize}
        \item 关键控制点：能够实施控制的、对防止和消除食品安全危害或将其降低到可接受水平所必需的某一步骤
        \item \textbf{CCP的三种控制形式：}
        \begin{enumerate}
            \item \textbf{防止危害发生：}如pH $<$ 4.6可抑制肉毒梭菌生长
            \item \textbf{消除危害：}如加热杀菌工艺
            \item \textbf{降低到可接受水平：}如冷冻降低微生物生长速率
        \end{enumerate}
        \item \textbf{CCP判定树（Decision Tree）：}通过逻辑推理判断题，逐一回答四问，确定某步骤是否为CCP
    \end{itemize}

    \item \textbf{原理三：建立关键限值（Critical Limit, CL）}
    \begin{itemize}
        \item \textbf{关键限值（CL）：}在CCP上设定的\keyword{最大值和/或最小值}，用于将危害控制在一定范围内
        \item 举例：巴氏消毒最低72$^\circ$C、15秒；罐头pH上限4.6
        \item \textbf{操作限值（OL, Operating Limit）：}企业自定义的更严格的内控标准（比CL更严）
        \item OL的三大优点：
        \begin{enumerate}
            \item 质量优化
            \item 避免超出关键限值
            \item 容忍正常操作误差
        \end{enumerate}
    \end{itemize}

    \item \textbf{原理四：建立CCP监控（Monitoring, M）系统}
    \begin{itemize}
        \item \textbf{监控的四大要素：}
        \begin{enumerate}
            \item \textbf{What（监控什么？）}——如温度、时间、pH值
            \item \textbf{How（怎样监控？）}——如温度计、计时器、pH计
            \item \textbf{Frequency（监控频次？）}——如连续监控/批批检测/定期抽检
            \item \textbf{Who（谁监控？）}——指定经过培训的责任人员
        \end{enumerate}
        \item 监控目的：及时发现CCP是否偏离关键限值，以便立即采取纠正措施
    \end{itemize}

    \item \textbf{原理五：建立纠偏程序（Corrective Action, CA）}
    \begin{itemize}
        \item 当监控发现CCP偏离关键限值时，\keyword{立即}采取的措施
        \item 内容包括：
        \begin{enumerate}
            \item 隔离和评估受影响的产品
            \item 纠正偏离原因，恢复控制
            \item 防止再次发生
            \item 记录全部纠偏活动
        \end{enumerate}
    \end{itemize}

    \item \textbf{原理六：建立验证程序（Verification, V）}
    \begin{itemize}
        \item \textbf{验证的三个维度：}
        \begin{enumerate}
            \item \textbf{适宜性（Suitability）：}HACCP计划是否适合企业实际生产情况？
            \item \textbf{一致性（Consistency）：}实际执行是否与HACCP计划一致？
            \item \textbf{有效性（Effectiveness）：}HACCP体系是否确实有效控制了危害？
        \end{enumerate}
        \item 验证方式：HACCP计划审查、CCP记录审查、现场观察、抽样检测
    \end{itemize}

    \item \textbf{原理七：建立记录保持和文件化（Record-keeping, R）}
    \begin{itemize}
        \item 所有HACCP相关活动必须有\keyword{完整、准确的记录}
        \item 文件包括：HACCP计划文本、危害分析表、CCP监控记录、纠偏记录、验证记录
        \item 记录保存时限：根据法规要求和产品保质期确定
    \end{itemize}
\end{enumerate}
\end{keybox}

\begin{tipbox}
\textbf{七大原理记忆口诀：}"危害分析是关键，确定CCP定界限，监控纠偏验证录"——HA $\to$ CCP $\to$ CL $\to$ M $\to$ CA $\to$ V $\to$ R。
\end{tipbox}

\subsubsection{HACCP体系建立的六大步骤}

\begin{keybox}
\textbf{HACCP体系建立的六大步骤（掌握*）：}

\begin{enumerate}[leftmargin=*]
    \item \textbf{组建HACCP小组（Form HACCP Team）}
    \begin{itemize}
        \item 多学科人员组成（生产技术、质量管理、卫生、工程维修等）
        \item 指定HACCP小组组长
        \item 对成员进行HACCP知识培训
    \end{itemize}

    \item \textbf{全面描述产品（Describe Product Fully）}
    \begin{itemize}
        \item 产品名称、成分、加工方式
        \item 包装形式、储存条件、保质期
        \item 预期用途和消费对象（是否特殊人群？）
    \end{itemize}

    \item \textbf{收集制定HACCP计划所需全部信息（Collect All Information）}
    \begin{itemize}
        \item 原料和服务信息
        \item 加工过程参数
        \item 相关法规和标准
        \item 同类产品的历史安全数据
    \end{itemize}

    \item \textbf{制定工艺流程图并现场验证（Develop and Verify Process Flow Diagram）}
    \begin{itemize}
        \item 绘制从原料接收到成品出厂的完整流程图
        \item \keyword{必须在生产现场进行验证和确认}
        \item 标注所有步骤的时间、温度等关键参数
    \end{itemize}

    \item \textbf{按照七大原理编写正式HACCP计划（Write HACCP Plan）}
    \begin{itemize}
        \item 逐一按七大原理完成危害分析、CCP确定、关键限值设定等
        \item 编制HACCP计划表
        \item 形成正式书面文件
    \end{itemize}

    \item \textbf{全面验证HACCP计划（Comprehensively Verify HACCP Plan）}
    \begin{itemize}
        \item 首次验证：确认HACCP计划科学有效
        \item 重新验证：生产工艺变更时、新危害出现时、发生食品安全事故后
        \item 定期验证：至少每年一次
    \end{itemize}
\end{enumerate}
\end{keybox}

\subsubsection{HACCP核心概念深入解析}

\begin{keybox}
\textbf{危害分析（HA）中的关键概念——掌握：}
\begin{itemize}[leftmargin=*]
    \item \textbf{危害（Hazard）：}食品中可能对健康产生不良影响的\keyword{生物、化学或物理因素}，或食品本身的不安全状态
    \item \textbf{风险（Risk）：}发生危害的\keyword{概率（Probability）}
    \item \textbf{严重性（Severity）：}发生危害后对健康造成损害的\keyword{严重程度}
    \item 风险分析 = 概率 $\times$ 严重性
\end{itemize}
\end{keybox}

\begin{keybox}
\textbf{CCP的三个层次（掌握）：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{CCP1（第一类关键控制点）：}该步骤能\keyword{完全消除危害}（如加热杀菌、金属探测）
    \item \textbf{CCP2（第二类关键控制点）：}该步骤能\keyword{降低但无法完全消除危害}（如冷藏、酸化）
\end{enumerate}
\end{keybox}

\begin{exambox}
\textbf{CCP判定树的逻辑判断四问：}
\begin{enumerate}[leftmargin=*]
    \item Q1：该步骤是否存在显著危害？有预防控制措施吗？（否 $\to$ 不是CCP，停止）
    \item Q2：该步骤是否专门设计用于消除危害或将其降低至可接受水平？（是 $\to$ CCP）
    \item Q3：该步骤是否存在污染超过可接受水平或不可接受程度增加的可能？（否 $\to$ 不是CCP）
    \item Q4：后续步骤是否能消除危害或将其降低至可接受水平？（是 $\to$ 不是CCP；否 $\to$ CCP）
\end{enumerate}
\end{exambox}

    \item \textbf{SSOP} = 标准化卫生操作程序（具体落实GMP的卫生要求）

\section{食品经营的监督管理}

\subsection{食品经营许可制度}

\begin{itemize}[leftmargin=*]
    \item 从事食品生产、食品销售、餐饮服务须依法取得许可
    \item 食品经营许可证由县级以上市场监督管理部门颁发
\end{itemize}

\subsection{食品召回制度}

\begin{keybox}
\textbf{食品召回制度（了解）：}
食品生产经营者发现其生产经营的食品不安全时，应立即\keyword{停止生产、主动召回}，并通知相关生产经营者和消费者，记录召回和通知情况。
\end{keybox}

\textbf{召回的两种类型：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{主动召回（Voluntary Recall）：}企业自行发现不安全食品后主动实施
    \item \textbf{责令召回（Mandatory Recall）：}监管部门发现不安全食品后责令企业召回
\end{enumerate}

\subsection{"三新食品"审批制度}

\begin{keybox}
\textbf{"三新食品"审批：}
\begin{enumerate}[leftmargin=*]
    \item \textbf{新食品原料（Novel Food Raw Materials）：}如新资源食品
    \item \textbf{新的食品添加剂品种（New Food Additive Varieties）}
    \item \textbf{新的食品相关产品新品种（New Food-related Product Varieties）}
\end{enumerate}
由国家卫生健康委员会审批。
\end{keybox}

\subsection{经营场所卫生要求}

\begin{itemize}[leftmargin=*]
    \item \textbf{选址要求：}远离污染源（化工厂、垃圾场等）
    \item \textbf{建筑结构：}坚固耐用，地面防滑防水不渗水
    \item \textbf{防鼠设施：}外门设1.5m高度防鼠板；库房设"两夹一盒"（挡鼠板+鼠夹+毒饵盒）
    \item \textbf{通风良好：}保持干燥，防止霉菌生长
    \item \textbf{用水卫生：}饮用水符合GB 5749生活饮用水卫生标准
\end{itemize}
